29/// Every declaration in one style-rule body, plus a warning for each
30/// one that fell outside the subset. `first_line` narrows the subset
31/// to what `::first-line` takes.
32pub(super) fn declarations(
33    input: &mut Parser<'_, '_>,
34    sheet: &str,
35    first_line: bool,
36) -> (Vec<(Declaration, Importance)>, Vec<Written>, Vec<Warning>) {
37    let mut properties = Properties {
38        first_line,
39        sheet: sheet.to_string(),
40    };
41    let mut kept = Vec::new();
42    let mut written = Vec::new();
43    let mut warnings = Vec::new();
44    for result in RuleBodyParser::new(input, &mut properties) {
45        match result {
46            Ok(parsed) => {
47                let importance = parsed.importance;
48                written.push(parsed.written(kept.len()));
49                kept.extend(parsed.declarations.into_iter().map(|d| (d, importance)))
50            }
51            Err((error, _)) => warnings.push(warning(sheet, &error)),
52        }
53    }
54    (kept, written, warnings)
55}

57/// One declaration read: what it expanded to, and how it was written.
58pub(super) struct Parsed<D> {
59    pub(super) declarations: Vec<D>,
60    pub(super) importance: Importance,
61    pub(super) property: String,
62    pub(super) value: String,
63    pub(super) position: SheetPosition,
64}

66impl<D> Parsed<D> {
67    /// How it was written, over the longhands it adds after the `at`
68    /// a rule already holds.
69    pub(super) fn written(&self, at: usize) -> Written {
70        Written {
71            property: self.property.clone(),
72            value: self.value.clone(),
73            important: self.importance == Importance::Important,
74            longhands: at..at + self.declarations.len(),
75            position: self.position.clone(),
76        }
77    }
78}

80/// Pins a declaration's diagnostic to where the declaration began.
81/// A reader looks for the property name, not for the token the parser
82/// gave up on.
83pub(super) fn at<'i: 't, 't, T>(
84    start: &ParserState,
85    parse: impl FnOnce(&mut Parser<'i, 't>) -> Result<T, ParseError<'i, StyleError<'i>>>,
86) -> impl FnOnce(&mut Parser<'i, 't>) -> Result<T, ParseError<'i, StyleError<'i>>> {
87    let location = start.source_location();
88    move |input| {
89        parse(input).map_err(|error| ParseError {
90            kind: error.kind,
91            location,
92        })
93    }
94}

96/// The declaration parser for style rules.
97struct Properties {
98    /// Whether the rule this body belongs to selects `::first-line`.
99    first_line: bool,
100    /// What diagnostics call the sheet, which a url carries with it
101    /// so that a missing image names where it was asked for.
102    sheet: String,
103}
104
105/// What one declaration expands to: a shorthand is several longhands.
106type Longhands = Parsed<Declaration>;
107
108impl<'i> DeclarationParser<'i> for Properties {
109    type Declaration = Longhands;
110    type Error = StyleError<'i>;
111
112    fn parse_value<'t>(
113        &mut self,
114        name: CowRcStr<'i>,
115        input: &mut Parser<'i, 't>,
116        start: &ParserState,
117    ) -> Result<Self::Declaration, ParseError<'i, Self::Error>> {
118        at(start, |input| {
119            let custom = name.starts_with("--");
120            if !custom
121                && self.first_line
122                && !FIRST_LINE_PROPERTIES.contains(&&*name.to_ascii_lowercase())
123            {
124                return Err(input.new_custom_error(StyleError::NotOnFirstLine(name.clone())));
125            }
126            let from = input.position();
127            let origin = || position(&self.sheet, start.source_location());
128            let mut declarations = if custom {
129                vec![Declaration::Custom(Custom {
130                    name: name.to_string(),
131                    value: raw(input),
132                    origin: origin(),
133                })]
134            } else if mentions_var(input) {
135                let Some(spec) = Spec::find(PROPERTIES, &name) else {
136                    return Err(
137                        input.new_custom_error(StyleError::UnsupportedProperty(name.clone()))
138                    );
139                };
140                vec![Declaration::Pending(Pending {
141                    property: spec.name.to_string(),
142                    value: raw(input),
143                    origin: origin(),
144                })]
145            } else {
146                property(&name, input)?
147            };
148            let value = input.slice_from(from).trim().to_string();
149            written_at(&mut declarations, &self.sheet, start);
150            let importance = if input.try_parse(cssparser::parse_important).is_ok() {
151                Importance::Important
152            } else {
153                Importance::Normal
154            };
155            input.expect_exhausted()?;
156            Ok(Parsed {
157                declarations,
158                importance,
159                property: if custom {
160                    name.to_string()
161                } else {
162                    name.to_ascii_lowercase()
163                },
164                value,
165                position: SheetPosition::at(&self.sheet, start.source_location()),
166            })
167        })(input)
168    }
169}

193/// One property the parser reads: the name it matches, what the
194/// value may be, and the reader that turns the value into
195/// declarations. The tables of these are the whole of what the
196/// engine claims to understand, and the description in `subset` is
197/// read off them.
198///
199/// `syntax` is CSS value-definition syntax. A bare word in it is a
200/// keyword, `<name>` a type, `name()` a function. `examples` are
201/// values that parse, one for each form the syntax allows.
202pub(crate) struct Spec<D> {
203    pub name: &'static str,
204    /// Whether a child starts from the parent's value.
205    pub inherited: bool,
206    pub syntax: &'static str,
207    pub examples: &'static [&'static str],
208    pub(super) read: Reader<D>,
209}

211/// Reads one declaration's value, from after the colon to the end of
212/// the declaration.
213type Reader<D> = for<'i, 't> fn(
214    &CowRcStr<'i>,
215    &mut Parser<'i, 't>,
216) -> Result<Vec<D>, ParseError<'i, StyleError<'i>>>;
217
218impl<D> Spec<D> {
219    /// The spec for `name`, whatever case it was written in.
220    pub(crate) fn find<'a>(specs: &'a [Spec<D>], name: &str) -> Option<&'a Spec<D>> {
221        specs
222            .iter()
223            .find(|spec| spec.name.eq_ignore_ascii_case(name))
224    }
225
226    /// Reads a declaration of this property.
227    pub(crate) fn read<'i>(
228        &self,
229        name: &CowRcStr<'i>,
230        input: &mut Parser<'i, '_>,
231    ) -> Result<Vec<D>, ParseError<'i, StyleError<'i>>> {
232        (self.read)(name, input)
233    }
234}

236/// Pins every url in a declaration to where the declaration was
237/// written. Probing happens long after parsing, so the position has
238/// to travel with the url rather than be looked up again.
239pub(super) fn written_at(declarations: &mut [Declaration], sheet: &str, start: &ParserState) {
240    for declaration in declarations {
241        if let Declaration::BackgroundImage(Some(url)) = declaration {
242            url.origin = Some(position(sheet, start.source_location()));
243        }
244    }
245}

247/// Reads one longhand: `parse` for the value, `wrap` for the
248/// declaration it becomes.
249pub(super) fn longhand<'i, T, D>(
250    name: &CowRcStr<'i>,
251    input: &mut Parser<'i, '_>,
252    parse: fn(&mut Parser<'i, '_>) -> Option<T>,
253    wrap: fn(T) -> D,
254) -> Result<Vec<D>, ParseError<'i, StyleError<'i>>> {
255    let value = keyword_or(input, parse, || StyleError::UnsupportedValue(name.clone()))?;
256    Ok(vec![wrap(value)])
257}

259/// Reads a one-to-four value shorthand (`margin`, `padding`,
260/// `border-width`) into its four longhands.
261pub(super) fn sides<'i, T: Copy, D>(
262    name: &CowRcStr<'i>,
263    input: &mut Parser<'i, '_>,
264    parse: fn(&mut Parser<'_, '_>) -> Option<T>,
265    wrap: fn(Edge, T) -> D,
266) -> Result<Vec<D>, ParseError<'i, StyleError<'i>>> {
267    let edges = edges(input, parse)
268        .ok_or_else(|| input.new_custom_error(StyleError::UnsupportedValue(name.clone())))?;
269    Ok(edges
270        .into_iter()
271        .map(|(edge, value)| wrap(edge, value))
272        .collect())
273}

275/// Reads `border` or one of its per-edge longhands: a width, a style
276/// and a colour in any order, any of them left out. What is left out
277/// goes back to its initial value, as the shorthand asks.
278fn border<'i>(
279    name: &CowRcStr<'i>,
280    input: &mut Parser<'i, '_>,
281    edges: &[Edge],
282) -> Result<Vec<Declaration>, ParseError<'i, StyleError<'i>>> {
283    let (mut width, mut style, mut color) = (None, None, None);
284    loop {
285        if width.is_none()
286            && let Some(value) = input.try_parse(|input| line_width(input).ok_or(())).ok()
287        {
288            width = Some(value);
289        } else if style.is_none()
290            && let Some(value) = input.try_parse(|input| line_style(input).ok_or(())).ok()
291        {
292            style = Some(value);
293        } else if color.is_none()
294            && let Some(value) = input.try_parse(|input| self::color(input).ok_or(())).ok()
295        {
296            color = Some(value);
297        } else {
298            break;
299        }
300    }
301    if width.is_none() && style.is_none() && color.is_none() {
302        return Err(input.new_custom_error(StyleError::UnsupportedValue(name.clone())));
303    }
304    let width = width.unwrap_or(Length::Points(MEDIUM));
305    let style = style.unwrap_or(BorderStyle::None);
306    Ok(edges
307        .iter()
308        .flat_map(|edge| {
309            [
310                Declaration::BorderWidth(*edge, width),
311                Declaration::BorderStyle(*edge, style),
312                Declaration::BorderColor(*edge, color),
313            ]
314        })
315        .collect())
316}

318/// Reads `text-decoration`: the rules, a style, a colour and a
319/// thickness in any order, any of them left out. What is left out
320/// goes back to its initial value, as the shorthand asks.
321fn text_decoration<'i>(
322    name: &CowRcStr<'i>,
323    input: &mut Parser<'i, '_>,
324) -> Result<Vec<Declaration>, ParseError<'i, StyleError<'i>>> {
325    let (mut line, mut style, mut color, mut thickness) = (None, None, None, None);
326    loop {
327        if let Ok(one) = input.try_parse(|input| decoration_keyword(input).ok_or(())) {
328            let drawn: &mut DecorationLine = line.get_or_insert(DecorationLine::NONE);
329            drawn.under |= one.under;
330            drawn.over |= one.over;
331            drawn.through |= one.through;
332        } else if style.is_none()
333            && let Ok(value) = input.try_parse(|input| decoration_style(input).ok_or(()))
334        {
335            style = Some(value);
336        } else if thickness.is_none()
337            && let Ok(value) = input.try_parse(|input| decoration_thickness(input).ok_or(()))
338        {
339            thickness = Some(value);
340        } else if color.is_none()
341            && let Ok(value) = input.try_parse(|input| decoration_color(input).ok_or(()))
342        {
343            color = Some(value);
344        } else {
345            break;
346        }
347    }
348    if line.is_none() && style.is_none() && color.is_none() && thickness.is_none() {
349        return Err(input.new_custom_error(StyleError::UnsupportedValue(name.clone())));
350    }
351    Ok(vec![
352        Declaration::TextDecorationLine(line.unwrap_or(DecorationLine::NONE)),
353        Declaration::TextDecorationStyle(style.unwrap_or_default()),
354        Declaration::TextDecorationColor(color.flatten()),
355        Declaration::TextDecorationThickness(thickness.flatten()),
356    ])
357}
