25/// One node's resolved style: what every downstream pass reads.
26#[derive(Debug, Clone, PartialEq, Serialize)]
27pub struct ComputedStyle {
28    /// The face this style shapes with, resolved against the
29    /// registry. Family, slope and weight chose it; layout only needs
30    /// the answer.
31    pub font_id: u16,
32    /// The families asked for, in order, kept for diagnostics and for
33    /// the snapshot the style tree is reviewed through.
34    pub font_family: Vec<Family>,
35    /// Font size in points.
36    pub font_size: f32,
37    /// Upright or italic.
38    pub font_style: FontStyle,
39    /// Weight on the CSS 1–1000 scale.
40    pub font_weight: u16,
41    /// What text and rules are painted in.
42    pub color: Color,
43    /// Line height as a unitless multiple of the font size.
44    pub line_height: f32,
45    /// Extra advance after every glyph, in points, from
46    /// `letter-spacing`.
47    pub letter_spacing: f32,
48    /// Which capitals the run is drawn with.
49    pub font_variant_caps: FontVariantCaps,
50    /// The features the sheet asked the face for, from
51    /// `font-feature-settings`. They are asked for after the ones the
52    /// `font-variant` longhands name, so a tag in both is set here.
53    #[serde(skip_serializing_if = "Vec::is_empty")]
54    pub font_feature_settings: Vec<FeatureSetting>,
55    /// Which ligatures the run is set with.
56    #[serde(skip_serializing_if = "default_ligatures")]
57    pub font_variant_ligatures: FontVariantLigatures,
58    /// Which figures the run is set with.
59    #[serde(skip_serializing_if = "default_numeric")]
60    pub font_variant_numeric: FontVariantNumeric,
61    /// Which alternate glyphs the run is set with.
62    #[serde(skip_serializing_if = "default_alternates")]
63    pub font_variant_alternates: FontVariantAlternates,
64    /// What the run's letters are transformed to before shaping.
65    pub text_transform: TextTransform,
66    /// Which rules are drawn across the run, from
67    /// `text-decoration-line`.
68    #[serde(skip_serializing_if = "undecorated")]
69    pub text_decoration_line: DecorationLine,
70    /// What those rules are painted in, from
71    /// `text-decoration-color`. `None` is the colour of the text.
72    #[serde(skip_serializing_if = "Option::is_none")]
73    pub text_decoration_color: Option<Color>,
74    /// How each rule is drawn, from `text-decoration-style`.
75    #[serde(skip_serializing_if = "solid")]
76    pub text_decoration_style: DecorationStyle,
77    /// How thick each rule is, in points, from
78    /// `text-decoration-thickness`. `None` is the thickness the face
79    /// declares.
80    #[serde(skip_serializing_if = "Option::is_none")]
81    pub text_decoration_thickness: Option<f32>,
82    /// How lines fill the measure.
83    pub text_align: TextAlign,
84    /// What justification opens up to fill it.
85    pub text_justify: TextJustify,
86    /// Which marks may hang past the measure.
87    pub hanging_punctuation: HangingPunctuation,
88    /// First-line indent in points.
89    pub text_indent: f32,
90    /// Whether words may break at syllable boundaries.
91    pub hyphens: Hyphens,
92    /// Lines that must be left at the bottom of a fragment.
93    pub orphans: u16,
94    /// Lines that must be moved to the top of the next one.
95    pub widows: u16,
96    /// The named page this element's pages take, from `page`.
97    pub page: Option<String>,
98    /// What the element paints in place of children it has none of:
99    /// the ornament a thematic break is set with.
100    pub content: Content,
101    /// The named strings this element sets when the flow reaches it,
102    /// from `string-set`. This is where a running head's text comes
103    /// from.
104    #[serde(skip_serializing_if = "Vec::is_empty")]
105    pub string_set: Vec<StringSet>,
106    /// The folio the page this element opens takes, from
107    /// `counter-reset: page`.
108    #[serde(skip_serializing_if = "Option::is_none")]
109    pub counter_reset: Option<u32>,
110    /// The number the first note under this element takes, from
111    /// `counter-reset: note`. On the footnote area it restarts the
112    /// numbering on every page.
113    #[serde(skip_serializing_if = "Option::is_none")]
114    pub note_reset: Option<u32>,
115    /// Lines an initial letter is sunk over, from `initial-letter`.
116    /// Fewer than two is no drop cap.
117    pub initial_letter: u16,
118    /// Whether the element sits in the flow or against the page.
119    #[serde(skip_serializing_if = "in_flow")]
120    pub position: Position,
121    /// What the insets say, for an element that is against the page.
122    #[serde(skip_serializing_if = "unplaced")]
123    pub inset: Edges<Inset>,
124    /// Which layer the block paints in, from `z-index`. Higher paints
125    /// later, and `auto` is layer 0.
126    #[serde(skip_serializing_if = "ground_layer")]
127    pub z_index: i32,
128    /// How much of what the block paints shows, from `opacity`: 1 is
129    /// all of it and 0 is none. The blocks inside it show that much
130    /// of their own.
131    #[serde(skip_serializing_if = "opaque")]
132    pub opacity: f32,
133    /// How what the element draws is mixed with what is under it,
134    /// from `mix-blend-mode`.
135    #[serde(skip_serializing_if = "BlendMode::is_normal")]
136    pub mix_blend_mode: BlendMode,
137    /// The mode of the nearest element around this one that names
138    /// one. An element is part of what the elements around it draw.
139    #[serde(skip)]
140    pub blend_around: BlendMode,
141    /// Which side of this element the prose sets on, from `wrap-flow`.
142    #[serde(skip_serializing_if = "wraps_nothing")]
143    pub wrap_flow: WrapFlow,
144    /// The contour the prose sets around, from `shape-outside`.
145    #[serde(skip_serializing_if = "wraps_its_box")]
146    pub shape_outside: ShapeOutside,
147    /// How far the prose keeps off that contour, in points, from
148    /// `shape-margin`.
149    #[serde(skip_serializing_if = "no_shape_margin")]
150    pub shape_margin: f32,
151    /// Margins in points.
152    pub margin: Edges,
153    /// Padding in points, between the border and the content.
154    #[serde(skip_serializing_if = "no_padding")]
155    pub padding: Edges,
156    /// The four border edges.
157    #[serde(skip_serializing_if = "no_border")]
158    pub border: Edges<Border>,
159    /// How far each corner of the border box is rounded, from
160    /// `border-radius`. The background and the border follow the
161    /// corners.
162    #[serde(skip_serializing_if = "square")]
163    pub border_radius: BorderRadius,
164    /// What is painted behind the block: a tint, and an image over
165    /// it.
166    #[serde(skip_serializing_if = "no_background")]
167    pub background: Background,
168    /// What the decoration does where a page break splits the block.
169    #[serde(skip_serializing_if = "sliced")]
170    pub box_decoration_break: BoxDecorationBreak,
171    /// The width the box is asked to take, from `width`. A table reads
172    /// it off the cells of its first row to size its columns.
173    #[serde(skip_serializing_if = "auto_width")]
174    pub width: Width,
175    /// The height the content box is asked to take, from `height`.
176    #[serde(skip_serializing_if = "auto_width")]
177    pub height: Width,
178    /// The least height the content box takes, from `min-height`.
179    #[serde(skip_serializing_if = "auto_width")]
180    pub min_height: Width,
181    /// The most width an image takes, from `max-width`. `Auto` is
182    /// `none`.
183    #[serde(skip_serializing_if = "auto_width")]
184    pub max_width: Width,
185    /// The most height an image takes, from `max-height`. `Auto` is
186    /// `none`.
187    #[serde(skip_serializing_if = "auto_width")]
188    pub max_height: Width,
189    /// Whether a table's cells share their borders, from
190    /// `border-collapse`.
191    #[serde(skip_serializing_if = "separate")]
192    pub border_collapse: BorderCollapse,
193    /// The marker a list item is set with, from `list-style-type`.
194    #[serde(skip_serializing_if = "disc")]
195    pub list_style_type: ListStyleType,
196    /// Where a page break falls before this element.
197    pub break_before: Break,
198    /// Where one falls after it.
199    pub break_after: Break,
200    /// Whether this element may be split across pages.
201    pub break_inside: Break,
202    /// Whether this element is set across every column of a divided
203    /// page, from `column-span`.
204    #[serde(skip_serializing_if = "one_column")]
205    pub column_span: ColumnSpan,
206    /// The custom properties in force, by name, each with every
207    /// `var()` in its value already replaced. A child inherits them.
208    #[serde(skip_serializing_if = "BTreeMap::is_empty")]
209    pub custom: BTreeMap<String, String>,
210}

315    /// Applies one declaration. `parent_size` is what `em` and
316    /// percentages in `font-size` measure against; `root_size` is
317    /// what `rem` does.
318    pub fn apply(&mut self, declaration: &Declaration, parent_size: f32, root_size: f32) {
319        match declaration {
320            Declaration::FontFamily(families) => self.font_family = families.clone(),
321            Declaration::FontSize(length) => {
322                self.font_size = length.to_points(parent_size, root_size).max(0.0)
323            }
324            Declaration::FontStyle(style) => self.font_style = *style,
325            Declaration::FontWeight(weight) => self.font_weight = *weight,
326            Declaration::Color(color) => self.color = *color,
327            Declaration::LineHeight(line_height) => {
328                self.line_height = line_height.to_multiple(self.font_size, root_size)
329            }
330            Declaration::LetterSpacing(length) => {
331                self.letter_spacing = length.to_points(self.font_size, root_size)
332            }
333            Declaration::FontVariantCaps(caps) => self.font_variant_caps = *caps,
334            Declaration::FontFeatureSettings(settings) => {
335                self.font_feature_settings = settings.clone()
336            }
337            Declaration::FontVariantLigatures(ligatures) => {
338                self.font_variant_ligatures = *ligatures
339            }
340            Declaration::FontVariantNumeric(numeric) => self.font_variant_numeric = *numeric,
341            Declaration::FontVariantAlternates(alternates) => {
342                self.font_variant_alternates = *alternates
343            }
344            Declaration::TextTransform(transform) => self.text_transform = *transform,
345            Declaration::TextDecorationLine(line) => self.text_decoration_line = *line,
346            Declaration::TextDecorationColor(color) => self.text_decoration_color = *color,
347            Declaration::TextDecorationStyle(style) => self.text_decoration_style = *style,
348            Declaration::TextDecorationThickness(length) => {
349                self.text_decoration_thickness =
350                    length.map(|length| length.to_points(self.font_size, root_size).max(0.0))
351            }
352            Declaration::TextAlign(align) => self.text_align = *align,
353            Declaration::TextJustify(justify) => self.text_justify = *justify,
354            Declaration::HangingPunctuation(hanging) => self.hanging_punctuation = *hanging,
355            Declaration::TextIndent(length) => {
356                self.text_indent = length.to_points(self.font_size, root_size)
357            }
358            Declaration::Hyphens(hyphens) => self.hyphens = *hyphens,
359            Declaration::Orphans(lines) => self.orphans = *lines,
360            Declaration::Widows(lines) => self.widows = *lines,
361            Declaration::Page(name) => self.page = name.clone(),
362            Declaration::Content(content) => self.content = content.clone(),
363            Declaration::StringSet(sets) => self.string_set = sets.clone(),
364            Declaration::CounterReset(counters) => {
365                self.counter_reset = counters.page;
366                self.note_reset = counters.note;
367            }
368            Declaration::InitialLetter(lines) => self.initial_letter = *lines,
369            Declaration::Position(position) => self.position = *position,
370            Declaration::Inset(edge, length) => {
371                *self.inset.edge(*edge) = match length {
372                    None => Inset::Auto,
373                    Some(Length::Percent(percent)) => Inset::Percent(*percent),
374                    Some(length) => Inset::Points(length.to_points(self.font_size, root_size)),
375                }
376            }
377            Declaration::ZIndex(layer) => self.z_index = *layer,
378            Declaration::Opacity(opacity) => self.opacity = *opacity,
379            Declaration::MixBlendMode(mode) => self.mix_blend_mode = *mode,
380            Declaration::WrapFlow(wrap) => self.wrap_flow = *wrap,
381            Declaration::ShapeOutside(shape) => {
382                self.shape_outside = match shape {
383                    ShapeSource::None => ShapeOutside::None,
384                    ShapeSource::Auto => ShapeOutside::Auto,
385                    ShapeSource::Polygon(points) => ShapeOutside::Polygon(
386                        points
387                            .iter()
388                            .map(|(x, y)| ShapePoint {
389                                x: Coord::of(*x, self.font_size, root_size),
390                                y: Coord::of(*y, self.font_size, root_size),
391                            })
392                            .collect(),
393                    ),
394                }
395            }
396            Declaration::ShapeMargin(length) => {
397                self.shape_margin = length.to_points(self.font_size, root_size).max(0.0)
398            }
399            Declaration::Margin(edge, length) => {
400                *self.margin.edge(*edge) = length.to_points(self.font_size, root_size)
401            }
402            Declaration::Padding(edge, length) => {
403                *self.padding.edge(*edge) = length.to_points(self.font_size, root_size).max(0.0)
404            }
405            Declaration::BorderStyle(edge, style) => self.border.edge(*edge).style = *style,
406            Declaration::BorderWidth(edge, length) => {
407                self.border.edge(*edge).width = length.to_points(self.font_size, root_size).max(0.0)
408            }
409            Declaration::BorderColor(edge, color) => self.border.edge(*edge).color = *color,
410            Declaration::BorderRadius(corner, x, y) => {
411                *self.border_radius.corner(*corner) = CornerRadius {
412                    x: Coord::of(*x, self.font_size, root_size),
413                    y: Coord::of(*y, self.font_size, root_size),
414                }
415            }
416            Declaration::BackgroundColor(color) => self.background.color = *color,
417            Declaration::BackgroundImage(url) => self.background.image = url.clone(),
418            Declaration::BackgroundRepeat(repeat) => self.background.repeat = *repeat,
419            Declaration::BackgroundSize(size) => {
420                self.background.size = computed_size(*size, self.font_size, root_size)
421            }
422            Declaration::BackgroundPosition(x, y) => {
423                self.background.position = BackgroundPosition {
424                    x: Coord::of(*x, self.font_size, root_size),
425                    y: Coord::of(*y, self.font_size, root_size),
426                }
427            }
428            Declaration::BoxDecorationBreak(value) => self.box_decoration_break = *value,
429            Declaration::Width(width) => self.width = self.size(*width, root_size),
430            Declaration::Height(height) => self.height = self.size(*height, root_size),
431            Declaration::MinHeight(height) => self.min_height = self.size(*height, root_size),
432            Declaration::MaxWidth(width) => self.max_width = self.size(*width, root_size),
433            Declaration::MaxHeight(height) => self.max_height = self.size(*height, root_size),
434            Declaration::BorderCollapse(value) => self.border_collapse = *value,
435            Declaration::ListStyleType(value) => self.list_style_type = *value,
436            Declaration::BreakBefore(value) => self.break_before = *value,
437            Declaration::BreakAfter(value) => self.break_after = *value,
438            Declaration::BreakInside(value) => self.break_inside = *value,
439            Declaration::ColumnSpan(value) => self.column_span = *value,
440            // The cascade resolves both before anything is applied.
441            Declaration::Custom(_) | Declaration::Pending(_) => {}
442        }
443    }

566    /// What this style, computed for `::first-line`, changes about
567    /// the element it was computed beside.
568    pub fn first_line_over(&self, element: &ComputedStyle) -> crate::lines::FirstLine {
569        fn set<T>(differs: bool, value: T) -> Option<T> {
570            differs.then_some(value)
571        }
572        let face = crate::lines::Face {
573            family: set(self.font_family != element.font_family, self.font_id),
574            italic: set(
575                self.font_style != element.font_style,
576                self.font_style == FontStyle::Italic,
577            ),
578            weight: set(self.font_weight != element.font_weight, self.font_weight),
579        };
580        crate::lines::FirstLine {
581            face: (face != crate::lines::Face::default()).then_some(face),
582            size: set(self.font_size != element.font_size, self.font_size),
583            letter_spacing: set(
584                self.letter_spacing != element.letter_spacing,
585                self.letter_spacing,
586            ),
587            caps: set(
588                self.font_variant_caps != element.font_variant_caps,
589                self.font_variant_caps,
590            ),
591            transform: set(
592                self.text_transform != element.text_transform,
593                self.text_transform,
594            ),
595            color: set(self.color != element.color, self.color),
596            decoration: set(self.decoration() != element.decoration(), self.decoration()),
597        }
598    }

600    /// This block's border box inside `measure`, laid out at `x` from
601    /// the enclosing content box's leading edge: `(leading edge,
602    /// width)`, with only the margins taken off.
603    pub fn border_box(&self, x: f32, measure: f32) -> (f32, f32) {
604        (
605            x + self.margin.left,
606            (measure - self.margin.inline()).max(0.0),
607        )
608    }

610    /// Whether prose is set around this element rather than over it:
611    /// the sheet has taken it out of the flow, and it excludes a side.
612    ///
613    /// This is the one place a contour is read, so it is also what
614    /// decides whether one is traced. An element still in the flow
615    /// takes a band of its own and nothing sets beside it, so a
616    /// contour on it would answer a question nobody asks.
617    pub fn excludes(&self) -> bool {
618        self.position == Position::Absolute && self.wrap_flow != WrapFlow::Auto
619    }

621    /// This block's content box inside `measure`: `(leading edge, the
622    /// measure its lines break to)`. Margin, border and padding come
623    /// off both edges, and the leading edge moves in by what they
624    /// take on the left.
625    pub fn content_box(&self, x: f32, measure: f32) -> (f32, f32) {
626        let border = self.border.widths();
627        let leading = self.margin.left + border.left + self.padding.left;
628        let trailing = self.margin.right + border.right + self.padding.right;
629        (x + leading, (measure - leading - trailing).max(0.0))
630    }

632    /// The box this style paints around the runs of an inline
633    /// element, and `None` where it paints none and takes no width.
634    ///
635    /// Padding counts even where nothing is painted over it: it is
636    /// width, and the line breaks against it.
637    pub fn inline_box(&self) -> Option<crate::lines::InlineBox> {
638        let padding = self.padding;
639        let border = self.border.widths();
640        let paints = self.background.paints() || self.border.paints();
641        let spaced = padding != Edges::all(0.0);
642        (paints || spaced).then_some(crate::lines::InlineBox {
643            padding,
644            border,
645            cloned: self.box_decoration_break == BoxDecorationBreak::Clone,
646        })
647    }

699/// What the cascade makes of a written `background-size`: `em` and
700/// `rem` against the font size in force, a percentage kept as one,
701/// because the box it measures against is sized in the layout pass.
702pub(crate) fn computed_size(size: SizeSource, font_size: f32, root_size: f32) -> BackgroundSize {
703    let axis =
704        |length: Option<Length>| length.map(|length| Coord::of(length, font_size, root_size));
705    match size {
706        SizeSource::Auto => BackgroundSize::Auto,
707        SizeSource::Cover => BackgroundSize::Cover,
708        SizeSource::Contain => BackgroundSize::Contain,
709        SizeSource::Fixed(width, height) => BackgroundSize::Fixed {
710            width: axis(width),
711            height: axis(height),
712        },
713    }
714}
