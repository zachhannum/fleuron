1//! Layout: box construction, inline layout, fragmentation.
2//!
3//! ```text
4//! content + style ─► box tree ─► line layout ─► fragmentation ─► pages
5//! ```
6//!
7//! v0.2 folds the middle of the pipeline into one pass: each section
8//! becomes fragments (each block with the style the tree computed for
9//! it, via `lines::LineLayout`), and the paginator flows those
10//! fragments into page content boxes. Nothing here decides what
11//! anything looks like — the style tree was told, and this asks it.
12//!
13//! A fragment is what the flow can move: one line, one image, one
14//! ornament. Where a page may end is decided when the fragments are
15//! built — orphans, widows, `break-inside`, an ornament that must
16//! keep the prose around it — and the flow only stacks and, when
17//! something does not fit, walks back to the last place a break was
18//! allowed.
19//!
20//! The stages have a file each: `fragment` is what the flow moves,
21//! `build` turns blocks into fragments, `cap` sets the initial letter
22//! beside them, `list` sets a list an item at a time with the marker
23//! of each, `table` sets a table a row at a time, `flow` stacks
24//! fragments into pages, `exclusion` places the images and blocks the
25//! sheet anchored and wraps prose around them, `background` puts art
26//! behind a box, `inline` paints the box an inline element takes on a
27//! line, `furniture` paints the margin boxes, and `text` turns a
28//! shaped line into paint ops.

79/// How many times a book whose notes are numbered by page is laid

85/// One book through the whole pipeline: lines laid out, flowed into
86/// pages, everything the output needs assembled.
87///
88/// A single run over a session that retains nothing. It keeps one
89/// section's lines at a time, which is what a process that renders a
90/// book once and exits wants. A live preview uses `Session` instead.
91///
92/// `assets` is the images the host probed. A book with none of them
93/// passes [`Assets::none`].
94pub fn layout_book(
95    book: &Book,
96    styles: &StyleTree,
97    registry: &FontRegistry,
98    assets: &Assets,
99) -> LayoutOutput {
100    Session::once(book, styles, registry, assets).into_output()
101}

122/// The pagination pass: content in, `Page`s of `DrawItem`s out.
123///
124/// Fragments stack from the top of the content box. One that does not
125/// fit ends the page — at the last point a break was allowed, which
126/// may be several fragments back — and a fragment taller than a whole
127/// page overflows it.
128pub struct Paginator<'a> {
129    registry: &'a FontRegistry,
130    styles: &'a StyleTree,
131    assets: &'a Assets,
132    /// What the trace stage made of the assets the sheet asked to
133    /// wrap around.
134    contours: &'a Contours,
135    lines: LineLayout<'a>,
136    /// The syllable patterns `hyphens: auto` breaks by.
137    patterns: Cell<Patterns>,
138    /// The declared language there are no patterns for, which the
139    /// first paragraph that asks to be hyphenated complains about.
140    unknown: RefCell<Option<String>>,
141    /// What fragmentation had to complain about. Recorded once per
142    /// message: a book that scales the same image twice has one
143    /// problem, not two.
144    warnings: RefCell<Vec<Warning>>,
145    /// The characters set with no glyph. Recorded once for each
146    /// character and source, so the same message stands once under
147    /// every file that has the character.
148    uncovered: RefCell<Vec<UncoveredWarning>>,
149    /// Whether the sheet anchors anything to the page, answered once.
150    wraps: OnceCell<bool>,
151    /// How many times the flow set a paragraph again beside an image.
152    rebreaks: Cell<u32>,
153    /// What the references in the book resolve against.
154    references: RefCell<References>,
155    /// What the notes of the book are numbered.
156    notes: RefCell<Numbering>,
157    /// How many times the book was laid out again to print the pages
158    /// its references name.
159    settles: Cell<u32>,
160}

162/// A character set with no glyph, and what the run says about it.
163///
164/// Kept apart from the other warnings, which are recorded once for
165/// each message. This one is recorded once for each character and
166/// source.
167#[derive(Debug, Clone, PartialEq)]
168pub(crate) struct UncoveredWarning {
169    pub(crate) character: char,
170    /// The file the character was written in.
171    pub(crate) source: Option<String>,
172    pub(crate) warning: Warning,
173}

175impl UncoveredWarning {
176    /// The warnings of `found`, in order, once for each character
177    /// and source.
178    pub(crate) fn warnings<'f>(
179        found: impl IntoIterator<Item = &'f UncoveredWarning>,
180    ) -> Vec<Warning> {
181        let mut seen = HashSet::new();
182        found
183            .into_iter()
184            .filter(|found| seen.insert((found.source.as_deref(), found.character)))
185            .map(|found| found.warning.clone())
186            .collect()
187    }
188}

235impl Paginator<'_> {
236    /// Takes the hyphenation patterns from the language the book
237    /// declares. `paginate` reads them from the book it is handed. A
238    /// caller that builds one section's fragments on its own sets
239    /// them here.
240    ///
241    /// A language with no patterns leaves every word whole, rather
242    /// than breaking one language's words at another's syllables,
243    /// and the first paragraph that asks for hyphenation says so.
244    pub fn language(&self, metadata: &Metadata) {
245        let patterns = Patterns::of(metadata);
246        self.patterns.set(patterns);
247        *self.unknown.borrow_mut() = metadata
248            .language()
249            .filter(|_| patterns == Patterns::NONE)
250            .map(str::to_string);
251    }
252
253    /// What `hyphens: auto` has to break with, complaining the first
254    /// time it is asked for and there is nothing behind the language
255    /// the book declares.
256    fn patterns(&self) -> Patterns {
257        if let Some(tag) = self.unknown.borrow().as_deref() {
258            self.warn(
259                format!(
260                    "No hyphenation patterns for `{tag}`. Hyphenation is skipped for that \
261                     language."
262                ),
263                None,
264            );
265        }
266        self.patterns.get()
267    }
268
269    /// What fragmentation had to complain about.
270    pub fn warnings(&self) -> Vec<Warning> {
271        let mut warnings = self.flow_warnings();
272        let uncovered = self.uncovered.borrow();
273        warnings.extend(uncovered.iter().map(|found| found.warning.clone()));
274        warnings
275    }

277    /// The same, without the characters set with no glyph.
278    pub(crate) fn flow_warnings(&self) -> Vec<Warning> {
279        self.warnings.borrow().clone()
280    }

282    /// The characters set with no glyph, once for each character and
283    /// source.
284    pub(crate) fn uncovered(&self) -> Vec<UncoveredWarning> {
285        self.uncovered.borrow().clone()
286    }

288    /// Files the characters shaping found no glyph for under
289    /// `source`, the file they were written in.
290    fn uncover(&self, source: Option<&str>) {
291        let mut uncovered = self.uncovered.borrow_mut();
292        for found in self.lines.take_uncovered() {
293            let seen = uncovered
294                .iter()
295                .any(|seen| seen.character == found.character && seen.source.as_deref() == source);
296            if seen {
297                continue;
298            }
299            let face = self
300                .registry
301                .font_ref(found.font_id)
302                .map(|entry| entry.name.clone())
303                .unwrap_or_default();
304            let at = origin(source, found.position);
305            uncovered.push(UncoveredWarning {
306                character: found.character,
307                source: source.map(str::to_string),
308                warning: Warning {
309                    message: format!(
310                        "{face} has no glyph for `{}` (U+{:04X}).",
311                        found.character, found.character as u32,
312                    ),
313                    origin: (!at.is_empty()).then_some(at),
314                },
315            });
316        }
317    }

342    /// A book that anchors nothing never sets a paragraph twice, so

345        *self.wraps.get_or_init(|| {

353    /// Records one diagnostic, once. A book that hits the same
354    /// problem on every page has one problem.
355    fn warn(&self, message: String, origin: Option<String>) {
356        let mut warnings = self.warnings.borrow_mut();
357        if !warnings.iter().any(|seen| seen.message == message) {
358            warnings.push(Warning { message, origin });
359        }
360    }

362    /// The style a note's reference is set in: the style of the
363    /// note, and the face's superior figures over it. The engine
364    /// asks for them, the way it asks for small capitals, because
365    /// `font-feature-settings` inherits and a rule on the note would
366    /// set the prose of the note in superiors as well.
367    ///
368    /// A face with no superior figures sets the reference on the
369    /// baseline at the size the note gives it, and says so once.
370    fn superior(&self, mut style: ParagraphStyle) -> ParagraphStyle {
371        const SUPS: [u8; 4] = *b"sups";
372        if self.registry.has_feature(style.font_id, SUPS) {
373            style.features.push(FeatureSetting::new(SUPS, 1));
374            return style;
375        }
376        let family = self
377            .registry
378            .font_ref(style.font_id)
379            .map(|entry| entry.family.clone())
380            .unwrap_or_default();
381        self.warn(
382            format!(
383                "{family} has no superior figures. The reference of a note stands on the \
384                 baseline."
385            ),
386            None,
387        );
388        style
389    }

391    /// Says so where the host supplied no image for a url. The table
392    /// complains about a url it probed and refused. A url the table
393    /// was never offered means the host supplied nothing at all.
394    /// What one box paints behind its content, complaining where the
395    /// sheet named an image nothing answers for.
396    fn backdrop(&self, background: &Background) -> Backdrop {
397        let found = background.image.as_ref().and_then(|url| {
398            let found = self.assets.lookup(&url.value);
399            if found.is_none() {
400                self.missing(&url.value, url.origin.clone().unwrap_or_default());
401            }
402            found
403        });
404        Backdrop::of(background, found)
405    }

407    fn missing(&self, url: &str, origin: String) {
408        if !self.assets.probed(url) {
409            self.warn(
410                format!("No image was supplied for {url}. The image is skipped."),
411                (!origin.is_empty()).then_some(origin),
412            );
413        }
414    }

416    /// Flows one book into numbered, side-tagged pages.
417    ///
418    /// A section's fragments are built, flowed, and released before
419    /// the next one is measured: what exists at once is the book's
420    /// pages, not every line it was ever broken into.
421    ///
422    /// A book whose references print pages is laid out twice: once to
423    /// find the page each element lands on, and once to print it.
424    pub fn paginate(&self, book: &Book) -> Vec<Page> {
425        self.paginated(book).pages
426    }

428    /// The same, with where each id landed and the box of each block.
429    pub(crate) fn paginated(&self, book: &Book) -> Paged {
430        self.language(&book.metadata);
431        if self.styles.refers() {
432            self.refer(References::of(book));
433        }
434        self.number(Numbering::of(book, self.styles));
435        let mut paged = self.pass(book);
436        if self.styles.numbers_notes_per_page() {
437            paged = self.settle_notes(book, paged);
438        }
439        if self.styles.counts_pages() {
440            let found = landed(&paged);
441            let resolved = self.references.borrow().landed(found.clone());
442            self.refer(resolved);
443            self.settles.set(self.settles.get() + 1);
444            paged = self.pass(book);
445            let references = self.references.borrow();
446            let printed: BTreeSet<_> = book
447                .sections
448                .iter()
449                .flat_map(|section| Named::in_section(section, self.styles, &references).pages)
450                .collect();
451            for warning in moved(&found, &landed(&paged), &printed, &references) {
452                self.warn(warning.message, warning.origin);
453            }
454        }
455        self.paint(&mut paged.pages, &paged.infos);
456        paged
457    }

459    /// Numbers the notes by the page their references were set on,
460    /// and lays the book out again to print those numbers.
461    ///
462    /// A number of another width moves the line its reference is on,
463    /// which can move a note onto another page and number it again.
464    /// So the book is laid out until the numbering stops changing,
465    /// and the numbering of the last pass stands where it does not.
466    fn settle_notes(&self, book: &Book, mut paged: Paged) -> Paged {
467        let start = self.styles.first_note_number();
468        for _ in 0..NOTE_PASSES {
469            let numbered = self.notes.borrow().on_pages(&paged.notes, start);
470            if numbered == *self.notes.borrow() {
471                return paged;
472            }
473            self.number(numbered);
474            self.settles.set(self.settles.get() + 1);
475            paged = self.pass(book);
476        }
477        self.warn(note::UNSETTLED.to_string(), None);
478        paged
479    }

481    /// One pass over the whole book, stopping short of the furniture.
482    fn pass(&self, book: &Book) -> Paged {
483        let anchored = self.anchored(book, |index| {
484            Cow::Owned(self.section_fragments(&book.sections[index]))
485        });
486        let mut flow = Flow::new(self, anchored);
487        for section in &book.sections {
488            let fragments = self.section_fragments(section);
489            flow.section(section, &fragments);
490        }
491        flow.finish()
492    }

494    /// The boxes the sheet anchors, each on the page its anchor
495    /// landed on and against the block its insets measure from.
496    ///
497    /// A box inside a positioned block is laid out a second time: the
498    /// first layout breaks its lines to what the page area leaves it,
499    /// and the settling pass answers which block they measure from and
500    /// how wide that block is.
501    fn anchored<'f>(
502        &self,
503        book: &Book,
504        mut fragments: impl FnMut(usize) -> Cow<'f, [Fragment]>,
505    ) -> AnchoredBoxes {
506        let boxes = self.anchored_boxes(book, &[]);
507        if boxes.is_empty() {
508            return AnchoredBoxes::default();
509        }
510        let mut settled = self.settle(book, boxes, &mut fragments);
511        if let Some(within) = settled.narrowed() {
512            let boxes = self.anchored_boxes(book, &within);
513            settled.relaid(boxes, within);
514        }
515        settled
516    }

518    /// Fragments in, numbered pages out: fragmentation and page
519    /// assembly, one `Vec<Fragment>` per section of `book`. Nothing
520    /// here measures — every fragment arrives with its box decided.
521    pub fn flow(&self, book: &Book, sections: &[Vec<Fragment>]) -> Vec<Page> {
522        let mut paged = self.fragment(book, sections.iter().map(Vec::as_slice));
523        self.paint(&mut paged.pages, &paged.infos);
524        paged.pages
525    }
526
527    /// The same, stopping short of the furniture: pages as the flow
528    /// settled them, and what each one needs to paint its own.
529    pub(crate) fn fragment<'f>(
530        &self,
531        book: &Book,
532        sections: impl IntoIterator<Item = &'f [Fragment]>,
533    ) -> Paged {
534        let sections: Vec<&[Fragment]> = sections.into_iter().collect();
535        let anchored = self.anchored(book, |index| Cow::Borrowed(sections[index]));
536        let mut flow = Flow::new(self, anchored);
537        for (section, fragments) in book.sections.iter().zip(&sections) {
538            flow.section(section, fragments);
539        }
540        flow.finish()
541    }

570    /// Pagination is line layout then flow, and splitting it that way
571    /// changes nothing: the harness times the two halves separately,
572    /// which is only worth doing while their composition is the whole.
573    #[test]
574    fn the_stages_compose_into_what_paginate_does() {
575        let book = book_of(vec![
576            section(long_prose(30)),
577            section([vec![heading("Two")], long_prose(24)].concat()),
578        ]);
579        let styles = crate::style::defaults(&book, registry());
580        let paginator = Paginator::new(registry(), &styles);
581
582        let staged: Vec<Vec<Fragment>> = book
583            .sections
584            .iter()
585            .map(|section| paginator.section_fragments(section))
586            .collect();
587        let by_stage = paginator.flow(&book, &staged);
588        let in_one = paginator.paginate(&book);
589
590        assert!(in_one.len() > 2, "a book worth splitting");
591        assert_eq!(by_stage.len(), in_one.len());
592        for (staged, whole) in by_stage.iter().zip(&in_one) {
593            assert_eq!(staged.number, whole.number);
594            assert_eq!(staged.side, whole.side);
595            assert_eq!(format!("{:?}", staged.items), format!("{:?}", whole.items));
596        }
597    }
