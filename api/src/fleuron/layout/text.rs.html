12    /// One fragment as paint ops. `x` is the leading edge its own `x`
13    /// is measured from, and `top` is where the top of its box falls.
14    pub(super) fn fragment_items(&self, fragment: &Fragment, x: f32, top: f32) -> Vec<DrawItem> {
15        let (x, top) = (x + fragment.offset.0, top + fragment.offset.1);
16        let mut items = Vec::new();
17        for marker in fragment.markers.iter().flat_map(|markers| markers.iter()) {
18            // A marker sits on the baseline of the line its item opens
19            // with. Beside anything else, it sits at the top.
20            let baseline = match &fragment.piece {
21                Piece::Line { line, .. } => line.box_.baseline,
22                _ => marker.line.box_.baseline,
23            };
24            items.append(&mut self.text_items(
25                &marker.line,
26                x + marker.x,
27                top + baseline,
28                fragment.layer,
29            ));
30        }
31        items.append(&mut match &fragment.piece {
32            Piece::Line { line, cap } => {
33                let baseline = top + line.box_.baseline;
34                let mut items = self.inline_items(line, x + fragment.x, baseline, fragment.layer);
35                items.append(&mut self.text_items(line, x + fragment.x, baseline, fragment.layer));
36                if let Some(cap) = cap {
37                    items.append(&mut self.text_items(
38                        &cap.line,
39                        x + cap.x,
40                        baseline + cap.drop,
41                        fragment.layer,
42                    ));
43                }
44                items
45            }
46            Piece::Image {
47                width,
48                height,
49                asset,
50                blend,
51            } => vec![DrawItem::Image {
52                x: x + fragment.x,
53                y: top,
54                w: *width,
55                h: *height,
56                asset: *asset,
57                alpha: 255,
58                layer: fragment.layer,
59                blend: *blend,
60            }],
61            // A row's items were painted with the layers of the
62            // table they belong to.
63            Piece::Row(row) => {
64                let mut items = row.items.clone();
65                shift(&mut items, x + fragment.x, top);
66                items
67            }
68            Piece::Float(float) => vec![DrawItem::Image {
69                x: x + fragment.x + float.x,
70                y: top + float.y,
71                w: float.width,
72                h: float.height,
73                asset: float.asset,
74                alpha: 255,
75                layer: fragment.layer,
76                blend: float.blend,
77            }],
78            Piece::Blank | Piece::Anchor(_) => Vec::new(),
79        });
80        fade(&mut items, fragment.opacity);
81        items
82    }

84    /// One string as a single shaped line: page furniture, and the
85    /// ornaments and initial letters that are content but not prose.
86    pub(super) fn line_of(&self, text: &str, style: &ParagraphStyle) -> Option<Line> {
87        // One line holds no hard break: the newline a heading's break
88        // reads as is a word space here.
89        let spaced;
90        let text = if text.contains('\n') {
91            spaced = text.replace('\n', " ");
92            spaced.as_str()
93        } else {
94            text
95        };
96        let runs = self.lines.shape(text, style)?;
97        let box_ = self.lines.line_box(&runs, style);
98        Some(Line::of(runs, box_))
99    }

101    /// Design units per em of a face, for the one conversion that
102    /// takes shaped advances into points.
103    fn upem(&self, font_id: u16) -> f32 {
104        self.registry
105            .metrics(font_id)
106            .map(|m| m.units_per_em as f32)
107            .unwrap_or(1000.0)
108    }

110    /// A line's width in points. Runs of different sizes each convert
111    /// against their own face: font units do not commute across sizes.
112    /// What hangs into a margin is not part of the width, which is the
113    /// point of hanging it.
114    pub(super) fn line_width(&self, line: &Line) -> f32 {
115        line.runs.iter().map(|run| self.run_width(run)).sum::<f32>()
116            - line.overhang
117            - line.protrusion
118    }
119
120    /// One run in points, the edges of the inline boxes that open and
121    /// close on it included: what it takes across the line.
122    fn run_width(&self, run: &crate::lines::ShapedRun) -> f32 {
123        run.lead + run.advance as f32 / self.upem(run.font_id) * run.size + run.trail
124    }
125
126    /// One span's width in points. Runs of different sizes each
127    /// convert against their own face: font units do not commute
128    /// across sizes. A band hangs into the margins at its own two
129    /// ends, never at a span boundary inside it.
130    pub(super) fn span_width(&self, line: &Line, index: usize) -> f32 {
131        let span = &line.spans[index];
132        let ink = line.runs[span.runs.clone()]
133            .iter()
134            .map(|run| self.run_width(run))
135            .sum::<f32>();
136        let overhang = if index + 1 == line.spans.len() {
137            line.overhang
138        } else {
139            0.0
140        };
141        let protrusion = if index == 0 { line.protrusion } else { 0.0 };
142        ink - overhang - protrusion
143    }

145    /// One line as paint ops in `layer`: every run a
146    /// `DrawItem::Text` at the baseline, glyphs placed at their
147    /// accumulated advances, and each span of the line opened at its
148    /// own origin.
149    pub(super) fn text_items(
150        &self,
151        line: &Line,
152        x: f32,
153        baseline: f32,
154        layer: i32,
155    ) -> Vec<DrawItem> {
156        let mut items = Vec::new();
157        for span in line.spans.iter() {
158            let mut x_cursor = x + span.offset;
159            for run in &line.runs[span.runs.clone()] {
160                let upem = self.upem(run.font_id);
161                // The leading edges of the inline boxes that open on
162                // this run come before its first glyph.
163                x_cursor += run.lead;
164                let mut glyphs = Vec::with_capacity(run.glyphs.len());
165                let mut glyph_x = x_cursor;
166                for (shaped, range) in run.glyphs.iter().zip(run.glyph_ranges()) {
167                    glyphs.push(Glyph {
168                        id: shaped.id,
169                        x: glyph_x,
170                        range,
171                    });
172                    glyph_x += shaped.x_advance as f32 / upem * run.size;
173                }
174                items.push(DrawItem::Text {
175                    x: x_cursor,
176                    y: baseline,
177                    font_id: run.font_id,
178                    size: run.size,
179                    width: glyph_x - x_cursor,
180                    text: run.text.clone(),
181                    source: run.source.clone(),
182                    source_map: run.source_map.clone(),
183                    origin: run.origin.clone(),
184                    pseudo_element: run.pseudo_element,
185                    features: run.features.clone(),
186                    color: run.color,
187                    glyphs,
188                    layer,
189                    blend: run.blend,
190                });
191                items.append(&mut self.decoration_items(
192                    run,
193                    x_cursor,
194                    baseline,
195                    glyph_x - x_cursor,
196                    layer,
197                ));
198                x_cursor = glyph_x + run.trail;
199            }
200        }
201        items
202    }

204    /// The rules `text-decoration` draws across one run, each a rect
205    /// over the advance the run's glyphs took. They come after the
206    /// glyphs, so a rule is painted over the letters it crosses.
207    ///
208    /// A run is what one line holds, so a decoration a line break
209    /// splits is drawn on both lines, each at that line's own width.
210    fn decoration_items(
211        &self,
212        run: &ShapedRun,
213        x: f32,
214        baseline: f32,
215        width: f32,
216        layer: i32,
217    ) -> Vec<DrawItem> {
218        let decoration = run.decoration;
219        if !decoration.draws() || width <= 0.0 {
220            return Vec::new();
221        }
222        let metrics = self.registry.metrics(run.font_id);
223        let color = decoration.color.unwrap_or(run.color);
224        let mut items = Vec::new();
225        let mut rules = Vec::new();
226        if decoration.line.over {
227            rules.push(overline(metrics, run.size));
228        }
229        if decoration.line.through {
230            rules.push(line_through(metrics, run.size));
231        }
232        if decoration.line.under {
233            rules.push(underline(metrics, run.size));
234        }
235        for (top, thickness) in rules {
236            let thickness = decoration.thickness.unwrap_or(thickness).max(0.0);
237            if thickness <= 0.0 {
238                continue;
239            }
240            for rule in 0..decoration.style.rules() {
241                items.push(DrawItem::Rect {
242                    x,
243                    y: baseline + top + rule as f32 * thickness * 2.0,
244                    w: width,
245                    h: thickness,
246                    color,
247                    layer,
248                    blend: run.blend,
249                });
250            }
251        }
252        items
253    }

256/// Where one rule sits under a face's own baseline, and how thick it
257/// is, both in points at `size`. The offset is positive downward,
258/// which is the direction the page measures in.
259fn placed(metrics: Option<FontMetricsTable>, rule: Option<Rule>, size: f32) -> Option<(f32, f32)> {
260    let metrics = metrics?;
261    let rule = rule?;
262    let scale = size / metrics.units_per_em.max(1) as f32;
263    Some((-rule.offset as f32 * scale, rule.thickness as f32 * scale))
264}
265
266/// A rule under the text, where the face's `post` table puts it.
267fn underline(metrics: Option<FontMetricsTable>, size: f32) -> (f32, f32) {
268    placed(metrics, metrics.and_then(|m| m.underline), size)
269        .unwrap_or((size * FALLBACK_UNDERLINE, size * FALLBACK_THICKNESS))
270}
271
272/// A rule across the text, where the face's `OS/2` table puts it.
273fn line_through(metrics: Option<FontMetricsTable>, size: f32) -> (f32, f32) {
274    placed(metrics, metrics.and_then(|m| m.strikeout), size)
275        .unwrap_or((-size * FALLBACK_STRIKEOUT, size * FALLBACK_THICKNESS))
276}
277
278/// A rule over the text. No table declares one, so it sits at the
279/// ascent, as thick as the underline is.
280fn overline(metrics: Option<FontMetricsTable>, size: f32) -> (f32, f32) {
281    let thickness = underline(metrics, size).1;
282    let top = match metrics {
283        Some(metrics) => -(metrics.ascender as f32) * size / metrics.units_per_em.max(1) as f32,
284        None => -size * FALLBACK_ASCENT,
285    };
286    (top, thickness)
287}

289/// Where the rules fall on a face that declares none, as fractions
290/// of the em. A face without a `post` table is rare, and a book is
291/// better served by a rule a little out of place than by none.
292const FALLBACK_UNDERLINE: f32 = 0.1;
293const FALLBACK_STRIKEOUT: f32 = 0.25;
294const FALLBACK_THICKNESS: f32 = 0.05;
295const FALLBACK_ASCENT: f32 = 0.8;

304    /// A run's `text` is what was drawn, and its `source` is what was
305    /// written. The two painters read different fields for different
306    /// reasons: one draws characters and hands `text` to a browser,
307    /// the other maps glyphs back and reads `source`. A run with
308    /// only the manuscript in it would have the preview set a
309    /// chapter title in the case the export does not.
310    #[test]
311    fn a_run_says_both_what_was_drawn_and_what_was_written() {
312        let pages = paginate_styled(
313            "h1 { text-transform: uppercase }",
314            vec![section(vec![heading("A Voyage to Lilliput")])],
315        );
316        let title = pages[0]
317            .items
318            .iter()
319            .find_map(|item| match item {
320                DrawItem::Text {
321                    text,
322                    source,
323                    source_map,
324                    glyphs,
325                    ..
326                } => Some((text, source, source_map, glyphs)),
327                _ => None,
328            })
329            .expect("the chapter set its title");
330        let (text, source, source_map, glyphs) = title;
331        assert_eq!(
332            text, "A VOYAGE TO LILLIPUT",
333            "the title was not drawn in capitals"
334        );
335        assert_eq!(
336            source, "A Voyage to Lilliput",
337            "the title lost its manuscript"
338        );
339        assert_eq!(
340            source_map.len(),
341            text.len() + 1,
342            "the map does not cover every byte boundary of what was drawn",
343        );
344        // Every glyph's range indexes what was drawn, and taken
345        // through the map it reads back the manuscript entire.
346        let read: String = glyphs
347            .iter()
348            .map(|glyph| {
349                let from = source_map[glyph.range.start as usize] as usize;
350                let to = source_map[glyph.range.end as usize] as usize;
351                &source[from..to]
352            })
353            .collect();
354        assert_eq!(read, *source, "the glyphs do not read the manuscript back");
355
356        // A book nothing transformed says nothing twice over.
357        let plain = paginate(vec![section(vec![heading("A Voyage to Lilliput")])]);
358        assert!(
359            plain[0].items.iter().all(|item| !matches!(
360                item,
361                DrawItem::Text { source, .. } if !source.is_empty()
362            )),
363            "an untransformed run has a source of its own",
364        );
365    }

367    /// Glyph positions accumulate advances: the first glyph paints at
368    /// the item origin and each subsequent glyph sits one shaped
369    /// advance past its predecessor, in points.
370    #[test]
371    fn glyphs_are_placed_at_their_advances() {
372        let pages = paginate(vec![section(vec![paragraph("hello")])]);
373        let DrawItem::Text {
374            x, glyphs, size, ..
375        } = &pages[0].items[0]
376        else {
377            panic!("expected text");
378        };
379        assert_eq!(glyphs.len(), 5);
380        assert!((glyphs[0].x - *x).abs() < 1e-4);
381        let shaped = registry().shape(0, "hello").unwrap();
382        let upem = registry().metrics(0).unwrap().units_per_em as f32;
383        let mut expected_x = *x;
384        for (glyph, shaped_glyph) in glyphs.iter().zip(&shaped) {
385            assert!(
386                (glyph.x - expected_x).abs() < 1e-3,
387                "glyph at {}, expected {expected_x}",
388                glyph.x
389            );
390            expected_x += shaped_glyph.x_advance as f32 / upem * size;
391        }
392    }
