1//! What an inline element paints on one line: the tint behind its
2//! runs, and the border around them.
3//!
4//! A block has one border box per page and an inline element has one
5//! per line. Line layout works the extent out, because it is the
6//! stage that knows where the runs of one element fell; this turns
7//! each of them into paint ops, on the page the line landed on.
8
9use crate::lines::Line;
10use crate::pages::{DrawItem, Radius};
11use crate::style::Edges;
12
13use super::Paginator;
14use super::flow::{box_items, mix};
15
16impl Paginator<'_> {
17    /// The boxes the inline elements on one line paint there, in
18    /// `layer`. `x` is the line's own leading edge and `baseline` its
19    /// baseline.
20    ///
21    /// They come out outermost first, and before the line's glyphs:
22    /// a chip is behind the word it tints.
23    pub(super) fn inline_items(
24        &self,
25        line: &Line,
26        x: f32,
27        baseline: f32,
28        layer: i32,
29    ) -> Vec<DrawItem> {
30        let mut items = Vec::new();
31        for fragment in &line.boxes {
32            let (w, h) = (fragment.width, fragment.above + fragment.below);
33            if w <= 0.0 || h <= 0.0 {
34                continue;
35            }
36            let style = self.styles.style(fragment.node);
37            let backdrop = self.backdrop(&style.background);
38            let ink = |edge: crate::style::Border| edge.color.unwrap_or(style.color);
39            let colors = Edges {
40                top: ink(style.border.top),
41                right: ink(style.border.right),
42                bottom: ink(style.border.bottom),
43                left: ink(style.border.left),
44            };
45            // The box goes on past an edge the line break left open,
46            // so that edge is not drawn and its corners are square.
47            let mut border = style.border.widths();
48            let mut radii = style.border_radius.resolve(w, h);
49            if !fragment.opens {
50                border.left = 0.0;
51                (radii.top_left, radii.bottom_left) = (Radius::SQUARE, Radius::SQUARE);
52            }
53            if !fragment.closes {
54                border.right = 0.0;
55                (radii.top_right, radii.bottom_right) = (Radius::SQUARE, Radius::SQUARE);
56            }
57            let offset = line
58                .spans
59                .get(fragment.span)
60                .map(|span| span.offset)
61                .unwrap_or_default();
62            let mut painted = box_items(
63                x + offset + fragment.x,
64                baseline - fragment.above,
65                w,
66                h,
67                radii,
68                border,
69                colors,
70                &backdrop,
71                layer,
72            );
73            mix(&mut painted, style.blend());
74            items.append(&mut painted);
75        }
76        items
77    }
78}

80#[cfg(test)]
81mod tests {
82    use crate::layout::testing::{paginate_styled, quote, section, tagged_paragraph};
83    use crate::pages::DrawItem;
84    use crate::style::Color;
85
86    /// A quotation with a tint of its own, holding a paragraph with a
87    /// tinted tag in it.
88    const NESTED_CSS: &str = "blockquote { background-color: #f4f1ea; padding: 6pt } \
89                              code { background-color: #858585; padding: 2pt 4pt }";
90
91    /// Part: a box an inline element paints goes over the background
92    /// of the block around it and under the glyphs it sits behind.
93    #[test]
94    fn an_inline_box_paints_between_the_block_and_its_glyphs() {
95        let pages = paginate_styled(
96            NESTED_CSS,
97            vec![section(vec![quote(vec![tagged_paragraph(
98                "roll for ",
99                "2d6",
100                " and add the modifier",
101            )])])],
102        );
103        let items = &pages[0].items;
104        let rect = |ink: Color| {
105            items
106                .iter()
107                .position(|item| matches!(item, DrawItem::Rect { color, .. } if *color == ink))
108                .unwrap_or_else(|| panic!("nothing was painted in {}", ink.to_hex()))
109        };
110        let block = rect(Color::rgb(0xf4, 0xf1, 0xea));
111        let chip = rect(Color::rgb(0x85, 0x85, 0x85));
112        let tag = items
113            .iter()
114            .position(|item| matches!(item, DrawItem::Text { text, .. } if text == "2d6"))
115            .expect("the tag was set");
116        assert!(
117            block < chip && chip < tag,
118            "the order was block {block}, chip {chip}, tag {tag}",
119        );
120    }
