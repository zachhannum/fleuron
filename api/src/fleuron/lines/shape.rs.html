13    /// One string as shaped runs, set the way `style` asks for it.
14    /// The counterpart of `layout` for text that is not broken into
15    /// lines: page furniture, an ornament, an initial letter.
16    pub fn shape(&self, text: &str, style: &ParagraphStyle) -> Option<Vec<ShapedRun>> {
17        let upem = self.registry.metrics(style.font_id)?.units_per_em as f32;
18        let mut flat = FlatParagraph::new();
19        flat.push_styled(text, style, self.small_caps(style));
20        let shaped = self.shape_spans(&flat, style, upem);
21        Some(cut_runs(&flat, &shaped, 0, flat.text.len()))
22    }
23
24    /// Shapes each span. A glyph's cluster is an offset into the
25    /// span, which indexes the paragraph text once offset by the
26    /// span's start. A span set at another size measures in its own
27    /// font units, so `scale` takes it into the paragraph's.
28    ///
29    /// Tracking is added here, to the last glyph of every cluster, so
30    /// every pass downstream measures it without being told about it.
31    pub(super) fn shape_spans(
32        &self,
33        flat: &FlatParagraph,
34        style: &ParagraphStyle,
35        upem: f32,
36    ) -> Vec<ShapedSpan> {
37        flat.spans
38            .iter()
39            .map(|span| {
40                let mut glyphs = self
41                    .registry
42                    .shape_with(span.font_id, &flat.text[span.range.clone()], &span.features)
43                    .unwrap_or_default();
44                let track = self.tracking_units(span);
45                if track != 0 {
46                    for at in 0..glyphs.len() {
47                        let last = glyphs
48                            .get(at + 1)
49                            .is_none_or(|next| next.cluster != glyphs[at].cluster);
50                        if last {
51                            glyphs[at].x_advance =
52                                (glyphs[at].x_advance as i64 + track).max(0) as u32;
53                        }
54                    }
55                }
56                let scale = self.scale(span.font_id, span.size, style, upem);
57                ShapedSpan {
58                    range: span.range.clone(),
59                    scale,
60                    track,
61                    tracking: track as f32 * scale,
62                    glyphs,
63                }
64            })
65            .collect()
66    }
67
68    /// A span's tracking in its own font units.
69    fn tracking_units(&self, span: &StyleSpan) -> i64 {
70        if span.tracking == 0.0 || span.size <= 0.0 {
71            return 0;
72        }
73        let upem = self
74            .registry
75            .metrics(span.font_id)
76            .map(|m| m.units_per_em as f32)
77            .unwrap_or(1000.0);
78        (span.tracking / span.size * upem).round() as i64
79    }
80
81    /// What a span's own font units are worth in the paragraph's.
82    /// One em of a 6pt face is not one em of an 11pt one, and the
83    /// measure is written in the paragraph's.
84    fn scale(&self, font_id: u16, size: f32, style: &ParagraphStyle, upem: f32) -> f32 {
85        let span_upem = self
86            .registry
87            .metrics(font_id)
88            .map(|m| m.units_per_em as f32)
89            .unwrap_or(upem);
90        if span_upem <= 0.0 || style.size <= 0.0 {
91            return 1.0;
92        }
93        size / span_upem * upem / style.size
94    }

96    /// Draws the hyphen a break inside a word leaves behind.
97    ///
98    /// The break was charged for it when it was chosen, so it is
99    /// drawn in the same face the charge was read from: a hyphen
100    /// taken from one face and paid for out of another is a line
101    /// that measures one width and paints a different one.
102    pub(super) fn hyphenate(&self, runs: &mut Vec<ShapedRun>, style: &ParagraphStyle) {
103        let Some(id) = self.registry.char_glyph(style.font_id, '-') else {
104            return;
105        };
106        let advance = self
107            .registry
108            .advance_width(style.font_id, id)
109            .unwrap_or_default() as u32;
110        let ending = runs
111            .last()
112            .map(|run| run.text_start + run.text.len() as u32)
113            .unwrap_or_default();
114        // The hyphen takes the last run's colour, because colour
115        // costs no width, and the paragraph's face, because that is
116        // where its width was charged.
117        let color = runs.last().map_or(style.color, |run| run.color);
118        // The hyphen belongs to the word it broke, so a rule drawn
119        // across that word is drawn across it too.
120        let decoration = runs.last().map_or(style.decoration, |run| run.decoration);
121        let blend = runs.last().map_or(style.blend, |run| run.blend);
122        // The hyphen belongs to the word it broke, so it sits inside
123        // whatever box that word is in.
124        let inline = runs.last().and_then(|run| run.inline);
125        match runs.last_mut() {
126            Some(run) if run.font_id == style.font_id && run.size == style.size => {
127                run.text.push('-');
128                // A hyphen the breaker drew stands for nothing the
129                // author wrote, so it maps to an empty stretch of the
130                // source and extraction reads straight past it.
131                if let Some(end) = run.source_map.last().copied() {
132                    run.source_map.push(end);
133                }
134                run.glyphs.push(ShapedGlyph {
135                    id,
136                    x_advance: advance,
137                    cluster: ending,
138                });
139                run.advance += advance;
140            }
141            // A break inside an emphasised word: the hyphen is the
142            // paragraph's own, so it goes in a run of its own rather
143            // than into a face it was not measured in.
144            _ => runs.push(ShapedRun {
145                font_id: style.font_id,
146                size: style.size,
147                text: "-".to_string(),
148                source: String::new(),
149                source_map: Vec::new(),
150                text_start: ending,
151                origin: None,
152                pseudo_element: None,
153                inline,
154                lead: 0.0,
155                trail: 0.0,
156                features: Features::NONE,
157                color,
158                decoration,
159                blend,
160                glyphs: vec![ShapedGlyph {
161                    id,
162                    x_advance: advance,
163                    cluster: ending,
164                }],
165                advance,
166            }),
167        }
168    }
169
170    pub(super) fn hyphen_advance(&self, style: &ParagraphStyle) -> u32 {
171        self.registry
172            .char_glyph(style.font_id, '-')
173            .and_then(|g| self.registry.advance_width(style.font_id, g))
174            .unwrap_or(0) as u32
175    }
176}

178/// One shaped span, its glyph clusters still relative to the span's
179/// own text.
180pub(super) struct ShapedSpan {
181    /// Byte range of the span in the paragraph text.
182    pub(super) range: Range<usize>,
183    /// What one of this span's font units is worth in the
184    /// paragraph's.
185    pub(super) scale: f32,
186    /// Tracking charged after each of its clusters, in its own font
187    /// units.
188    pub(super) track: i64,
189    /// The same in the paragraph's font units.
190    pub(super) tracking: f32,
191    pub(super) glyphs: Vec<ShapedGlyph>,
192}
193
194impl ShapedSpan {
195    /// The glyphs whose clusters fall in `[start, end)`, clusters
196    /// rebased to the paragraph text.
197    pub(super) fn glyphs_in(&self, start: usize, end: usize) -> Vec<ShapedGlyph> {
198        self.glyphs
199            .iter()
200            .filter(|g| {
201                let cluster = self.range.start + g.cluster as usize;
202                cluster >= start && cluster < end
203            })
204            .map(|g| ShapedGlyph {
205                cluster: g.cluster + self.range.start as u32,
206                ..*g
207            })
208            .collect()
209    }
210}

222    /// The hyphen is painted as well as charged: the run has the
223    /// character and a glyph for it, and the last line has neither.
224    #[test]
225    fn a_hyphenated_line_paints_the_hyphen_it_paid_for() {
226        let text = "extraordinarily";
227        let lines = layout_body_opts(text, 53.0, hyphenated());
228        let first = &lines[0];
229        assert_eq!(line_text(first), "extraordi-");
230
231        let run = first.runs.last().expect("the line has no runs");
232        assert_eq!(
233            run.glyphs.len(),
234            run.text.chars().count(),
235            "the hyphen is in the text and not in the glyphs",
236        );
237        let ranges = run.glyph_ranges();
238        let last = ranges.last().expect("the run has no glyphs");
239        assert_eq!(
240            &run.text[last.start as usize..last.end as usize],
241            "-",
242            "the last glyph does not stand for the hyphen",
243        );
244
245        // Charged and drawn are the same number: the line's width
246        // covers the glyph it ends with.
247        let hyphen = run.glyphs.last().expect("the run has no glyphs").x_advance;
248        assert!(hyphen > 0, "the hyphen has no advance");
249        assert_eq!(
250            first.width,
251            first.runs.iter().map(|r| r.advance).sum::<u32>(),
252            "the line's width and its runs disagree",
253        );
254
255        let last_line = lines.last().expect("no lines");
256        assert!(
257            !line_text(last_line).ends_with('-'),
258            "the last line was hyphenated: {:?}",
259            line_text(last_line),
260        );
261    }

366    /// Letter-spacing is advance on the shaper's glyphs: every gap
367    /// between two glyphs opens by the tracking, and the last glyph
368    /// keeps its own advance, so a tracked title measures the tracking
369    /// times one fewer than its glyphs.
370    #[test]
371    fn letter_spacing_opens_the_gaps_between_glyphs() {
372        let plain = layout_style("HANDGLOVES", 400.0, &body());
373        let tracking = 0.08 * body().size;
374        let tracked = layout_style(
375            "HANDGLOVES",
376            400.0,
377            &ParagraphStyle {
378                letter_spacing: tracking,
379                ..body()
380            },
381        );
382        let glyphs = plain[0].runs[0].glyphs.len();
383        assert_eq!(glyphs, 10, "the title shaped to one glyph a letter");
384        let units = tracking / body().size * units_per_em() as f32;
385        assert_eq!(
386            tracked[0].width - plain[0].width,
387            (units * (glyphs - 1) as f32).round() as u32,
388            "tracking did not open exactly the gaps between the glyphs",
389        );
390        for (loose, tight) in tracked[0].runs[0]
391            .glyphs
392            .iter()
393            .zip(&plain[0].runs[0].glyphs)
394            .take(glyphs - 1)
395        {
396            assert_eq!(
397                loose.x_advance - tight.x_advance,
398                units.round() as u32,
399                "a glyph has no tracking of its own",
400            );
401        }
402    }

404    /// Small capitals come out of the face where the face has them:
405    /// the run stays at the size around it and draws glyphs the plain
406    /// text does not. A face with no substitutions of its own gets a
407    /// synthesis instead, the letters raised to capitals and set at a
408    /// fraction of the size, and the capitals already there are left
409    /// alone.
410    #[test]
411    fn small_caps_take_the_feature_or_a_synthesis() {
412        let style = ParagraphStyle {
413            caps: FontVariantCaps::SmallCaps,
414            ..body()
415        };
416        let plain = layout_style("hello", 400.0, &body());
417        let feature = layout_style("hello", 400.0, &style);
418        assert_eq!(feature[0].runs.len(), 1, "the feature split the run");
419        assert_eq!(feature[0].runs[0].size, body().size);
420        assert_ne!(
421            feature[0].runs[0]
422                .glyphs
423                .iter()
424                .map(|g| g.id)
425                .collect::<Vec<_>>(),
426            plain[0].runs[0]
427                .glyphs
428                .iter()
429                .map(|g| g.id)
430                .collect::<Vec<_>>(),
431            "the face's small capitals drew the lowercase glyphs",
432        );
433
434        let bare = crate::fonts::registry_without_substitutions();
435        assert!(!bare.has_small_caps(0), "the face still substitutes");
436        let inlines = vec![Inline::Text {
437            id: NodeId::UNASSIGNED,
438            value: "hi Ho".to_string(),
439            attributes: Attributes::default(),
440            position: None,
441            span: None,
442        }];
443        let synthesized =
444            LineLayout::new(&bare).layout(&inlines, &style, 400.0, LineBreakOptions::default());
445        let runs: Vec<(f32, &str, &str)> = synthesized[0]
446            .runs
447            .iter()
448            .map(|run| (run.size, run.text.as_str(), run.source.as_str()))
449            .collect();
450        assert_eq!(
451            runs,
452            vec![
453                (body().size * SMALL_CAPS_RATIO, "HI", "hi"),
454                (body().size, " H", ""),
455                (body().size * SMALL_CAPS_RATIO, "O", "o"),
456            ],
457            "the synthesis did not raise what was lowercase, leave the rest, \
458             and keep what the author wrote",
459        );
460    }
