17/// What one section's lines were built from: the file it came from,
18/// its content, and the style every node in it resolved to. Node ids
19/// are deliberately absent, because they renumber globally on every
20/// edit and a chapter nothing touched would then miss its own
21/// cache. So is the stretch of the source a node was read from,
22/// which moves whenever a byte above it does and changes no line.
23///
24/// A section's references reach past it, to the elements they name.
25/// The words of each, and whether an element carries the id at all,
26/// are known before anything is laid out, so they are part of it. So
27/// is the number of each note it holds, which is what the reference
28/// of that note prints in a line.
29pub(super) fn section_key(
30    section: &Section,
31    styles: &StyleTree,
32    against: Against,
33    assets: Option<usize>,
34    hyphenation: (Patterns, Option<&str>),
35    references: &References,
36    notes: &Numbering,
37) -> u64 {
38    let mut hasher = DefaultHasher::new();
39    let h = &mut hasher;
40    against.hash_into(h);
41    assets.hash(h);
42    hyphenation.hash(h);
43    (&section.source, &section.title, section.position).hash(h);
44    hash_node(section.id, styles, h);
45    hash_blocks(&section.blocks, styles, h);
46    for note in notes_in_blocks(&section.blocks) {
47        notes.number(inline_id(note)).hash(h);
48    }
49    if styles.refers() {
50        let named = Named::in_section(section, styles, references);
51        (named.pages.len(), named.texts.len()).hash(h);
52        for node in named.pages.iter().chain(&named.texts) {
53            references.text(*node).hash(h);
54        }
55    }
56    hasher.finish()
57}

59/// What one section's lines were built from on the pass that prints
60/// the pages its references name: what they were built from on the
61/// pass before, and the folio each of those references prints.
62pub(super) fn settled_key(key: u64, pages: &[NodeId], found: &BTreeMap<NodeId, u32>) -> u64 {
63    let mut hasher = DefaultHasher::new();
64    key.hash(&mut hasher);
65    // The folio, not the node: ids move when an earlier section changes.
66    for node in pages {
67        found.get(node).hash(&mut hasher);
68    }
69    hasher.finish()
70}

72fn hash_blocks(blocks: &[Block], styles: &StyleTree, h: &mut DefaultHasher) {
73    for block in blocks {
74        match block {
75            Block::Heading {
76                id,
77                level,
78                inlines,
79                position,
80                attributes: _,
81                span: _,
82            } => {
83                (0u8, level, position).hash(h);
84                hash_node(*id, styles, h);
85                hash_inlines(inlines, styles, h);
86            }
87            Block::Paragraph {
88                id,
89                inlines,
90                position,
91                attributes: _,
92                span: _,
93            } => {
94                (1u8, position).hash(h);
95                hash_node(*id, styles, h);
96                hash_inlines(inlines, styles, h);
97            }
98            Block::Blockquote {
99                id,
100                blocks,
101                position,
102                attributes: _,
103                span: _,
104            } => {
105                (2u8, position).hash(h);
106                hash_node(*id, styles, h);
107                hash_blocks(blocks, styles, h);
108            }
109            Block::ThematicBreak {
110                id,
111                position,
112                attributes: _,
113                span: _,
114            } => {
115                (3u8, position).hash(h);
116                hash_node(*id, styles, h);
117            }
118            Block::PageBreak {
119                id,
120                position,
121                attributes: _,
122                span: _,
123            } => {
124                (7u8, position).hash(h);
125                hash_node(*id, styles, h);
126            }
127            Block::ColumnBreak {
128                id,
129                position,
130                attributes: _,
131                span: _,
132            } => {
133                (8u8, position).hash(h);
134                hash_node(*id, styles, h);
135            }
136            Block::Image {
137                id,
138                url,
139                alt,
140                position,
141                attributes: _,
142                span: _,
143            } => {
144                (4u8, url, alt, position).hash(h);
145                hash_node(*id, styles, h);
146            }
147            Block::CodeBlock {
148                id,
149                info,
150                text,
151                position,
152                attributes: _,
153                span: _,
154            } => {
155                (9u8, info, text, position).hash(h);
156                hash_node(*id, styles, h);
157            }
158            Block::List {
159                id,
160                ordered,
161                start,
162                tight,
163                items,
164                position,
165                attributes: _,
166                span: _,
167            } => {
168                (6u8, position, ordered, start, tight, items.len()).hash(h);
169                hash_node(*id, styles, h);
170                for item in items {
171                    item.position.hash(h);
172                    hash_node(item.id, styles, h);
173                    hash_blocks(&item.blocks, styles, h);
174                }
175            }
176            Block::Table {
177                id,
178                head,
179                body,
180                position,
181                attributes: _,
182                span: _,
183            } => {
184                (5u8, position, head.len()).hash(h);
185                hash_node(*id, styles, h);
186                for row in rows(head, body) {
187                    (row.position, row.cells.len()).hash(h);
188                    hash_node(row.id, styles, h);
189                    for cell in &row.cells {
190                        (cell.position, cell.align).hash(h);
191                        hash_node(cell.id, styles, h);
192                        hash_blocks(&cell.blocks, styles, h);
193                    }
194                }
195            }
196        }
197    }
198}

200fn hash_inlines(inlines: &[Inline], styles: &StyleTree, h: &mut DefaultHasher) {
201    for inline in inlines {
202        match inline {
203            Inline::Text {
204                id,
205                value,
206                position,
207                attributes: _,
208                span: _,
209            } => {
210                (0u8, value, position).hash(h);
211                hash_node(*id, styles, h);
212            }
213            Inline::Break {
214                id,
215                position,
216                attributes: _,
217                span: _,
218            } => {
219                (5u8, position).hash(h);
220                hash_node(*id, styles, h);
221            }
222            Inline::Emphasis {
223                id,
224                children,
225                position,
226                attributes: _,
227                span: _,
228            } => {
229                (1u8, position).hash(h);
230                hash_node(*id, styles, h);
231                hash_inlines(children, styles, h);
232            }
233            Inline::Strong {
234                id,
235                children,
236                position,
237                attributes: _,
238                span: _,
239            } => {
240                (2u8, position).hash(h);
241                hash_node(*id, styles, h);
242                hash_inlines(children, styles, h);
243            }
244            Inline::Code {
245                id,
246                value,
247                position,
248                attributes: _,
249                span: _,
250            } => {

252                hash_node(*id, styles, h);

255                id,
256                url,
257                children,
258                position,
259                attributes: _,
260                span: _,
261            } => {
262                (4u8, url, position).hash(h);
263                hash_node(*id, styles, h);

268                children,
269                position,

272            } => {

278                id,
279                children,
280                position,
281                attributes: _,
282                span: _,
283            } => {
284                (9u8, position).hash(h);
285                hash_node(*id, styles, h);

288            Inline::Span {
289                id,
290                children,
291                position,
292                attributes: _,
293                span: _,
294            } => {
295                (6u8, position).hash(h);
296                hash_node(*id, styles, h);
297                hash_inlines(children, styles, h);
298            }
299            Inline::Note {
300                id,
301                blocks,
302                position,
303                attributes: _,
304                span: _,
305            } => {
306                (7u8, position).hash(h);
307                hash_node(*id, styles, h);
308                hash_blocks(blocks, styles, h);
309            }
310        }
311    }

314/// One node's resolved styling, and its pseudo-elements beside it.
315fn hash_node(id: NodeId, styles: &StyleTree, h: &mut DefaultHasher) {
316    hash_layout(styles.style(id), h);
317    for pseudo in [
318        styles.first_letter(id),
319        styles.first_line(id),
320        styles.before(id),
321        styles.after(id),
322    ] {
323        match pseudo {
324            Some(style) => {
325                1u8.hash(h);
326                hash_layout(style, h);
327            }
328            None => 0u8.hash(h),
329        }
330    }
331}

333/// Everything a fragment is built from. Destructured field by field
334/// on purpose: a property added to `ComputedStyle` stops compiling
335/// here until somebody says which stage it belongs to.
336pub(super) fn hash_layout(style: &ComputedStyle, h: &mut DefaultHasher) {
337    let ComputedStyle {
338        font_id,
339        // The families asked for chose the face; layout reads only
340        // the answer.
341        font_family: _,
342        font_size,
343        font_style: _,
344        font_weight: _,
345        // No line moves for it. The runs the broken lines hold
346        // carry the colour, so a sheet that recolours them has to
347        // break them again.
348        color,
349        line_height,
350        letter_spacing,
351        font_variant_caps,
352        text_transform,
353        // A rule across a run is painted over the advance the
354        // glyphs took, so nothing about it moves a line. The runs
355        // carry it, which is why a sheet that draws one has to break
356        // them again.
357        text_decoration_line,
358        text_decoration_color,
359        text_decoration_style,
360        text_decoration_thickness,
361        text_align,
362        text_justify,
363        hanging_punctuation,
364        text_indent,
365        hyphens,
366        orphans,
367        widows,
368        // The named page is settled when a page opens, not when a
369        // line breaks.
370        page: _,
371        content,
372        string_set,
373        counter_reset,
374        note_reset,
375        initial_letter,
376        // Whether a block is in the flow decides whether the section
377        // holds fragments for it or an anchor. A relative block's
378        // fragments carry the offset its insets give them.
379        position,
380        inset,
381        // The layer and the opacity travel on the fragments a block
382        // emits.
383        z_index,
384        opacity,
385        mix_blend_mode,
386        blend_around,
387        wrap_flow: _,
388        shape_outside: _,
389        shape_margin: _,
390        margin,
391        padding,
392        border,
393        // A decoration carries its corners to the page.
394        border_radius,
395        background,
396        box_decoration_break,
397        width,
398        // A block taller than its content emits the space below it.
399        height,
400        min_height,
401        max_width,
402        max_height,
403        border_collapse,
404        list_style_type,
405        break_before,
406        break_after,
407        break_inside,
408        // A spanning block breaks to the whole content box.
409        column_span,
410        // What a custom property holds reaches layout only through the
411        // properties that read it, which are hashed here already.
412        custom: _,
413        // A feature draws other glyphs, and other glyphs take other
414        // advances, so a line breaks where the features put it.
415        font_feature_settings,
416        font_variant_ligatures,
417        font_variant_numeric,
418        font_variant_alternates,
419    } = style;
420    (border_collapse, list_style_type).hash(h);
421    for size in [width, height, min_height, max_width, max_height] {
422        match size {
423            Width::Auto => 0u8.hash(h),
424            Width::Points(points) => (1u8, points.to_bits()).hash(h),
425            Width::Percent(percent) => (2u8, percent.to_bits()).hash(h),
426        }
427    }
428    (font_id, font_size.to_bits(), line_height.to_bits(), color).hash(h);
429    (letter_spacing.to_bits(), font_variant_caps, text_transform).hash(h);
430    (
431        font_feature_settings,
432        font_variant_ligatures,
433        font_variant_numeric,
434        font_variant_alternates,
435    )
436        .hash(h);
437    (
438        text_decoration_line,
439        text_decoration_color,
440        text_decoration_style,
441        text_decoration_thickness.map(f32::to_bits),
442    )
443        .hash(h);
444    (text_align, text_justify, hanging_punctuation).hash(h);
445    (text_indent.to_bits(), hyphens, orphans, widows).hash(h);
446    (
447        content,
448        string_set,
449        counter_reset,
450        note_reset,
451        initial_letter,
452    )
453        .hash(h);
454    (position, z_index, opacity.to_bits()).hash(h);
455    (mix_blend_mode, blend_around).hash(h);
456    hash_insets(*inset, h);
457    (break_before, break_after, break_inside, column_span).hash(h);
458    hash_background(background, h);
459    box_decoration_break.hash(h);
460    hash_edges(*margin, h);
461    hash_edges(*padding, h);
462    hash_edges(border.widths(), h);
463    for edge in [border.top, border.right, border.bottom, border.left] {
464        edge.color.hash(h);
465    }
466    for corner in [
467        border_radius.top_left,
468        border_radius.top_right,
469        border_radius.bottom_right,
470        border_radius.bottom_left,
471    ] {
472        for axis in [corner.x, corner.y] {
473            match axis {
474                crate::style::Coord::Points(points) => (0u8, points.to_bits()).hash(h),
475                crate::style::Coord::Percent(percent) => (1u8, percent.to_bits()).hash(h),
476            }
477        }
478    }
479}

481/// What one box paints behind its content. The tint and the image
482/// both reach the display structure through the flow, and neither
483/// moves a line, but a section keyed without them would serve a page
484/// still painted the way it was.
485pub(super) fn hash_background(background: &Background, h: &mut DefaultHasher) {
486    let Background {
487        color,
488        image,
489        repeat,
490        size,
491        position,
492    } = background;
493    (color, image, repeat).hash(h);
494    match size {
495        BackgroundSize::Auto => 0u8.hash(h),
496        BackgroundSize::Cover => 1u8.hash(h),
497        BackgroundSize::Contain => 2u8.hash(h),
498        BackgroundSize::Fixed { width, height } => {
499            3u8.hash(h);
500            for axis in [width, height] {
501                hash_coord(*axis, h);
502            }
503        }
504    }
505    hash_coord(Some(position.x), h);
506    hash_coord(Some(position.y), h);
507}

509/// One coordinate, or the absence of one: `f32` is not `Hash`, and a
510/// percentage and a length of the same number are different answers.
511fn hash_coord(coord: Option<Coord>, h: &mut DefaultHasher) {
512    match coord {
513        None => 0u8.hash(h),
514        Some(Coord::Points(points)) => (1u8, points.to_bits()).hash(h),
515        Some(Coord::Percent(percent)) => (2u8, percent.to_bits()).hash(h),
516    }
517}

519pub(super) fn hash_geometry(geometry: PageGeometry, h: &mut DefaultHasher) {
520    let PageGeometry {
521        width,
522        height,
523        margin,
524        columns,
525        align_content,
526    } = geometry;
527    (width.to_bits(), height.to_bits(), align_content).hash(h);
528    hash_edges(margin, h);
529    let Columns {
530        count,
531        width,
532        gap,
533        rule,
534    } = columns;
535    (count, width.map(f32::to_bits), gap.to_bits()).hash(h);
536    let ColumnRule { style, width } = rule;
537    (style, width.to_bits()).hash(h);
538}

540/// What one contour is: which of the three `shape-outside` says, and
541/// the points where it says a polygon.
542pub(super) fn hash_shape(shape: &ShapeOutside, h: &mut DefaultHasher) {
543    match shape {
544        ShapeOutside::None => 0u8.hash(h),
545        ShapeOutside::Auto => 1u8.hash(h),
546        ShapeOutside::Polygon(points) => {
547            2u8.hash(h);
548            for point in points {
549                for coord in [point.x, point.y] {
550                    match coord {
551                        Coord::Points(points) => (0u8, points.to_bits()).hash(h),
552                        Coord::Percent(percent) => (1u8, percent.to_bits()).hash(h),
553                    }
554                }
555            }
556        }
557    }
558}

560pub(super) fn hash_edges(edges: Edges, h: &mut DefaultHasher) {
561    let Edges {
562        top,
563        right,
564        bottom,
565        left,
566    } = edges;
567    [top, right, bottom, left].map(f32::to_bits).hash(h);
568}

570pub(super) fn hash_insets(insets: Edges<Inset>, h: &mut DefaultHasher) {
571    for inset in [insets.top, insets.right, insets.bottom, insets.left] {

574            Inset::Points(points) => (1u8, points.to_bits()).hash(h),
575            Inset::Percent(percent) => (2u8, percent.to_bits()).hash(h),
