1//! The content tree as XHTML: one document per section, with the
2//! element names the sheets already select.
3//!
4//! The mapping follows the element tree the cascade matches against,
5//! so a selector that reaches a node in the PDF reaches the same node
6//! here. Two names have no element in HTML: the book is the `body`,
7//! and a note is an `aside` after the section, with a numbered link
8//! where the note was written.
9

21/// The attribute a block element carries its node id in.
22pub const NODE: &str = "data-node";

24/// What a node is to the ids the plan gives out.
25#[derive(Debug, Clone, Copy, PartialEq, Eq)]
26enum Role {
27    Heading,
28    /// A block that is an element in its document.
29    Block,
30    Note,
31    Other,
32}
33
34/// Where each element that something links to is: the document it is
35/// in, and the id it carries there.
36#[derive(Debug, Default)]
37pub struct Plan {
38    /// Each section's document, in reading order.
39    pub hrefs: Vec<String>,
40    /// The document of each section.
41    sections: BTreeMap<NodeId, usize>,
42    /// The document and id of every element that carries an id.
43    ids: BTreeMap<NodeId, (usize, String)>,
44}
45
46impl Plan {
47    /// Names each section's document and gives an id to every node
48    /// that has one written, every block, every note and every node a
49    /// link reaches.
50    pub fn of(book: &Book, anchors: &Anchors) -> Plan {
51        let mut targets = BTreeSet::new();
52        for section in &book.sections {
53            let source = section.source.as_deref();
54            each_block(&section.blocks, &mut |node| {
55                if let Visit::Link(url) = node
56                    && let LinkTarget::Node(target) = anchors.resolve(url, source)
57                {
58                    targets.insert(target);
59                }
60            });
61        }
62
63        let mut plan = Plan::default();
64        for (doc, section) in book.sections.iter().enumerate() {
65            plan.hrefs.push(format!("section-{:03}.xhtml", doc + 1));
66            plan.sections.insert(section.id, doc);
67
68            // Every id the document writes is taken before any is
69            // made up, so a made-up one never takes a written one.
70            let mut written: Vec<(NodeId, String)> = Vec::new();
71            let mut wanted: Vec<(NodeId, Role)> = Vec::new();
72            let mut take = |id: NodeId, attributes: &Attributes, role: Role| {
73                if let Some(name) = attributes.id.as_deref() {
74                    written.push((id, name.to_string()));
75                } else if role != Role::Other || targets.contains(&id) {
76                    wanted.push((id, role));
77                }
78            };
79            take(section.id, &section.attributes, Role::Block);
80            each_block(&section.blocks, &mut |node| {
81                if let Visit::Node(id, attributes, role) = node {
82                    take(id, attributes, role);
83                }
84            });
85
86            let mut taken = BTreeSet::new();
87            for (id, name) in written {
88                if taken.insert(name.clone()) {
89                    plan.ids.insert(id, (doc, name));
90                }
91            }
92            let mut notes = 0;
93            for (id, role) in wanted {
94                let stem = match role {
95                    Role::Note => {
96                        notes += 1;
97                        format!("note-{notes}")
98                    }
99                    Role::Heading | Role::Block | Role::Other => format!("n{}", id.get()),
100                };
101                let mut name = stem.clone();
102                let mut again = 1;
103                while !taken.insert(name.clone()) {
104                    again += 1;
105                    name = format!("{stem}-{again}");
106                }
107                plan.ids.insert(id, (doc, name));
108            }
109        }
110        plan
111    }
112
113    /// The id a node carries in its document, if it carries one.
114    fn id(&self, node: NodeId) -> Option<&str> {
115        self.ids.get(&node).map(|(_, id)| id.as_str())
116    }
117
118    /// Where a link to `node` goes: its document, and its id there
119    /// unless it is a section, which is the whole document.
120    pub fn href(&self, node: NodeId) -> Option<String> {
121        if let Some(doc) = self.sections.get(&node) {
122            return Some(self.hrefs[*doc].clone());
123        }
124        let (doc, id) = self.ids.get(&node)?;
125        Some(format!("{}#{id}", self.hrefs[*doc]))
126    }
127}

129/// One node the walk over a section passes, in document order.
130enum Visit<'a> {
131    Node(NodeId, &'a Attributes, Role),
132    Link(&'a str),
133}
134
135fn each_block<'a>(blocks: &'a [Block], f: &mut dyn FnMut(Visit<'a>)) {
136    for block in blocks {
137        match block {
138            Block::Heading {
139                id,
140                inlines,
141                attributes,
142                ..
143            } => {
144                f(Visit::Node(*id, attributes, Role::Heading));
145                each_inline(inlines, f);
146            }
147            Block::Paragraph {
148                id,
149                inlines,
150                attributes,
151                ..
152            } => {
153                f(Visit::Node(*id, attributes, Role::Block));
154                each_inline(inlines, f);
155            }
156            Block::Blockquote {
157                id,
158                blocks,
159                attributes,
160                ..
161            } => {
162                f(Visit::Node(*id, attributes, Role::Block));
163                each_block(blocks, f);
164            }
165            Block::CodeBlock { id, attributes, .. }
166            | Block::ThematicBreak { id, attributes, .. }
167            | Block::Image { id, attributes, .. } => f(Visit::Node(*id, attributes, Role::Block)),
168            // A break has no element, so it has an id only when a
169            // link reaches it or the author wrote one.
170            Block::PageBreak { id, attributes, .. } | Block::ColumnBreak { id, attributes, .. } => {
171                f(Visit::Node(*id, attributes, Role::Other))
172            }
173            Block::List {
174                id,
175                items,
176                attributes,
177                ..
178            } => {
179                f(Visit::Node(*id, attributes, Role::Block));
180                for item in items {
181                    f(Visit::Node(item.id, &item.attributes, Role::Block));
182                    each_block(&item.blocks, f);
183                }
184            }
185            Block::Table {
186                id,
187                head,
188                body,
189                attributes,
190                ..
191            } => {
192                f(Visit::Node(*id, attributes, Role::Block));
193                for row in rows(head, body) {
194                    f(Visit::Node(row.id, &row.attributes, Role::Block));
195                    for cell in &row.cells {
196                        f(Visit::Node(cell.id, &cell.attributes, Role::Block));
197                        each_block(&cell.blocks, f);
198                    }
199                }
200            }
201        }
202    }
203}
204
205fn each_inline<'a>(inlines: &'a [Inline], f: &mut dyn FnMut(Visit<'a>)) {
206    for inline in inlines {
207        match inline {
208            // A run of text and a break are no elements, so a name on
209            // one has nothing to go on.
210            Inline::Text { .. } | Inline::Break { .. } => {}
211            Inline::Code { id, attributes, .. } => f(Visit::Node(*id, attributes, Role::Other)),
212            Inline::Link {
213                id,
214                url,
215                children,
216                attributes,
217                ..
218            } => {
219                f(Visit::Node(*id, attributes, Role::Other));
220                f(Visit::Link(url));
221                each_inline(children, f);
222            }
223            Inline::Emphasis {
224                id,
225                children,
226                attributes,
227                ..
228            }
229            | Inline::Strong {
230                id,
231                children,
232                attributes,
233                ..
234            }
235            | Inline::Strikethrough {
236                id,
237                children,
238                attributes,
239                ..
240            }
241            | Inline::Highlight {
242                id,
243                children,
244                attributes,
245                ..
246            }
247            | Inline::Span {
248                id,
249                children,
250                attributes,
251                ..
252            } => {
253                f(Visit::Node(*id, attributes, Role::Other));
254                each_inline(children, f);
255            }
256            Inline::Note {
257                id,
258                blocks,
259                attributes,
260                ..
261            } => {
262                f(Visit::Node(*id, attributes, Role::Note));
263                each_block(blocks, f);
264            }
265        }
266    }
267}

270pub struct Context<'a> {

272    pub anchors: &'a Anchors,
273    /// The book's language, for the root element.
274    pub language: Option<&'a str>,

278}

280/// One section as a document: its title, for the head, and the whole
281/// file.
282pub fn document(section: &Section, title: &str, cx: &mut Context<'_>) -> String {
283    let mut writer = Writer {
284        out: String::new(),
285        source: section.source.as_deref(),
286        notes: Vec::new(),
287        numbers: BTreeMap::new(),
288        cx,
289    };
290    writer.open("section", section.id, &section.attributes, &[]);
291    writer.out.push('\n');
292    writer.blocks(&section.blocks, false);
293    writer.out.push_str("</section>\n");
294    writer.notes();
295    let body = writer.out;
296    let mut out = String::new();
297    head(&mut out, title, cx.language, Some("book.css"));
298    out.push_str("<body>\n");
299    out.push_str(&body);
300    out.push_str("</body>\n</html>\n");
301    out
302}

304/// The top of an XHTML document, up to the end of its head.
305pub fn head(out: &mut String, title: &str, language: Option<&str>, sheet: Option<&str>) {
306    out.push_str("<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n<!DOCTYPE html>\n");
307    out.push_str(
308        "<html xmlns=\"http://www.w3.org/1999/xhtml\" xmlns:epub=\"http://www.idpf.org/2007/ops\"",
309    );
310    if let Some(language) = language {
311        out.push_str(" xml:lang=\"");
312        xml::attribute(out, language);
313        out.push_str("\" lang=\"");
314        xml::attribute(out, language);
315        out.push('"');
316    }
317    out.push_str(">\n<head>\n<title>");
318    xml::text(out, title);
319    out.push_str("</title>\n");
320    if let Some(sheet) = sheet {
321        out.push_str("<link rel=\"stylesheet\" type=\"text/css\" href=\"");
322        xml::attribute(out, sheet);
323        out.push_str("\"/>\n");
324    }
325    out.push_str("</head>\n");
326}

328struct Writer<'s, 'c, 'a> {
329    out: String,
330    source: Option<&'s str>,
331    /// The notes written so far whose own text is still to come, in
332    /// the order their references were written.
333    notes: Vec<&'s Inline>,
334    /// The number each note's reference printed.
335    numbers: BTreeMap<NodeId, usize>,
336    cx: &'c mut Context<'a>,
337}
338
339impl<'s> Writer<'s, '_, '_> {
340    /// Writes the opening tag of a block with the node's id, the node
341    /// itself and its classes, and any other attributes after them.
342    fn open(&mut self, tag: &str, node: NodeId, attributes: &Attributes, extra: &[(&str, &str)]) {
343        self.tag(tag, node, attributes, extra, true);
344    }
345
346    /// The same for an inline, which does not carry its node.
347    fn open_inline(
348        &mut self,
349        tag: &str,
350        node: NodeId,
351        attributes: &Attributes,
352        extra: &[(&str, &str)],
353    ) {
354        self.tag(tag, node, attributes, extra, false);
355    }
356
357    fn tag(
358        &mut self,
359        tag: &str,
360        node: NodeId,
361        attributes: &Attributes,
362        extra: &[(&str, &str)],
363        block: bool,
364    ) {
365        self.out.push('<');
366        self.out.push_str(tag);
367        if let Some(id) = self.cx.plan.id(node) {
368            self.attribute("id", id);
369        }
370        // The id can be the author's, or a made-up one that gave way
371        // to the author's, so the node is written apart from it.
372        if block {
373            self.attribute(NODE, &node.get().to_string());
374        }
375        if !attributes.classes.is_empty() {
376            self.attribute("class", &attributes.classes.join(" "));
377        }
378        for (name, value) in extra {
379            self.attribute(name, value);
380        }
381        self.out.push('>');
382    }
383
384    fn attribute(&mut self, name: &str, value: &str) {
385        self.out.push(' ');
386        self.out.push_str(name);
387        self.out.push_str("=\"");
388        xml::attribute(&mut self.out, value);
389        self.out.push('"');
390    }
391
392    fn close(&mut self, tag: &str) {
393        self.out.push_str("</");
394        self.out.push_str(tag);
395        self.out.push('>');
396    }
397
398    fn warn(&mut self, message: String, position: Option<SourcePos>) {
399        let at = origin(self.source, position);
400        self.cx.warnings.push(Warning {
401            message,
402            origin: (!at.is_empty()).then_some(at),
403        });
404    }
405
406    /// `bare` is an item of a tight list, whose paragraphs with no
407    /// names of their own are its text rather than elements.
408    fn blocks(&mut self, blocks: &'s [Block], bare: bool) {
409        for block in blocks {
410            let before = self.out.len();
411            self.block(block, bare);
412            if self.out.len() > before {
413                self.out.push('\n');
414            }
415        }
416    }
417
418    fn block(&mut self, block: &'s Block, bare: bool) {
419        match block {
420            Block::Heading {
421                id,
422                level,
423                inlines,
424                attributes,
425                ..
426            } => {
427                let tag = heading(*level);
428                self.open(tag, *id, attributes, &[]);
429                self.inlines(inlines);
430                self.close(tag);
431            }
432            Block::Paragraph {
433                inlines,
434                attributes,
435                ..
436            } if bare && attributes.is_empty() => self.inlines(inlines),
437            Block::Paragraph {
438                id,
439                inlines,
440                attributes,
441                ..
442            } => {
443                self.open("p", *id, attributes, &[]);
444                self.inlines(inlines);
445                self.close("p");
446            }
447            Block::Blockquote {
448                id,
449                blocks,
450                attributes,
451                ..
452            } => {
453                self.open("blockquote", *id, attributes, &[]);
454                self.out.push('\n');
455                self.blocks(blocks, false);
456                self.close("blockquote");
457            }
458            Block::CodeBlock {
459                id,
460                text,
461                attributes,
462                ..
463            } => {
464                self.open("pre", *id, attributes, &[]);
465                xml::text(&mut self.out, text);
466                self.close("pre");
467            }
468            Block::ThematicBreak { id, attributes, .. } => {
469                self.open("hr", *id, attributes, &[]);
470                self.close("hr");
471            }
472            // A reading system makes its own pages and columns, so a
473            // break between them has nothing to write.
474            Block::PageBreak { .. } | Block::ColumnBreak { .. } => {}
475            Block::Image {
476                id,
477                url,
478                alt,
479                attributes,
480                position,
481                ..
482            } => self.image(*id, url, alt, attributes, *position),
483            Block::List {
484                id,
485                ordered,
486                start,
487                tight,
488                items,
489                attributes,
490                ..
491            } => {
492                let tag = if *ordered { "ol" } else { "ul" };
493                let start = start.to_string();
494                let extra: &[(&str, &str)] = if *ordered && *start != *"1" {
495                    &[("start", &start)]
496                } else {
497                    &[]
498                };
499                self.open(tag, *id, attributes, extra);
500                self.out.push('\n');
501                for item in items {
502                    self.item(item, *tight);
503                }
504                self.close(tag);
505            }
506            Block::Table {
507                id,
508                head,
509                body,
510                attributes,
511                ..
512            } => {
513                self.open("table", *id, attributes, &[]);
514                self.out.push('\n');
515                for (group, rows, cell) in [("thead", head, "th"), ("tbody", body, "td")] {
516                    if rows.is_empty() {
517                        continue;
518                    }
519                    self.out.push('<');
520                    self.out.push_str(group);
521                    self.out.push_str(">\n");
522                    for row in rows {
523                        self.open("tr", row.id, &row.attributes, &[]);
524                        for content in &row.cells {
525                            let align = content.align.map(|align| match align {
526                                Alignment::Left => "left",
527                                Alignment::Center => "center",
528                                Alignment::Right => "right",
529                            });
530                            let extra: &[(&str, &str)] = match &align {
531                                Some(align) => &[("data-align", align)],
532                                None => &[],
533                            };
534                            self.open(cell, content.id, &content.attributes, extra);
535                            self.blocks(&content.blocks, false);
536                            self.close(cell);
537                        }
538                        self.close("tr");
539                        self.out.push('\n');
540                    }
541                    self.close(group);
542                    self.out.push('\n');
543                }
544                self.close("table");
545            }
546        }
547    }
548
549    fn item(&mut self, item: &'s ListItem, tight: bool) {
550        self.open("li", item.id, &item.attributes, &[]);
551        let mut first = true;
552        for block in &item.blocks {
553            if !first {
554                self.out.push('\n');
555            }
556            first = false;
557            self.block(block, tight);
558        }
559        self.close("li");
560        self.out.push('\n');
561    }
562
563    fn image(
564        &mut self,
565        id: NodeId,
566        url: &str,
567        alt: &str,
568        attributes: &Attributes,
569        position: Option<SourcePos>,
570    ) {
571        let images = self.cx.images;
572        match self
573            .cx
574            .resources
575            .resolve(url, Kind::Image, |url| images.load(url))
576        {
577            Ok(href) => {
578                let href = href.to_string();
579                self.open("img", id, attributes, &[("src", &href), ("alt", alt)]);
580                self.close("img");
581            }
582            Err(refused) => self.warn(refusal("Image", url, refused), position),
583        }
584    }
585
586    fn inlines(&mut self, inlines: &'s [Inline]) {
587        for inline in inlines {
588            self.inline(inline);
589        }
590    }
591
592    fn inline(&mut self, inline: &'s Inline) {
593        match inline {
594            Inline::Text { value, .. } => xml::text(&mut self.out, value),
595            Inline::Break { .. } => self.out.push_str("<br/>"),
596            Inline::Code {
597                id,
598                value,
599                attributes,
600                ..
601            } => {
602                self.open_inline("code", *id, attributes, &[]);
603                xml::text(&mut self.out, value);
604                self.close("code");
605            }
606            Inline::Emphasis {
607                id,
608                children,
609                attributes,
610                ..
611            } => self.wrap("em", *id, attributes, children),
612            Inline::Strong {
613                id,
614                children,
615                attributes,
616                ..
617            } => self.wrap("strong", *id, attributes, children),
618            Inline::Strikethrough {
619                id,
620                children,
621                attributes,
622                ..
623            } => self.wrap("s", *id, attributes, children),
624            Inline::Highlight {
625                id,
626                children,
627                attributes,
628                ..
629            } => self.wrap("mark", *id, attributes, children),
630            Inline::Span {
631                id,
632                children,
633                attributes,
634                ..
635            } => self.wrap("span", *id, attributes, children),
636            Inline::Link {
637                id,
638                url,
639                children,
640                attributes,
641                position,
642                ..
643            } => {
644                let href = match self.cx.anchors.resolve(url, self.source) {
645                    LinkTarget::Outside => Some(url.clone()),
646                    LinkTarget::Node(node) => self.cx.plan.href(node),
647                    LinkTarget::Missing => None,
648                };
649                if href.is_none() {
650                    self.warn(
651                        format!("`{url}` names nothing in the book. The text is not a link."),
652                        *position,
653                    );
654                }
655                let extra: &[(&str, &str)] = match &href {
656                    Some(href) => &[("href", href)],
657                    None => &[],
658                };
659                self.open_inline("a", *id, attributes, extra);
660                self.inlines(children);
661                self.close("a");
662            }
663            Inline::Note { id, .. } => {
664                let number = self.numbers.len() + 1;
665                self.numbers.insert(*id, number);
666                self.notes.push(inline);
667                let target = format!("#{}", self.cx.plan.id(*id).unwrap_or_default());
668                self.out
669                    .push_str("<a epub:type=\"noteref\" role=\"doc-noteref\"");
670                self.attribute("href", &target);
671                self.out.push('>');
672                self.out.push_str(&number.to_string());
673                self.out.push_str("</a>");
674            }
675        }
676    }
677
678    fn wrap(&mut self, tag: &str, id: NodeId, attributes: &Attributes, children: &'s [Inline]) {
679        self.open_inline(tag, id, attributes, &[]);
680        self.inlines(children);
681        self.close(tag);
682    }
683
684    /// The text of every note, after the section, in the order the
685    /// references were written. A note written inside a note comes
686    /// after the one that holds it.
687    fn notes(&mut self) {
688        let mut next = 0;
689        while let Some(note) = self.notes.get(next).copied() {
690            next += 1;
691            let Inline::Note {
692                id,
693                blocks,
694                attributes,
695                ..
696            } = note
697            else {
698                continue;
699            };
700            self.open(
701                "aside",
702                *id,
703                attributes,
704                &[("epub:type", "footnote"), ("role", "doc-footnote")],
705            );
706            self.out.push('\n');
707            self.blocks(blocks, false);
708            self.close("aside");
709            self.out.push('\n');
710        }
711    }
712}

726/// The warning for a file the container could not take.
727pub fn refusal(what: &str, url: &str, refused: Refused) -> String {
728    match refused {
729        Refused::Missing => {
730            format!("{what} {url} did not load. It is left out of the EPUB.")
731        }
732        Refused::Unknown => {
733            format!("{what} {url} is not a type an EPUB can hold. It is left out of the EPUB.")
734        }
735    }
736}

746/// Every heading of the book with words in it, in reading order. A
747/// heading inside a quotation, a list, a table or a note is part of
748/// that and not of the book's outline.
749pub fn headings(book: &Book, plan: &Plan) -> Vec<Heading> {
750    let mut out = Vec::new();
751    for section in &book.sections {
752        for block in &section.blocks {
753            let Block::Heading {
754                id, level, inlines, ..
755            } = block
756            else {
757                continue;
758            };
759            let title = text(inlines)
760                .split_whitespace()
761                .collect::<Vec<_>>()
762                .join(" ");
763            if let (false, Some(href)) = (title.is_empty(), plan.href(*id)) {
764                out.push(Heading {
765                    level: u8::from(*level),
766                    title,
767                    href,
768                });
769            }
770        }
771    }
772    out
773}
