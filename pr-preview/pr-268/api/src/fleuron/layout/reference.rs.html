1//! Cross-references: the text the sheet generates around an inline
2//! element, and the page the node a link reaches lands on.
3//!
4//! A page number is known after pagination, and the reference that
5//! prints it is set before. So a book whose references print pages is
6//! laid out twice. The first pass sets a placeholder where each page
7//! number goes, and reads off the page each node landed on. The second
8//! sets the numbers the first found. The second pass ships, and where
9//! it moved an element the first one found, the run says so.
10

26/// What a page number is set as on the pass that finds the pages:
27/// three figures, as wide as most page numbers a book prints.
28const PLACEHOLDER: &str = "000";
29
30/// What the references in one book resolve against.
31#[derive(Debug, Clone, Default)]
32pub(crate) struct References {
33    /// What the links reach. `None` where the book is not known, so
34    /// nothing can be missing from it.
35    known: Option<Arc<Known>>,
36    /// The folio each node landed on in the pass before this one.
37    /// `None` on the pass that finds them.
38    folios: Option<Arc<BTreeMap<NodeId, u32>>>,
39}
40
41#[derive(Debug)]
42struct Known {
43    anchors: Anchors,
44    /// The text of each node a link can reach.
45    texts: BTreeMap<NodeId, String>,
46}
47
48impl References {
49    /// What the links in one book reach, each node with its text.
50    pub(crate) fn of(book: &Book) -> References {
51        let anchors = book.anchors();
52        let mut texts = BTreeMap::new();
53        for section in &book.sections {
54            if anchors.reaches(section.id) {
55                texts.insert(section.id, section_text(section));
56            }
57            texts_of(&section.blocks, &anchors, &mut texts);
58        }
59        References {
60            known: Some(Arc::new(Known { anchors, texts })),
61            folios: None,
62        }
63    }
64
65    /// The same, with the folio each node landed on.
66    pub(crate) fn landed(&self, folios: BTreeMap<NodeId, u32>) -> References {
67        References {
68            known: self.known.clone(),
69            folios: Some(Arc::new(folios)),
70        }
71    }
72
73    /// The text of one node a link can reach.
74    pub(crate) fn text(&self, node: NodeId) -> Option<&str> {
75        self.known.as_ref()?.texts.get(&node).map(String::as_str)
76    }
77
78    /// What a warning calls one node a link can reach.
79    fn name(&self, node: NodeId) -> String {
80        self.known
81            .as_ref()
82            .and_then(|known| known.anchors.name(node))
83            .map_or_else(|| format!("node {}", node.get()), str::to_string)
84    }
85
86    /// The text one `content` value generates on an element whose
87    /// `href` is `href`, written in `source`, or why it generates none.
88    fn resolve(
89        &self,
90        content: &Content,
91        href: Option<&str>,
92        source: Option<&str>,
93    ) -> Result<String, Unresolved> {
94        let pieces = match content {
95            Content::Text(text) => return Ok(text.clone()),
96            Content::Pieces(pieces) => pieces,
97            _ => return Ok(String::new()),
98        };
99        let mut generated = String::new();
100        for piece in pieces {
101            match piece {
102                ContentPiece::Text(text) => generated.push_str(text),
103                ContentPiece::TargetCounter { target, style } => {
104                    let node = self.target(target, href, source)?;
105                    match (&self.folios, node) {
106                        (Some(folios), Some(node)) => {
107                            let folio = folios
108                                .get(&node)
109                                .ok_or_else(|| Unresolved::Unplaced(self.name(node)))?;
110                            generated.push_str(&style.format(*folio));
111                        }
112                        _ => generated.push_str(PLACEHOLDER),
113                    }
114                }
115                ContentPiece::TargetText { target } => {
116                    if let Some(node) = self.target(target, href, source)? {
117                        generated.push_str(self.text(node).unwrap_or_default());
118                    }
119                }
120            }
121        }
122        Ok(generated)
123    }
124
125    /// The node one target reaches from `source`. `None` where the
126    /// book is not known.
127    fn target(
128        &self,
129        target: &Target,
130        href: Option<&str>,
131        source: Option<&str>,
132    ) -> Result<Option<NodeId>, Unresolved> {
133        let url = target.url(href).ok_or(Unresolved::NotALink)?;
134        let Some(known) = &self.known else {
135            return Ok(None);
136        };
137        match known.anchors.resolve(url, source) {
138            LinkTarget::Node(node) => Ok(Some(node)),
139            LinkTarget::Outside => Err(Unresolved::Outside),
140            LinkTarget::Missing => Err(Unresolved::Missing(url.to_string())),
141        }
142    }
143}

145/// Why a reference generates nothing.
146enum Unresolved {
147    /// `attr(href url)` on an element that is not a link.
148    NotALink,
149    /// A url outside the book, which a sheet that styles every link
150    /// meets on purpose.
151    Outside,
152    /// A url that reaches nothing in the book.
153    Missing(String),
154    /// A node that reached no page, by its name.
155    Unplaced(String),
156}
157
158impl Unresolved {
159    fn message(&self) -> Option<String> {
160        Some(match self {
161            Unresolved::NotALink => "`attr(href url)` is used on an element that is not a link. \
162                                     Nothing is generated."
163                .to_string(),
164            Unresolved::Outside => return None,
165            Unresolved::Missing(url) => {
166                format!("`{url}` names nothing in the book. Nothing is generated.")
167            }
168            Unresolved::Unplaced(name) => {
169                format!("`{name}` is on no page. Nothing is generated.")
170            }
171        })
172    }
173}

175/// Takes the text of one node a link can reach, unless an element
176/// before it took the node already.
177fn claim(
178    texts: &mut BTreeMap<NodeId, String>,
179    anchors: &Anchors,
180    node: NodeId,
181    text: impl FnOnce() -> String,
182) {
183    if anchors.reaches(node) {
184        texts.entry(node).or_insert_with(text);
185    }
186}
187
188/// The text of every node a link can reach, in these blocks and
189/// everything inside them.
190fn texts_of(blocks: &[Block], anchors: &Anchors, texts: &mut BTreeMap<NodeId, String>) {
191    for block in blocks {
192        claim(texts, anchors, block_id(block), || block_text(block));
193        match block {
194            Block::Heading { inlines, .. } | Block::Paragraph { inlines, .. } => {
195                inline_texts(inlines, anchors, texts)
196            }
197            Block::Blockquote { blocks, .. } => texts_of(blocks, anchors, texts),
198            Block::List { items, .. } => {
199                for item in items {
200                    claim(texts, anchors, item.id, || blocks_text(&item.blocks));
201                    texts_of(&item.blocks, anchors, texts);
202                }
203            }
204            Block::Table { head, body, .. } => {
205                for row in rows(head, body) {
206                    claim(texts, anchors, row.id, || row_text(row));
207                    for cell in &row.cells {
208                        claim(texts, anchors, cell.id, || blocks_text(&cell.blocks));
209                        texts_of(&cell.blocks, anchors, texts);
210                    }
211                }
212            }
213            Block::CodeBlock { .. }
214            | Block::ThematicBreak { .. }
215            | Block::PageBreak { .. }
216            | Block::ColumnBreak { .. }
217            | Block::Image { .. } => {}
218        }
219    }
220}
221
222/// The same, over the inlines of one block.
223fn inline_texts(inlines: &[Inline], anchors: &Anchors, texts: &mut BTreeMap<NodeId, String>) {
224    for inline in inlines {
225        claim(texts, anchors, inline_id(inline), || {
226            text(std::slice::from_ref(inline))
227        });
228        if let Inline::Emphasis { children, .. }
229        | Inline::Strong { children, .. }
230        | Inline::Link { children, .. }
231        | Inline::Strikethrough { children, .. }
232        | Inline::Span { children, .. } = inline
233        {
234            inline_texts(children, anchors, texts);
235        }
236    }
237}

239/// What `target-text()` prints for a link to a whole source: the
240/// section's title, or the words of the heading it opens with.
241fn section_text(section: &Section) -> String {
242    section
243        .title
244        .clone()
245        .unwrap_or_else(|| match section.blocks.first() {
246            Some(block @ Block::Heading { .. }) => block_text(block),
247            _ => String::new(),
248        })
249}
250
251/// What `target-text()` prints for one block: its words, markup
252/// discarded, or an image's description.
253fn block_text(block: &Block) -> String {
254    match block {
255        Block::Heading { inlines, .. } | Block::Paragraph { inlines, .. } => text(inlines),
256        Block::Image { alt, .. } => alt.clone(),
257        Block::CodeBlock { text: code, .. } => code.clone(),
258        Block::Blockquote { blocks, .. } => blocks_text(blocks),
259        Block::List { items, .. } => item_blocks(items)
260            .map(blocks_text)
261            .collect::<Vec<_>>()
262            .join(" "),
263        Block::Table { head, body, .. } => {
264            rows(head, body).map(row_text).collect::<Vec<_>>().join(" ")
265        }
266        Block::ThematicBreak { .. } | Block::PageBreak { .. } | Block::ColumnBreak { .. } => {
267            String::new()
268        }
269    }
270}

272/// The words of one table row, a space between each two cells.
273fn row_text(row: &Row) -> String {
274    row.cells
275        .iter()
276        .map(|cell| blocks_text(&cell.blocks))
277        .collect::<Vec<_>>()
278        .join(" ")
279}
280
281/// The words of several blocks, a space between each two.
282fn blocks_text(blocks: &[Block]) -> String {
283    blocks.iter().map(block_text).collect::<Vec<_>>().join(" ")
284}

286/// The nodes one section's references reach, in the order they are
287/// written.
288#[derive(Debug, Default)]
289pub(crate) struct Named {
290    /// Those whose page a reference prints.
291    pub(crate) pages: Vec<NodeId>,
292    /// Those whose text a reference prints.
293    pub(crate) texts: Vec<NodeId>,
294}
295
296impl Named {
297    /// What the references in one section reach.
298    pub(crate) fn in_section(
299        section: &Section,
300        styles: &StyleTree,
301        references: &References,
302    ) -> Named {
303        let mut named = Named::default();
304        let source = section.source.as_deref();
305        named.content(styles.before(section.id), None, references, source);
306        named.blocks(&section.blocks, styles, references, source);
307        named.content(styles.after(section.id), None, references, source);
308        named
309    }
310
311    /// What one pseudo-element's `content` reaches, on an element
312    /// whose `href` is `href`.
313    fn content(
314        &mut self,
315        pseudo: Option<&ComputedStyle>,
316        href: Option<&str>,
317        references: &References,
318        source: Option<&str>,
319    ) {
320        let Some(Content::Pieces(pieces)) = pseudo.map(|pseudo| &pseudo.content) else {
321            return;
322        };
323        for piece in pieces {
324            let (list, target) = match piece {
325                ContentPiece::Text(_) => continue,
326                ContentPiece::TargetCounter { target, .. } => (&mut self.pages, target),
327                ContentPiece::TargetText { target } => (&mut self.texts, target),
328            };
329            if let Ok(Some(node)) = references.target(target, href, source) {
330                list.push(node);
331            }
332        }
333    }
334
335    fn blocks(
336        &mut self,
337        blocks: &[Block],
338        styles: &StyleTree,
339        references: &References,
340        source: Option<&str>,
341    ) {
342        for block in blocks {
343            let id = block_id(block);
344            self.content(styles.before(id), None, references, source);
345            match block {
346                Block::Heading { inlines, .. } | Block::Paragraph { inlines, .. } => {
347                    self.inlines(inlines, styles, references, source)
348                }
349                Block::Blockquote { blocks, .. } => self.blocks(blocks, styles, references, source),
350                Block::List { items, .. } => {
351                    for item in items {
352                        self.content(styles.before(item.id), None, references, source);
353                        self.blocks(&item.blocks, styles, references, source);
354                        self.content(styles.after(item.id), None, references, source);
355                    }
356                }
357                Block::Table { head, body, .. } => {
358                    for cell in rows(head, body).flat_map(|row| &row.cells) {
359                        self.blocks(&cell.blocks, styles, references, source);
360                    }
361                }
362                Block::CodeBlock { .. }
363                | Block::ThematicBreak { .. }
364                | Block::PageBreak { .. }
365                | Block::ColumnBreak { .. }
366                | Block::Image { .. } => {}
367            }
368            self.content(styles.after(id), None, references, source);
369        }
370    }
371
372    fn inlines(
373        &mut self,
374        inlines: &[Inline],
375        styles: &StyleTree,
376        references: &References,
377        source: Option<&str>,
378    ) {
379        for inline in inlines {
380            let (href, children) = match inline {
381                Inline::Text { .. } | Inline::Break { .. } => continue,
382                Inline::Code { .. } => (None, None),
383                Inline::Note { blocks, .. } => {
384                    self.blocks(blocks, styles, references, source);
385                    (None, None)
386                }
387                Inline::Emphasis { children, .. }
388                | Inline::Strong { children, .. }
389                | Inline::Strikethrough { children, .. }
390                | Inline::Span { children, .. } => (None, Some(children)),
391                Inline::Link { url, children, .. } => (Some(url.as_str()), Some(children)),
392            };
393            let id = inline_id(inline);
394            self.content(styles.before(id), href, references, source);
395            self.content(styles.after(id), href, references, source);
396            if let Some(children) = children {
397                self.inlines(children, styles, references, source);
398            }
399        }
400    }
401}

403/// The folio each node landed on, off one pass's pages.
404pub(crate) fn landed(paged: &Paged) -> BTreeMap<NodeId, u32> {
405    let numbers = folios(&paged.infos);
406    paged
407        .targets
408        .iter()
409        .map(|(node, area)| (*node, numbers[area.page as usize]))
410        .collect()
411}

413/// A warning for every node a reference prints the page of that the
414/// second pass set on another page than the first found it on.
415pub(crate) fn moved(
416    found: &BTreeMap<NodeId, u32>,
417    landed: &BTreeMap<NodeId, u32>,
418    printed: &BTreeSet<NodeId>,
419    references: &References,
420) -> Vec<Warning> {
421    printed
422        .iter()
423        .filter_map(|node| {
424            let (was, now) = (found.get(node)?, landed.get(node)?);
425            (was != now).then(|| Warning {
426                message: format!(
427                    "The page printed for `{}` is {was}, and the element is on page {now}.",
428                    references.name(*node),
429                ),
430                origin: None,
431            })
432        })
433        .collect()
434}

436impl Paginator<'_> {
437    /// The text one `content` value generates on an element whose
438    /// `href` is `href`, written in `source` at `position`. `None`
439    /// where a reference in it reaches nothing, which warns unless the
440    /// url is outside the book.
441    pub(super) fn generate(
442        &self,
443        content: &Content,
444        href: Option<&str>,
445        source: Option<&str>,
446        position: Option<SourcePos>,
447    ) -> Option<String> {
448        let resolved = self.references.borrow().resolve(content, href, source);
449        match resolved {
450            Ok(text) => Some(text),
451            Err(unresolved) => {
452                if let Some(message) = unresolved.message() {
453                    let at = origin(source, position);
454                    self.warn(message, (!at.is_empty()).then_some(at));
455                }
456                None
457            }
458        }
459    }
460}

462/// The style tree, with the text each pseudo-element generates
463/// resolved against the book's references. Line layout asks this for
464/// the style of every inline of a paragraph.
465pub(super) struct Referring<'r, 'a> {
466    pub(super) paginator: &'r Paginator<'a>,
467    /// The file the paragraph was read from, for diagnostics and for
468    /// the links that name no file.
469    pub(super) source: Option<&'r str>,
470}
471
472impl InlineStyles for Referring<'_, '_> {
473    fn style(&self, id: NodeId, block: &ParagraphStyle) -> ParagraphStyle {
474        InlineStyles::style(self.paginator.styles, id, block)
475    }
476
477    fn inline_box(&self, id: NodeId) -> Option<InlineBox> {
478        InlineStyles::inline_box(self.paginator.styles, id)
479    }
480
481    fn generated(&self, inline: &Inline) -> Generated {
482        let styles = self.paginator.styles;
483        let id = inline_id(inline);
484        let href = match inline {
485            Inline::Link { url, .. } => Some(url.as_str()),
486            _ => None,
487        };
488        let text = |pseudo: Option<&ComputedStyle>| {
489            let pseudo = pseudo?;
490            let text = self.paginator.generate(
491                &pseudo.content,
492                href,
493                self.source,
494                inline_position(inline),
495            )?;
496            Some((text, pseudo.paragraph()))
497        };
498        Generated {
499            before: text(styles.before(id)),
500            after: text(styles.after(id)),
501        }
502    }
503
504    fn call(&self, note: &Inline) -> Option<(String, ParagraphStyle)> {
505        let id = inline_id(note);
506        let style = self.paginator.styles.style(id);
507        let number = self.paginator.note_number(id);
508        let call = style.list_style_type.numeral(number)?;
509        Some((call, self.paginator.superior(style.paragraph())))
510    }
511}
