13/// Flattened paragraph content: the text as it is shaped, the text
14/// the author wrote under it, and, per style span, the byte range it
15/// covers. Style boundaries are segmentation boundaries — a shaped
16/// run never spans two fonts.
17#[derive(Default)]
18pub(super) struct FlatParagraph {
19    /// What is shaped, measured and broken. `text-transform` and
20    /// synthesized small capitals have already been applied.
21    pub(super) text: String,
22    /// What the author wrote. Kept only from the point something
23    /// first transforms; until then `text` is the source.
24    source: String,
25    /// How much of `source` the shaped text up to each of its bytes
26    /// accounts for. Kept alongside `source`, and rising with it, so
27    /// the stretch between two boundaries is the source that the
28    /// text between them was made from.
29    map: Vec<u32>,
30    /// Whether anything has been written that differs from its
31    /// source.
32    transformed: bool,
33    /// Whether the next character opens a word: what `capitalize`
34    /// reads.
35    word_start: bool,
36    /// The style spans, in document order.
37    pub(super) spans: Vec<StyleSpan>,
38    /// The inline elements that paint a box, in the order they open.
39    pub(super) boxes: Vec<BoxSpan>,
40    /// The innermost inline element open, which the spans written
41    /// now belong to.
42    inline: Option<NodeId>,
43    /// Which node each stretch of the source was written in, in
44    /// document order.
45    origins: Vec<Origin>,
46    /// Bytes of the source still to be passed over before anything
47    /// is written: what a drop cap took out of the paragraph.
48    skip: usize,
49    /// The id of the paragraph's `::first-line`, and how far into the
50    /// shaped text its style reaches.
51    first_line: Option<(NodeId, usize)>,
52    /// Stretches of the shaped text that are set as they were
53    /// written, so no hyphen is put inside one. An inline code span
54    /// is one.
55    pub(super) literal: Vec<Range<usize>>,
56}

58/// One stretch of the paragraph's source, and the node it was
59/// written in. Stretches are contiguous, so one origin runs to where
60/// the next begins.
61struct Origin {
62    /// `None` on text the sheet generated, which no node holds.
63    node: Option<NodeId>,
64    /// Where the stretch starts in the paragraph's source.
65    start: u32,
66    /// Where it starts in the node's own text, which is past the
67    /// letter wherever a drop cap took one.
68    node_start: u32,
69    /// Where the node's text starts in the manuscript, or where the
70    /// node does when `whole` is set.
71    position: Option<SourcePos>,
72    /// Whether the node's text runs over more than one line of the
73    /// manuscript, so that a byte of it is only as near as the node.
74    whole: bool,
75}

89/// One span of uniform shaping: a face, a size, the tracking after
90/// each of its clusters, and the features it is shaped with.
91pub(super) struct StyleSpan {
92    pub(super) font_id: u16,
93    pub(super) size: f32,
94    /// Extra advance after each cluster, in points.
95    pub(super) tracking: f32,
96    pub(super) features: Features,
97    pub(super) color: Color,
98    /// What is drawn across the span.
99    pub(super) decoration: TextDecoration,

101    pub(super) inline: Option<NodeId>,

106impl StyleSpan {
107    /// Whether two spans are set the same way, the text they cover
108    /// aside. An inline element is part of that: a box is painted
109    /// around the runs of one element, so a run never spans two.
110    fn same_style(&self, other: &StyleSpan) -> bool {
111        self.font_id == other.font_id
112            && self.size == other.size
113            && self.tracking == other.tracking
114            && self.features == other.features
115            && self.color == other.color
116            && self.decoration == other.decoration
117            && self.inline == other.inline
118    }
119}

121/// How a span gets its small capitals.
122#[derive(Debug, Clone, Copy, PartialEq, Eq)]
123pub(super) enum SmallCaps {
124    /// It does not: the text is set in the letters it was written in.
125    Off,
126    /// The face draws them, and the shaper is asked for `smcp`.
127    Feature,
128    /// The face does not, so lowercase letters are set as capitals at
129    /// a reduced size.
130    Synthesized,
131}

143    /// What the author wrote under one stretch of the shaped text,
144    /// and how much of it the text up to each byte accounts for.
145    ///
146    /// Both are empty where the two stretches are the same word for
147    /// word, which is every run of a book nothing transformed and
148    /// the untouched runs of a line that has one.
149    pub(super) fn source_of(&self, range: Range<usize>) -> (String, Vec<u32>) {
150        if !self.transformed {
151            return (String::new(), Vec::new());
152        }
153        let at = |byte: usize| self.source_at(byte);
154        let (from, to) = (at(range.start), at(range.end).max(at(range.start)));
155        let source = &self.source[from..to];
156        if source == &self.text[range.clone()] {
157            return (String::new(), Vec::new());
158        }
159        let map = (range.start..=range.end)
160            .map(|byte| (at(byte).clamp(from, to) - from) as u32)
161            .collect();
162        (source.to_string(), map)
163    }
164
165    /// Where one byte of the shaped text falls in the source it was
166    /// made from. The two run together until something transforms,
167    /// and the map holds them together after that.
168    fn source_at(&self, byte: usize) -> usize {
169        if !self.transformed {
170            return byte.min(self.text.len());
171        }
172        match self.map.get(byte) {
173            Some(offset) => *offset as usize,
174            None => self.source.len(),
175        }
176    }
177
178    /// How much source the paragraph has been written from so far.
179    fn source_len(&self) -> usize {
180        if self.transformed {
181            self.source.len()
182        } else {
183            self.text.len()
184        }
185    }

187    /// Opens a stretch of the source written in `node`, starting
188    /// `node_start` bytes into that node's own text.
189    fn open(&mut self, node: Option<NodeId>, node_start: usize) {
190        self.open_at(node, node_start, None, false);
191    }

193    /// The same, for a node the manuscript holds at `position`.
194    fn open_at(
195        &mut self,
196        node: Option<NodeId>,
197        node_start: usize,
198        position: Option<SourcePos>,
199        whole: bool,
200    ) {
201        self.origins.push(Origin {
202            node,
203            start: self.source_len() as u32,
204            node_start: node_start as u32,
205            position,
206            whole,
207        });
208    }

210    /// The line and column one byte of the shaped text was written
211    /// at. Columns count bytes, as the positions on the nodes do.
212    ///
213    /// `None` where the manuscript holds no such byte: text the sheet
214    /// generated, and a tree a host built without positions.
215    pub(super) fn written_at(&self, byte: usize) -> Option<SourcePos> {
216        let at = self.source_at(byte);
217        let index = self
218            .origins
219            .partition_point(|origin| origin.start as usize <= at)
220            .checked_sub(1)?;
221        let origin = &self.origins[index];
222        let position = origin.position?;
223        if origin.whole {
224            return Some(position);
225        }
226        let within = origin.node_start + (at as u32).saturating_sub(origin.start);
227        Some(SourcePos {
228            column: position.column + within,
229            ..position
230        })
231    }
232
233    /// The stretch of the source one byte of the shaped text was
234    /// written in: the node, where the stretch starts in that node's
235    /// own text, and the source it covers.
236    ///
237    /// `None` where no node was walked: page furniture, an ornament,
238    /// text the sheet generated.
239    fn origin_at(&self, byte: usize) -> Option<(NodeId, u32, Range<usize>)> {
240        let at = self.source_at(byte);
241        let index = self
242            .origins
243            .partition_point(|origin| origin.start as usize <= at)
244            .checked_sub(1)?;
245        let origin = &self.origins[index];
246        let end = self.origins[index + 1..]
247            .first()
248            .map_or(self.source_len(), |next| next.start as usize);
249        Some((origin.node?, origin.node_start, origin.start as usize..end))
250    }
251
252    /// The node one stretch of the shaped text was written in, and
253    /// the bytes of that node's own text it stands for, running from
254    /// `from` to wherever `to` reaches.
255    ///
256    /// `after` is the node the stretch before this one was written
257    /// in. A stretch opening a node opens at that node's first byte,
258    /// wherever the break before it fell, so the space a break
259    /// swallowed at a node boundary belongs to the node that holds
260    /// it. Both ends stop at the node.
261    pub(super) fn origin_of(
262        &self,
263        after: Option<NodeId>,
264        from: usize,
265        to: usize,
266    ) -> Option<SourceRange> {
267        let (node, node_start, stretch) = self.origin_at(from)?;
268        let start = if after == Some(node) {
269            self.source_at(from).clamp(stretch.start, stretch.end)
270        } else {
271            stretch.start
272        };
273        let end = self.source_at(to).clamp(start, stretch.end);
274        let at = |source: usize| node_start + (source - stretch.start) as u32;
275        Some(SourceRange {
276            node,
277            range: at(start)..at(end),
278        })
279    }

281    /// The pseudo-element a run that starts at `from` and stands for
282    /// `origin` was cut from: the `::before` or `::after` that
283    /// generated its text, or else the `::first-line` whose style
284    /// reaches it.
285    pub(super) fn pseudo_element_of(
286        &self,
287        origin: Option<&SourceRange>,
288        from: usize,
289    ) -> Option<NodeId> {
290        if let Some(origin) = origin.filter(|origin| origin.node.pseudo_element().is_some()) {
291            return Some(origin.node);
292        }
293        self.first_line
294            .filter(|(_, reach)| from < *reach)
295            .map(|(id, _)| id)
296    }

298    /// Starts keeping the source text, backfilling what has been
299    /// written so far, all of which stood as it was written.
300    fn start_mapping(&mut self) {
301        if self.transformed {
302            return;
303        }
304        self.transformed = true;
305        self.source.push_str(&self.text);
306        self.map.extend(0..self.text.len() as u32);
307    }

309    /// Appends a stretch of text nothing transformed.
310    fn push_verbatim(&mut self, value: &str) {
311        if self.transformed {
312            let at = self.source.len() as u32;
313            self.map
314                .extend((0..value.len() as u32).map(|byte| at + byte));
315            self.source.push_str(value);
316        }
317        self.text.push_str(value);
318    }

320    /// Appends one character shaped as something other than itself.
321    ///
322    /// The whole character is accounted for at the first byte it was
323    /// written as, so the first glyph of the pair `ß` shapes as
324    /// stands for the letter and the second stands for nothing.
325    /// Extraction walks the glyphs in order and reads the source back
326    /// once.
327    fn push_mapped(&mut self, letter: char, written: &str) {
328        self.start_mapping();
329        let at = self.source.len() as u32;
330        self.source.push(letter);
331        let after = self.source.len() as u32;
332        self.map.push(at);
333        self.map
334            .extend(std::iter::repeat_n(after, written.len() - 1));
335        self.text.push_str(written);
336    }

338    /// Appends one styled stretch of text: `text-transform` first,
339    /// then small capitals over what it produced, and the spans they
340    /// come to.
341    pub(super) fn push_styled(&mut self, value: &str, style: &ParagraphStyle, caps: SmallCaps) {
342        if style.transform == TextTransform::None && caps != SmallCaps::Synthesized {
343            let start = self.text.len();
344            self.push_verbatim(value);
345            if let Some(last) = value.chars().next_back() {
346                self.word_start = !continues_word(last);
347            }
348            self.span(style, caps, false, start);
349            return;
350        }
351        // A span breaks where the reduced size does, so the letters a
352        // synthesis raised are shaped apart from the ones it left.
353        let mut open: Option<(bool, usize)> = None;
354        let mut written = String::new();
355        for letter in value.chars() {
356            written.clear();
357            let mut changed = style.transform.write(letter, self.word_start, &mut written);
358            self.word_start = !continues_word(letter);
359            let small = raise(caps, &mut written, &mut changed);
360            if open.map(|(was, _)| was) != Some(small) {
361                if let Some((was, at)) = open {
362                    self.span(style, caps, was, at);
363                }
364                open = Some((small, self.text.len()));
365            }
366            if changed {
367                self.push_mapped(letter, &written);
368            } else {
369                self.push_verbatim(&written);
370            }
371        }
372        if let Some((was, at)) = open {
373            self.span(style, caps, was, at);
374        }
375    }

377    /// Records the span that ends where the text now does.
378    fn span(&mut self, style: &ParagraphStyle, caps: SmallCaps, small: bool, start: usize) {
379        if start >= self.text.len() {
380            return;
381        }
382        let span = StyleSpan {
383            font_id: style.font_id,
384            size: if small {
385                style.size * SMALL_CAPS_RATIO
386            } else {
387                style.size
388            },
389            tracking: style.letter_spacing,
390            features: Features::new(caps == SmallCaps::Feature, style.features.clone()),
391            color: style.color,
392            decoration: style.decoration,
393            inline: self.inline,
394            range: start..self.text.len(),
395        };
396        self.spans.push(span);
397    }

399    /// Opens the inline element `id`, with the box it paints where it
400    /// paints one, and answers what closing it puts back.
401    fn open_inline(&mut self, id: NodeId, box_: Option<InlineBox>) -> Open {
402        let was = self.inline;
403        self.inline = Some(id);
404        let at = box_.map(|box_| {
405            self.boxes.push(BoxSpan {
406                node: id,
407                box_,
408                range: self.text.len()..self.text.len(),
409            });
410            self.boxes.len() - 1
411        });
412        Open { was, at }
413    }
414
415    /// Closes it again, over the text written since.
416    fn close_inline(&mut self, open: Open) {
417        if let Some(at) = open.at {
418            self.boxes[at].range.end = self.text.len();
419        }
420        self.inline = open.was;
421    }

423    /// Joins two adjacent spans set the same way, from `from` to the
424    /// end. Text written a character at a time is one run of one
425    /// style as much as text written in a stretch, and a run shapes
426    /// as a whole: kerning and ligatures do not reach across a span.
427    fn merge_from(&mut self, from: usize) {
428        let mut at = from + 1;
429        while at < self.spans.len() {
430            let joins = self.spans[at - 1].range.end == self.spans[at].range.start
431                && self.spans[at - 1].same_style(&self.spans[at]);
432            if joins {
433                self.spans[at - 1].range.end = self.spans[at].range.end;
434                self.spans.remove(at);
435            } else {
436                at += 1;
437            }
438        }
439    }

442/// One open inline element: the one it was opened inside, and the box
443/// it pushed, where it paints one.
444struct Open {
445    was: Option<NodeId>,
446    at: Option<usize>,
447}

449/// Raises one character to a capital where a synthesis has to draw a
450/// small one, and answers whether it is set at the reduced size. Only
451/// what was lowercase is: a face's own small capitals leave the word
452/// space and the comma the size they were.
453fn raise(caps: SmallCaps, written: &mut String, changed: &mut bool) -> bool {
454    if caps != SmallCaps::Synthesized || !written.chars().any(char::is_lowercase) {
455        return false;
456    }
457    *written = written.to_uppercase();
458    *changed = true;
459    true
460}

462/// Whether a character continues a word rather than ending it.
463/// `capitalize` raises the letter after every other kind, which is
464/// why `well-known` comes out with two capitals and `don't` with one.
465fn continues_word(letter: char) -> bool {
466    letter.is_alphanumeric() || letter == '\'' || letter == '\u{2019}'
467}

469/// What a synthesized small capital is set at, as a fraction of the
470/// size around it. A face's own small caps sit a little above the
471/// x-height, and four-fifths of the cap height is about where that
472/// lands.
473pub(super) const SMALL_CAPS_RATIO: f32 = 0.8;

475impl LineLayout<'_> {
476    /// Flattens a paragraph's inlines, each one styled as the tree
477    /// says. A style boundary is a span boundary, so a run never
478    /// spans two faces.
479    pub(super) fn flatten(
480        &self,
481        inlines: &[Inline],
482        style: &ParagraphStyle,
483        styles: &dyn InlineStyles,
484        lead: Lead,
485    ) -> FlatParagraph {
486        let mut flat = FlatParagraph::new();
487        flat.skip = lead.taken;
488        flat.first_line = lead.pseudo_element.map(|id| (id, lead.reach()));
489        self.walk_inlines(inlines, style, styles, lead, &mut flat);
490        flat
491    }
492
493    /// Flattens the text of the generated box `node`, as that box's
494    /// own text.
495    pub(super) fn flatten_generated(
496        &self,
497        text: &str,
498        node: NodeId,
499        style: &ParagraphStyle,
500    ) -> FlatParagraph {
501        let mut flat = FlatParagraph::new();
502        let lead = Lead::default();
503        let value = text.replace('\n', " ");
504        if !value.is_empty() {
505            flat.open(Some(node), 0);
506            self.push_run(&mut flat, &value, style, lead);
507        }
508        flat
509    }
510
511    /// Flattens preformatted text as the text of `node`: its newlines
512    /// and its spaces stand as the author wrote them.
513    pub(super) fn flatten_preformatted(
514        &self,
515        text: &str,
516        node: NodeId,
517        position: Option<SourcePos>,
518        style: &ParagraphStyle,
519    ) -> FlatParagraph {
520        let mut flat = FlatParagraph::new();
521        if !text.is_empty() {
522            flat.open_at(Some(node), 0, position, true);
523            self.push_run(&mut flat, text, style, Lead::default());
524        }
525        flat
526    }
527
528    fn walk_inlines(
529        &self,
530        inlines: &[Inline],
531        style: &ParagraphStyle,
532        styles: &dyn InlineStyles,
533        lead: Lead,
534        flat: &mut FlatParagraph,
535    ) {
536        for inline in inlines {
537            match inline {
538                Inline::Text {
539                    id,
540                    value,
541                    position,
542                    ..
543                } => self.push_text(flat, *id, value, *position, style, lead),
544                Inline::Break { .. } => self.push_break(flat, style, lead),
545                Inline::Code {
546                    id,
547                    value,
548                    position,
549                    span,
550                    ..
551                } => {
552                    // The node starts at its opening backticks, and
553                    // its text after them.
554                    let ticks = span.map_or(0, |span| {
555                        span.width().saturating_sub(value.len() as u32) / 2
556                    });
557                    let position = position.map(|position| SourcePos {
558                        column: position.column + ticks,
559                        ..position
560                    });
561                    let generated = styles.generated(inline);
562                    let open = flat.open_inline(*id, styles.inline_box(*id));
563                    self.push_generated(flat, generated.before, *id, PseudoElement::Before, lead);
564                    let from = flat.text.len();
565                    let code = styles.style(*id, style);
566                    self.push_text(flat, *id, value, position, &code, lead);
567                    flat.literal.push(from..flat.text.len());
568                    self.push_generated(flat, generated.after, *id, PseudoElement::After, lead);
569                    flat.close_inline(open);
570                }
571                // The note itself is set at the foot of the page.
572                // What stands in the line is its reference. The box
573                // the note paints is around the note there, not
574                // around the reference here, so the reference takes
575                // none of it.
576                Inline::Note { id, .. } => {
577                    let Some((call, style)) = styles.call(inline) else {
578                        continue;
579                    };
580                    let open = flat.open_inline(*id, None);
581                    flat.open(None, 0);
582                    self.push_run(flat, &call, &style, lead);
583                    flat.close_inline(open);
584                }
585                Inline::Emphasis { id, children, .. }
586                | Inline::Strong { id, children, .. }
587                | Inline::Link { id, children, .. }
588                | Inline::Strikethrough { id, children, .. }
589                | Inline::Span { id, children, .. } => {
590                    let generated = styles.generated(inline);
591                    let open = flat.open_inline(*id, styles.inline_box(*id));
592                    self.push_generated(flat, generated.before, *id, PseudoElement::Before, lead);
593                    self.walk_inlines(children, &styles.style(*id, style), styles, lead, flat);
594                    self.push_generated(flat, generated.after, *id, PseudoElement::After, lead);
595                    flat.close_inline(open);
596                }
597            }
598        }
599    }

601    /// Appends text the sheet generated as `which` of `element`. No
602    /// content node holds it, so the runs it lands in name the
603    /// pseudo-element, and a drop cap passes over none of it.
604    fn push_generated(
605        &self,
606        flat: &mut FlatParagraph,
607        generated: Option<(String, ParagraphStyle)>,
608        element: NodeId,
609        which: PseudoElement,
610        lead: Lead,
611    ) {
612        let Some((mut value, style)) = generated else {
613            return;
614        };
615        if value.is_empty() {
616            return;
617        }
618        // Generated text holds no hard break: the newline a heading's
619        // break reads as is a word space here.
620        if value.contains('\n') {
621            value = value.replace('\n', " ");
622        }
623        flat.open(Some(element.pseudo(which)), 0);
624        self.push_run(flat, &value, &style, lead);
625    }

627    /// Appends a hard break as the newline the breaker ends a line
628    /// at. No node's text holds the newline, and a drop cap passes
629    /// over it as it does any other byte.
630    fn push_break(&self, flat: &mut FlatParagraph, style: &ParagraphStyle, lead: Lead) {
631        if flat.skip > 0 {
632            flat.skip -= 1;
633            return;
634        }
635        flat.open(None, 0);
636        self.push_run(flat, "\n", style, lead);
637    }

639    /// Appends one node's text, in the opening style as far as it
640    /// reaches and in `style` after that. What a drop cap took is
641    /// passed over here, so the paragraph starts at the letter after
642    /// the one the cap holds.
643    ///
644    /// The switch to `style` is on the length of the shaped text,
645    /// which is what the extent a break gave is written in. Where the
646    /// opening style still applies the text goes in a character at a
647    /// time, so the switch can fall inside a word; the spans it
648    /// writes merge back into one, and the whole of the opening line
649    /// shapes together.
650    fn push_text(
651        &self,
652        flat: &mut FlatParagraph,
653        node: NodeId,
654        value: &str,
655        position: Option<SourcePos>,
656        style: &ParagraphStyle,
657        lead: Lead,
658    ) {
659        let mut at = flat.skip.min(value.len());
660        while !value.is_char_boundary(at) {
661            at += 1;
662        }
663        flat.skip = flat.skip.saturating_sub(at);
664        let (node_start, value) = (at, &value[at..]);
665        if value.is_empty() {
666            return;
667        }
668        flat.open_at(Some(node), node_start, position, false);
669        self.push_run(flat, value, style, lead);
670    }
671
672    /// Appends one stretch of text, in the opening style as far as
673    /// it reaches and in `style` after that.
674    fn push_run(&self, flat: &mut FlatParagraph, value: &str, style: &ParagraphStyle, lead: Lead) {
675        let (opening, extent) = match lead.style {
676            Some(first) if flat.text.len() < lead.reach() => {
677                (first.over(style, self.registry), lead.reach())
678            }
679            _ => {
680                flat.push_styled(value, style, self.small_caps(style));
681                return;
682            }
683        };
684        let (opening_caps, caps) = (self.small_caps(&opening), self.small_caps(style));
685        let from = flat.spans.len();
686        for (at, letter) in value.char_indices() {
687            if flat.text.len() >= extent {
688                flat.push_styled(&value[at..], style, caps);
689                break;
690            }
691            flat.push_styled(&value[at..at + letter.len_utf8()], &opening, opening_caps);
692        }
693        flat.merge_from(from);
694    }
695
696    /// Where a style's small capitals come from: the face's own where
697    /// it has them, and a synthesis where it does not.
698    pub(super) fn small_caps(&self, style: &ParagraphStyle) -> SmallCaps {
699        match style.caps {
700            FontVariantCaps::Normal => SmallCaps::Off,
701            FontVariantCaps::SmallCaps if self.registry.has_small_caps(style.font_id) => {
702                SmallCaps::Feature
703            }
704            FontVariantCaps::SmallCaps => SmallCaps::Synthesized,
705        }
706    }
707}

772    /// `text-transform` changes what is shaped, and the run says so
773    /// twice over: `text` is what was drawn, which is what a painter
774    /// that draws characters draws, and `source` is what the author
775    /// wrote, which is what extraction reads back.
776    #[test]
777    fn text_transform_shapes_one_text_and_reports_another() {
778        let style = ParagraphStyle {
779            transform: TextTransform::Uppercase,
780            ..body()
781        };
782        let lines = layout_style("Lilliput", 400.0, &style);
783        assert_eq!(
784            line_text(&lines[0]),
785            "LILLIPUT",
786            "the run was not drawn in capitals"
787        );
788        assert_eq!(
789            lines[0].runs[0].source, "Lilliput",
790            "the run lost the source"
791        );
792        let shouted = layout_style("LILLIPUT", 400.0, &body());
793        assert_eq!(
794            lines[0].runs[0]
795                .glyphs
796                .iter()
797                .map(|g| g.id)
798                .collect::<Vec<_>>(),
799            shouted[0].runs[0]
800                .glyphs
801                .iter()
802                .map(|g| g.id)
803                .collect::<Vec<_>>(),
804            "the transform did not reach the shaper",
805        );
806
807        // A mapping that is not one for one: `ß` shapes as two
808        // capitals, and both of them stand for the one letter.
809        let lines = layout_style("Straße", 400.0, &style);
810        let run = &lines[0].runs[0];
811        assert_eq!(
812            (run.text.as_str(), run.source.as_str()),
813            ("STRASSE", "Straße")
814        );
815        assert_eq!(run.glyphs.len(), 7, "STRASSE is seven glyphs");
816        let sharp = run.source.find('ß').expect("the source keeps its ß") as u32;
817        let through = |range: Range<u32>| {
818            run.source_map[range.start as usize]..run.source_map[range.end as usize]
819        };
820        let ranges = run.glyph_ranges();
821        assert_eq!(
822            (through(ranges[4].clone()), through(ranges[5].clone())),
823            (sharp..sharp + 2, sharp + 2..sharp + 2),
824            "the pair of capitals does not stand for the ß once",
825        );
826        assert_eq!(
827            run.source_map.last().copied(),
828            Some(run.source.len() as u32),
829            "the map does not run to the end of the source",
830        );
831    }
