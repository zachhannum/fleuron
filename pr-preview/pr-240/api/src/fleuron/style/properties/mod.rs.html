1//! The supported property vocabulary, and what one node computes to.
2//!
3//! Everything the engine can honour has a variant here; everything
4//! else is a diagnostic. Lengths compute to points, the engine's one
5//! unit, and `line-height` computes to a unitless multiple of the
6//! font size whatever it was written as.
7//!
8//! The vocabulary is split by kind: `value` holds the simple values,
9//! `edges` the four edges of a box, `background` what is painted
10//! behind one, `page` the page box, `counter` what a margin box
11//! holds, `exclusion` a block placed against the page, and
12//! `computed` what one node comes to.

50/// One declaration the engine understood. The cascade applies these
51/// in order, so a later one simply overwrites what an earlier one
52/// said about the same property.
53#[derive(Debug, Clone, PartialEq)]
54pub enum Declaration {
55    FontFamily(Vec<Family>),
56    Color(Color),
57    FontSize(Length),
58    FontStyle(FontStyle),
59    FontWeight(u16),
60    LineHeight(LineHeight),
61    LetterSpacing(Length),
62    FontVariantCaps(FontVariantCaps),
63    /// The features a sheet asked the face for, in the order written.
64    FontFeatureSettings(Vec<FeatureSetting>),
65    FontVariantLigatures(FontVariantLigatures),
66    FontVariantNumeric(FontVariantNumeric),
67    FontVariantAlternates(FontVariantAlternates),
68    TextTransform(TextTransform),
69    TextDecorationLine(DecorationLine),
70    /// `None` is `currentcolor`.
71    TextDecorationColor(Option<Color>),
72    TextDecorationStyle(DecorationStyle),
73    /// `None` is `auto`, which is the thickness the face declares.
74    TextDecorationThickness(Option<Length>),
75    TextAlign(TextAlign),
76    TextJustify(TextJustify),
77    TextIndent(Length),
78    HangingPunctuation(HangingPunctuation),
79    Hyphens(Hyphens),
80    Orphans(u16),
81    Widows(u16),
82    Page(Option<String>),
83    Content(Content),
84    StringSet(Vec<StringSet>),
85    CounterReset(counter::CounterReset),
86    InitialLetter(u16),
87    Position(Position),
88    Inset(Edge, Option<Length>),
89    ZIndex(i32),
90    /// From 0 to 1.
91    Opacity(f32),
92    MixBlendMode(BlendMode),
93    WrapFlow(WrapFlow),
94    Float(Float),
95    Clear(Clear),
96    ShapeOutside(ShapeSource),
97    ShapeMargin(Length),
98    Margin(Edge, Length),
99    Padding(Edge, Length),
100    BorderStyle(Edge, BorderStyle),
101    BorderWidth(Edge, Length),
102    BorderColor(Edge, Option<Color>),
103    /// Along the top or bottom edge, then along the left or right.
104    BorderRadius(Corner, Length, Length),
105    BackgroundColor(Option<Color>),
106    BackgroundImage(Option<Url>),
107    BackgroundRepeat(BackgroundRepeat),
108    BackgroundSize(SizeSource),
109    /// Across the box, then down it.
110    BackgroundPosition(Length, Length),
111    BoxDecorationBreak(BoxDecorationBreak),
112    Width(Option<Length>),
113    Height(Option<Length>),
114    MinHeight(Option<Length>),
115    /// `None` is `none`.
116    MaxWidth(Option<Length>),
117    /// `None` is `none`.
118    MaxHeight(Option<Length>),
119    BorderCollapse(BorderCollapse),
120    ListStyleType(ListStyleType),
121    BreakBefore(Break),
122    BreakAfter(Break),
123    BreakInside(Break),
124    ColumnSpan(ColumnSpan),
125    /// A custom property, whose value is read only through `var()`.
126    Custom(Custom),
127    /// A value that reads a custom property, parsed once the cascade
128    /// knows what the property holds.
129    Pending(Pending),
130}

132/// A custom property as written: a name that starts with `--`, and a
133/// value kept as CSS text.
134#[derive(Debug, Clone, PartialEq)]
135pub struct Custom {
136    /// The name, `--` included, in the case it was written in.
137    pub name: String,
138    /// The value, without `!important`.
139    pub value: String,
140    /// Where the declaration was written, as a warning names it.
141    pub origin: String,
142}
143
144/// A declaration whose value holds `var()`. The property is known and
145/// the value is kept as CSS text until each name it reads is known.
146#[derive(Debug, Clone, PartialEq)]
147pub struct Pending {
148    /// The property, in lowercase. A shorthand stays a shorthand.
149    pub property: String,
150    /// The value, without `!important`.
151    pub value: String,
152    /// Where the declaration was written, as a warning names it.
153    pub origin: String,
154}
155
156impl Declaration {
157    /// The longhand this declaration sets, as CSS names it. Of two
158    /// declarations that name the same longhand, the later one in
159    /// cascade order is the one that counts. A pending declaration
160    /// names the property it was written for, which can be a
161    /// shorthand.
162    pub fn property(&self) -> &str {
163        let edge = |edge: &Edge, [top, right, bottom, left]: [&'static str; 4]| match edge {
164            Edge::Top => top,
165            Edge::Right => right,
166            Edge::Bottom => bottom,
167            Edge::Left => left,
168        };
169        match self {
170            Declaration::FontFamily(_) => "font-family",
171            Declaration::Color(_) => "color",
172            Declaration::FontSize(_) => "font-size",
173            Declaration::FontStyle(_) => "font-style",
174            Declaration::FontWeight(_) => "font-weight",
175            Declaration::LineHeight(_) => "line-height",
176            Declaration::LetterSpacing(_) => "letter-spacing",
177            Declaration::FontVariantCaps(_) => "font-variant-caps",
178            Declaration::FontFeatureSettings(_) => "font-feature-settings",
179            Declaration::FontVariantLigatures(_) => "font-variant-ligatures",
180            Declaration::FontVariantNumeric(_) => "font-variant-numeric",
181            Declaration::FontVariantAlternates(_) => "font-variant-alternates",
182            Declaration::TextTransform(_) => "text-transform",
183            Declaration::TextDecorationLine(_) => "text-decoration-line",
184            Declaration::TextDecorationColor(_) => "text-decoration-color",
185            Declaration::TextDecorationStyle(_) => "text-decoration-style",
186            Declaration::TextDecorationThickness(_) => "text-decoration-thickness",
187            Declaration::TextAlign(_) => "text-align",
188            Declaration::TextJustify(_) => "text-justify",
189            Declaration::TextIndent(_) => "text-indent",
190            Declaration::HangingPunctuation(_) => "hanging-punctuation",
191            Declaration::Hyphens(_) => "hyphens",
192            Declaration::Orphans(_) => "orphans",
193            Declaration::Widows(_) => "widows",
194            Declaration::Page(_) => "page",
195            Declaration::Content(_) => "content",
196            Declaration::StringSet(_) => "string-set",
197            Declaration::CounterReset(_) => "counter-reset",
198            Declaration::InitialLetter(_) => "initial-letter",
199            Declaration::Position(_) => "position",
200            Declaration::Inset(side, _) => edge(side, ["top", "right", "bottom", "left"]),
201            Declaration::ZIndex(_) => "z-index",
202            Declaration::Opacity(_) => "opacity",
203            Declaration::MixBlendMode(_) => "mix-blend-mode",
204            Declaration::WrapFlow(_) => "wrap-flow",
205            Declaration::Float(_) => "float",
206            Declaration::Clear(_) => "clear",
207            Declaration::ShapeOutside(_) => "shape-outside",
208            Declaration::ShapeMargin(_) => "shape-margin",
209            Declaration::Margin(side, _) => edge(
210                side,
211                ["margin-top", "margin-right", "margin-bottom", "margin-left"],
212            ),
213            Declaration::Padding(side, _) => edge(
214                side,
215                [
216                    "padding-top",
217                    "padding-right",
218                    "padding-bottom",
219                    "padding-left",
220                ],
221            ),
222            Declaration::BorderWidth(side, _) => edge(
223                side,
224                [
225                    "border-top-width",
226                    "border-right-width",
227                    "border-bottom-width",
228                    "border-left-width",
229                ],
230            ),
231            Declaration::BorderStyle(side, _) => edge(
232                side,
233                [
234                    "border-top-style",
235                    "border-right-style",
236                    "border-bottom-style",
237                    "border-left-style",
238                ],
239            ),
240            Declaration::BorderColor(side, _) => edge(
241                side,
242                [
243                    "border-top-color",
244                    "border-right-color",
245                    "border-bottom-color",
246                    "border-left-color",
247                ],
248            ),
249            Declaration::BorderRadius(corner, ..) => match corner {
250                Corner::TopLeft => "border-top-left-radius",
251                Corner::TopRight => "border-top-right-radius",
252                Corner::BottomRight => "border-bottom-right-radius",
253                Corner::BottomLeft => "border-bottom-left-radius",
254            },
255            Declaration::BackgroundColor(_) => "background-color",
256            Declaration::BackgroundImage(_) => "background-image",
257            Declaration::BackgroundRepeat(_) => "background-repeat",
258            Declaration::BackgroundSize(_) => "background-size",
259            Declaration::BackgroundPosition(..) => "background-position",
260            Declaration::BoxDecorationBreak(_) => "box-decoration-break",
261            Declaration::Width(_) => "width",
262            Declaration::Height(_) => "height",
263            Declaration::MinHeight(_) => "min-height",
264            Declaration::MaxWidth(_) => "max-width",
265            Declaration::MaxHeight(_) => "max-height",
266            Declaration::BorderCollapse(_) => "border-collapse",
267            Declaration::ListStyleType(_) => "list-style-type",
268            Declaration::BreakBefore(_) => "break-before",
269            Declaration::BreakAfter(_) => "break-after",
270            Declaration::BreakInside(_) => "break-inside",
271            Declaration::ColumnSpan(_) => "column-span",
272            Declaration::Custom(custom) => &custom.name,
273            Declaration::Pending(pending) => &pending.property,
274        }
275    }
