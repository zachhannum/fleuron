20    /// Compiles the styling again and classifies what moved.
21    pub(super) fn recompile(&mut self) {
22        let Some(sheets) = &self.sheets else {
23            return;
24        };
25        let styles = sheets.compile(&self.book, self.registry.get());
26        self.stages.style += 1;
27        let prints = Prints::of(&styles, self.images);
28        self.stale = self.stale.max(self.prints.against(&prints));
29        self.prints = prints;
30        self.section_local = section_local(&styles);
31        self.styles = Cow::Owned(styles);
32    }
33
34    /// Runs the stages the last change invalidated, and no others.
35    pub(super) fn update(&mut self) {
36        if self.stale >= Stale::Trace {
37            self.trace();
38        }
39        if self.retain {
40            if self.stale >= Stale::Break {
41                self.notes = Numbering::of(&self.book, &self.styles);
42                self.rebreak();
43            }
44            if self.stale >= Stale::Flow {
45                self.reflow();
46            }
47            if self.stale >= Stale::Paint {
48                self.repaint();
49            }
50        } else if self.stale != Stale::Nothing {
51            self.run_once();
52        }
53        self.stale = Stale::Nothing;
54        self.collect_warnings();
55    }

57    /// Traces the contours the cascade asks for, keeping the ones an
58    /// earlier run already traced.
59    fn trace(&mut self) {
60        self.stages.trace += self
61            .contours
62            .update(&self.book, &self.styles, self.assets.get());
63    }

65    /// Breaks the sections whose lines the cache cannot answer for,
66    /// and keeps the rest as they stand.
67    fn rebreak(&mut self) {
68        let against = Against::of(&self.styles, self.images);
69        // An image that arrives is a box that was not reserved
70        // before it, so the sections are broken again around it. The
71        // table only grows, so a count that moved is an image.
72        let supplied = self.images.then(|| self.assets.get().assets().len());
73        self.references = if self.styles.refers() {
74            References::of(&self.book)
75        } else {
76            References::default()
77        };
78        let mut previous: Vec<Option<Cached>> = std::mem::take(&mut self.lines)
79            .into_iter()
80            .map(Some)
81            .collect();
82        let mut spare: HashMap<u64, Vec<usize>> = HashMap::new();
83        if self.section_local {
84            for (index, cached) in previous.iter().enumerate() {
85                let key = cached.as_ref().expect("nothing is taken yet").key;
86                spare.entry(key).or_default().push(index);
87            }
88        }
89        let mut fresh = Vec::with_capacity(self.book.sections.len());
90        for section in &self.book.sections {
91            let key = section_key(
92                section,
93                &self.styles,
94                against,
95                supplied,
96                hyphenation(&self.book.metadata),
97                &self.references,
98                &self.notes,
99            );
100            let kept = spare
101                .get_mut(&key)
102                .and_then(|slots| slots.pop())
103                .and_then(|slot| previous[slot].take());
104            fresh.push(match kept.filter(|cached| cached.renumbers(section.id)) {
105                Some(mut cached) => {
106                    cached.renumber(section.id);
107                    cached
108                }
109                None => {
110                    // One paginator per section, so the warnings it
111                    // collects are the ones this section raised.
112                    let paginator = Paginator::with_contours(
113                        self.registry.get(),
114                        &self.styles,
115                        self.assets.get(),
116                        &self.contours,
117                    );
118                    paginator.language(&self.book.metadata);
119                    paginator.refer(self.references.clone());
120                    paginator.number(self.notes.clone());
121                    let fragments = paginator.section_fragments(section);
122                    self.stages.lines += 1;
123                    Cached {
124                        key,
125                        section: section.id,
126                        fragments,
127                        warnings: paginator.warnings(),
128                    }
129                }
130            });
131        }
132        self.lines = fresh;
133        self.flow_warnings.clear();
134        for cached in &self.lines {
135            for warning in &cached.warnings {
136                if !self
137                    .flow_warnings
138                    .iter()
139                    .any(|seen| seen.message == warning.message)
140                {
141                    self.flow_warnings.push(warning.clone());
142                }
143            }
144        }
145    }

147    /// Flows the cached lines into pages. Nothing here measures, but
148    /// for the sections whose references print a page, which are
149    /// built again once the pages are known.
150    fn reflow(&mut self) {
151        let mut paged = self.fragment(false);
152        if self.styles.numbers_notes_per_page() {
153            paged = self.settle_notes(paged);
154        }
155        if self.styles.counts_pages() {
156            paged = self.settle(&paged);
157        } else {
158            self.settled.clear();
159            self.settle_warnings.clear();
160        }
161        self.infos = paged.infos;
162        self.boxes = paged.boxes;
163        self.place(paged.pages, &paged.targets);
164    }

166    /// Takes the pages of a flow, with the links on them and the
167    /// outline of the book's headings.
168    fn place(&mut self, mut pages: Vec<Page>, targets: &BTreeMap<NodeId, PageBox>) {
169        let (registry, assets) = (self.registry.get(), self.assets.get());
170        let (navigation, warnings) = navigation(&self.book, &mut pages, targets, registry);
171        self.link_warnings = warnings;
172        let output = self
173            .output
174            .get_or_insert_with(|| blank_output(registry, assets));
175        output.pages = pages;
176        output.navigation = navigation;
177    }

179    /// Fragments the book over each section's lines: the lines of the
180    /// pass that finds the pages, or, where `settled` asks for them
181    /// and a section has them, the lines of the pass that prints them.
182    fn fragment(&mut self, settled: bool) -> Paged {
183        let paginator = Paginator::with_contours(
184            self.registry.get(),
185            &self.styles,
186            self.assets.get(),
187            &self.contours,
188        );
189        let sections = self.lines.iter().enumerate().map(|(index, cached)| {
190            let kept = settled
191                .then(|| self.settled.get(index).and_then(Option::as_ref))
192                .flatten();
193            kept.unwrap_or(cached).fragments.as_slice()
194        });
195        let paged = paginator.fragment(&self.book, sections);
196        self.stages.flow += 1;
197        paged
198    }

200    /// Numbers the notes by the page their references were set on,
201    /// and breaks the sections that hold one again to print those
202    /// numbers.
203    ///
204    /// A number of another width moves the line its reference is on,
205    /// which can move a note onto another page and number it again.
206    /// So the pass runs until the numbering stops changing, and the
207    /// numbering of the last pass stands where it does not.
208    fn settle_notes(&mut self, mut paged: Paged) -> Paged {
209        let start = self.styles.first_note_number();
210        for _ in 0..NOTE_PASSES {
211            let numbered = self.notes.on_pages(&paged.notes, start);
212            if numbered == self.notes {
213                return paged;
214            }
215            self.notes = numbered;
216            self.rebreak();
217            self.stages.settle += 1;
218            paged = self.fragment(false);
219        }
220        self.flow_warnings.push(Warning {
221            message: UNSETTLED.to_string(),
222            origin: None,
223        });
224        paged
225    }
226
227    /// Lays the book out again with the folio each reference prints,
228    /// read off the pages of the pass before. Only the sections whose
229    /// references print a page are built again, and one that prints
230    /// the folios it printed last time keeps the lines it had.
231    fn settle(&mut self, paged: &Paged) -> Paged {
232        let found = landed(paged);
233        let resolved = self.references.landed(found.clone());
234        let mut spare: HashMap<u64, Vec<Cached>> = HashMap::new();
235        for cached in std::mem::take(&mut self.settled).into_iter().flatten() {
236            if self.section_local {
237                spare.entry(cached.key).or_default().push(cached);
238            }
239        }
240        let mut printed = BTreeSet::new();
241        let mut warnings: Vec<Warning> = Vec::new();
242        let mut settled = Vec::with_capacity(self.book.sections.len());
243        for (section, first) in self.book.sections.iter().zip(&self.lines) {
244            let named = Named::in_section(section, &self.styles, &self.references);
245            if named.pages.is_empty() {
246                settled.push(None);
247                continue;
248            }
249            let key = settled_key(first.key, &named.pages, &found);
250            printed.extend(named.pages);
251            let cached = match spare
252                .get_mut(&key)
253                .and_then(Vec::pop)
254                .filter(|cached| cached.renumbers(section.id))
255            {
256                Some(mut cached) => {
257                    cached.renumber(section.id);
258                    cached
259                }
260                None => {
261                    let paginator = Paginator::with_contours(
262                        self.registry.get(),
263                        &self.styles,
264                        self.assets.get(),
265                        &self.contours,
266                    );
267                    paginator.language(&self.book.metadata);
268                    paginator.refer(resolved.clone());
269                    paginator.number(self.notes.clone());
270                    let fragments = paginator.section_fragments(section);
271                    self.stages.lines += 1;
272                    Cached {
273                        key,
274                        section: section.id,
275                        fragments,
276                        warnings: paginator.warnings(),
277                    }
278                }
279            };
280            for warning in &cached.warnings {
281                if !warnings.iter().any(|seen| seen.message == warning.message) {
282                    warnings.push(warning.clone());
283                }
284            }
285            settled.push(Some(cached));
286        }
287        self.settled = settled;
288        self.stages.settle += 1;
289        let paged = self.fragment(true);
290        warnings.extend(moved(&found, &landed(&paged), &printed, &self.references));
291        self.settle_warnings = warnings;
292        paged
293    }

295    /// Repaints the furniture over pages the flow already settled.
296    fn repaint(&mut self) {
297        let paginator = Paginator::with_contours(
298            self.registry.get(),
299            &self.styles,
300            self.assets.get(),
301            &self.contours,
302        );
303        if let Some(output) = &mut self.output {
304            paginator.paint(&mut output.pages, &self.infos);
305            self.stages.paint += 1;
306        }
307    }

309    /// The whole pipeline, one section's lines alive at a time.
310    fn run_once(&mut self) {
311        let paginator = Paginator::with_contours(
312            self.registry.get(),
313            &self.styles,
314            self.assets.get(),
315            &self.contours,
316        );
317        let paged = paginator.paginated(&self.book);
318        let passes = 1 + paginator.settles();
319        self.stages.lines += self.book.sections.len() as u32 * passes;
320        self.stages.flow += passes;
321        self.stages.settle += paginator.settles();
322        self.stages.paint += 1;
323        self.flow_warnings = paginator.warnings();
324        self.place(paged.pages, &paged.targets);
325    }
326
327    /// Everything the run has to complain about, in the order the
328    /// stages raised it.
329    fn collect_warnings(&mut self) {
330        let mut warnings = self.source_warnings.clone();
331        warnings.extend(self.styles.warnings().iter().cloned());
332        warnings.extend(self.assets.get().warnings().iter().cloned());
333        warnings.extend(self.contours.warnings().iter().cloned());
334        warnings.extend(self.flow_warnings.iter().cloned());
335        for warning in self.settle_warnings.iter().chain(&self.link_warnings) {
336            if !warnings.iter().any(|seen| seen.message == warning.message) {
337                warnings.push(warning.clone());
338            }
339        }
340        if let Some(output) = &mut self.output {
341            output.warnings = warnings;
342        }
343    }

795    /// A chapter read from `source` that opens on a heading carrying
796    /// `id`, whose first paragraph links to `to`, over eight
797    /// paragraphs of prose tagged `tag`.
798    fn linked_chapter(source: &str, title: &str, id: &str, to: &str, tag: &str) -> Section {
799        let words = |value: &str| Inline::Text {
800            id: NodeId::UNASSIGNED,
801            value: value.into(),
802            attributes: Attributes::default(),
803            position: None,
804            span: None,
805        };
806        let mut blocks = vec![
807            Block::Heading {
808                id: NodeId::UNASSIGNED,
809                level: HeadingLevel::H1,
810                inlines: vec![words(title)],
811                attributes: Attributes {
812                    id: Some(id.into()),
813                    classes: Vec::new(),
814                },
815                position: None,
816                span: None,
817            },
818            Block::Paragraph {
819                id: NodeId::UNASSIGNED,
820                inlines: vec![
821                    words("See "),
822                    Inline::Link {
823                        id: NodeId::UNASSIGNED,
824                        url: to.into(),
825                        children: vec![words("there")],
826                        attributes: Attributes::default(),
827                        position: None,
828                        span: None,
829                    },
830                    words("."),
831                ],
832                attributes: Attributes::default(),
833                position: None,
834                span: None,
835            },
836        ];
837        blocks.extend(prose(tag, 8));
838        section(source, blocks)
839    }

863    fn heading_folio(session: &mut Session<'_>, section: usize) -> u32 {
864        let node = crate::content::block_id(&session.book().sections[section].blocks[0]);

870    /// Every page number the book's references print, in reading
871    /// order.
872    fn printed(output: &crate::LayoutOutput) -> Vec<u32> {
873        let whole = painted(output).join(" ");
874        whole
875            .split("(page ")
876            .skip(1)
877            .map(|rest| {
878                let digits: String = rest.chars().take_while(char::is_ascii_digit).collect();
879                digits.parse().expect("a number follows every reference")
880            })
881            .collect()
882    }
