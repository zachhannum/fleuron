1//! A list: each item a block inside the list's content box, and the
2//! marker of each item to the left of it, on the baseline of the
3//! item's first line.
4
5use crate::content::{ListItem, NodeId, PseudoElement, SourcePos};
6use crate::style::{Break, ComputedStyle};
7
8use super::build::{Builder, children};
9use super::fragment::{Fragment, Marker, Piece};
10
11impl Builder<'_, '_> {
12    /// One list, at `x` from the content box's leading edge and
13    /// breaking to `measure`. `start` is the number of its first item.
14    pub(super) fn list(
15        &mut self,
16        id: NodeId,
17        start: u32,
18        items: &[ListItem],
19        position: Option<SourcePos>,
20        x: f32,
21        measure: f32,
22    ) {
23        let styles = self.paginator.styles;
24        let style = styles.style(id).clone();
25        let open = self.open(id, &style, &[], x, measure);
26        let (inner, narrowed) = style.content_box(x, measure);
27        self.pseudo(id, PseudoElement::Before, position, inner, narrowed);
28        let last = items.len().saturating_sub(1);
29        for (index, item) in items.iter().enumerate() {
30            // A page does not end after the first item or before the
31            // last one, so neither is left alone on a page.
32            if index == 1 || (index > 0 && index == last) {
33                self.ask(Break::Avoid);
34            }
35            self.item(item, start.saturating_add(index as u32), inner, narrowed);
36        }
37        self.pseudo(id, PseudoElement::After, position, inner, narrowed);
38        self.close(&style, open);
39    }
40
41    /// One item of a list, numbered `number`.
42    fn item(&mut self, item: &ListItem, number: u32, x: f32, measure: f32) {
43        let styles = self.paginator.styles;
44        if item.attributes.id.is_some() {
45            self.name(item.id);
46        }
47        let style = styles.style(item.id).clone();
48        let open = self.open(item.id, &style, &[], x, measure);
49        let (inner, narrowed) = style.content_box(x, measure);
50        self.blocks(
51            children(styles, item.id, &item.blocks, item.position),
52            inner,
53            narrowed,
54        );
55        if let Some(marker) = self.marker(&style, number, x, measure) {
56            self.hang(open, marker);
57        }
58        self.close(&style, open);
59    }
60
61    /// The marker of the item numbered `number`, shaped in the item's
62    /// style. It ends where the item's border box starts.
63    fn marker(&self, style: &ComputedStyle, number: u32, x: f32, measure: f32) -> Option<Marker> {
64        let text = style.list_style_type.marker(number)?;
65        let line = self.paginator.line_of(&text, &style.paragraph())?;
66        let (left, _) = style.border_box(x, measure);
67        let x = left - self.paginator.line_width(&line);
68        Some(Marker { line, x })
69    }
70
71    /// Puts a marker on the first fragment its item emitted, from
72    /// `start`. An item that emitted nothing emits the height of the
73    /// marker to set it on.
74    pub(super) fn hang(&mut self, start: usize, marker: Marker) {
75        let first = self.fragments[start..]
76            .iter_mut()
77            .find(|fragment| !matches!(fragment.piece, Piece::Anchor(_)));
78        match first {
79            Some(fragment) => fragment
80                .markers
81                .get_or_insert_with(Box::default)
82                .insert(0, marker),
83            None => {
84                let mut blank = Fragment::plain(0.0, marker.line.box_.height, Piece::Blank);
85                blank.markers = Some(Box::new(vec![marker]));
86                self.emit(&mut true, blank);
87            }
88        }
89    }
90}
