1//! fleuron-wasm: layout in a worker, bytes out.
2//!
3//! The door between the engine and a host: markdown or a content
4//! tree and CSS text in, one buffer out: the postcard display structure,
5//! PDF bytes on the export path, or an EPUB with its warnings. The engine never touches the
6//! DOM, opens nothing, and reads no clock.
7//!
8//! The surface is a [`Session`] rather than a function, because the
9//! event a preview is built around is a small change to one input
10//! while the others stand. Inputs cross when they change: a face is
11//! registered once and stays registered, a manuscript crosses once
12//! and a keystroke replaces the one file it touched, and the
13//! author's styling crosses as named sheets on its own.
14//! [`render`] is the batch case, and it is the same session used
15//! once.
16//!
17//! Nothing here deals with generations or workers. A render runs to
18//! completion, so a superseded one is dropped by the host that asked
19//! for it rather than interrupted half-way through a stage; the
20//! protocol that does the dropping is in the TypeScript beside this
21//! crate.
22

36/// The version the display structure is encoded at. A host reads the same
37/// number off the front of every buffer and refuses one it does not
38/// know.
39#[wasm_bindgen(js_name = wireVersion)]
40pub fn wire_version() -> u16 {
41    wire::VERSION
42}

44/// A retained pipeline, kept in the module between calls.
45///
46/// The session keeps the content tree, the styling and every stage
47/// between them and the page, so a second render pays for what
48/// changed and not for the book. What the stages cost, and what
49/// survives which edit, is the engine's own contract.
50#[wasm_bindgen]
51pub struct Session {
52    engine: Engine<'static>,
53    reading: Options,
54    /// What reading each source complained about, kept per source so
55    /// that replacing one file replaces its complaints and no
56    /// others.
57    complaints: BTreeMap<String, Vec<Warning>>,
58    /// The classes and id the host named each source's sections by,
59    /// kept so that a source read again keeps them.
60    names: BTreeMap<String, Attributes>,
61}

103    /// Which markdown the sources are written in: `fleuron`,
104    /// `commonmark`, `gfm` or `obsidian`. `fleuron` is what a session

118    /// Where a source's sections begin: at a heading of this level
119    /// or shallower, or nowhere at all when the level is zero, which
120    /// makes each file one section.
121    ///
122    /// Sections are what a page can start on, so this sets a book's
123    /// page count before any styling does.

126        self.reading.sections = match level {

128            level => Sections::AtHeading(HeadingLevel::try_from(level).map_err(js_error)?),
129        };
130        Ok(())

133    /// Reads one markdown source as the whole book, its frontmatter
134    /// the book's metadata.
135    ///
136    /// Everything below box construction is invalidated: this is the
137    /// manuscript arriving, not an edit to it.
138    #[wasm_bindgen(js_name = setMarkdown)]
139    pub fn set_markdown(&mut self, name: &str, text: &str) {
140        let metadata = fleuron_markdown::frontmatter(text);
141        let (sections, complaints) = fleuron_markdown::to_sections(text, name, &self.reading);
142        self.engine
143            .set_content(fleuron_markdown::assemble(metadata, sections));
144        self.complaints.clear();
145        self.names.clear();
146        self.complain(name, complaints);
147    }

149    /// Reads several markdown sources as one book, in the order
150    /// given.
151    ///
152    /// A lone source is the whole book, so its frontmatter is the
153    /// book's. Several are chapters: each file's frontmatter belongs
154    /// to the section it became, and the book is left unnamed rather
155    /// than named after whichever chapter came first, which is what
156    /// [`Session::set_metadata`] is for.
157    ///
158    /// `attributes`, when given, holds one entry per source: the
159    /// classes and id its sections take, as JSON in the shape
160    /// [`Session::set_source_attributes`] reads, or an empty string
161    /// for none.
162    #[wasm_bindgen(js_name = setSources)]
163    pub fn set_sources(
164        &mut self,
165        names: Vec<String>,
166        texts: Vec<String>,
167        attributes: Option<Vec<String>>,
168    ) -> Result<(), JsError> {
169        if names.len() != texts.len() {
170            return Err(JsError::new(&format!(
171                "{} sources named and {} handed over",
172                names.len(),
173                texts.len()
174            )));
175        }
176        let attributes = attributes.unwrap_or_default();
177        if !attributes.is_empty() && attributes.len() != names.len() {
178            return Err(JsError::new(&format!(
179                "{} sources named and {} named by attributes",
180                names.len(),
181                attributes.len()
182            )));
183        }
184        let metadata = match texts.as_slice() {
185            [whole] => fleuron_markdown::frontmatter(whole),
186            _ => Metadata::default(),
187        };
188        self.names.clear();
189        for (name, json) in names.iter().zip(&attributes) {
190            if !json.is_empty() {
191                self.names.insert(name.clone(), attributes_from(json)?);
192            }
193        }
194        self.complaints.clear();
195        let mut sections = Vec::new();
196        for (name, text) in names.iter().zip(&texts) {
197            let (mut read, complaints) = fleuron_markdown::to_sections(text, name, &self.reading);
198            self.name(name, &mut read);
199            sections.extend(read);
200            self.complaints.insert(name.clone(), complaints);
201        }
202        self.engine
203            .set_content(fleuron_markdown::assemble(metadata, sections));
204        self.engine.set_source_order(names);
205        self.reflect();
206        Ok(())
207    }

209    /// Drops every section that came from one source, and the

211    #[wasm_bindgen(js_name = removeMarkdown)]
212    pub fn remove_markdown(&mut self, name: &str) {
213        self.engine.remove_source(name);
214        self.complaints.remove(name);
215        self.names.remove(name);
216        self.reflect();
217    }

219    /// Names every section read from one source by classes and an id,
220    /// from JSON: `{"classes": ["chapter"], "id": "chapter-twelve"}`.
221    /// `{}` takes the names away.
222    ///
223    /// A sheet reaches the sections as `section.chapter` and
224    /// `section#chapter-twelve`. The names stay with the source's name
225    /// rather than its place in the book, so a rule written against
226    /// them still matches after a reorder. The text is not touched,
227    /// so every byte `nodeAt` and `nodeSource` answer with stays where
228    /// it was, and the source keeps the names when it is read again.
229    #[wasm_bindgen(js_name = setSourceAttributes)]
230    pub fn set_source_attributes(&mut self, name: &str, json: &str) -> Result<(), JsError> {
231        let attributes = attributes_from(json)?;
232        self.engine.set_source_attributes(name, &attributes);
233        if attributes.is_empty() {
234            self.names.remove(name);
235        } else {
236            self.names.insert(name.to_string(), attributes);
237        }
238        Ok(())
239    }

242    /// object for whatever else a frontend read.
243    ///
244    /// A book read from several sources has no frontmatter of its
245    /// own, so this is how it gets a title. A book renamed between
246    /// renders re-runs no stage: the PDF writer reads the name, and

248    /// the hyphenation patterns. A book that declares a different

250    #[wasm_bindgen(js_name = setMetadata)]
251    pub fn set_metadata(&mut self, json: &str) -> Result<(), JsError> {
252        let metadata: Metadata = serde_json::from_str(json).map_err(js_error)?;
253        self.engine.set_metadata(metadata);
254        Ok(())
255    }

257    /// Replaces every section that came from one source, reparsing
258    /// that source alone. A source whose text has no blocks keeps its
259    /// place in the book for the text it is given next. A name the
260    /// book has not seen appends instead, which is how a new file
261    /// arrives.
262    ///
263    /// This is the keystroke path: one file crosses, one file is
264    /// read, and every other section keeps the lines it already has.
265    #[wasm_bindgen(js_name = updateMarkdown)]
266    pub fn update_markdown(&mut self, name: &str, text: &str) {
267        let (mut sections, complaints) = fleuron_markdown::to_sections(text, name, &self.reading);
268        self.name(name, &mut sections);
269        self.engine.replace_source(name, sections);
270        self.complain(name, complaints);
271    }

273    /// Sets the book from a content tree, as JSON.
274    ///
275    /// Markdown is the way in; this is the door for a host with a
276    /// structured source of its own. Node ids are the engine's and
277    /// are assigned on the way in, so a tree built by hand needs
278    /// none.
279    #[wasm_bindgen(js_name = setContent)]
280    pub fn set_content(&mut self, json: &str) -> Result<(), JsError> {
281        let book: Book = serde_json::from_str(json).map_err(js_error)?;
282        self.engine.set_content(book);
283        self.complaints.clear();
284        self.names.clear();
285        self.engine.set_source_warnings(Vec::new());
286        Ok(())
287    }

289    /// Sets the author styling from named sheets, in cascade order,
290    /// over the built-in one. Later sheets win, and a warning names
291    /// the sheet its declaration was written in.
292    ///
293    /// A host that builds its styling out of layers hands over the
294    /// layers.
295    ///
296    /// Which stages this costs is the change's own business: a
297    /// colour repaints nothing, page geometry re-fragments over the
298    /// lines already broken, and only the measure or the face breaks
299    /// them again.
300    #[wasm_bindgen(js_name = setStyle)]
301    pub fn set_style(&mut self, names: Vec<String>, texts: Vec<String>) -> Result<(), JsError> {
302        if names.len() != texts.len() {
303            return Err(JsError::new(&format!(
304                "{} sheets named and {} handed over",
305                names.len(),
306                texts.len()
307            )));
308        }
309        let sources: Vec<Source<'_>> = names
310            .iter()
311            .zip(&texts)
312            .map(|(name, css)| Source::author(name, css))
313            .collect();
314        self.engine.set_style(Stylesheets::parse(&sources));
315        Ok(())
316    }

349    /// The display structure, postcard-encoded, version first.
350    ///
351    /// `first` and `count` ask for a slice of the book's pages rather
352    /// than all of them; leaving either out answers with the whole
353    /// book, same as before ranges existed.
354    pub fn preview(&mut self, first: Option<u32>, count: Option<u32>) -> Result<Vec<u8>, JsError> {
355        let output = self.engine.preview();
356        match (first, count) {
357            (Some(first), Some(count)) => {
358                wire::encode_range(output, first as usize, count as usize)
359            }
360            _ => wire::encode(output),
361        }
362        .map_err(js_error)
363    }

390    /// The same EPUB as its files, not zipped, in the wire's
391    /// encoding: the version, the warnings, the documents in reading
392    /// order, each with its path and the node of its section, then
393    /// each file with its path in the container, its media type and
394    /// its bytes.
395    ///
396    /// The files are the entries of the zip [`Session::export_epub`]
397    /// returns, in the same order, for a host that loads them into a
398    /// browser frame one by one.
399    #[wasm_bindgen(js_name = exportEpubFiles)]
400    pub fn export_epub_files(&self) -> Result<Vec<u8>, JsError> {
401        let files = self.epub_files();
402        wire::encode_epub_files(&wire::EpubFiles {
403            warnings: files.warnings,
404            spine: files
405                .spine
406                .into_iter()
407                .map(|entry| wire::EpubSpineEntry {
408                    path: entry.path,
409                    section: entry.section,
410                })
411                .collect(),
412            files: files
413                .files
414                .into_iter()
415                .map(|file| wire::EpubFile {
416                    path: file.path,
417                    media_type: file.media_type,
418                    bytes: file.bytes,
419                })
420                .collect(),
421        })
422        .map_err(js_error)
423    }

478    /// Where each of these nodes' content is set, as JSON: one
479    /// answer per node, in the order asked about, each
480    /// `{"first": 38, "last": 52, "at": 37, "count": 15}` or `null`.
481    ///
482    /// This is the direction a reflow invalidates. A new face
483    /// repaginates the book, and a host that puts the reader back
484    /// where they were, or turns to a chapter, or names the chapter
485    /// on screen, asks where a node went. No page crosses to answer
486    /// it.
487    ///
488    /// `first` and `last` are the folios, which is what a host puts
489    /// on screen. `at` and `count` are where those pages fall in the
490    /// book, counting from 0, which is what
491    /// [`Session::preview`]'s own `first` and `count` take. A page
492    /// counter that restarts makes the two differ.
493    ///
494    /// A node covers itself and everything under it, so a heading
495    /// answers with the page its own text is on. `null` for a node
496    /// the book does not hold, one the engine synthesized, or one
497    /// whose content reaches no page.
498    #[wasm_bindgen(js_name = nodeFolios)]
499    pub fn node_folios(&mut self, nodes: Vec<u32>) -> Result<String, JsError> {
500        let nodes: Vec<NodeId> = nodes.into_iter().map(NodeId::new).collect();
501        serde_json::to_string(&self.engine.folios(&nodes)).map_err(js_error)
502    }

504    /// What styled one node and where it landed, as JSON: the element
505    /// the node stands for and the elements around it, the rules that
506    /// matched it in cascade order with the declarations that won, the
507    /// computed value of every property, and its border box on each
508    /// page it reaches. A text node answers for the element that holds
509    /// it. A pseudo-element answers for itself, and `pseudoElement`
510    /// names it.
511    ///
512    /// The engine runs the cascade again for the one element, over the
513    /// sheets it laid the book out with, so a host matches no selector
514    /// of its own. `undefined` for a node the engine synthesized, or
515    /// one the book does not hold.
516    pub fn inspect(&mut self, node: u32) -> Result<Option<String>, JsError> {
517        self.engine
518            .inspect(NodeId::new(node))
519            .map(|inspection| serde_json::to_string(&inspection))
520            .transpose()
521            .map_err(js_error)
522    }

568    /// so that what the frontend said comes back on the display structure
569    /// beside what the styling and the layout said.

571        self.complaints.insert(name.to_string(), complaints);
572        self.reflect();
573    }

575    /// The EPUB [`Session::export_epub`] encodes, with the sources'
576    /// complaints in front of the writer's own.
577    pub fn epub(&self) -> fleuron_epub::Epub {
578        let files = self.epub_files();
579        fleuron_epub::Epub {
580            bytes: files.zip(),
581            warnings: files.warnings,
582        }
583    }
584
585    /// The files of that EPUB, not zipped, with the same warnings.
586    pub fn epub_files(&self) -> fleuron_epub::Files {
587        let unstyled;
588        let sheets = match self.engine.sheets() {
589            Some(sheets) => sheets,
590            None => {
591                unstyled = Stylesheets::parse(&[]);
592                &unstyled
593            }
594        };
595        let mut files = fleuron_epub::files(
596            self.engine.book(),
597            sheets,
598            self.engine.images(),
599            self.engine.font_files(),
600        );
601        let mut warnings: Vec<Warning> = self.complaints.values().flatten().cloned().collect();
602        warnings.append(&mut files.warnings);
603        files.warnings = warnings;
604        files
605    }
606
607    /// Hands the engine every source's complaints as they stand.
608    fn reflect(&mut self) {
609        self.engine
610            .set_source_warnings(self.complaints.values().flatten().cloned().collect());
611    }
612
613    /// Gives the sections just read from one source the names the host
614    /// set on it. A source the host named nothing keeps what its
615    /// frontmatter said.
616    fn name(&self, source: &str, sections: &mut [Section]) {
617        if let Some(attributes) = self.names.get(source) {
618            for section in sections {
619                section.attributes = attributes.clone();
620            }
621        }
622    }
623}
624
625fn attributes_from(json: &str) -> Result<Attributes, JsError> {
626    serde_json::from_str(json).map_err(js_error)
627}

629/// One markdown source and one stylesheet, laid out once: the batch
630/// case, over the same session a live preview keeps.
631#[wasm_bindgen]
632pub fn render(markdown: &str, css: &str) -> Result<Vec<u8>, JsError> {
633    once(markdown, css)?.preview(None, None)
634}
635
636/// The same, as PDF bytes.
637#[wasm_bindgen(js_name = renderPdf)]
638pub fn render_pdf(markdown: &str, css: &str) -> Result<Vec<u8>, JsError> {
639    once(markdown, css)?.export_pdf()
640}
641
642fn once(markdown: &str, css: &str) -> Result<Session, JsError> {
643    let mut session = Session::new()?;
644    session.set_markdown("book.md", markdown);
645    if !css.is_empty() {
646        session.set_style(vec!["author.css".to_string()], vec![css.to_string()])?;
647    }
648    Ok(session)
649}
