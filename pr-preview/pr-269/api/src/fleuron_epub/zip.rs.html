1//! The container: a zip archive with every timestamp pinned, so one
2//! book always makes the same bytes.
3
4/// 1980-01-01 00:00:00, the earliest time a zip entry can carry.
5const DOS_DATE: u16 = (1 << 5) | 1;
6const DOS_TIME: u16 = 0;
7
8/// Version 2.0: deflate, and no feature past it.
9const VERSION: u16 = 20;
10
11/// Bit 11: the entry name is UTF-8.
12const UTF8: u16 = 1 << 11;
13
14/// How an entry's bytes are held.
15#[derive(Debug, Clone, Copy, PartialEq, Eq)]
16pub enum Method {
17    /// As they are. The `mimetype` entry has to be held this way.
18    Stored,
19    /// Deflated.
20    Deflated,
21}
22
23impl Method {
24    fn code(self) -> u16 {
25        match self {
26            Method::Stored => 0,
27            Method::Deflated => 8,
28        }
29    }
30}
31
32/// One entry already in the archive, as the central directory lists
33/// it.
34struct Entry {
35    name: String,
36    method: Method,
37    crc: u32,
38    compressed: u32,
39    size: u32,
40    offset: u32,
41}
42
43/// An archive being written, entry by entry, in the order the entries
44/// are added.
45#[derive(Default)]
46pub struct Archive {
47    bytes: Vec<u8>,
48    entries: Vec<Entry>,
49}
50
51impl Archive {
52    /// Adds one entry.
53    pub fn add(&mut self, name: &str, data: &[u8], method: Method) {
54        let held = match method {
55            Method::Stored => data.to_vec(),
56            Method::Deflated => miniz_oxide::deflate::compress_to_vec(data, 9),
57        };
58        let entry = Entry {
59            name: name.to_string(),
60            method,
61            crc: crc32fast::hash(data),
62            compressed: held.len() as u32,
63            size: data.len() as u32,
64            offset: self.bytes.len() as u32,
65        };
66        let out = &mut self.bytes;
67        u32le(out, 0x0403_4b50);
68        u16le(out, VERSION);
69        header(out, &entry);
70        u16le(out, 0);
71        out.extend_from_slice(entry.name.as_bytes());
72        out.extend_from_slice(&held);
73        self.entries.push(entry);
74    }
75
76    /// The archive, with its central directory written after the
77    /// entries.
78    pub fn finish(mut self) -> Vec<u8> {
79        let start = self.bytes.len() as u32;
80        let out = &mut self.bytes;
81        for entry in &self.entries {
82            u32le(out, 0x0201_4b50);
83            u16le(out, VERSION);
84            u16le(out, VERSION);
85            header(out, entry);
86            // Extra field, comment, disk, internal and external
87            // attributes.
88            u16le(out, 0);
89            u16le(out, 0);
90            u16le(out, 0);
91            u16le(out, 0);
92            u32le(out, 0);
93            u32le(out, entry.offset);
94            out.extend_from_slice(entry.name.as_bytes());
95        }
96        let size = out.len() as u32 - start;
97        let count = self.entries.len() as u16;
98        u32le(out, 0x0605_4b50);
99        u16le(out, 0);
100        u16le(out, 0);
101        u16le(out, count);
102        u16le(out, count);
103        u32le(out, size);
104        u32le(out, start);
105        u16le(out, 0);
106        self.bytes
107    }
108}
109
110/// The fields a local header and a central directory entry share,
111/// from the flags to the length of the name.
112fn header(out: &mut Vec<u8>, entry: &Entry) {
113    let flags = if entry.name.is_ascii() { 0 } else { UTF8 };
114    u16le(out, flags);
115    u16le(out, entry.method.code());
116    u16le(out, DOS_TIME);
117    u16le(out, DOS_DATE);
118    u32le(out, entry.crc);
119    u32le(out, entry.compressed);
120    u32le(out, entry.size);
121    u16le(out, entry.name.len() as u16);
122}
123
124fn u16le(out: &mut Vec<u8>, value: u16) {
125    out.extend_from_slice(&value.to_le_bytes());
126}
127
128fn u32le(out: &mut Vec<u8>, value: u32) {
129    out.extend_from_slice(&value.to_le_bytes());
130}

132#[cfg(test)]
133mod tests {
134    use super::*;
135
136    fn archive() -> Vec<u8> {
137        let mut archive = Archive::default();
138        archive.add("mimetype", b"application/epub+zip", Method::Stored);
139        archive.add(
140            "EPUB/a.txt",
141            "a line of prose\n".repeat(40).as_bytes(),
142            Method::Deflated,
143        );
144        archive.finish()
145    }
146
147    /// The first entry is the media type, held as it is, so the
148    /// bytes at offset 38 name the format.
149    #[test]
150    fn the_first_entry_is_stored_and_readable_at_a_fixed_offset() {
151        let bytes = archive();
152        assert_eq!(&bytes[0..4], b"PK\x03\x04");
153        assert_eq!(u16::from_le_bytes([bytes[8], bytes[9]]), 0, "stored");
154        assert_eq!(&bytes[30..38], b"mimetype");
155        assert_eq!(&bytes[38..58], b"application/epub+zip");
156    }
157
158    /// Every entry carries the same date, so two runs match.
159    #[test]
160    fn two_archives_of_the_same_entries_are_the_same_bytes() {
161        assert_eq!(archive(), archive());
162        let bytes = archive();
163        assert_eq!(u16::from_le_bytes([bytes[12], bytes[13]]), DOS_DATE);
164    }
165
166    /// A deflated entry inflates back to what went in.
167    #[test]
168    fn a_deflated_entry_inflates_to_its_data() {
169        let bytes = archive();
170        let second = 38 + 20;
171        assert_eq!(&bytes[second..second + 4], b"PK\x03\x04");
172        let compressed = u32::from_le_bytes(bytes[second + 18..second + 22].try_into().unwrap());
173        let data = second + 30 + "EPUB/a.txt".len();
174        let inflated =
175            miniz_oxide::inflate::decompress_to_vec(&bytes[data..data + compressed as usize])
176                .unwrap();
177        assert_eq!(inflated, "a line of prose\n".repeat(40).as_bytes());
178    }
179}
