13/// Flattened paragraph content: the text as it is shaped, the text
14/// the author wrote under it, and, per style span, the byte range it
15/// covers. Style boundaries are segmentation boundaries — a shaped
16/// run never spans two fonts.
17#[derive(Default)]
18pub(super) struct FlatParagraph {
19    /// What is shaped, measured and broken. `text-transform` and
20    /// synthesized small capitals have already been applied.
21    pub(super) text: String,
22    /// What the author wrote. Kept only from the point something
23    /// first transforms; until then `text` is the source.
24    source: String,
25    /// How much of `source` the shaped text up to each of its bytes
26    /// accounts for. Kept alongside `source`, and rising with it, so
27    /// the stretch between two boundaries is the source that the
28    /// text between them was made from.
29    map: Vec<u32>,
30    /// Whether anything has been written that differs from its
31    /// source.
32    transformed: bool,
33    /// Whether the next character opens a word: what `capitalize`
34    /// reads.
35    word_start: bool,
36    /// The style spans, in document order.
37    pub(super) spans: Vec<StyleSpan>,
38    /// The inline elements that paint a box, in the order they open.
39    pub(super) boxes: Vec<BoxSpan>,
40    /// The innermost inline element open, which the spans written
41    /// now belong to.
42    inline: Option<NodeId>,
43    /// Which node each stretch of the source was written in, in
44    /// document order.
45    origins: Vec<Origin>,
46    /// Bytes of the source still to be passed over before anything
47    /// is written: what a drop cap took out of the paragraph.
48    skip: usize,
49    /// The id of the paragraph's `::first-line`, and how far into the
50    /// shaped text its style reaches.
51    first_line: Option<(NodeId, usize)>,
52    /// Stretches of the shaped text that are set as they were
53    /// written, so no hyphen is put inside one. An inline code span
54    /// is one.
55    pub(super) literal: Vec<Range<usize>>,
56}

58/// One stretch of the paragraph's source, and the node it was
59/// written in. Stretches are contiguous, so one origin runs to where
60/// the next begins.
61struct Origin {
62    /// `None` on text the sheet generated, which no node holds.
63    node: Option<NodeId>,
64    /// Where the stretch starts in the paragraph's source.
65    start: u32,
66    /// Where it starts in the node's own text, which is past the
67    /// letter wherever a drop cap took one.
68    node_start: u32,
69}

83/// One span of uniform shaping: a face, a size, the tracking after
84/// each of its clusters, and the features it is shaped with.
85pub(super) struct StyleSpan {
86    pub(super) font_id: u16,
87    pub(super) size: f32,
88    /// Extra advance after each cluster, in points.
89    pub(super) tracking: f32,

91    pub(super) color: Color,

93    pub(super) decoration: TextDecoration,

101    /// Whether two spans are set the same way, the text they cover
102    /// aside. An inline element is part of that: a box is painted
103    /// around the runs of one element, so a run never spans two.
104    fn same_style(&self, other: &StyleSpan) -> bool {
105        self.font_id == other.font_id
106            && self.size == other.size
107            && self.tracking == other.tracking
108            && self.features == other.features
109            && self.color == other.color
110            && self.decoration == other.decoration
111            && self.inline == other.inline
112    }
113}

115/// How a span gets its small capitals.
116#[derive(Debug, Clone, Copy, PartialEq, Eq)]
117pub(super) enum SmallCaps {
118    /// It does not: the text is set in the letters it was written in.
119    Off,
120    /// The face draws them, and the shaper is asked for `smcp`.
121    Feature,
122    /// The face does not, so lowercase letters are set as capitals at
123    /// a reduced size.
124    Synthesized,
125}

137    /// What the author wrote under one stretch of the shaped text,
138    /// and how much of it the text up to each byte accounts for.
139    ///
140    /// Both are empty where the two stretches are the same word for
141    /// word, which is every run of a book nothing transformed and
142    /// the untouched runs of a line that has one.
143    pub(super) fn source_of(&self, range: Range<usize>) -> (String, Vec<u32>) {
144        if !self.transformed {
145            return (String::new(), Vec::new());
146        }
147        let at = |byte: usize| self.source_at(byte);
148        let (from, to) = (at(range.start), at(range.end).max(at(range.start)));
149        let source = &self.source[from..to];
150        if source == &self.text[range.clone()] {
151            return (String::new(), Vec::new());
152        }
153        let map = (range.start..=range.end)
154            .map(|byte| (at(byte).clamp(from, to) - from) as u32)
155            .collect();
156        (source.to_string(), map)
157    }
158
159    /// Where one byte of the shaped text falls in the source it was
160    /// made from. The two run together until something transforms,
161    /// and the map holds them together after that.
162    fn source_at(&self, byte: usize) -> usize {
163        if !self.transformed {
164            return byte.min(self.text.len());
165        }
166        match self.map.get(byte) {
167            Some(offset) => *offset as usize,
168            None => self.source.len(),
169        }
170    }
171
172    /// How much source the paragraph has been written from so far.
173    fn source_len(&self) -> usize {
174        if self.transformed {
175            self.source.len()
176        } else {
177            self.text.len()
178        }
179    }

181    /// Opens a stretch of the source written in `node`, starting
182    /// `node_start` bytes into that node's own text.
183    fn open(&mut self, node: Option<NodeId>, node_start: usize) {
184        self.origins.push(Origin {
185            node,
186            start: self.source_len() as u32,
187            node_start: node_start as u32,
188        });
189    }
190
191    /// The stretch of the source one byte of the shaped text was
192    /// written in: the node, where the stretch starts in that node's
193    /// own text, and the source it covers.
194    ///
195    /// `None` where no node was walked: page furniture, an ornament,
196    /// text the sheet generated.
197    fn origin_at(&self, byte: usize) -> Option<(NodeId, u32, Range<usize>)> {
198        let at = self.source_at(byte);
199        let index = self
200            .origins
201            .partition_point(|origin| origin.start as usize <= at)
202            .checked_sub(1)?;
203        let origin = &self.origins[index];
204        let end = self.origins[index + 1..]
205            .first()
206            .map_or(self.source_len(), |next| next.start as usize);
207        Some((origin.node?, origin.node_start, origin.start as usize..end))
208    }
209
210    /// The node one stretch of the shaped text was written in, and
211    /// the bytes of that node's own text it stands for, running from
212    /// `from` to wherever `to` reaches.
213    ///
214    /// `after` is the node the stretch before this one was written
215    /// in. A stretch opening a node opens at that node's first byte,
216    /// wherever the break before it fell, so the space a break
217    /// swallowed at a node boundary belongs to the node that holds
218    /// it. Both ends stop at the node.
219    pub(super) fn origin_of(
220        &self,
221        after: Option<NodeId>,
222        from: usize,
223        to: usize,
224    ) -> Option<SourceRange> {
225        let (node, node_start, stretch) = self.origin_at(from)?;
226        let start = if after == Some(node) {
227            self.source_at(from).clamp(stretch.start, stretch.end)
228        } else {
229            stretch.start
230        };
231        let end = self.source_at(to).clamp(start, stretch.end);
232        let at = |source: usize| node_start + (source - stretch.start) as u32;
233        Some(SourceRange {
234            node,
235            range: at(start)..at(end),
236        })
237    }

239    /// The pseudo-element a run that starts at `from` and stands for
240    /// `origin` was cut from: the `::before` or `::after` that
241    /// generated its text, or else the `::first-line` whose style
242    /// reaches it.
243    pub(super) fn pseudo_element_of(
244        &self,
245        origin: Option<&SourceRange>,
246        from: usize,
247    ) -> Option<NodeId> {
248        if let Some(origin) = origin.filter(|origin| origin.node.pseudo_element().is_some()) {
249            return Some(origin.node);
250        }
251        self.first_line
252            .filter(|(_, reach)| from < *reach)
253            .map(|(id, _)| id)
254    }

256    /// Starts keeping the source text, backfilling what has been
257    /// written so far, all of which stood as it was written.
258    fn start_mapping(&mut self) {
259        if self.transformed {
260            return;
261        }
262        self.transformed = true;
263        self.source.push_str(&self.text);
264        self.map.extend(0..self.text.len() as u32);
265    }

267    /// Appends a stretch of text nothing transformed.
268    fn push_verbatim(&mut self, value: &str) {
269        if self.transformed {
270            let at = self.source.len() as u32;
271            self.map
272                .extend((0..value.len() as u32).map(|byte| at + byte));
273            self.source.push_str(value);
274        }
275        self.text.push_str(value);
276    }

278    /// Appends one character shaped as something other than itself.
279    ///
280    /// The whole character is accounted for at the first byte it was
281    /// written as, so the first glyph of the pair `ß` shapes as
282    /// stands for the letter and the second stands for nothing.
283    /// Extraction walks the glyphs in order and reads the source back
284    /// once.
285    fn push_mapped(&mut self, letter: char, written: &str) {
286        self.start_mapping();
287        let at = self.source.len() as u32;
288        self.source.push(letter);
289        let after = self.source.len() as u32;
290        self.map.push(at);
291        self.map
292            .extend(std::iter::repeat_n(after, written.len() - 1));
293        self.text.push_str(written);
294    }

296    /// Appends one styled stretch of text: `text-transform` first,
297    /// then small capitals over what it produced, and the spans they
298    /// come to.
299    pub(super) fn push_styled(&mut self, value: &str, style: &ParagraphStyle, caps: SmallCaps) {
300        if style.transform == TextTransform::None && caps != SmallCaps::Synthesized {
301            let start = self.text.len();
302            self.push_verbatim(value);
303            if let Some(last) = value.chars().next_back() {
304                self.word_start = !continues_word(last);
305            }
306            self.span(style, caps, false, start);
307            return;
308        }
309        // A span breaks where the reduced size does, so the letters a
310        // synthesis raised are shaped apart from the ones it left.
311        let mut open: Option<(bool, usize)> = None;
312        let mut written = String::new();
313        for letter in value.chars() {
314            written.clear();
315            let mut changed = style.transform.write(letter, self.word_start, &mut written);
316            self.word_start = !continues_word(letter);
317            let small = raise(caps, &mut written, &mut changed);
318            if open.map(|(was, _)| was) != Some(small) {
319                if let Some((was, at)) = open {
320                    self.span(style, caps, was, at);
321                }
322                open = Some((small, self.text.len()));
323            }
324            if changed {
325                self.push_mapped(letter, &written);
326            } else {
327                self.push_verbatim(&written);
328            }
329        }
330        if let Some((was, at)) = open {
331            self.span(style, caps, was, at);
332        }
333    }

335    /// Records the span that ends where the text now does.
336    fn span(&mut self, style: &ParagraphStyle, caps: SmallCaps, small: bool, start: usize) {
337        if start >= self.text.len() {
338            return;
339        }
340        let span = StyleSpan {
341            font_id: style.font_id,
342            size: if small {
343                style.size * SMALL_CAPS_RATIO
344            } else {
345                style.size
346            },
347            tracking: style.letter_spacing,
348            features: Features::new(caps == SmallCaps::Feature, style.features.clone()),
349            color: style.color,
350            decoration: style.decoration,
351            inline: self.inline,
352            range: start..self.text.len(),
353        };
354        self.spans.push(span);
355    }

357    /// Opens the inline element `id`, with the box it paints where it
358    /// paints one, and answers what closing it puts back.
359    fn open_inline(&mut self, id: NodeId, box_: Option<InlineBox>) -> Open {
360        let was = self.inline;
361        self.inline = Some(id);
362        let at = box_.map(|box_| {
363            self.boxes.push(BoxSpan {
364                node: id,
365                box_,
366                range: self.text.len()..self.text.len(),
367            });
368            self.boxes.len() - 1
369        });
370        Open { was, at }
371    }
372
373    /// Closes it again, over the text written since.
374    fn close_inline(&mut self, open: Open) {
375        if let Some(at) = open.at {
376            self.boxes[at].range.end = self.text.len();
377        }
378        self.inline = open.was;
379    }

381    /// Joins two adjacent spans set the same way, from `from` to the
382    /// end. Text written a character at a time is one run of one
383    /// style as much as text written in a stretch, and a run shapes
384    /// as a whole: kerning and ligatures do not reach across a span.
385    fn merge_from(&mut self, from: usize) {
386        let mut at = from + 1;
387        while at < self.spans.len() {
388            let joins = self.spans[at - 1].range.end == self.spans[at].range.start
389                && self.spans[at - 1].same_style(&self.spans[at]);
390            if joins {
391                self.spans[at - 1].range.end = self.spans[at].range.end;
392                self.spans.remove(at);
393            } else {
394                at += 1;
395            }
396        }
397    }

400/// One open inline element: the one it was opened inside, and the box

407/// Raises one character to a capital where a synthesis has to draw a
408/// small one, and answers whether it is set at the reduced size. Only
409/// what was lowercase is: a face's own small capitals leave the word
410/// space and the comma the size they were.
411fn raise(caps: SmallCaps, written: &mut String, changed: &mut bool) -> bool {
412    if caps != SmallCaps::Synthesized || !written.chars().any(char::is_lowercase) {
413        return false;
414    }
415    *written = written.to_uppercase();
416    *changed = true;
417    true
418}

420/// Whether a character continues a word rather than ending it.
421/// `capitalize` raises the letter after every other kind, which is
422/// why `well-known` comes out with two capitals and `don't` with one.
423fn continues_word(letter: char) -> bool {
424    letter.is_alphanumeric() || letter == '\'' || letter == '\u{2019}'
425}

427/// What a synthesized small capital is set at, as a fraction of the
428/// size around it. A face's own small caps sit a little above the
429/// x-height, and four-fifths of the cap height is about where that
430/// lands.
431pub(super) const SMALL_CAPS_RATIO: f32 = 0.8;

433impl LineLayout<'_> {
434    /// Flattens a paragraph's inlines, each one styled as the tree
435    /// says. A style boundary is a span boundary, so a run never
436    /// spans two faces.
437    pub(super) fn flatten(
438        &self,
439        inlines: &[Inline],
440        style: &ParagraphStyle,
441        styles: &dyn InlineStyles,
442        lead: Lead,
443    ) -> FlatParagraph {
444        let mut flat = FlatParagraph::new();
445        flat.skip = lead.taken;
446        flat.first_line = lead.pseudo_element.map(|id| (id, lead.reach()));
447        self.walk_inlines(inlines, style, styles, lead, &mut flat);
448        flat
449    }

451    /// Flattens the text of the generated box `node`, as that box's
452    /// own text.
453    pub(super) fn flatten_generated(
454        &self,
455        text: &str,
456        node: NodeId,
457        style: &ParagraphStyle,
458    ) -> FlatParagraph {
459        let mut flat = FlatParagraph::new();
460        let lead = Lead::default();
461        let value = text.replace('\n', " ");
462        if !value.is_empty() {
463            flat.open(Some(node), 0);
464            self.push_run(&mut flat, &value, style, lead);
465        }
466        flat
467    }

469    /// Flattens preformatted text as the text of `node`: its newlines

471    pub(super) fn flatten_preformatted(
472        &self,
473        text: &str,
474        node: NodeId,
475        style: &ParagraphStyle,

481        }

485    fn walk_inlines(
486        &self,
487        inlines: &[Inline],
488        style: &ParagraphStyle,
489        styles: &dyn InlineStyles,
490        lead: Lead,
491        flat: &mut FlatParagraph,
492    ) {
493        for inline in inlines {
494            match inline {
495                Inline::Text { id, value, .. } => self.push_text(flat, *id, value, style, lead),
496                Inline::Break { .. } => self.push_break(flat, style, lead),
497                Inline::Code { id, value, .. } => {
498                    let generated = styles.generated(inline);
499                    let open = flat.open_inline(*id, styles.inline_box(*id));
500                    self.push_generated(flat, generated.before, *id, PseudoElement::Before, lead);
501                    let from = flat.text.len();
502                    self.push_text(flat, *id, value, &styles.style(*id, style), lead);
503                    flat.literal.push(from..flat.text.len());
504                    self.push_generated(flat, generated.after, *id, PseudoElement::After, lead);
505                    flat.close_inline(open);
506                }
507                // The note itself is set at the foot of the page.
508                // What stands in the line is its reference. The box
509                // the note paints is around the note there, not
510                // around the reference here, so the reference takes
511                // none of it.
512                Inline::Note { id, .. } => {
513                    let Some((call, style)) = styles.call(inline) else {
514                        continue;
515                    };
516                    let open = flat.open_inline(*id, None);
517                    flat.open(None, 0);
518                    self.push_run(flat, &call, &style, lead);
519                    flat.close_inline(open);
520                }
521                Inline::Emphasis { id, children, .. }
522                | Inline::Strong { id, children, .. }
523                | Inline::Link { id, children, .. }
524                | Inline::Strikethrough { id, children, .. }
525                | Inline::Highlight { id, children, .. }
526                | Inline::Span { id, children, .. } => {
527                    let generated = styles.generated(inline);
528                    let open = flat.open_inline(*id, styles.inline_box(*id));
529                    self.push_generated(flat, generated.before, *id, PseudoElement::Before, lead);
530                    self.walk_inlines(children, &styles.style(*id, style), styles, lead, flat);
531                    self.push_generated(flat, generated.after, *id, PseudoElement::After, lead);
532                    flat.close_inline(open);
533                }
534            }
535        }
536    }

538    /// Appends text the sheet generated as `which` of `element`. No
539    /// content node holds it, so the runs it lands in name the
540    /// pseudo-element, and a drop cap passes over none of it.
541    fn push_generated(
542        &self,
543        flat: &mut FlatParagraph,
544        generated: Option<(String, ParagraphStyle)>,
545        element: NodeId,
546        which: PseudoElement,
547        lead: Lead,
548    ) {
549        let Some((mut value, style)) = generated else {
550            return;
551        };
552        if value.is_empty() {
553            return;
554        }
555        // Generated text holds no hard break: the newline a heading's
556        // break reads as is a word space here.
557        if value.contains('\n') {
558            value = value.replace('\n', " ");
559        }
560        flat.open(Some(element.pseudo(which)), 0);
561        self.push_run(flat, &value, &style, lead);
562    }

564    /// Appends a hard break as the newline the breaker ends a line
565    /// at. No node's text holds the newline, and a drop cap passes
566    /// over it as it does any other byte.
567    fn push_break(&self, flat: &mut FlatParagraph, style: &ParagraphStyle, lead: Lead) {
568        if flat.skip > 0 {
569            flat.skip -= 1;
570            return;
571        }
572        flat.open(None, 0);
573        self.push_run(flat, "\n", style, lead);
574    }

576    /// Appends one node's text, in the opening style as far as it
577    /// reaches and in `style` after that. What a drop cap took is
578    /// passed over here, so the paragraph starts at the letter after
579    /// the one the cap holds.
580    ///
581    /// The switch to `style` is on the length of the shaped text,
582    /// which is what the extent a break gave is written in. Where the
583    /// opening style still applies the text goes in a character at a
584    /// time, so the switch can fall inside a word; the spans it
585    /// writes merge back into one, and the whole of the opening line
586    /// shapes together.
587    fn push_text(
588        &self,
589        flat: &mut FlatParagraph,
590        node: NodeId,
591        value: &str,
592        style: &ParagraphStyle,
593        lead: Lead,
594    ) {
595        let mut at = flat.skip.min(value.len());
596        while !value.is_char_boundary(at) {
597            at += 1;
598        }
599        flat.skip = flat.skip.saturating_sub(at);
600        let (node_start, value) = (at, &value[at..]);
601        if value.is_empty() {
602            return;
603        }
604        flat.open(Some(node), node_start);
605        self.push_run(flat, value, style, lead);
606    }

608    /// Appends one stretch of text, in the opening style as far as
609    /// it reaches and in `style` after that.
610    fn push_run(&self, flat: &mut FlatParagraph, value: &str, style: &ParagraphStyle, lead: Lead) {
611        let (opening, extent) = match lead.style {
612            Some(first) if flat.text.len() < lead.reach() => {
613                (first.over(style, self.registry), lead.reach())
614            }
615            _ => {
616                flat.push_styled(value, style, self.small_caps(style));
617                return;
618            }
619        };
620        let (opening_caps, caps) = (self.small_caps(&opening), self.small_caps(style));
621        let from = flat.spans.len();
622        for (at, letter) in value.char_indices() {
623            if flat.text.len() >= extent {
624                flat.push_styled(&value[at..], style, caps);
625                break;
626            }
627            flat.push_styled(&value[at..at + letter.len_utf8()], &opening, opening_caps);
628        }
629        flat.merge_from(from);
630    }

632    /// Where a style's small capitals come from: the face's own where
633    /// it has them, and a synthesis where it does not.
634    pub(super) fn small_caps(&self, style: &ParagraphStyle) -> SmallCaps {
635        match style.caps {
636            FontVariantCaps::Normal => SmallCaps::Off,
637            FontVariantCaps::SmallCaps if self.registry.has_small_caps(style.font_id) => {
638                SmallCaps::Feature
639            }
640            FontVariantCaps::SmallCaps => SmallCaps::Synthesized,
641        }
642    }
643}

708    /// `text-transform` changes what is shaped, and the run says so
709    /// twice over: `text` is what was drawn, which is what a painter
710    /// that draws characters draws, and `source` is what the author
711    /// wrote, which is what extraction reads back.
712    #[test]
713    fn text_transform_shapes_one_text_and_reports_another() {
714        let style = ParagraphStyle {
715            transform: TextTransform::Uppercase,
716            ..body()
717        };
718        let lines = layout_style("Lilliput", 400.0, &style);
719        assert_eq!(
720            line_text(&lines[0]),
721            "LILLIPUT",
722            "the run was not drawn in capitals"
723        );
724        assert_eq!(
725            lines[0].runs[0].source, "Lilliput",
726            "the run lost the source"
727        );
728        let shouted = layout_style("LILLIPUT", 400.0, &body());
729        assert_eq!(
730            lines[0].runs[0]
731                .glyphs
732                .iter()
733                .map(|g| g.id)
734                .collect::<Vec<_>>(),
735            shouted[0].runs[0]
736                .glyphs
737                .iter()
738                .map(|g| g.id)
739                .collect::<Vec<_>>(),
740            "the transform did not reach the shaper",
741        );
742
743        // A mapping that is not one for one: `ß` shapes as two
744        // capitals, and both of them stand for the one letter.
745        let lines = layout_style("Straße", 400.0, &style);
746        let run = &lines[0].runs[0];
747        assert_eq!(
748            (run.text.as_str(), run.source.as_str()),
749            ("STRASSE", "Straße")
750        );
751        assert_eq!(run.glyphs.len(), 7, "STRASSE is seven glyphs");
752        let sharp = run.source.find('ß').expect("the source keeps its ß") as u32;
753        let through = |range: Range<u32>| {
754            run.source_map[range.start as usize]..run.source_map[range.end as usize]
755        };
756        let ranges = run.glyph_ranges();
757        assert_eq!(
758            (through(ranges[4].clone()), through(ranges[5].clone())),
759            (sharp..sharp + 2, sharp + 2..sharp + 2),
760            "the pair of capitals does not stand for the ß once",
761        );
762        assert_eq!(
763            run.source_map.last().copied(),
764            Some(run.source.len() as u32),
765            "the map does not run to the end of the source",
766        );
767    }
