12impl Paginator<'_> {
13    /// The initial letter one paragraph opens with, and the inlines
14    /// left after it is taken out.
15    ///
16    /// The sink is what `initial-letter` asked for: the cap's own cap
17    /// height spans that many lines, so its baseline lands on the
18    /// last of them and its top on the first line's cap height.
19    pub(super) fn drop_cap(
20        &self,
21        id: NodeId,
22        computed: &ComputedStyle,
23        inlines: &[Inline],
24    ) -> Option<(Cap, usize)> {
25        let cap_style = self.styles.first_letter(id)?;
26        let sink = cap_style.initial_letter as usize;
27        if sink < 2 {
28            return None;
29        }
30        let initial = take_initial(inlines)?;
31        let body = computed.paragraph();
32        let cap_metrics = self.registry.metrics(cap_style.font_id)?;
33        let cap_units = cap_height(cap_metrics);
34        if cap_units <= 0.0 {
35            return None;
36        }
37        let body_metrics = self.registry.metrics(body.font_id)?;
38        let body_cap = cap_height(body_metrics) / body_metrics.units_per_em as f32 * body.size;
39        let sunk = (sink - 1) as f32 * self.lines.strut(&body).height() + body_cap;
40        let style = ParagraphStyle {
41            size: sunk * cap_metrics.units_per_em as f32 / cap_units,
42            ..cap_style.paragraph()
43        };
44        let mut line = self.line_of(&initial.text, &style)?;
45        // The cap stands for the letter it was taken from, so a
46        // cursor on the manuscript's first word lands on it.
47        if let Some(run) = line.runs.first_mut() {
48            run.origin = Some(initial.origin);
49            run.pseudo_element = Some(id.pseudo(PseudoElement::FirstLetter));
50        }
51        // A word space of the body text separates the cap from the
52        // lines it is sunk into.
53        let gutter = self
54            .registry
55            .char_glyph(body.font_id, ' ')
56            .and_then(|glyph| self.registry.advance_width(body.font_id, glyph))
57            .unwrap_or(0) as f32
58            / body_metrics.units_per_em as f32
59            * body.size;
60        Some((
61            Cap {
62                reserved: self.line_width(&line) + gutter,
63                line,
64                lines: sink,
65            },
66            initial.taken,
67        ))
68    }
69}

71/// The initial letter of one paragraph, sized and shaped, with the
72/// text it was taken out of.
73#[derive(Debug, Clone)]
74pub(super) struct Cap {
75    /// The letter, shaped at the size the sink works out to.
76    pub(super) line: Line,
77    /// Width the lines beside it give up, the gutter included.
78    pub(super) reserved: f32,
79    /// Lines it is sunk over.
80    pub(super) lines: usize,
81}
82
83/// The text a drop cap is set from, where it was written, and how far
84/// into the paragraph the rest of the prose starts.
85struct Initial {
86    /// The first letter with the punctuation on either side of it.
87    text: String,
88    /// The node the cap starts in, and the bytes of it the cap holds:
89    /// its text and whatever space stood before it.
90    origin: SourceRange,
91    /// Bytes of the paragraph's text the cap holds.
92    taken: usize,
93}
94
95/// The first letter of a run of inlines and the punctuation next to
96/// it, wherever the markup has put them: a paragraph opening in
97/// italic still opens with a letter.
98fn take_initial(inlines: &[Inline]) -> Option<Initial> {
99    fn texts<'a>(inlines: &'a [Inline], out: &mut Vec<(Option<NodeId>, &'a str)>) {
100        for inline in inlines {
101            match inline {
102                Inline::Text { id, value, .. } | Inline::Code { id, value, .. } => {
103                    out.push((Some(*id), value))
104                }
105                Inline::Break { .. } => out.push((None, "\n")),
106                // The note is set at the foot of the page, so an
107                // initial letter never comes out of one.
108                Inline::Note { .. } => {}
109                Inline::Emphasis { children, .. }
110                | Inline::Strong { children, .. }
111                | Inline::Link { children, .. }
112                | Inline::Strikethrough { children, .. }
113                | Inline::Highlight { children, .. }
114                | Inline::Span { children, .. } => texts(children, out),
115            }
116        }
117    }
118    let mut pieces = Vec::new();
119    texts(inlines, &mut pieces);
120
121    let mut text = String::new();
122    let mut origin: Option<SourceRange> = None;
123    let mut lettered = false;
124    let mut taken = 0;
125    let mut before = 0;
126    'pieces: for (node, value) in pieces {
127        for (at, character) in value.char_indices() {
128            if text.is_empty() && character.is_whitespace() {
129                continue;
130            }
131            let joins = match Kind::of(character) {
132                Kind::Punctuation => true,
133                Kind::Letter => !std::mem::replace(&mut lettered, true),
134                Kind::Other => false,
135            };
136            let Some(node) = node.filter(|_| joins) else {
137                break 'pieces;
138            };
139            let end = (at + character.len_utf8()) as u32;
140            let origin = origin.get_or_insert(SourceRange { node, range: 0..0 });
141            if origin.node == node {
142                origin.range.end = end;
143            }
144            text.push(character);
145            taken = before + end as usize;
146        }
147        before += value.len();
148    }
149    Some(Initial {
150        text,
151        origin: origin.filter(|_| lettered)?,
152        taken,
153    })
154}

156/// What a character is to `::first-letter`.
157enum Kind {
158    /// Opening, closing, quotation, and other punctuation. Dashes and
159    /// connectors are not in it.
160    Punctuation,
161    /// A letter, a number, or a symbol.
162    Letter,
163    Other,
164}
165
166impl Kind {
167    fn of(character: char) -> Kind {
168        match get_general_category(character).abbreviation().as_bytes() {
169            [b'P', b's' | b'e' | b'i' | b'f' | b'o'] => Kind::Punctuation,
170            [b'L' | b'N' | b'S', _] => Kind::Letter,
171            _ => Kind::Other,
172        }
173    }
174}

176/// A face's cap height, falling back to its ascender when the file
177/// declares none.
178fn cap_height(metrics: crate::fonts::FontMetricsTable) -> f32 {
179    if metrics.cap_height > 0 {
180        metrics.cap_height as f32
181    } else {
182        metrics.ascender as f32
183    }
184}

208    /// Acceptance: punctuation before the first letter joins the drop
209    /// cap, and the first line opens after the letter.
210    #[test]
211    fn punctuation_before_the_first_letter_joins_the_drop_cap() {
212        let prose = format!(
213            "\u{201C}Sir,\u{201D} said he, {}",
214            "my father had a small estate in nottinghamshire ".repeat(12)
215        );
216        let pages = paginate_styled(
217            "p::first-letter { initial-letter: 3 }",
218            vec![section(vec![paragraph(&prose)])],
219        );
220        let lines = content_lines(&pages[0]);
221        let cap: Vec<&str> = lines
222            .iter()
223            .flat_map(|(_, runs)| runs)
224            .filter(|(_, size, _)| *size > body_size())
225            .map(|(_, _, text)| *text)
226            .collect();
227        assert_eq!(cap, ["\u{201C}S"]);
228        assert!(
229            lines[0].1[0].2.starts_with("ir,"),
230            "the first line does not open after the letter: {:?}",

279    fn a_drop_cap_absorbs_the_indent() {

347    /// Acceptance: a three-line drop cap sits on the third baseline,
348    /// its top on the first line's cap height, and the lines beside
349    /// it are set to the measure it left them.
350    #[test]
351    fn a_drop_cap_aligns_to_the_third_baseline_and_shortens_three_lines() {
352        let pages = paginate_styled(
353            "p::first-letter { initial-letter: 3 }",
354            vec![section(vec![paragraph(
355                &"my father had a small estate in nottinghamshire ".repeat(12),
356            )])],
357        );
358        let lines = content_lines(&pages[0]);
359        assert!(lines.len() > 4, "not enough lines to sink into");
360
361        let body = ua().root();
362        let (left, _) = origin_of(&pages[0]);
363        // The cap is the one run set larger than the body.
364        let (cap_index, cap) = lines
365            .iter()
366            .enumerate()
367            .find(|(_, (_, runs))| runs[0].1 > body.font_size)
368            .map(|(index, (baseline, runs))| (index, (*baseline, runs[0])))
369            .expect("a drop cap paints");
370        let (cap_baseline, (cap_x, cap_size, _)) = cap;
371
372        // Its baseline is the third line's, and it starts at the
373        // content box's own leading edge.
374        let prose: Vec<&ContentLine<'_>> = lines
375            .iter()
376            .enumerate()
377            .filter(|(index, _)| *index != cap_index)
378            .map(|(_, line)| line)
379            .collect();
380        assert!(
381            (cap_baseline - prose[2].0).abs() < 1e-3,
382            "the cap is not on the third baseline"
383        );
384        assert!(
385            (cap_x - left).abs() < 1e-3,
386            "the cap is not at the leading edge"
387        );
388
389        // Its top sits on the first line's cap height.
390        let cap_height = |font: u16, size: f32| {
391            let metrics = registry().metrics(font).unwrap();
392            metrics.cap_height as f32 / metrics.units_per_em as f32 * size
393        };
394        let top = cap_baseline - cap_height(body.font_id, cap_size);
395        assert!(
396            (top - (prose[0].0 - cap_height(body.font_id, body.font_size))).abs() < 1e-2,
397            "the cap's top is not the first line's cap height",
398        );
399
400        // The three lines beside it start past the cap and are set to
401        // the measure it left them; the fourth is back at the full one.
402        let sunk = prose[0].1[0].0;
403        assert!(sunk > left, "the first line was not moved aside");
404        for line in prose.iter().take(3) {
405            assert!(
406                (line.1[0].0 - sunk).abs() < 1e-3,
407                "a sunk line is not set to the shortened measure",
408            );
409        }
410        assert!(
411            (prose[3].1[0].0 - left).abs() < 1e-3,
412            "the fourth line did not go back to the full measure",
413        );
414        for page in &pages {
415            for item in &page.items {
416                let DrawItem::Text { glyphs, .. } = item else {
417                    continue;
418                };
419                for glyph in glyphs {
420                    assert!(
421                        glyph.x <= left + master(Situation::Body(page.side)).geometry.measure(),
422                        "a sunk line ran past the measure",
423                    );
424                }
425            }
426        }
427    }
