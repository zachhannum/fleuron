1//! The style tree: resolved styling.
2//!
3//! Styling enters as CSS. A built-in user-agent stylesheet supplies
4//! the defaults; author CSS cascades over it. What comes out is one
5//! computed style per content node and one page master per situation
6//! a page can be in, and that is the whole of what layout is told.
7//!
8//! ```text
9//! content tree + CSS ─► style tree ─► box tree ─► …
10//! ```
11//!
12//! Font faces reach the registry here too: `@font-face` names a
13//! source, the host loader hands back bytes, and every node's face is
14//! an id by the time layout sees it.

111/// The situation a page finds itself in, which is what `@page`
112/// selects on beyond the page's name.
113#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize)]
114#[serde(rename_all = "snake_case")]
115pub enum Situation {
116    /// The page a page group opens on: `@page :first`.
117    First(Side),
118    /// Any later page of the group.
119    Body(Side),
120    /// A page inserted to square the sheet: `@page :blank`.
121    Blank,
122}
123
124impl Situation {
125    /// The side of the spread this situation falls on. Blanks are
126    /// versos: a book never opens a leaf to a blank right-hand page.
127    pub fn side(self) -> Side {
128        match self {
129            Situation::First(side) | Situation::Body(side) => side,
130            Situation::Blank => Side::Verso,
131        }
132    }
133}

204    #[serde(skip)]

206    /// `::first-line` style index by raw node id, where there is one.
207    #[serde(skip)]
208    first_line_by_node: Vec<Option<u32>>,
209    /// `::before` style index by raw node id, where it generates text.
210    #[serde(skip)]
211    before_by_node: Vec<Option<u32>>,
212    /// `::after` style index by raw node id, where it generates text.
213    #[serde(skip)]
214    after_by_node: Vec<Option<u32>>,

216    #[serde(skip)]

228    /// The computed style of one content node, or of one
229    /// pseudo-element. An id the tree does not know — an unassigned
230    /// one, or a node from another book — gets the root style, which
231    /// is the book's own defaults.
232    pub fn style(&self, id: NodeId) -> &ComputedStyle {
233        if let Some((element, which)) = id.pseudo_element() {
234            return self
235                .pseudo_style(element, which)
236                .unwrap_or_else(|| self.style(NodeId::UNASSIGNED));
237        }
238        let index = self
239            .by_node
240            .get(id.get() as usize)
241            .copied()
242            .unwrap_or_default();
243        &self.styles[index as usize]
244    }

376/// Resolves every master the book can use: the unnamed default, plus
377/// each named page, in each situation a page can be in.
378fn resolve_masters(
379    pages: &[(u8, PageRule)],
380    styles: &[ComputedStyle],
381    root: &ComputedStyle,
382    registry: &FontRegistry,
383    warnings: &mut Vec<Warning>,
384) -> Vec<PageMaster> {
385    let mut names: Vec<Option<String>> = vec![None];
386    for style in styles {
387        if let Some(name) = &style.page
388            && !names.iter().any(|known| known.as_deref() == Some(name))
389        {
390            names.push(Some(name.clone()));
391        }
392    }
393    let situations = [
394        Situation::First(Side::Recto),
395        Situation::First(Side::Verso),
396        Situation::Body(Side::Recto),
397        Situation::Body(Side::Verso),
398        Situation::Blank,
399    ];
400    let mut masters = Vec::new();
401    for name in names {
402        for situation in situations {
403            masters.push(PageMaster {
404                page: name.clone(),
405                situation,
406                style: resolve_page(
407                    pages,
408                    PageQuery {
409                        name: name.as_deref(),
410                        situation,
411                    },
412                    root,
413                    registry,
414                    warnings,
415                ),
416            });
417        }
418    }
419    masters
420}

422/// One page master: every `@page` rule that selects it, applied in
423/// specificity then source order.
424fn resolve_page(
425    pages: &[(u8, PageRule)],
426    query: PageQuery<'_>,
427    root: &ComputedStyle,
428    registry: &FontRegistry,
429    warnings: &mut Vec<Warning>,
430) -> PageStyle {
431    let mut matching: Vec<&(u8, PageRule)> = pages
432        .iter()
433        .filter(|(_, rule)| selects(rule, query))
434        .collect();
435    // `@page` cascades like any other rule: origin first, then
436    // specificity. The rules are already in source order, and a stable
437    // sort leaves the later of two equal rules on top.
438    matching.sort_by_key(|(level, rule)| (*level, rule.specificity()));
439
440    let root_size = root.font_size;
441    // `column-gap: normal` is one em of the book's own size, and the
442    // page box is not text, so the em it means is the root's.
443    let mut geometry = PageGeometry {
444        width: 612.0,
445        height: 792.0,
446        margin: Edges::all(0.0),
447        columns: Columns::undivided(root_size),
448        align_content: AlignContent::Start,
449        bleed: None,
450        marks: Marks::NONE,
451    };
452    let initial = geometry;
453    let mut background = Background::NONE;
454    let mut boxes: BTreeMap<MarginBox, MarginBoxStyle> = BTreeMap::new();
455    for (_, rule) in matching {
456        for declaration in &rule.declarations {
457            let PageDeclaration::Pending(pending) = declaration else {
458                apply_page(&mut geometry, &mut background, declaration, root_size);
459                continue;
460            };
461            let read = sheet::read_pending_page;
462            match substitution::substituted(pending, &root.custom, read, warnings) {
463                Some(declarations) => {
464                    for declaration in &declarations {
465                        apply_page(&mut geometry, &mut background, declaration, root_size);
466                    }
467                }
468                None => {
469                    for like in sheet::page_longhands(&pending.property) {
470                        reset_page(&mut geometry, &mut background, &like, &initial);
471                    }
472                }
473            }
474        }
475        for margin in &rule.boxes {
476            let entry = boxes.entry(margin.which).or_insert_with(|| MarginBoxStyle {
477                which: margin.which,
478                content: Content::None,
479                style: root.inherit(),
480            });
481            for declaration in &margin.declarations {
482                let MarginDeclaration::Pending(pending) = declaration else {
483                    apply_margin(entry, declaration, root_size);
484                    continue;
485                };
486                let read = sheet::read_pending_margin;
487                match substitution::substituted(pending, &root.custom, read, warnings) {
488                    Some(declarations) => {
489                        for declaration in &declarations {
490                            apply_margin(entry, declaration, root_size);
491                        }
492                    }
493                    None => {
494                        let base = root.inherit();
495                        for like in sheet::margin_longhands(&pending.property) {
496                            match like {
497                                MarginDeclaration::Style(like) => entry.style.reset(&like, &base),
498                                _ => entry.content = Content::None,
499                            }
500                        }
501                    }
502                }
503            }
504            let (font_id, warning) = resolve_face(&entry.style, registry);
505            entry.style.font_id = font_id;
506            report(warnings, warning);
507        }
508    }
509    PageStyle {
510        geometry,
511        background,
512        boxes: boxes.into_values().collect(),
513    }
514}

516/// Applies one declaration of a `@page` body.
517fn apply_page(
518    geometry: &mut PageGeometry,
519    background: &mut Background,
520    declaration: &PageDeclaration,
521    root_size: f32,
522) {
523    match declaration {
524        PageDeclaration::Size(width, height) => {
525            geometry.width = *width;
526            geometry.height = *height;
527        }
528        PageDeclaration::Margin(edge, length) => {
529            *geometry.margin.edge(*edge) = length.to_points(root_size, root_size)
530        }
531        PageDeclaration::BackgroundColor(color) => background.color = *color,
532        PageDeclaration::BackgroundImage(url) => background.image = url.clone(),
533        PageDeclaration::BackgroundRepeat(repeat) => background.repeat = *repeat,
534        PageDeclaration::BackgroundSize(size) => {
535            background.size = properties::computed_size(*size, root_size, root_size)
536        }
537        PageDeclaration::BackgroundPosition(x, y) => {
538            background.position = BackgroundPosition {
539                x: Coord::of(*x, root_size, root_size),
540                y: Coord::of(*y, root_size, root_size),
541            }
542        }
543        PageDeclaration::ColumnCount(count) => geometry.columns.count = *count,
544        PageDeclaration::ColumnWidth(width) => {
545            geometry.columns.width = width.map(|width| width.to_points(root_size, root_size))
546        }
547        PageDeclaration::ColumnGap(gap) => {
548            geometry.columns.gap = gap
549                .map(|gap| gap.to_points(root_size, root_size))
550                .unwrap_or(root_size)
551                .max(0.0)
552        }
553        PageDeclaration::ColumnRuleWidth(width) => {
554            geometry.columns.rule.width = width.to_points(root_size, root_size)
555        }
556        PageDeclaration::ColumnRuleStyle(style) => geometry.columns.rule.style = *style,
557        PageDeclaration::AlignContent(align) => geometry.align_content = *align,
558        PageDeclaration::Bleed(bleed) => {
559            geometry.bleed = bleed.map(|bleed| bleed.to_points(root_size, root_size))
560        }
561        PageDeclaration::Marks(marks) => geometry.marks = *marks,
562        PageDeclaration::Pending(_) => {}
563    }
564}

566/// Sets what `like` declares on a page back to its initial value.
567fn reset_page(
568    geometry: &mut PageGeometry,
569    background: &mut Background,
570    like: &PageDeclaration,
571    initial: &PageGeometry,
572) {
573    match like {
574        PageDeclaration::Size(..) => {
575            geometry.width = initial.width;
576            geometry.height = initial.height;
577        }
578        PageDeclaration::Margin(edge, _) => {
579            *geometry.margin.edge(*edge) = initial.margin.get(*edge)
580        }
581        PageDeclaration::BackgroundColor(_) => background.color = Background::NONE.color,
582        PageDeclaration::BackgroundImage(_) => background.image = None,
583        PageDeclaration::BackgroundRepeat(_) => background.repeat = Background::NONE.repeat,
584        PageDeclaration::BackgroundSize(_) => background.size = Background::NONE.size,
585        PageDeclaration::BackgroundPosition(..) => background.position = Background::NONE.position,
586        PageDeclaration::ColumnCount(_) => geometry.columns.count = initial.columns.count,
587        PageDeclaration::ColumnWidth(_) => geometry.columns.width = initial.columns.width,
588        PageDeclaration::ColumnGap(_) => geometry.columns.gap = initial.columns.gap,
589        PageDeclaration::ColumnRuleWidth(_) => {
590            geometry.columns.rule.width = initial.columns.rule.width
591        }
592        PageDeclaration::ColumnRuleStyle(_) => {
593            geometry.columns.rule.style = initial.columns.rule.style
594        }
595        PageDeclaration::AlignContent(_) => geometry.align_content = initial.align_content,
596        PageDeclaration::Bleed(_) => geometry.bleed = initial.bleed,
597        PageDeclaration::Marks(_) => geometry.marks = initial.marks,
598        PageDeclaration::Pending(_) => {}
599    }
600}

602/// Applies one declaration of a page margin box.
603fn apply_margin(entry: &mut MarginBoxStyle, declaration: &MarginDeclaration, root_size: f32) {
604    match declaration {
605        MarginDeclaration::Content(content) => entry.content = content.clone(),
606        MarginDeclaration::Style(style) => entry.style.apply(style, root_size, root_size),
607        MarginDeclaration::Pending(_) => {}
608    }

625/// Whether one `@page` rule selects the page a query describes.
626fn selects(rule: &PageRule, query: PageQuery<'_>) -> bool {
627    if rule.name.is_some() && rule.name.as_deref() != query.name {
628        return false;
629    }
630    if rule.first && !matches!(query.situation, Situation::First(_)) {
631        return false;
632    }
633    if rule.blank && query.situation != Situation::Blank {
634        return false;
635    }
636    match rule.side {
637        Some(side) => side == query.situation.side(),
638        None => true,
639    }
640}

642/// Parsed stylesheets: the built-in sheet, then whatever the host
643/// added, ready to compile against a book.
644///
645/// Parsing is separate from compiling because a run styles several
646/// books from one set of sheets, and separate from font loading
647/// because loading is the one step that reaches outside the engine.
648#[derive(Debug)]
649pub struct Stylesheets {
650    sheets: Vec<Sheet>,
651    /// What parsing said, then what font loading said.
652    warnings: Vec<Warning>,
653    /// How many of `warnings` parsing raised.
654    parsed: usize,
655}
656
657impl Stylesheets {
658    /// Parses the built-in sheet, then `sources` in the order given.
659    pub fn parse(sources: &[Source<'_>]) -> Stylesheets {
660        let built_in = Source::user_agent("user-agent.css", USER_AGENT_CSS);
661        let mut sheets = Vec::new();
662        let mut warnings = Vec::new();
663        for source in std::iter::once(&built_in).chain(sources) {
664            let (sheet, mut sheet_warnings) = sheet::parse(source);
665            warnings.append(&mut sheet_warnings);
666            sheets.push(sheet);
667        }
668        let parsed = warnings.len();
669        Stylesheets {
670            sheets,
671            warnings,
672            parsed,
673        }
674    }
675
676    /// Registers every `@font-face` the loader resolves, under the
677    /// family the sheet gave it rather than the one inside the file:
678    /// a stylesheet's name for a face is the one its selectors use.
679    ///
680    /// Run this before compiling — a face the registry does not have
681    /// is a face no computed style can resolve to. Running it again
682    /// replaces what the last run warned about.
683    pub fn load_fonts(&mut self, registry: &mut FontRegistry, loader: &dyn FontLoader) {
684        let warnings = self
685            .sheets
686            .iter()
687            .flat_map(|sheet| &sheet.faces)
688            .flat_map(|face| register_face(face, registry, loader))
689            .collect();
690        self.set_font_warnings(warnings);
691    }
692
693    /// What the last font loading warned about.
694    pub(crate) fn font_warnings(&self) -> &[Warning] {
695        &self.warnings[self.parsed..]
696    }
697
698    /// Replaces what font loading warned about, for sheets whose faces
699    /// a session already registered from an earlier copy of them.
700    pub(crate) fn set_font_warnings(&mut self, warnings: Vec<Warning>) {
701        self.warnings.truncate(self.parsed);
702        self.warnings.extend(warnings);
703    }
704
705    /// Hashes every `@font-face` the sheets declare, and what `file`
706    /// says about the bytes behind each url: two sheets that hash the
707    /// same register the same faces.
708    pub(crate) fn hash_faces(
709        &self,
710        h: &mut impl std::hash::Hasher,
711        file: impl Fn(&str) -> Option<u64>,
712    ) {
713        use std::hash::Hash;
714        for face in self.sheets.iter().flat_map(|sheet| &sheet.faces) {
715            face.family.hash(h);
716            face.style.hash(h);
717            face.weight.hash(h);
718            for src in &face.src {
719                match src {
720                    Src::Url(url) => (url, file(url)).hash(h),
721                    Src::Local(name) => name.hash(h),
722                }
723            }
724        }
725    }
726
727    /// What parsing and font loading had to complain about.
728    pub fn warnings(&self) -> &[Warning] {
729        &self.warnings
730    }
731
732    /// Every sheet as it was written, the built-in sheet first: what
733    /// a writer reads that hands the CSS to a renderer of its own
734    /// rather than laying it out.
735    ///
736    /// Only what the parser accepted is here. A rule or a declaration
737    /// outside the subset is in [`Stylesheets::warnings`] instead.
738    pub fn written(&self) -> Vec<WrittenSheet> {
739        self.sheets
740            .iter()
741            .map(|sheet| WrittenSheet {
742                origin: sheet.origin,
743                rules: sheet
744                    .rules
745                    .iter()
746                    .map(|rule| WrittenRule {
747                        selectors: rule
748                            .selectors
749                            .slice()
750                            .iter()
751                            .map(|selector| selector.to_css_string())
752                            .collect(),
753                        declarations: rule.written.clone(),
754                        position: rule.position.clone(),
755                    })
756                    .collect(),
757                pages: sheet
758                    .pages
759                    .iter()
760                    .map(|page| WrittenPage {
761                        selector: page.selector.clone(),
762                        position: page.position.clone(),
763                    })
764                    .collect(),
765                faces: sheet.faces.clone(),
766            })
767            .collect()
768    }
769
770    /// Compiles one book's styling against the faces in the registry.
771    pub fn compile(&self, book: &Book, registry: &FontRegistry) -> StyleTree {
772        cascade(book, &self.sheets, registry, self.warnings.clone())
773    }
774}

815/// Registers one `@font-face`, or says why it could not be.
816fn register_face(
817    face: &FontFace,
818    registry: &mut FontRegistry,
819    loader: &dyn FontLoader,
820) -> Vec<Warning> {
821    let mut warnings = Vec::new();
822    for src in &face.src {
823        let Src::Url(url) = src else {
824            continue;
825        };
826        let Some(bytes) = loader.load(url) else {
827            continue;
828        };
829        match FontSource::from_bytes(bytes) {
830            Ok(mut source) => {
831                source.family = face.family.to_lowercase();
832                source.declared = declared(face);
833                return match registry.add(source) {
834                    Ok(_) => warnings,
835                    Err(error) => {
836                        warnings.push(Warning {
837                            message: face_failure(&face.family, &error),
838                            origin: Some(url.clone()),
839                        });
840                        warnings
841                    }
842                };
843            }
844            Err(error) => warnings.push(Warning {
845                message: face_failure(&face.family, &error),
846                origin: Some(url.clone()),
847            }),
848        }
849    }
850    warnings.push(Warning {
851        message: format!("No src url loaded for {}. {FALLBACK}", face.family),
852        origin: None,
853    });
854    warnings
855}
856
857/// What happens to text that asked for a face the registry has not
858/// got.
859const FALLBACK: &str = "Falling back to another registered font.";
860
861/// Why one `@font-face` file did not register. The two causes read
862/// apart, because the fix for each one is different.
863fn face_failure(family: &str, error: &FontError) -> String {
864    match error {
865        FontError::MissingName => {
866            format!("{family} has no family name and was not registered. {FALLBACK}")
867        }
868        FontError::Parse => format!("{family} could not be read as a font. {FALLBACK}"),
869    }
870}

872/// What a `@font-face` declared its source to be. A sheet that
873/// declares neither slope nor weight is naming a family, not a cut,
874/// and the file names the cuts.
875fn declared(face: &FontFace) -> Option<FaceAttributes> {
876    let (style, weight) = (face.style, face.weight);
877    (style.is_some() || weight.is_some()).then(|| FaceAttributes {
878        italic: style == Some(FontStyle::Italic),
879        weight: weight.unwrap_or(FaceAttributes::REGULAR.weight),
880    })
881}

883/// Matching and the cascade: every element gets the declarations that
884/// match it, in cascade order, applied over what it inherited.
885fn cascade(
886    book: &Book,
887    sheets: &[Sheet],
888    registry: &FontRegistry,
889    mut warnings: Vec<Warning>,
890) -> StyleTree {
891    let elements = ElementTree::build(book);
892    warnings.extend(repeated_ids(book));
893    let mut caches = SelectorCaches::default();
894
895    let mut styles: Vec<ComputedStyle> = Vec::new();
896    let mut nodes = Vec::new();
897    let mut computed: Vec<u32> = Vec::with_capacity(elements.nodes().len());
898    let mut first_letters: Vec<Option<u32>> = Vec::with_capacity(elements.nodes().len());
899    let mut first_lines: Vec<Option<u32>> = Vec::with_capacity(elements.nodes().len());
900    let mut befores: Vec<Option<u32>> = Vec::with_capacity(elements.nodes().len());
901    let mut afters: Vec<Option<u32>> = Vec::with_capacity(elements.nodes().len());
902    let mut max_id = 0u32;
903    // Most sheets name neither, and a pass that no rule can match is
904    // a pass over every rule for every element.
905    let named = |which: &PseudoElement| {
906        sheets.iter().any(|sheet| {
907            sheet.rules.iter().any(|rule| {
908                rule.selectors
909                    .slice()
910                    .iter()
911                    .any(|selector| selector.pseudo_element() == Some(which))
912            })
913        })
914    };
915    let (wants_before, wants_after) = (named(&PseudoElement::Before), named(&PseudoElement::After));
916
917    for (index, node) in elements.nodes().iter().enumerate() {
918        let parent = node
919            .parent
920            .map(|parent| styles[computed[parent] as usize].clone())
921            .unwrap_or_else(ComputedStyle::initial);
922        let mut style = parent.inherit();
923
924        let matched = applicable(sheets, &elements, index, &mut caches, None);
925        let root_size = styles.first().map(|style| style.font_size);
926        let parent_size = parent.font_size;
927        let hint = node.align.map(align_hint);
928        apply_all(
929            &mut style,
930            sheets,
931            &matched,
932            hint.as_ref(),
933            parent_size,
934            root_size,
935            &mut warnings,
936        );
937        let (font_id, warning) = resolve_face(&style, registry);
938        style.font_id = font_id;
939        report(&mut warnings, warning);
940        report(&mut warnings, synthesized_small_caps(&style, registry));
941        report_all(&mut warnings, missing_features(&style, registry));
942
943        // A pseudo-element cascades over the style of the element it
944        // belongs to, so each is a second matching pass rather than a
945        // second element.
946        let mut pseudo_style = |which, warnings: &mut Vec<Warning>| {
947            let matched = applicable(sheets, &elements, index, &mut caches, Some(which));
948            (!matched.is_empty()).then(|| {
949                let mut pseudo = style.inherit();
950                apply_all(
951                    &mut pseudo,
952                    sheets,
953                    &matched,
954                    None,
955                    style.font_size,
956                    root_size,
957                    warnings,
958                );
959                let (font_id, warning) = resolve_face(&pseudo, registry);
960                pseudo.font_id = font_id;
961                report(warnings, warning);
962                report(warnings, synthesized_small_caps(&pseudo, registry));
963                report_all(warnings, missing_features(&pseudo, registry));
964                pseudo
965            })
966        };
967        let first_letter = pseudo_style(&PseudoElement::FirstLetter, &mut warnings);
968        let first_line = pseudo_style(&PseudoElement::FirstLine, &mut warnings);
969        let before = wants_before
970            .then(|| pseudo_style(&PseudoElement::Before, &mut warnings))
971            .flatten();
972        let after = wants_after
973            .then(|| pseudo_style(&PseudoElement::After, &mut warnings))
974            .flatten();
975        // `content: none`, the initial value, generates nothing.
976        let [before, after] = [(before, "::before"), (after, "::after")].map(|(pseudo, name)| {
977            let pseudo = pseudo.filter(|pseudo| pseudo.content != Content::None)?;
978            if INLINE_ELEMENTS.contains(&node.name) || BLOCK_ELEMENTS.contains(&node.name) {
979                return Some(pseudo);
980            }
981            let message = format!(
982                "Unsupported pseudo-element `{name}` on `{}`. Nothing is generated.",
983                node.name
984            );
985            if !warnings.iter().any(|seen| seen.message == message) {
986                warnings.push(Warning {
987                    message,
988                    origin: None,
989                });
990            }
991            None
992        });
993
994        let index_of_style = intern(&mut styles, style);
995        let index_of_initial = first_letter.map(|initial| intern(&mut styles, initial));
996        let index_of_line = first_line.map(|line| intern(&mut styles, line));
997        let index_of_before = before.map(|before| intern(&mut styles, before));
998        let index_of_after = after.map(|after| intern(&mut styles, after));
999        computed.push(index_of_style);
1000        first_letters.push(index_of_initial);
1001        first_lines.push(index_of_line);
1002        befores.push(index_of_before);
1003        afters.push(index_of_after);
1004        nodes.push(NodeStyle {
1005            id: node.id.get(),
1006            element: node.name,
1007            style: index_of_style,
1008            first_letter: index_of_initial,
1009            first_line: index_of_line,
1010            before: index_of_before,
1011            after: index_of_after,
1012        });
1013        max_id = max_id.max(node.id.get());
1014    }
1015
1016    // A paragraph with no element of its own takes the style of an
1017    // anonymous box inside its item: what the item passes on, and
1018    // nothing of its own.
1019    let mut anonymous = Vec::with_capacity(elements.anonymous().len());
1020    for (id, item) in elements.anonymous() {
1021        let inherited = styles[computed[*item] as usize].inherit();
1022        anonymous.push((id.get() as usize, intern(&mut styles, inherited)));
1023        max_id = max_id.max(id.get());
1024    }
1025
1026    let root = computed.first().copied().unwrap_or(0);
1027    let mut by_node = vec![root; max_id as usize + 1];
1028    let mut initial_by_node = vec![None; max_id as usize + 1];
1029    let mut first_line_by_node = vec![None; max_id as usize + 1];
1030    let mut before_by_node = vec![None; max_id as usize + 1];
1031    let mut after_by_node = vec![None; max_id as usize + 1];
1032    for (index, node) in elements.nodes().iter().enumerate() {
1033        let id = node.id.get() as usize;
1034        by_node[id] = computed[index];
1035        initial_by_node[id] = first_letters[index];
1036        first_line_by_node[id] = first_lines[index];
1037        before_by_node[id] = befores[index];
1038        after_by_node[id] = afters[index];
1039    }
1040    for (id, style) in anonymous {
1041        by_node[id] = style;
1042    }
1043    by_node[0] = root;
1044    let generated = || {
1045        befores
1046            .iter()
1047            .chain(&afters)
1048            .flatten()
1049            .map(|index| &styles[*index as usize].content)
1050    };
1051    let counts_pages = generated().any(Content::counts_pages);
1052    let refers = generated().any(|content| matches!(content, Content::Pieces(_)));
1053
1054    if styles.is_empty() {
1055        styles.push(ComputedStyle::initial());
1056    }
1057    let pages: Vec<(u8, PageRule)> = sheets
1058        .iter()
1059        .flat_map(|sheet| {
1060            let level = level(sheet.origin, Importance::Normal);
1061            sheet.pages.iter().cloned().map(move |rule| (level, rule))
1062        })
1063        .collect();
1064    let masters = resolve_masters(
1065        &pages,
1066        &styles,
1067        &styles[root as usize].clone(),
1068        registry,
1069        &mut warnings,
1070    );
1071
1072    StyleTree {
1073        styles,
1074        nodes,
1075        masters,
1076        by_node,
1077        initial_by_node,
1078        first_line_by_node,
1079        before_by_node,
1080        after_by_node,
1081        counts_pages,
1082        refers,
1083        notes: elements.notes(),
1084        warnings,
1085    }
1086}

1101/// The ids of these blocks and everything inside them, in document
1102/// order, warning on each id already taken.
1103fn named_blocks<'a>(
1104    blocks: &'a [Block],
1105    source: Option<&str>,
1106    first: &mut BTreeMap<&'a str, String>,
1107    warnings: &mut Vec<Warning>,
1108) {
1109    for block in blocks {
1110        claim(
1111            block_attributes(block),
1112            origin(source, block_position(block)),
1113            first,
1114            warnings,
1115        );
1116        match block {
1117            Block::Heading { inlines, .. } | Block::Paragraph { inlines, .. } => {
1118                named_inlines(inlines, source, first, warnings)
1119            }
1120            Block::Blockquote { blocks, .. } => named_blocks(blocks, source, first, warnings),
1121            Block::List { items, .. } => {
1122                for item in items {
1123                    claim(
1124                        &item.attributes,
1125                        origin(source, item.position),
1126                        first,
1127                        warnings,
1128                    );
1129                    named_blocks(&item.blocks, source, first, warnings);
1130                }
1131            }
1132            Block::Table { head, body, .. } => {
1133                for row in rows(head, body) {
1134                    claim(
1135                        &row.attributes,
1136                        origin(source, row.position),
1137                        first,
1138                        warnings,
1139                    );
1140                    for cell in &row.cells {
1141                        claim(
1142                            &cell.attributes,
1143                            origin(source, cell.position),
1144                            first,
1145                            warnings,
1146                        );
1147                        named_blocks(&cell.blocks, source, first, warnings);
1148                    }
1149                }
1150            }
1151            Block::CodeBlock { .. }
1152            | Block::ThematicBreak { .. }
1153            | Block::PageBreak { .. }
1154            | Block::ColumnBreak { .. }
1155            | Block::Image { .. } => {}
1156        }
1157    }
1158}

1160/// The same, over the inlines of one block.
1161fn named_inlines<'a>(
1162    inlines: &'a [Inline],
1163    source: Option<&str>,
1164    first: &mut BTreeMap<&'a str, String>,
1165    warnings: &mut Vec<Warning>,
1166) {

1175            Inline::Text { .. } | Inline::Code { .. } | Inline::Break { .. } => {}
1176            Inline::Note { blocks, .. } => named_blocks(blocks, source, first, warnings),
1177            Inline::Emphasis { children, .. }
1178            | Inline::Strong { children, .. }
1179            | Inline::Link { children, .. }
1180            | Inline::Strikethrough { children, .. }
1181            | Inline::Highlight { children, .. }
1182            | Inline::Span { children, .. } => named_inlines(children, source, first, warnings),
1183        }
1184    }

1187/// Takes one node's id, or reports the element that has it already.
1188fn claim<'a>(
1189    attributes: &'a Attributes,
1190    at: String,
1191    first: &mut BTreeMap<&'a str, String>,
1192    warnings: &mut Vec<Warning>,
1193) {
1194    let Some(id) = attributes.id.as_deref() else {
1195        return;
1196    };
1197    match first.get(id) {
1198        Some(taken) => warnings.push(Warning {
1199            message: format!("Id `{id}` is duplicated. Applied at {taken}."),

1201        }),
1202        None => {

1204        }
1205    }
1206}

1208/// Where one applicable declaration lives, sorted into cascade order:
1209/// origin and importance, then specificity, then source order.
1210type Applicable = (u8, u32, usize, usize, usize);
1211
1212/// Every declaration that matches one element, in cascade order.
1213/// `pseudo` selects the pass: `None` matches the element itself,
1214/// `Some(_)` matches only rules ending in that pseudo-element.
1215fn applicable(
1216    sheets: &[Sheet],
1217    elements: &ElementTree,
1218    index: usize,
1219    caches: &mut SelectorCaches,
1220    pseudo: Option<&PseudoElement>,
1221) -> Vec<Applicable> {
1222    let mode = match pseudo {
1223        Some(_) => MatchingMode::ForStatelessPseudoElement,
1224        None => MatchingMode::Normal,
1225    };
1226    let mut applicable = Vec::new();
1227    for (sheet_index, sheet) in sheets.iter().enumerate() {
1228        for (rule_index, rule) in sheet.rules.iter().enumerate() {
1229            let specificity = rule
1230                .selectors
1231                .slice()
1232                .iter()
1233                .filter(|selector| selector.pseudo_element() == pseudo)
1234                .filter(|selector| {
1235                    let mut context = MatchingContext::new(
1236                        mode,
1237                        None,
1238                        caches,
1239                        QuirksMode::NoQuirks,
1240                        NeedsSelectorFlags::No,
1241                        MatchingForInvalidation::No,
1242                    );
1243                    matches_selector(selector, 0, None, &elements.at(index), &mut context)
1244                })
1245                .map(|selector| selector.specificity())
1246                .max();
1247            let Some(specificity) = specificity else {
1248                continue;
1249            };
1250            for (order, (_, importance)) in rule.declarations.iter().enumerate() {
1251                applicable.push((
1252                    level(sheet.origin, *importance),
1253                    specificity,
1254                    sheet_index,
1255                    rule_index,
1256                    order,
1257                ));
1258            }
1259        }
1260    }
1261    applicable.sort_unstable();
1262    applicable
1263}

1265/// Applies matched declarations in cascade order.
1266///
1267/// `hint` is what the source wrote on the element itself, as HTML's
1268/// `align` is written on a cell. It goes in after the built-in sheet
1269/// and before every rule of the author's, whatever its specificity.
1270fn apply_all(
1271    style: &mut ComputedStyle,
1272    sheets: &[Sheet],
1273    applicable: &[Applicable],
1274    hint: Option<&properties::Declaration>,
1275    parent_size: f32,
1276    root_size: Option<f32>,
1277    warnings: &mut Vec<Warning>,
1278) {
1279    let root_size = root_size.unwrap_or(parent_size);
1280    let declaration = |(_, _, sheet, rule, order): &Applicable| {
1281        &sheets[*sheet].rules[*rule].declarations[*order].0
1282    };
1283    // Custom properties resolve first, because a declaration anywhere
1284    // in the cascade can read one declared after it.
1285    let declared: Vec<&Custom> = applicable
1286        .iter()
1287        .filter_map(|at| match declaration(at) {
1288            Declaration::Custom(custom) => Some(custom),
1289            _ => None,
1290        })
1291        .collect();
1292    if !declared.is_empty() {
1293        style.custom = substitution::resolve(&style.custom, &declared, warnings);
1294    }
1295    let base = applicable
1296        .iter()
1297        .any(|at| matches!(declaration(at), Declaration::Pending(_)))
1298        .then(|| style.clone());
1299    let mut hint = hint;
1300    for at in applicable {
1301        if at.0 > 0
1302            && let Some(declaration) = hint.take()
1303        {
1304            style.apply(declaration, parent_size, root_size);
1305        }
1306        match (declaration(at), &base) {
1307            (Declaration::Pending(pending), Some(base)) => {
1308                substitution::apply(style, pending, base, parent_size, root_size, warnings)
1309            }
1310            (declaration, _) => style.apply(declaration, parent_size, root_size),
1311        }
1312    }
1313    if let Some(declaration) = hint {
1314        style.apply(declaration, parent_size, root_size);
1315    }
1316}

1370/// The face a computed style shapes with: the first family in the
1371/// registry, at the nearest slope and weight it has.
1372///
1373/// A family the registry does not have is skipped — that is what a
1374/// font stack is for — but a stack that resolves nothing, or a
1375/// family with no cut at the slope asked for, falls back to a face
1376/// that is visibly not the one requested. Both say so.
1377fn resolve_face(style: &ComputedStyle, registry: &FontRegistry) -> (u16, Option<Warning>) {
1378    let want = FaceAttributes {
1379        italic: style.font_style == FontStyle::Italic,
1380        weight: style.font_weight,
1381    };
1382    for family in &style.font_family {
1383        let name = match family {
1384            Family::Named(name) => Some(name.clone()),
1385            Family::Generic(generic) => registry
1386                .generic(*generic)
1387                .and_then(|id| registry.font_ref(id).map(|entry| entry.family.clone())),
1388        };
1389        let Some(found) = name.as_deref().and_then(|name| registry.select(name, want)) else {
1390            continue;
1391        };
1392        let warning = (found.attributes.italic != want.italic).then(|| Warning {
1393            message: format!(
1394                "{} has no {} face. {} text falls back to {}.",
1395                name.unwrap_or_default(),
1396                slope(want.italic),
1397                sentence(slope(want.italic)),
1398                slope(found.attributes.italic),
1399            ),
1400            origin: None,
1401        });
1402        return (found.id, warning);
1403    }
1404    (
1405        0,
1406        Some(Warning {
1407            message: format!(
1408                "No registered font matches {}. The first registered font is used instead.",
1409                stack(&style.font_family),
1410            ),
1411            origin: None,
1412        }),
1413    )
1414}

1417/// per style that asks for them: an author who picked the face can
1418/// pick another one.
1419fn synthesized_small_caps(style: &ComputedStyle, registry: &FontRegistry) -> Option<Warning> {
1420    let asked = style.font_variant_caps == FontVariantCaps::SmallCaps;

1424            registry
1425                .font_ref(style.font_id)
1426                .map(|entry| entry.family.clone())
1427                .unwrap_or_else(|| stack(&style.font_family)),
1428        ),
1429        origin: None,
1430    })

1433/// The diagnostic for a face that carries none of a feature the
1434/// sheet asked for, one per style that asks for it. The run is set
1435/// without the feature; nothing is synthesized.
1436///
1437/// A setting of zero asks for a feature to be left off, which a face
1438/// that has nothing for it already does.
1439fn missing_features(style: &ComputedStyle, registry: &FontRegistry) -> Vec<Warning> {
1440    let family = || {
1441        registry
1442            .font_ref(style.font_id)
1443            .map(|entry| entry.family.clone())
1444            .unwrap_or_else(|| stack(&style.font_family))
1445    };
1446    style
1447        .features()
1448        .iter()
1449        .filter(|setting| setting.value != 0)
1450        .filter(|setting| !registry.has_feature(style.font_id, setting.tag))
1451        .map(|setting| Warning {
1452            message: format!(
1453                "{} has no `{}` feature. The text is set without it.",
1454                family(),
1455                String::from_utf8_lossy(&setting.tag),
1456            ),
1457            origin: None,
1458        })
1459        .collect()
1460}
