17/// The page a section's lines were broken against: the measure
18/// always, and the content height only for a book with an image or a
19/// percentage height in it. Those are the two things fragment building
20/// sizes against the page's own height, so a book without either never
21/// reads that height, and a page that grows taller leaves its prose
22/// broken where it was.
23///
24/// A book that anchors an image to the page or floats one carries the
25/// exclusions as well. The flow resolves their geometry. A section whose paragraphs
26/// can be set again beside an image keeps what it takes to set them.
27/// A book that gains its first image builds its sections again.
28///
29/// A book with a block that spans the columns carries the width of
30/// the content box, which is what that block breaks to.
31#[derive(Debug, Clone, Copy)]
32pub(super) struct Against {
33    measure: f32,
34    height: Option<f32>,
35    exclusions: Option<u64>,
36    width: Option<f32>,
37}
38
39impl Against {
40    pub(super) fn of(styles: &StyleTree, images: bool) -> Against {
41        let geometry = styles.default_page().geometry;
42        let mut anchored = DefaultHasher::new();
43        let mut any = false;
44        for style in styles
45            .styles()
46            .iter()
47            .filter(|style| style.position == Position::Absolute)
48        {
49            any = true;
50            style.position.hash(&mut anchored);
51            style.wrap_flow.hash(&mut anchored);
52            hash_shape(&style.shape_outside, &mut anchored);
53            style.shape_margin.to_bits().hash(&mut anchored);
54            hash_insets(style.inset, &mut anchored);
55            hash_edges(style.margin, &mut anchored);
56        }
57        for style in styles.styles() {
58            if style.float != Float::None {
59                any = true;
60            }
61            style.float.hash(&mut anchored);
62        }
63        Against {
64            measure: geometry.measure(),
65            height: (images || percent_heights(styles)).then(|| geometry.content_size().1),
66            exclusions: any.then(|| anchored.finish()),
67            width: spans(styles).then(|| geometry.content_size().0),
68        }
69    }
70
71    pub(super) fn hash_into(self, h: &mut DefaultHasher) {
72        (self.measure.to_bits(), self.height.map(f32::to_bits)).hash(h);
73        self.exclusions.hash(h);
74        self.width.map(f32::to_bits).hash(h);
75    }
76}

78/// Whether any block asks for a height as a percentage of the page.
79fn percent_heights(styles: &StyleTree) -> bool {
80    styles.styles().iter().any(|style| {
81        [style.height, style.min_height, style.max_height]
82            .iter()
83            .any(|size| matches!(size, Width::Percent(_)))
84    })
85}

87/// Whether any section of the book places an image.
88pub(super) fn has_images(book: &Book) -> bool {
89    fn walk(blocks: &[Block]) -> bool {
90        blocks.iter().any(|block| match block {
91            Block::Image { .. } => true,
92            Block::Blockquote { blocks, .. } => walk(blocks),
93            Block::Table { head, body, .. } => cell_blocks(head, body).any(walk),
94            _ => false,
95        })
96    }
97    book.sections.iter().any(|section| walk(&section.blocks))
98}

100/// What the compiled styling hashes to, split by the stage each part
101/// feeds. A style edit is classified by which of these moved.
102#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
103pub(super) struct Prints {
104    /// Which nodes ask for a contour to be traced, and from what.
105    trace: u64,
106    /// The measure, what every distinct style says about breaking,
107    /// and which style each node resolved to.
108    breaks: u64,
109    /// Page geometry, and the named page each element asks for.
110    flow: u64,
111    /// The margin boxes, and what they paint.
112    paint: u64,
113}
114
115impl Prints {
116    pub(super) fn of(styles: &StyleTree, images: bool) -> Prints {
117        // Both halves of what the stage keys on: which nodes name a
118        // contour, and which of them the flow reads one for. A sheet
119        // that anchors an image already asking for `auto` reaches the
120        // stage through the second.
121        let mut trace = DefaultHasher::new();
122        for style in styles.styles() {
123            hash_shape(&style.shape_outside, &mut trace);
124            style.excludes().hash(&mut trace);
125        }
126        hash_nodes(styles, &mut trace);
127
128        let mut breaks = DefaultHasher::new();
129        Against::of(styles, images).hash_into(&mut breaks);
130        for style in styles.styles() {
131            hash_layout(style, &mut breaks);
132        }
133        hash_nodes(styles, &mut breaks);
134
135        let mut flow = DefaultHasher::new();
136        for master in styles.masters() {
137            (&master.page, master.situation).hash(&mut flow);
138            hash_geometry(master.style.geometry, &mut flow);
139            hash_background(&master.style.background, &mut flow);
140        }
141        for style in styles.styles() {
142            style.page.hash(&mut flow);
143        }
144        hash_nodes(styles, &mut flow);
145
146        let mut paint = DefaultHasher::new();
147        for master in styles.masters() {
148            (&master.page, master.situation).hash(&mut paint);
149            hash_geometry(master.style.geometry, &mut paint);
150            hash_background(&master.style.background, &mut paint);
151            for box_ in &master.style.boxes {
152                (box_.which, &box_.content).hash(&mut paint);
153                hash_layout(&box_.style, &mut paint);
154            }
155        }
156
157        Prints {
158            trace: trace.finish(),
159            breaks: breaks.finish(),
160            flow: flow.finish(),
161            paint: paint.finish(),
162        }
163    }
164
165    /// The deepest stage a move from `self` to `fresh` invalidates.
166    pub(super) fn against(&self, fresh: &Prints) -> Stale {
167        if self.trace != fresh.trace {
168            Stale::Trace
169        } else if self.breaks != fresh.breaks {
170            Stale::Break
171        } else if self.flow != fresh.flow {
172            Stale::Flow
173        } else if self.paint != fresh.paint {
174            Stale::Paint
175        } else {
176            Stale::Nothing
177        }
178    }
179}

181/// Whether a section's lines may outlive an edit elsewhere.
182pub(super) fn section_local(styles: &StyleTree) -> bool {
183    uniform_measure(styles) && !paginated_prose(styles.styles())
184}

186/// Whether every master breaks to the same measure. One that does
187/// not makes where a line breaks depend on which page it lands on,
188/// and that on everything before it. A book with a block that spans
189/// the columns breaks to the width of the content box as well.
190fn uniform_measure(styles: &StyleTree) -> bool {
191    let spans = spans(styles);
192    let widths = |geometry: PageGeometry| {
193        (
194            geometry.measure().to_bits(),
195            spans.then(|| geometry.content_size().0.to_bits()),
196        )
197    };
198    let default = widths(styles.default_page().geometry);
199    styles
200        .masters()
201        .iter()
202        .all(|master| widths(master.style.geometry) == default)
203}
204
205/// Whether any block spans the columns of a divided page.
206fn spans(styles: &StyleTree) -> bool {
207    styles
208        .styles()
209        .iter()
210        .any(|style| style.column_span == ColumnSpan::All)
211}

213/// Whether any element generates text that only pagination resolves.
214/// `counter(page)` and `string()` in prose would make inline text
215/// depend on where the prose fell, which is a fixpoint rather than a
216/// cache invalidation.
217fn paginated_prose(styles: &[ComputedStyle]) -> bool {
218    styles
219        .iter()
220        .any(|style| matches!(style.content, Content::Counter(_) | Content::String(_)))
221}

223/// What the book's declared language decides: the patterns its words
224/// break by, and the tag the warning names where there are no
225/// patterns for it. Two languages without patterns break the same
226/// lines and warn about different tags.
227pub(super) fn hyphenation(metadata: &Metadata) -> (Patterns, Option<&str>) {
228    let patterns = Patterns::of(metadata);
229    let unknown = metadata.language().filter(|_| patterns == Patterns::NONE);
230    (patterns, unknown)
231}

247    /// Masters of different measures make where a line breaks depend
248    /// on which page it lands on. The session stops reusing lines
249    /// rather than serve breaks taken against another page's measure.
250    #[test]
251    fn asymmetric_page_margins_fall_back_to_full_re_breaking() {
252        let mut session = three_chapters();
253        session.set_style(sheets("@page :left { margin-left: 20pt }"));
254        session.preview();
255        assert!(
256            !session.reuses_sections(),
257            "two measures, and the cache still claims to be sound"
258        );
259
260        let broke = session.stages().lines;
261        session.replace_source("two.md", vec![section("two.md", prose("delta", 9))]);
262        session.preview();
263        assert_eq!(
264            session.stages().lines,
265            broke + 3,
266            "a section kept lines broken against a measure it may not land on"
267        );
268    }

270    /// A page background is what the flow paints when a page closes,
271    /// so a sheet that changes one flows the book again and leaves
272    /// every line where it was. Nothing is traced for it either: an
273    /// image behind a box is never decoded.
274    #[test]
275    fn a_page_background_flows_the_book_and_breaks_no_lines() {
276        let mut session =
277            crate::session::Session::owning(crate::fonts::bundled_registry().unwrap());
278        session.set_content(crate::session::testing::book(vec![
279            section("one.md", prose("alpha", 8)),
280            section("two.md", prose("beta", 8)),
281        ]));
282        session
283            .add_image("scan.png", crate::session::testing::gif(640, 960, 0))
284            .expect("an owning session registers");
285        session.preview();
286        let before = session.stages();
287
288        session.set_style(sheets("@page { background-image: url(scan.png) }"));
289        session.preview();
290        let after = session.stages();
291        assert_eq!(
292            after.lines, before.lines,
293            "a page background re-broke lines"
294        );
295        assert_eq!(after.trace, before.trace, "a background was traced");
296        assert!(after.flow > before.flow, "the page was not flowed again");
297    }

299    /// Inline text that depended on pagination would make breaking
300    /// depend on where the breaks fell. The parser keeps the page
301    /// counter out of element rules, so prose cannot ask for it, and
302    /// the precondition says so out loud rather than leaving it to
303    /// somebody's memory.
304    #[test]
305    fn generated_content_that_depends_on_pagination_closes_the_cache() {
306        let plain = ComputedStyle::initial();
307        let ornament = ComputedStyle {
308            content: Content::Text("\u{2766}".into()),
309            ..plain.clone()
310        };
311        assert!(!paginated_prose(&[plain.clone(), ornament]));
312
313        let folio = ComputedStyle {
314            content: Content::Counter(CounterStyle::Decimal),
315            ..plain.clone()
316        };
317        assert!(paginated_prose(&[plain.clone(), folio]));
318
319        let running = ComputedStyle {
320            content: Content::String("chapter".into()),
321            ..plain
322        };
323        assert!(paginated_prose(&[running]));
324    }
