pub const USER_AGENT_CSS: &str = "/* Fleuron\'s default stylesheet: a trade paperback at 6x9 inches, with\n   mirrored margins and a page number at the bottom. Any additional\n   CSS cascades over every rule here. */\n\nbook {\n  font-family: serif;\n  font-size: 11pt;\n  line-height: 1.4;\n  text-align: left;\n  hyphens: none;\n  orphans: 2;\n  widows: 2;\n}\n\n/* A chapter opens on a right-hand page. Naming its pages lets @page\n   tell an opening page from a continuation. */\nsection {\n  page: chapter;\n  break-before: recto;\n  counter-reset: note;\n}\n\nh1, h2, h3, h4, h5, h6 {\n  font-size: 18pt;\n  break-after: avoid;\n}\n\n/* An indent separates one paragraph from the next, so it belongs\n   only to a paragraph that follows another one. A paragraph after a\n   heading, a thematic break, or an image starts flush. */\np + p {\n  text-indent: 1.2em;\n}\n\n/* A quotation is narrower than the surrounding text and set off\n   from it. */\nblockquote {\n  margin: 1em 2em;\n}\n\n/* The title of a callout is bold. The paragraph under the title has\n   no indent. */\n.callout-title {\n  font-weight: bold;\n}\n\n.callout-title + p {\n  text-indent: 0;\n}\n\n/* A code block has space above and below it, and a monospace family.\n   The bundled family is the only face a book gets without\n   @font-face, so a book that wants a monospace face declares one. */\npre {\n  font-family: monospace;\n  margin: 1em 0;\n}\n\n/* A list has space above and below it. Its markers are in the\n   indent to the left of its items. */\nul, ol {\n  margin: 0.5em 0;\n  padding-left: 1.5em;\n}\n\nol {\n  list-style-type: decimal;\n}\n\n/* A list inside an item has no space above or below it, and its\n   markers are open circles. */\nli > :is(ul, ol) {\n  margin: 0;\n}\n\nli > ul {\n  list-style-type: circle;\n}\n\n/* An item of a loose list holds paragraphs, with space between them\n   and no indent. The items of a tight list have no space between\n   them. */\nli > p {\n  margin: 0.5em 0;\n  text-indent: 0;\n}\n\n/* A thematic break is a centered ornament. It belongs between\n   paragraphs, so it never opens or closes a page. */\nhr {\n  content: \"\\2766\";\n  text-align: center;\n  margin: 1em 0;\n  break-before: avoid;\n  break-after: avoid;\n}\n\n/* The notes of a page stand at the foot of its content box, under a\n   rule. */\nnotes {\n  margin-top: 1em;\n  padding-top: 0.4em;\n  border-top: thin solid;\n}\n\n/* A note is set smaller than the text, and starts flush in the\n   letters the book is set in, wherever its reference was written. The\n   number is the reference in the line and the mark beside the note,\n   and it hangs in the indent to the left of the note. */\nnote {\n  font-size: 9pt;\n  line-height: 1.2;\n  font-style: normal;\n  font-weight: normal;\n  text-indent: 0;\n  list-style-type: decimal;\n  padding-left: 1.5em;\n  margin-top: 0.4em;\n}\n\n/* A page break and a column break hold no text. The content after\n   one starts on a new page or in the next column. */\npagebreak {\n  break-after: page;\n}\n\ncolumnbreak {\n  break-after: column;\n}\n\n/* A section\'s opening heading is its chapter title. Storing it as a\n   named string lets a running head pick it up with\n   string(chapter). */\nsection > :is(h1, h2, h3, h4, h5, h6):first-child {\n  string-set: chapter content();\n}\n\n/* A table has a rule above it, a rule below it, and a rule under its\n   header row. Neighboring cells share one border. */\ntable {\n  margin: 1em 0;\n  border-collapse: collapse;\n  border-top: thin solid;\n  border-bottom: thin solid;\n}\n\n/* The text of a cell starts at the left of the cell, even where the\n   prose around the table is justified. Padding keeps the text off\n   the rules. */\nth, td {\n  padding: 0.25em 0.5em;\n  text-align: left;\n}\n\nth {\n  font-weight: bold;\n  border-bottom: thin solid;\n}\n\nem { font-style: italic }\nstrong { font-weight: bold }\ncode { font-family: monospace }\ns { text-decoration: line-through }\nmark { background-color: #ffff00 }\n\n@page {\n  size: 432pt 648pt;\n  margin: 54pt 42pt 54pt 54pt;\n  @bottom-center {\n    content: counter(page);\n    font-size: 9pt;\n    line-height: 1.4;\n  }\n}\n\n/* Margins mirror across the spread, so the wider one is always at\n   the spine. */\n@page :left {\n  margin-left: 42pt;\n  margin-right: 54pt;\n}\n\n@page :right {\n  margin-left: 54pt;\n  margin-right: 42pt;\n}\n\n/* A chapter\'s opening page counts toward the numbering but does not\n   show its number. */\n@page chapter:first {\n  @bottom-center { content: none }\n}\n";
