1//! A retained pipeline: every stage kept, and only what an edit
2//! changed run again.
3//!
4//! `layout_book` is a pure function of its inputs, so every call
5//! rebuilds every stage. A process that renders one book and exits
6//! wants that. A live preview does not, because the common event
7//! there is a small change to one input while the others stand. A
8//! session retains the output of each stage and works out the deepest
9//! stage an edit reaches: a colour serves the display structure back, page
10//! furniture repaints, `@page` geometry re-fragments over cached
11//! lines, and only the measure or the text itself breaks lines
12//! again.
13//!
14//! # Section-local lines
15//!
16//! Line *breaking* is section-local: where the breaks fall depends on
17//! the measure, the face and the text, not on where the section
18//! starts vertically. Line *placement* depends on position and
19//! belongs to fragmentation. So the cache stores breaks, shaped runs
20//! and advances, and no page coordinates at all. An edit to one
21//! chapter re-breaks that chapter and re-fragments the book, which
22//! for a whole novel costs about what tracking the pages that moved
23//! would.
24//!
25//! Two preconditions make that sound, and both are checked rather
26//! than remembered. The first is a uniform measure: masters that
27//! resolve different content widths make breaking depend on which
28//! page a line lands on. The second is that no inline text depends
29//! on pagination, which the parser guarantees: `counter(page)` is
30//! legal only inside a margin box. When either one fails, the session re-breaks
31//! everything instead of serving stale lines.
32//!
33//! A reference that prints a page is the one piece of inline text
34//! that pagination decides, and it is set on a pass of its own. The
35//! first pass sets a placeholder where the number goes, so its lines
36//! are section-local like any others. The second builds again only
37//! the sections whose references print a page, and keys each one by
38//! the folios it prints.
39//!
40//! The parts have a file each: `edit` is what a host changes,
41//! `output` is what it asks for, `inspect` answers what styled one
42//! node and where it landed, `stage` runs the stages, `invalidate`
43//! decides which of them an edit reaches, and `key` fingerprints one
44//! section.

69/// How many times each stage has run since the session was made.
70///
71/// A host reads these to see what an edit cost. The tests read them
72/// to prove what an edit did *not* cost, which a clock cannot show.
73#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
74pub struct Stages {
75    /// Style compilations: parse, match, cascade.
76    pub style: u32,
77    /// Images decoded to trace a contour. A book whose sheet names no
78    /// contour never decodes one, and a contour already traced is not
79    /// traced again.
80    pub trace: u32,
81    /// Sections broken into lines. One per section, per rebuild.
82    pub lines: u32,
83    /// Fragmentation and page assembly runs.
84    pub flow: u32,
85    /// Furniture paints: numbering and margin boxes.
86    pub paint: u32,
87    /// Second layout passes, which print the pages the references in
88    /// the book name. A book whose sheet prints no page number never
89    /// runs one.
90    pub settle: u32,
91}

93/// The deepest stage a change invalidates, which is the shallowest
94/// cache that survives it. Ordered: a deeper stage implies every
95/// stage under it.
96#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
97enum Stale {
98    /// Everything stands; the display structure is served as it is.
99    Nothing,
100    /// Numbering and margin boxes.
101    Paint,
102    /// Fragmentation, over lines that survived.
103    Flow,
104    /// Line breaking, and everything below it.
105    Break,
106    /// Tracing the contours the sheet asks for, which is above line
107    /// breaking because a contour that moved is a measure that moved.
108    Trace,
109}

111/// A table a session lays out against: a caller's, or one of its
112/// own.
113///
114/// A host with faces and images to lend keeps lending them. A worker
115/// has nowhere to keep either, since the module is all there is, so
116/// it hands them over once and adds to them through the session.
117enum Table<'a, T> {
118    Borrowed(&'a T),
119    Owned(Box<T>),
120}
121
122impl<T> Table<'_, T> {
123    fn get(&self) -> &T {
124        match self {
125            Table::Borrowed(table) => table,
126            Table::Owned(table) => table,
127        }
128    }
129
130    fn get_mut(&mut self) -> Option<&mut T> {
131        match self {
132            Table::Borrowed(_) => None,
133            Table::Owned(table) => Some(table),
134        }
135    }
136}

159/// One section's lines, and what building them had to complain about.
160struct Cached {
161    key: u64,
162    /// The id the section had when its lines were broken.
163    section: NodeId,
164    fragments: Vec<Fragment>,
165    warnings: Vec<Warning>,
166}
167
168impl Cached {
169    /// Whether the lines can be moved onto the ids the book hands out
170    /// now. The notes a line carries were built with ids of their
171    /// own, and moving those is not what `renumber` does, so a
172    /// section that holds one is broken again instead.
173    fn renumbers(&self, section: NodeId) -> bool {
174        section == self.section
175            || !self
176                .fragments
177                .iter()
178                .any(|fragment| fragment.notes.is_some())
179    }
180
181    /// Moves the source ranges onto the ids the book hands out now.
182    /// Ids renumber globally on every edit, so a chapter nothing
183    /// touched comes back out of the cache under a new number; its
184    /// nodes are dense and in document order from the section's own,
185    /// so all of them move by the same step.
186    fn renumber(&mut self, section: NodeId) {
187        let step = section.get() as i64 - self.section.get() as i64;
188        self.section = section;
189        if step == 0 {
190            return;
191        }
192        for fragment in &mut self.fragments {
193            if let Some(marks) = &mut fragment.marks {
194                for node in &mut marks.targets {
195                    *node = node.shifted(step);
196                }
197            }
198            if let Some(decorations) = &mut fragment.decorations {
199                for decoration in &mut decorations.opens {
200                    decoration.node = decoration.node.shifted(step);
201                }
202            }
203            if let Piece::Anchor(node) = &mut fragment.piece {
204                *node = node.shifted(step);
205            }
206            if let Piece::Row(row) = &mut fragment.piece {
207                for (node, _) in &mut row.boxes {
208                    *node = node.shifted(step);
209                }
210                for item in &mut row.items {
211                    if let crate::pages::DrawItem::Text {
212                        origin,
213                        pseudo_element,
214                        ..
215                    } = item
216                    {
217                        shift_run(origin, pseudo_element, step);
218                    }
219                }
220                continue;
221            }
222            let Piece::Line { line, cap } = &mut fragment.piece else {
223                continue;
224            };
225            let caps = cap.iter_mut().map(|cap| &mut cap.line);
226            for line in std::iter::once(line).chain(caps) {
227                for run in &mut line.runs {
228                    shift_run(&mut run.origin, &mut run.pseudo_element, step);
229                }
230            }
231        }
232    }
233}
234
235/// Moves the ids one run names by `step`.
236fn shift_run(origin: &mut Option<SourceRange>, pseudo_element: &mut Option<NodeId>, step: i64) {
237    if let Some(origin) = origin {
238        origin.node = origin.node.shifted(step);
239    }
240    if let Some(node) = pseudo_element {
241        *node = node.shifted(step);
242    }
243}

245/// A retained pipeline: content, styling, and every stage between
246/// them and the page.
247///
248/// ```
249/// # use fleuron::content::Book;
250/// # use fleuron::session::Session;
251/// # use fleuron::style::Stylesheets;
252/// # let registry = fleuron::fonts::bundled_registry().unwrap();
253/// let mut session = Session::new(&registry);
254/// session.set_content(Book::default());
255/// session.set_style(Stylesheets::parse(&[]));
256/// let pages = &session.preview().pages;
257/// ```
258///
259/// A computed style can only resolve to a face already in the
260/// registry. A session over a borrowed registry leaves it as the
261/// host made it, so a sheet that brings its own `@font-face` needs
262/// the host to load that face before the sheet is set. A session
263/// that owns its registry registers the faces the sheet declares from
264/// the files [`add_font_file`](Session::add_font_file) hands over.
265pub struct Session<'a> {
266    registry: Table<'a, FontRegistry>,
267    assets: Table<'a, Assets>,
268    fonts: faces::FontFiles,
269    /// What the trace stage made of the assets the sheet wraps prose
270    /// around. Empty for a sheet that names no contour.
271    contours: Contours,
272    book: Cow<'a, Book>,
273    /// The sources in the order the book holds them, one that has no
274    /// section among them.
275    order: Vec<String>,
276    /// The sheets the tree was compiled from. `None` on the one-shot
277    /// path, where the caller compiled the tree itself and nothing
278    /// will ask for another.
279    sheets: Option<Stylesheets>,
280    styles: Cow<'a, StyleTree>,
281    prints: Prints,
282    /// Whether the preconditions for reusing a section's lines are met.
283    section_local: bool,
284    /// Whether the book places an image, which is what makes the
285    /// page's height an input to breaking.
286    images: bool,
287    /// Whether the stages are kept between calls. This is the only
288    /// difference on the one-shot path, which keeps one section's
289    /// lines at a time and drops each as it is flowed.
290    retain: bool,
291    lines: Vec<Cached>,
292    /// What the book's references resolve against on the pass that
293    /// finds the pages.
294    references: References,
295    /// What the notes of the book are numbered.
296    notes: Numbering,
297    /// Each section's lines on the pass that prints the pages its
298    /// references name. `None` for a section with no such reference,
299    /// whose lines from the pass before stand.
300    settled: Vec<Option<Cached>>,
301    /// What that pass had to complain about: its own sections, and
302    /// every element it set on another page than the pass before.
303    settle_warnings: Vec<Warning>,
304    /// What the links had to complain about: a url that names nothing
305    /// in the book.
306    link_warnings: Vec<Warning>,
307    infos: Vec<PageInfo>,
308    /// The border box of each block, on each page it reaches.
309    boxes: Vec<(NodeId, PageBox)>,
310    output: Option<LayoutOutput>,
311    /// What building lines complained about, deduped in the order the
312    /// sections raised it.
313    flow_warnings: Vec<Warning>,
314    /// What a frontend had to say about the sources it read, which
315    /// happened upstream of every stage here.
316    source_warnings: Vec<Warning>,
317    stale: Stale,
318    stages: Stages,
319}

321impl<'a> Session<'a> {
322    /// A session over the faces in `registry`, with no content and
323    /// the built-in sheet alone.
324    pub fn new(registry: &'a FontRegistry) -> Session<'a> {
325        Session::with_assets(registry, no_assets())
326    }
327
328    /// The same, over images the host has already probed.
329    pub fn with_assets(registry: &'a FontRegistry, assets: &'a Assets) -> Session<'a> {
330        Session::over(Table::Borrowed(registry), Table::Borrowed(assets))
331    }
332
333    fn over(registry: Table<'a, FontRegistry>, assets: Table<'a, Assets>) -> Session<'a> {
334        let book = Book::default();
335        let sheets = Stylesheets::parse(&[]);
336        let styles = sheets.compile(&book, registry.get());
337        let fonts = faces::FontFiles::over(registry.get().len());
338        let mut session = Session {
339            registry,
340            assets,
341            fonts,
342            contours: Contours::none(),
343            book: Cow::Owned(book),
344            order: Vec::new(),
345            sheets: Some(sheets),
346            prints: Prints::of(&styles, false),
347            section_local: section_local(&styles),
348            images: false,
349            styles: Cow::Owned(styles),
350            retain: true,
351            lines: Vec::new(),
352            references: References::default(),
353            notes: Numbering::default(),
354            settled: Vec::new(),
355            settle_warnings: Vec::new(),
356            link_warnings: Vec::new(),
357            infos: Vec::new(),
358            boxes: Vec::new(),
359            output: None,
360            flow_warnings: Vec::new(),
361            source_warnings: Vec::new(),
362            stale: Stale::Trace,
363            stages: Stages {
364                style: 1,
365                ..Stages::default()
366            },
367        };
368        session.load_faces();
369        session
370    }
371
372    /// A session that owns the faces it lays out against, and takes
373    /// more through [`add_font`](Session::add_font).
374    ///
375    /// This is the shape a worker needs: font bytes cross the
376    /// boundary once, the module keeps them, and no caller on the
377    /// other side of the wall has a registry to lend.
378    pub fn owning(registry: FontRegistry) -> Session<'static> {
379        Session::over(
380            Table::Owned(Box::new(registry)),
381            Table::Owned(Box::new(Assets::none())),
382        )
383    }
384
385    /// The single run `layout_book` makes, over inputs the caller
386    /// owns and will not edit. Nothing is fingerprinted, because
387    /// nothing will be compared against it.
388    pub(crate) fn once(
389        book: &'a Book,
390        styles: &'a StyleTree,
391        registry: &'a FontRegistry,
392        assets: &'a Assets,
393    ) -> Session<'a> {
394        Session {
395            registry: Table::Borrowed(registry),
396            assets: Table::Borrowed(assets),
397            fonts: faces::FontFiles::over(registry.len()),
398            contours: Contours::none(),
399            book: Cow::Borrowed(book),
400            order: Vec::new(),
401            sheets: None,
402            styles: Cow::Borrowed(styles),
403            prints: Prints::default(),
404            section_local: false,
405            images: false,
406            retain: false,
407            lines: Vec::new(),
408            references: References::default(),
409            notes: Numbering::default(),
410            settled: Vec::new(),
411            settle_warnings: Vec::new(),
412            link_warnings: Vec::new(),
413            infos: Vec::new(),
414            boxes: Vec::new(),
415            output: None,
416            flow_warnings: Vec::new(),
417            source_warnings: Vec::new(),
418            stale: Stale::Trace,
419            stages: Stages::default(),
420        }
421    }
422}

424impl Session<'_> {
425    /// The session's own copy of the book, node ids assigned.
426    pub fn book(&self) -> &Book {
427        &self.book
428    }
429
430    /// The classes and the ids the blocks and inlines of one source
431    /// carry, or of the whole book when `source` is `None`, as the
432    /// book holds them now. A section's own names are not in the
433    /// answer.
434    pub fn names(&self, source: Option<&str>) -> Names {
435        self.book.names(source)
436    }
437
438    /// The compiled styling behind the last update.
439    pub fn styles(&self) -> &StyleTree {
440        &self.styles
441    }
442
443    /// The sheets the styling was compiled from, as the host last
444    /// set them.
445    pub fn sheets(&self) -> Option<&Stylesheets> {
446        self.sheets.as_ref()
447    }
448
449    /// The images this session holds, by the url each was registered
450    /// under.
451    pub fn images(&self) -> &Assets {
452        self.assets.get()
453    }
454
455    /// The files [`add_font_file`](Session::add_font_file) handed
456    /// over, by the url a `@font-face` names them by.
457    pub fn font_files(&self) -> &dyn FontLoader {
458        &self.fonts
459    }
460
461    /// The faces this session lays out against.
462    ///
463    /// A painter that has to draw with the same file the shaper used
464    /// reaches the bytes through here; the display structure names
465    /// ids, and the registry is what they index.
466    pub fn fonts(&self) -> &FontRegistry {
467        self.registry.get()
468    }
469
470    /// How many times each stage has run.
471    pub fn stages(&self) -> Stages {
472        self.stages
473    }
474
475    /// Whether a section's lines survive an edit elsewhere in the
476    /// book. This goes false when the styling breaks a precondition,
477    /// either masters of different measures or inline content that
478    /// depends on pagination, and everything is re-broken instead.
479    pub fn reuses_sections(&self) -> bool {
480        self.section_local
481    }
