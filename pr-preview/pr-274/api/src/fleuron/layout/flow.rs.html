19/// What one page needs to know to ask the style tree for its master:
20/// the named page in force, and the situation the page is in.

22pub(super) struct PageSlot {
23    name: Option<String>,
24    /// The page a section opens on: `@page :first`.
25    first: bool,
26    /// Inserted to square the sheet: `@page :blank`.
27    pub(super) blank: bool,

30impl PageSlot {
31    pub(super) fn query(&self, side: Side) -> PageQuery<'_> {
32        PageQuery {
33            name: self.name.as_deref(),
34            situation: match (self.blank, self.first) {
35                (true, _) => Situation::Blank,
36                (false, true) => Situation::First(side),
37                (false, false) => Situation::Body(side),

311    /// Places the item of `fragments` that starts at `index`, and
312    /// answers where the next one starts.
313    ///
314    /// A book with nothing anchored places one fragment at a time.
315    /// One with an image on the page places a paragraph at a time,
316    /// because an image narrows the bands the paragraph is set in.
317    /// The whole of it is then broken again.
318    pub(super) fn item(&mut self, fragments: &[Fragment], index: usize) -> usize {
319        match fragments[index].reflow.as_ref() {
320            Some(reflow) if !self.anchored.is_empty() => {
321                let end = paragraph_end(fragments, index, reflow);
322                self.paragraph(&fragments[index..end], reflow);
323                end
324            }
325            _ => {
326                self.place(&fragments[index]);
327                index + 1
328            }
329        }
330    }

385    /// Places one fragment, ending columns and pages as its break
386    /// point demands.
387    pub(super) fn place(&mut self, fragment: &Fragment) {
388        // An anchor is not placed. It binds to the next fragment that
389        // is, and takes the page that fragment ends on.
390        if let Piece::Anchor(node) = fragment.piece {
391            self.pending_anchors.push(node);
392            return;
393        }
394        if let Piece::Row(row) = &fragment.piece {
395            if row.opens {
396                self.header.clear();
397            }
398            if row.head {
399                // A header set again is text the engine wrote itself,
400                // so the runs that name a node still tile it once.
401                let mut items = row.items.clone();
402                for item in &mut items {
403                    if let DrawItem::Text {
404                        origin,
405                        pseudo_element,
406                        ..
407                    } = item
408                    {
409                        *origin = None;
410                        *pseudo_element = None;
411                    }
412                }
413                shift(&mut items, fragment.offset.0, fragment.offset.1);
414                fade(&mut items, fragment.opacity);
415                self.header.push((fragment.height, items));
416            }
417        }
418        if let BreakPoint::Forced(wanted) = fragment.break_before {
419            match wanted {
420                // A tier that opens here opens columns of its own.
421                Break::Column if fragment.spanning != self.tier().spanning => {}
422                Break::Column => self.break_column(),
423                _ => {
424                    self.close(Ending::Short);
425                    if let Break::Side(side) = wanted {
426                        self.square_to(side);
427                    }
428                }
429            }
430        }
431        // Nothing laid on the page yet means the page the section is
432        // waiting for is this one. Either way the section had its
433        // chance to claim one.
434        if self.placed.is_empty()
435            && let Some(slot) = self.pending_slot.take()
436        {
437            self.slot = slot;
438            self.remaster();
439        }
440        self.pending_slot = None;
441        // A fragment that does not fit ends the column. Where it ends
442        // is the last point a break was allowed — which may be
443        // several fragments back, and may be nowhere, in which case
444        // the break falls here whatever the cascade wanted.
445        let mut forced = false;
446        loop {
447            if fragment.spanning != self.tier().spanning {
448                self.open_tier(fragment.spanning, fragment.lead);
449            }
450            let opening = self.opening();
451            let lead = self.lead(fragment);
452            let room = self.room(fragment);
453            if opening || self.cursor + lead + fragment.fixed + fragment.height <= room {
454                let headed = opening && repeats(fragment) && self.head_rows();
455                self.emit(fragment, lead);
456                // A page cannot end between the header rows and the
457                // row they were set again above.
458                if headed && let Some(placed) = self.placed.last_mut() {
459                    placed.break_before = BreakPoint::Forbidden;
460                }
461                return;
462            }
463            let cut = if forced || fragment.break_before != BreakPoint::Forbidden {
464                self.placed.len()
465            } else {
466                self.back_up().unwrap_or(self.placed.len())
467            };
468            self.carry(cut);
469            forced = true;
470        }
471    }

473    /// Paints one fragment onto the page being built.
474    fn emit(&mut self, fragment: &Fragment, lead: f32) {
475        let (x, y) = self.origin();
476        let top = self.cursor + lead + fragment.fixed;
477        let items = if self.paints {
478            self.paginator.fragment_items(fragment, x, y + top)
479        } else {
480            Vec::new()
481        };
482        let boxes = match &fragment.piece {
483            Piece::Row(row) if self.paints => {
484                let mut boxes = row.boxes.clone();
485                let (dx, dy) = fragment.offset;
486                shift_boxes(&mut boxes, x + dx + fragment.x, y + top + dy);
487                boxes
488            }
489            _ => Vec::new(),
490        };
491        self.cursor = top + fragment.height;
492        let notes = fragment
493            .notes
494            .clone()
495            .map(|notes| *notes)
496            .unwrap_or_default();
497        let column = self.column;
498        self.notes
499            .extend(notes.iter().map(|note| (column, note.clone(), 0usize)));
500        self.placed.push(Placed {
501            section: self.section,
502            column: self.column,
503            tier: self.tiers.len() - 1,
504            top,
505            height: fragment.height,
506            break_before: fragment.break_before,
507            items,
508            marks: fragment.marks.clone(),
509            decorations: fragment.decorations.clone(),
510            anchors: std::mem::take(&mut self.pending_anchors),
511            repeats: repeats(fragment),
512            boxes,
513            notes,
514        });
515    }

517    /// Sets the header rows of the table being placed at the foot of
518    /// what the column holds, and answers whether there were any.
519    fn head_rows(&mut self) -> bool {
520        if self.header.is_empty() {
521            return false;
522        }
523        let (x, y) = self.origin();
524        for (height, items) in &self.header {
525            let top = self.cursor;
526            let mut items = if self.paints {
527                items.clone()
528            } else {
529                Vec::new()
530            };
531            shift(&mut items, x, y + top);
532            self.placed.push(Placed {
533                section: self.section,
534                column: self.column,
535                tier: self.tiers.len() - 1,
536                top,
537                height: *height,
538                break_before: BreakPoint::Forbidden,
539                items,
540                marks: None,
541                decorations: None,
542                anchors: Vec::new(),
543                repeats: false,
544                boxes: Vec::new(),
545                notes: Vec::new(),
546            });
547            self.cursor = top + height;
548        }
549        true
550    }

580    /// Closes the tier being filled and opens one `above` under the
581    /// foot of its tallest column, for blocks that span the columns or
582    /// for blocks that do not. The columns of the tier it closes keep
583    /// the heights they filled to.
584    pub(super) fn open_tier(&mut self, spanning: bool, above: f32) {
585        let tier = self.tier();
586        let foot = self.placed[tier.start..]
587            .iter()
588            .map(|placed| placed.top + placed.height)
589            .fold(tier.top, f32::max);
590        // A tier with nothing in it yet becomes the one asked for,
591        // where it stands.
592        let (top, above) = if tier.start == self.placed.len() {
593            self.tiers.pop();
594            (tier.top, tier.above)
595        } else {
596            (foot + above, above)
597        };
598        self.tiers.push(Tier {
599            top,
600            above,
601            spanning,
602            start: self.placed.len(),
603        });
604        self.column = 0;
605        self.column_start = self.placed.len();
606        self.cursor = top;
607    }

609    /// The last place above the foot of the column where a break was
610    /// allowed. Never its head: a column that carries everything on
611    /// it into the next one makes no progress.
612    ///
613    /// The head of a tier's first column is different where a tier
614    /// stands above it. A break there falls between the two tiers,
615    /// and where it is forbidden, the search goes on up the last
616    /// column of the tier above. A heading that spans the columns
617    /// then moves to the next page with the prose under it.
618    fn back_up(&self) -> Option<usize> {
619        let allowed = |index: usize| self.placed[index].break_before == BreakPoint::Allowed;
620        let (mut head, mut end, mut tier) =
621            (self.column_start, self.placed.len(), self.tiers.len() - 1);
622        loop {
623            if let Some(index) = (head + 1..end).rev().find(|index| allowed(*index)) {
624                return Some(index);
625            }
626            if tier == 0 || head != self.tiers[tier].start {
627                return None;
628            }
629            if head < self.placed.len() && allowed(head) {
630                return Some(head);
631            }
632            tier -= 1;
633            end = head;
634            let last = self.placed[end - 1].column;
635            head = (self.tiers[tier].start..end)
636                .find(|index| self.placed[*index].column == last)
637                .unwrap_or(end - 1);
638        }
639    }

641    /// Ends the column at `cut`, carrying what was below into the
642    /// next one, which is the next page's first where the column that
643    /// ended was the page's last. Carried fragments move; they are
644    /// never measured again.
645    ///
646    /// A cut at the head of the tier being filled, or above it, ends
647    /// the page: the columns of a tier share a top, so the next one
648    /// has no more room than this one. What came out of more than one
649    /// tier opens a tier for each on the next page, one under the
650    /// other.
651    fn carry(&mut self, cut: usize) {
652        let crossing = cut <= self.tier().start;
653        let mut carried = self.placed.split_off(cut);
654        let moved: usize = carried.iter().map(|placed| placed.notes.len()).sum();
655        self.notes.truncate(self.notes.len() - moved);
656        let from: Vec<(f32, f32)> = carried
657            .iter()
658            .map(|placed| self.column_origin(placed.column))
659            .collect();
660        let tiers = self.tiers.clone();
661        if crossing {
662            let kept = self.tiers.iter().filter(|tier| tier.start < cut).count();
663            self.tiers.truncate(kept);
664            self.close(Ending::Full);
665        } else {
666            self.advance(Ending::Full);
667        }
668        let Some(head) = carried.first().map(|placed| placed.top) else {
669            return;
670        };
671        let mut tier = carried[0].tier;
672        if self.tier().spanning != tiers[tier].spanning {
673            self.open_tier(tiers[tier].spanning, 0.0);
674        }
675        // A table's body row carried to the head of the column has the
676        // header rows set above it.
677        if carried[0].repeats && self.head_rows() {
678            carried[0].break_before = BreakPoint::Forbidden;
679        }
680        // The carried group starts at the head of the fresh column,
681        // and the space that was above it there is dropped. A tier
682        // under it keeps the space it had under the tier above.
683        let mut down = self.cursor - head;
684        for (mut placed, (from_x, from_y)) in carried.into_iter().zip(from) {
685            if placed.tier != tier {
686                tier = placed.tier;
687                self.open_tier(tiers[tier].spanning, tiers[tier].above);
688                down = self.cursor - tiers[tier].top;
689            }
690            let (to_x, to_y) = self.origin();
691            placed.top += down;
692            placed.column = self.column;
693            self.notes.extend(
694                placed
695                    .notes
696                    .iter()
697                    .map(|note| (placed.column, note.clone(), 0usize)),
698            );
699            placed.tier = self.tiers.len() - 1;
700            shift(&mut placed.items, to_x - from_x, to_y - from_y + down);
701            shift_boxes(&mut placed.boxes, to_x - from_x, to_y - from_y + down);
702            self.cursor = placed.top + placed.height;
703            self.placed.push(placed);
704        }
705    }

707    /// Moves to the next column, or ends the page when the column
708    /// that filled was its last. A tier of blocks that span the
709    /// columns is one column.
710    fn advance(&mut self, ending: Ending) {
711        let tier = self.tier();
712        if !tier.spanning && self.column + 1 < self.columns {
713            self.column += 1;
714            self.cursor = tier.top;
715            self.column_start = self.placed.len();
716        } else {
717            self.close(ending);
718        }
719    }

730    /// Ends the page being built, if anything is on it.
731    ///
732    /// What the page's fragments set takes effect here, not where
733    /// they were placed: a fragment moved onto the next page sets
734    /// its strings there instead, so a page's furniture only ever
735    /// reads what stood on it.
736    fn close(&mut self, ending: Ending) {
737        if self.placed.is_empty() {
738            return;
739        }
740        let opened = self.strings.clone();
741        let mut reset = None;
742        let mut placed = std::mem::take(&mut self.placed);
743        let within = self.containers(&placed);
744        self.land(&placed, within);
745        let areas = self.notes_areas(&placed);
746        let under = areas
747            .iter()
748            .map(|(_, area)| area.height)
749            .fold(0.0, f32::max);
750        if self.paints && ending == Ending::Short {
751            self.align(&mut placed, under);
752        }
753        // Backgrounds, borders, column rules and images go in front
754        // of the page's text, which is where the flow puts them. What
755        // the sheet asked to paint elsewhere is sorted into place
756        // once the page is assembled.
757        let mut items = Vec::new();
758        if self.paints {
759            items = self.page_background(self.pages.len(), &self.slot);
760            items.append(&mut self.decorate(&placed));
761            items.append(&mut self.rules(&placed));
762            items.append(&mut self.anchored_items());
763            let page = self.pages.len() as u32;
764            for (node, area) in self.anchored_areas() {
765                self.boxes.push((node, PageBox { page, ..area }));
766            }
767        }
768        self.notes_closed(&areas);
769        let index = self.pages.len();
770        let mut sections: Vec<NodeId> = Vec::new();
771        for placed in placed {
772            if sections.last() != Some(&placed.section) {
773                sections.push(placed.section);
774            }
775            let landed = placed
776                .marks
777                .as_ref()
778                .is_some_and(|marks| !marks.targets.is_empty())
779                .then(|| self.placed_area(&placed, index));
780            for (node, area) in placed.boxes {
781                let page = index as u32;
782                self.boxes.push((node, PageBox { page, ..area }));
783            }
784            if let Some(marks) = placed.marks {
785                for (name, value) in marks.strings {
786                    self.strings.insert(name, value);
787                }
788                reset = reset.or(marks.page_number);
789                if let Some(area) = landed {
790                    for id in marks.targets {
791                        self.targets.entry(id).or_insert(area);
792                    }
793                }
794            }
795            items.extend(placed.items);
796        }
797        // The notes go under the text of the column they were
798        // written in, so they are painted after it and the baselines
799        // of a column rise from its first line to its last note.
800        if self.paints {
801            let geometry = self.paginator.master(index, &self.slot).geometry;
802            let page = index as u32;
803            for (column, area) in &areas {
804                let (mut painted, boxes) = self
805                    .paginator
806                    .area_items(area, geometry.column_origin(*column));
807                items.append(&mut painted);
808                self.boxes.extend(
809                    boxes
810                        .into_iter()
811                        .map(|(node, area)| (node, PageBox { page, ..area })),
812                );
813            }
814        }
815        let mut page = self.paginator.blank_page(&self.slot);
816        page.side = Side::of_number(self.pages.len() as u32 + 1);
817        page.sections = sections;
818        page.items = items;
819        page.sort_by_layer();
820        let content_items = page.items.len();
821        self.pages.push(page);
822        self.infos.push(PageInfo {
823            slot: self.slot.clone(),
824            strings: opened,
825            reset,
826            content_items,
827        });
828        self.slot = self.pending_slot.take().unwrap_or(PageSlot {
829            first: false,
830            ..self.slot.clone()
831        });
832        self.cursor = 0.0;
833        self.column = 0;
834        self.column_start = 0;
835        self.tiers = vec![Tier::head(self.tier().spanning)];
836        self.remaster();
837        if let Some(settling) = self.settling.as_mut() {
838            settling.opened();
839        }
840    }

842    /// Moves what the page being closed holds down its content box, as
843    /// `align-content` asks, with the columns and boxes painted over
844    /// it. The margin boxes are painted later and stay where they are.
845    ///
846    /// A page whose prose wraps around a box stays at the top, because
847    /// its lines were broken beside the box where they stand.
848    fn align(&mut self, placed: &mut [Placed], under: f32) {
849        let index = self.pages.len();
850        let share = match self
851            .paginator
852            .master(index, &self.slot)
853            .geometry
854            .align_content
855        {
856            AlignContent::Start => return,
857            AlignContent::Center => 0.5,
858            AlignContent::End => 1.0,
859        };
860        if self.anchored.wraps(index) {
861            return;
862        }
863        let foot = placed
864            .iter()
865            .map(|placed| placed.top + placed.height)
866            .fold(0.0, f32::max);
867        let down = (self.height - under - foot).max(0.0) * share;
868        for placed in placed.iter_mut() {
869            placed.top += down;
870            shift(&mut placed.items, 0.0, down);
871            shift_boxes(&mut placed.boxes, 0.0, down);
872        }
873        for tier in &mut self.tiers {
874            tier.top += down;
875        }
876    }

879    fn square_to(&mut self, side: Side) {

884                blank: true,

893                slot: blank,
894                strings: self.strings.clone(),

1045    /// The border box every block on the page being closed takes
1046    /// there, with the column it is in, column by column and tier by
1047    /// tier.
1048    ///
1049    /// A column boundary cuts a block the way a page boundary does,
1050    /// so each column resolves on its own and what is still open at
1051    /// the foot of one carries into the next.
1052    fn resolve(&mut self, placed: &[Placed]) -> Vec<(u32, Painted)> {
1053        let mut resolved = Vec::new();
1054        let mut start = 0;
1055        let region = |placed: &Placed| (placed.tier, placed.column);
1056        for index in 1..=placed.len() {
1057            if index < placed.len() && region(&placed[index]) == region(&placed[start]) {
1058                continue;
1059            }
1060            let (tier, column) = region(&placed[start]);
1061            let top = self.tiers[tier].top;
1062            resolved.extend(
1063                self.resolve_column(&placed[start..index], top)
1064                    .into_iter()
1065                    .map(|painted| (column, painted)),
1066            );
1067            start = index;
1068        }
1069        resolved
1070    }

1072    /// The same over one column of one tier, whose top is `top`.
1073    ///
1074    /// A block whose first fragment landed in this column has its top
1075    /// edge here, and one whose last fragment did has its bottom; the
1076    /// ranges are contiguous, so a block with neither covers every
1077    /// fragment the column holds. What is still open when the column
1078    /// closes carries into the next.
1079    fn resolve_column(&mut self, placed: &[Placed], top: f32) -> Vec<Painted> {
1080        let mut boxes: Vec<Painted> = Vec::new();
1081        let mut open: Vec<usize> = Vec::new();
1082        for decoration in self.carried.drain(..) {
1083            open.push(boxes.len());
1084            boxes.push(Painted {
1085                decoration,
1086                top,
1087                bottom: 0.0,
1088                cut_above: true,
1089                cut_below: true,
1090            });
1091        }
1092        for entry in placed {
1093            let Some(decorations) = &entry.decorations else {
1094                continue;
1095            };
1096            for decoration in &decorations.opens {
1097                open.push(boxes.len());
1098                boxes.push(Painted {
1099                    top: entry.top - decoration.above,
1100                    decoration: decoration.clone(),
1101                    bottom: 0.0,
1102                    cut_above: false,
1103                    cut_below: true,
1104                });
1105            }
1106            for _ in 0..decorations.closes {
1107                let Some(index) = open.pop() else { continue };
1108                boxes[index].bottom = entry.top + entry.height + boxes[index].decoration.below;
1109                boxes[index].cut_below = false;
1110            }
1111        }
1112        let last = placed
1113            .last()
1114            .map(|entry| entry.top + entry.height)
1115            .unwrap_or(top);
1116        for index in open {
1117            boxes[index].bottom = last;
1118            self.carried.push(boxes[index].decoration.clone());
1119        }
1120        boxes
1121    }

1123    /// The rules down the gutters of the page being closed: in each
1124    /// tier of columns, one down each gutter the flow filled past,
1125    /// over the height of the taller of the two columns it divides.
1126    ///
1127    /// A tier the flow left in one column paints no rule, and neither
1128    /// does a tier of blocks that span the columns.
1129    fn rules(&self, placed: &[Placed]) -> Vec<DrawItem> {
1130        let geometry = self.paginator.master(self.pages.len(), &self.slot).geometry;
1131        let width = geometry.columns.rule.used();
1132        if width <= 0.0 || self.columns < 2 {
1133            return Vec::new();
1134        }
1135        let (_, top) = geometry.content_origin();
1136        let color = self.paginator.styles.root().color;
1137        let mut items = Vec::new();
1138        for (index, tier) in self.tiers.iter().enumerate() {
1139            if tier.spanning {
1140                continue;
1141            }
1142            let mut feet = vec![tier.top; self.columns as usize];
1143            let mut filled = vec![false; self.columns as usize];
1144            for entry in placed.iter().filter(|entry| entry.tier == index) {
1145                let column = entry.column as usize;
1146                feet[column] = feet[column].max(entry.top + entry.height);
1147                filled[column] = true;
1148            }
1149            items.extend(
1150                (1..self.columns as usize)
1151                    .filter(|column| filled[*column])
1152                    .map(|column| {
1153                        let gutter = geometry.column_origin(column as u32).0 - geometry.columns.gap;
1154                        DrawItem::Rect {
1155                            x: gutter + (geometry.columns.gap - width) / 2.0,
1156                            y: top + tier.top,
1157                            w: width,
1158                            h: feet[column - 1].max(feet[column]) - tier.top,
1159                            color,
1160                            layer: 0,
1161                        }
1162                    }),
1163            );
1164        }
1165        items
1166    }

1189/// One decorated block resolved against one page: the border box it
1190/// takes there, and which of its edges the page boundary cut.
1191pub(super) struct Painted {
1192    pub(super) decoration: Decoration,
1193    /// Top of the border box, from the page content box's top.
1194    pub(super) top: f32,
1195    /// Its bottom, the same way.
1196    pub(super) bottom: f32,
1197    /// Whether the block began on an earlier page.
1198    pub(super) cut_above: bool,
1199    /// Whether it goes on to the next one.
1200    pub(super) cut_below: bool,
1201}
1202
1203impl Painted {
1204    /// The border box on the page, as left, top, width and height:
1205    /// `origin` is the page's content box.
1206    pub(super) fn border_box(&self, origin: (f32, f32)) -> (f32, f32, f32, f32) {
1207        let (dx, dy) = self.decoration.offset;
1208        (
1209            origin.0 + self.decoration.x + dx,
1210            origin.1 + self.top + dy,
1211            self.decoration.width,
1212            self.bottom - self.top,
1213        )
1214    }
1215
1216    /// What this box paints, background first: `origin` is the page's
1217    /// content box.
1218    pub(super) fn items(&self, origin: (f32, f32)) -> Vec<DrawItem> {
1219        let (x, y, w, h) = self.border_box(origin);
1220        if !self.decoration.paints || w <= 0.0 || h <= 0.0 {
1221            return Vec::new();
1222        }
1223        let layer = self.decoration.layer;
1224        // `slice` leaves the two edges the break made open; `clone`
1225        // closes them.
1226        let closed = self.decoration.cloned;
1227        let (open_above, open_below) = (self.cut_above && !closed, self.cut_below && !closed);
1228        let border = self.decoration.border;
1229        let top = if open_above { 0.0 } else { border.top };
1230        let bottom = if open_below { 0.0 } else { border.bottom };
1231        // The box goes on past an open edge, so the corners on it are
1232        // square.
1233        let mut radii = self.decoration.radius.resolve(w, h);
1234        if open_above {
1235            (radii.top_left, radii.top_right) = (Radius::SQUARE, Radius::SQUARE);
1236        }
1237        if open_below {
1238            (radii.bottom_left, radii.bottom_right) = (Radius::SQUARE, Radius::SQUARE);
1239        }
1240        let widths = Edges {
1241            top,
1242            bottom,
1243            ..border
1244        };
1245        let mut items = box_items(
1246            x,
1247            y,
1248            w,
1249            h,
1250            radii,
1251            widths,
1252            self.decoration.colors,
1253            &self.decoration.backdrop,
1254            layer,
1255        );
1256        fade(&mut items, self.decoration.opacity);
1257        items
1258    }
1259}

1261/// What one box paints, backdrop first and border over it. `border`
1262/// is zero on an edge the box does not paint, and `radii` is square
1263/// at a corner it goes on past.
1264#[allow(clippy::too_many_arguments)]
1265pub(super) fn box_items(
1266    x: f32,
1267    y: f32,
1268    w: f32,
1269    h: f32,
1270    radii: Corners,
1271    border: Edges,
1272    colors: Edges<Color>,
1273    backdrop: &Backdrop,
1274    layer: i32,
1275) -> Vec<DrawItem> {
1276    let mut items = backdrop.rounded(x, y, w, h, radii, layer);
1277    if !radii.is_square() {
1278        items.extend(rings(x, y, w, h, radii, border, colors, layer));
1279        return items;
1280    }
1281    // The corners fall to the horizontal edges: a filled rect is all
1282    // a square box needs, and a mitre is a path.

1373/// The border of a rounded box, as rings. Where every edge that is
1374/// drawn has one colour, the border is one ring. Otherwise each edge
1375/// is a ring of its own, as wide as that edge and nothing on the other
1376/// three.
1377#[allow(clippy::too_many_arguments)]
1378fn rings(
1379    x: f32,
1380    y: f32,
1381    w: f32,
1382    h: f32,
1383    radii: Corners,
1384    widths: Edges,
1385    colors: Edges<Color>,
1386    layer: i32,
1387) -> Vec<DrawItem> {
1388    let ring = |ring: Edges, color: Color| DrawItem::Rounded {
1389        x,
1390        y,
1391        w,
1392        h,
1393        radii,
1394        ring,
1395        color,
1396        layer,
1397    };
1398    let none = Edges::all(0.0);
1399    let edges = [
1400        (
1401            Edges {
1402                top: widths.top,
1403                ..none
1404            },
1405            colors.top,
1406        ),
1407        (
1408            Edges {
1409                right: widths.right,
1410                ..none
1411            },
1412            colors.right,
1413        ),
1414        (
1415            Edges {
1416                bottom: widths.bottom,
1417                ..none
1418            },
1419            colors.bottom,
1420        ),
1421        (
1422            Edges {
1423                left: widths.left,
1424                ..none
1425            },
1426            colors.left,
1427        ),
1428    ];
1429    let drawn: Vec<(Edges, Color)> = edges
1430        .into_iter()
1431        .filter(|(edge, _)| edge.top + edge.right + edge.bottom + edge.left > 0.0)
1432        .collect();

1434        None => Vec::new(),
