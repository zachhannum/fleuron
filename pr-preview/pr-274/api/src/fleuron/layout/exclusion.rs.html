19impl Paginator<'_> {
20    /// The page each of `all` lands on, found by a flow that paints
21    /// nothing. `fragments` gives the fragments of the section at an
22    /// index of `book.sections`.
23    ///
24    /// The flow puts a box on a page when a page closes with the
25    /// anchor of that box on it. A page that a box changes is laid out
26    /// again, from the item that was being placed when that page
27    /// opened. A box that pushes its own anchor off its page goes on
28    /// the next page instead, and moves no more.
29    ///
30    /// A section stays in hand while a page that opened in it can
31    /// still be laid out again.
32    pub(super) fn settle<'f>(
33        &self,
34        book: &Book,
35        all: Vec<Anchored>,
36        mut fragments: impl FnMut(usize) -> Cow<'f, [Fragment]>,
37    ) -> AnchoredBoxes {
38        let mut flow = Flow::settling(
39            self,
40            AnchoredBoxes {
41                all,
42                by_page: BTreeMap::new(),
43            },
44        );
45        let mut held: BTreeMap<usize, Cow<'f, [Fragment]>> = BTreeMap::new();
46        let (mut section, mut item, mut opened) = (0, 0, false);
47        loop {
48            if let Some((at, from)) = flow.again() {
49                (section, item, opened) = (at, from, true);
50            }
51            let Some(content) = book.sections.get(section) else {
52                flow.close_book();
53                if flow.settling.as_ref().is_some_and(|s| s.redo.is_some()) {
54                    continue;
55                }
56                break;
57            };
58            let oldest = flow.oldest_section().unwrap_or(section);
59            held.retain(|index, _| *index >= oldest);
60            let built = held.entry(section).or_insert_with(|| fragments(section));
61            if !opened {
62                flow.open_section(content);
63                opened = true;
64            }
65            if item >= built.len() {
66                (section, item, opened) = (section + 1, 0, false);
67                continue;
68            }
69            flow.mark(section, item);
70            item = flow.item(built, item);
71        }
72        flow.anchored
73    }

75    /// The images and blocks the sheet anchored to the page, in
76    /// document order.
77    ///
78    /// A box inside a block that is anchored as well lands on the page
79    /// that block lands on.
80    ///
81    /// `within` is the containing block of the box at each index, as
82    /// the settling pass found it. A box with none of its own breaks
83    /// its lines to what the page area leaves it.
84    pub(super) fn anchored_boxes(&self, book: &Book, within: &[Option<Rect>]) -> Vec<Anchored> {
85        fn walk<'b>(
86            paginator: &Paginator,
87            boxes: impl IntoIterator<Item = Child<'b>>,
88            source: Option<&str>,
89            around: Around,
90            within: &[Option<Rect>],
91            anchored: &mut Vec<Anchored>,
92        ) {
93            let styles = paginator.styles;
94            for child in boxes {
95                let id = child.id();
96                let style = styles.style(id);
97                let lifted = style.position == Position::Absolute;
98                let opacity = around.opacity * style.opacity;
99                let node = if lifted {
100                    let node = around.node.unwrap_or(id);
101                    let area = within.get(anchored.len()).copied().flatten();
102                    anchored.extend(match child {
103                        Child::Block(Block::Image { url, position, .. }) => paginator
104                            .anchored_image(
105                                node,
106                                id,
107                                style,
108                                url,
109                                origin(source, *position),
110                                around,
111                            ),
112                        _ => Some(paginator.anchored_block(
113                            node,
114                            child,
115                            style,
116                            source,
117                            around,
118                            area.map(|area| area.w),
119                        )),
120                    });
121                    Some(node)
122                } else {
123                    around.node
124                };
125                let host = Around {
126                    node,
127                    opacity,
128                    ancestor: match style.position {
129                        Position::Static => around.ancestor,
130                        Position::Relative | Position::Absolute => Some(id),
131                    },
132                };
133                let Child::Block(block) = child else {
134                    continue;
135                };
136                let position = block_position(block);
137                let pseudo = |which| {
138                    styles
139                        .pseudo_element(id, which)
140                        .map(|id| Child::Generated(id, position))
141                };
142                match block {
143                    Block::Blockquote { blocks, .. } => walk(
144                        paginator,
145                        children(styles, id, blocks, position),
146                        source,
147                        host,
148                        within,
149                        anchored,
150                    ),
151                    Block::List { items, .. } => {
152                        walk(
153                            paginator,
154                            pseudo(PseudoElement::Before),
155                            source,
156                            host,
157                            within,
158                            anchored,
159                        );
160                        for item in items {
161                            walk(
162                                paginator,
163                                children(styles, item.id, &item.blocks, item.position),
164                                source,
165                                host,
166                                within,
167                                anchored,
168                            );
169                        }
170                        walk(
171                            paginator,
172                            pseudo(PseudoElement::After),
173                            source,
174                            host,
175                            within,
176                            anchored,
177                        );
178                    }
179                    Block::Table { head, body, .. } => {
180                        walk(
181                            paginator,
182                            pseudo(PseudoElement::Before),
183                            source,
184                            host,
185                            within,
186                            anchored,
187                        );
188                        for blocks in cell_blocks(head, body) {
189                            walk(
190                                paginator,
191                                blocks.iter().map(Child::Block),
192                                source,
193                                host,
194                                within,
195                                anchored,
196                            );
197                        }
198                        walk(
199                            paginator,
200                            pseudo(PseudoElement::After),
201                            source,
202                            host,
203                            within,
204                            anchored,
205                        );
206                    }
207                    // An image against the page is placed whole, with no
208                    // children in it.
209                    Block::Image { .. } if lifted => {}
210                    _ => walk(
211                        paginator,
212                        children(styles, id, &[], position),
213                        source,
214                        host,
215                        within,
216                        anchored,
217                    ),
218                }
219            }
220        }
221        let mut anchored = Vec::new();
222        for section in &book.sections {
223            walk(
224                self,
225                children(self.styles, section.id, &section.blocks, section.position),
226                section.source.as_deref(),
227                Around {
228                    node: None,
229                    opacity: self.styles.style(section.id).opacity,
230                    ancestor: None,
231                },
232                within,
233                &mut anchored,
234            );
235        }
236        anchored
237    }

247        id: NodeId,
248        style: &ComputedStyle,

292    /// One block anchored to the page, which lands on the page `node`
293    /// lands on, laid out on its own.
294    ///
295    /// Its lines break to the width that its insets leave of its
296    /// containing block, which is `area` wide, or of the page area
297    /// where it has none. With both insets set, that is the width
298    /// between them. With one set, it runs from that inset to the far
299    /// edge. With neither set, it is `geometry.measure()`.
300    fn anchored_block(
301        &self,
302        node: NodeId,
303        child: Child<'_>,
304        style: &ComputedStyle,
305        source: Option<&str>,
306        around: Around,
307        area: Option<f32>,
308    ) -> Anchored {
309        let geometry = self.styles.default_page().geometry;
310        let area = area.unwrap_or_else(|| geometry.content_size().0);
311        let (left, right) = (
312            style.inset.left.resolve(area),
313            style.inset.right.resolve(area),
314        );
315        let width = match (left, right) {
316            (None, None) => geometry.measure(),
317            _ => (area - left.unwrap_or(0.0) - right.unwrap_or(0.0)).max(0.0),
318        };
319        let mut builder = Builder::new(self, source);
320        builder.lifted = Some(child.id());
321        builder.opacity = around.opacity;
322        builder.blocks([child], 0.0, width);
323        let stacked = builder.stack();
324        if !stacked.notes.is_empty() {
325            self.warn(
326                "A note was written inside a block against the page. The note is left \
327                 out."
328                    .to_string(),
329                None,
330            );
331        }
332        let margin = style.margin;
333        let inner = (
334            (width - margin.inline()).max(0.0),
335            (stacked.height - margin.top - margin.bottom).max(0.0),
336        );
337        Anchored {
338            node,
339            ancestor: around.ancestor,
340            container: None,
341            measured: match (left, right) {
342                (None, None) => None,
343                _ => Some(area),
344            },
345            width,
346            height: stacked.height,
347            inset: style.inset,
348            wrap: style.wrap_flow,
349            shape: self.shape(style, None, inner),
350            layer: style.z_index,
351            opacity: style.opacity * around.opacity,
352            paint: Paint::Block(stacked.items, stacked.boxes),
353        }
354    }

356    /// The contour one anchored box's prose keeps clear of, in the
357    /// coordinates of the box its insets place. `size` is the box
358    /// inside its margins.
359    ///
360    /// `auto` is what the trace stage left, laid over the image
361    /// inside its margins. A block, and an image with no alpha from the
362    /// tracer, give their box. The percentages of a polygon are
363    /// percentages of the whole box, margins included.
364    fn shape(&self, style: &ComputedStyle, asset: Option<u32>, size: (f32, f32)) -> Option<Shape> {
365        // The one predicate the trace stage keys on as well, so that
366        // what is traced and what is read cannot drift apart.
367        if !style.excludes() {
368            return None;
369        }
370        let margin = style.margin;
371        let (width, height) = size;
372        let rings = match &style.shape_outside {
373            ShapeOutside::None => return None,
374            ShapeOutside::Auto => self
375                .contours
376                .get(asset?)?
377                .rings
378                .iter()
379                .map(|ring| {
380                    ring.iter()
381                        .map(|[x, y]| [margin.left + x * width, margin.top + y * height])
382                        .collect()
383                })
384                .collect(),
385            ShapeOutside::Polygon(points) => {
386                let box_ = (width + margin.inline(), height + margin.top + margin.bottom);
387                vec![
388                    points
389                        .iter()
390                        .map(|point| [point.x.to_points(box_.0), point.y.to_points(box_.1)])
391                        .collect(),
392                ]
393            }
394        };
395        Some(Shape {
396            rings,
397            margin: style.shape_margin,
398        })
399    }

487impl Anchored {
488    /// What it keeps to itself on a page of this geometry: its margin
489    /// box.
490    ///
491    /// An inset is a distance from the containing block: the padding
492    /// box of the nearest block around it that `position` moved, or
493    /// the page area, the box inside the margins, where there is no
494    /// such block. A negative inset reaches outside the containing
495    /// block. A percentage is a percentage of the width or the height
496    /// of the containing block. Where both insets of an axis are set,
497    /// the leading inset places the box. Where neither is set, the box
498    /// sits at the edge of the containing block.
499    fn rect(&self, geometry: PageGeometry) -> Rect {
500        let area = self.container.unwrap_or_else(|| page_area(geometry));
501        let (left, top) = (area.x, area.y);
502        let (width, height) = (area.w, area.h);
503        let (w, h) = (self.width, self.height);
504        let place =
505            |start: Option<f32>, end: Option<f32>, origin: f32, available: f32, size: f32| match (
506                start, end,
507            ) {
508                (Some(start), _) => origin + start,
509                (None, Some(end)) => origin + available - end - size,
510                (None, None) => origin,
511            };
512        Rect {
513            x: place(
514                self.inset.left.resolve(width),
515                self.inset.right.resolve(width),
516                left,
517                width,
518                w,
519            ),
520            y: place(
521                self.inset.top.resolve(height),
522                self.inset.bottom.resolve(height),
523                top,
524                height,
525                h,
526            ),
527            w,
528            h,
529        }
530    }

532    /// What it paints on a page of this geometry.
533    pub(super) fn items(&self, geometry: PageGeometry) -> Vec<DrawItem> {
534        let rect = self.rect(geometry);
535        match &self.paint {
536            Paint::Image {
537                asset,
538                x,
539                y,
540                width,
541                height,
542                ..
543            } => vec![DrawItem::Image {

546                w: *width,

548                asset: *asset,
549                alpha: crate::pages::fade(255, self.opacity),
550                layer: self.layer,

552            Paint::Block(items, _) => {
553                let mut items = items.clone();

558    }

561    /// image's own, or those of the blocks a block is made of.
562    pub(super) fn boxes(&self, geometry: PageGeometry) -> Vec<(NodeId, PageBox)> {
563        let rect = self.rect(geometry);
564        match &self.paint {
565            Paint::Image {
566                id,
567                x,
568                y,
569                width,
570                height,
571                ..
572            } => vec![(
573                *id,
574                PageBox {
575                    page: 0,
576                    x: rect.x + x,
577                    y: rect.y + y,
578                    width: *width,
579                    height: *height,
580                },
581            )],
582            Paint::Block(_, boxes) => {
583                let mut boxes = boxes.clone();
584                shift_boxes(&mut boxes, rect.x, rect.y);
585                boxes
586            }
587        }
588    }

591/// The images and blocks one book anchors, and the page each one
592/// landed on.
593///
594/// The flow gives the page that a box falls on. The sheet gives the
595/// place of the box on that page. [`Paginator::settle`] finds the
596/// pages, and the flow that paints keeps them.
597#[derive(Debug, Default)]
598pub(crate) struct AnchoredBoxes {
599    pub(super) all: Vec<Anchored>,
600    /// Which boxes a page carries, by page index, each list in
601    /// document order.
602    pub(super) by_page: BTreeMap<usize, Vec<usize>>,
603}

605impl AnchoredBoxes {
606    /// Whether the book anchors nothing.
607    pub(super) fn is_empty(&self) -> bool {
608        self.all.is_empty()
609    }
610
611    /// Whether any box measures its insets from a block rather than
612    /// from the page area.
613    pub(super) fn nested(&self) -> bool {
614        self.all.iter().any(|one| one.ancestor.is_some())
615    }
616
617    /// The containing block of each box, where the settling pass
618    /// found one that leaves a box a different width from the one its
619    /// lines were broken to. `None` where every box is as wide as it
620    /// was laid out.
621    pub(super) fn narrowed(&self) -> Option<Vec<Option<Rect>>> {
622        self.all
623            .iter()
624            .any(|one| match (one.measured, one.container) {
625                (Some(measured), Some(container)) => (container.w - measured).abs() > 1e-4,
626                _ => false,
627            })
628            .then(|| self.all.iter().map(|one| one.container).collect())
629    }
630
631    /// Takes the boxes of a second layout, which broke the lines of
632    /// each one to the width its containing block leaves it. The
633    /// containing blocks the settling pass found are kept.
634    pub(super) fn relaid(&mut self, boxes: Vec<Anchored>, within: Vec<Option<Rect>>) {
635        self.all = boxes;
636        for (one, container) in self.all.iter_mut().zip(within) {
637            one.container = container;
638        }
639    }
640
641    /// Whether the prose of the page at `index` wraps around a box.
642    pub(super) fn wraps(&self, index: usize) -> bool {
643        self.by_page
644            .get(&index)
645            .is_some_and(|boxes| boxes.iter().any(|at| self.all[*at].wrap != WrapFlow::Auto))
646    }

692impl Flow<'_, '_> {
693    /// Keeps the flow as it stands before the item at `item` of the
694    /// section at `section` is placed.
695    pub(super) fn mark(&mut self, section: usize, item: usize) {
696        if self.settling.is_none() {
697            return;
698        }
699        let checkpoint = Rc::new(self.checkpoint(section, item));
700        let Some(settling) = self.settling.as_mut() else {
701            return;
702        };
703        settling.page.get_or_insert_with(|| checkpoint.clone());
704        settling.before = Some(checkpoint);
705    }
706
707    /// The section that a page can be laid out again from.
708    pub(super) fn oldest_section(&self) -> Option<usize> {
709        let settling = self.settling.as_ref()?;
710        settling.page.as_ref().map(|page| page.section)
711    }

713    /// Puts the flow back where a page that a box changed opened, and
714    /// answers the item to place from there.
715    pub(super) fn again(&mut self) -> Option<(usize, usize)> {
716        let checkpoint = self.settling.as_mut()?.redo.take()?;
717        self.restore(&checkpoint);
718        let settling = self.settling.as_mut()?;
719        settling.page = Some(checkpoint.clone());
720        settling.before = Some(checkpoint.clone());
721        Some((checkpoint.section, checkpoint.item))
722    }

724    /// Moves boxes onto the page that is closing with `placed` on it,
725    /// and off it, and asks for the page again where that changes a
726    /// line.
727    ///
728    /// A box goes on the page that closes with its anchor on it. A box
729    /// on this page whose anchor is not on it pushed that anchor to a
730    /// later page, so it comes off. Once a page is asked for again,
731    /// nothing moves until the flow is back at that page.
732    pub(super) fn land(&mut self, placed: &[Placed], within: BTreeMap<NodeId, PageBox>) {
733        let index = self.pages.len();
734        let Some(settling) = self.settling.as_mut() else {
735            return;
736        };
737        if settling.redo.is_some() {
738            return;
739        }
740        let mut landed = Vec::new();
741        let mut pushed = Vec::new();
742        let bound: BTreeSet<NodeId> = placed
743            .iter()
744            .flat_map(|placed| placed.anchors.iter().copied())
745            .collect();
746        let mut again = false;
747        for (at, anchored) in self.anchored.all.iter().enumerate() {
748            let here = bound.contains(&anchored.node);
749            let landing = match settling.landings[at] {
750                Landing::Waiting if here => Landing::On(index),
751                Landing::On(page) if page == index && !here => Landing::Pushed {
752                    from: index + 1,
753                    passed: false,
754                },
755                Landing::Pushed { from, passed } if index >= from && (passed || here) => {
756                    Landing::Settled(index)
757                }
758                Landing::Pushed { from, .. } if here => Landing::Pushed { from, passed: true },
759                landing => landing,
760            };
761            let before = std::mem::replace(&mut settling.landings[at], landing);
762            let boxes = self.anchored.by_page.entry(index).or_default();
763            match (before, landing) {
764                (Landing::On(_), Landing::Pushed { .. }) => {
765                    boxes.retain(|other| *other != at);
766                    pushed.push(at);
767                }
768                (_, Landing::On(_) | Landing::Settled(_)) if before != landing => {
769                    let slot = boxes.partition_point(|other| *other < at);
770                    boxes.insert(slot, at);
771                    landed.push(at);
772                }
773                _ => continue,
774            }
775            again |= anchored.wrap != WrapFlow::Auto;
776        }
777        self.anchored.by_page.retain(|_, boxes| !boxes.is_empty());
778        if again {
779            settling.redo.clone_from(&settling.page);
780        }
781        self.contain(within, &landed, &pushed);
782    }

784    /// Gives each box that landed on the page being closed the
785    /// containing block its insets measure from: the padding box, on
786    /// this page, of the nearest block around it that `position`
787    /// moved. `within` is the border box of each block of the flow on
788    /// this page.
789    ///
790    /// A box keeps the first containing block it is given, the way it
791    /// keeps its page. A box that wraps prose moves the lines of the
792    /// page, and the block around it moves with them. A box that came
793    /// off this page gives its containing block up with it.
794    ///
795    /// A block against the page is the containing block of the boxes
796    /// inside it. It lands on the page its own anchor landed on, which
797    /// is the page the boxes inside it land on, so its own box is here
798    /// to measure from.
799    fn contain(
800        &mut self,
801        mut within: BTreeMap<NodeId, PageBox>,
802        landed: &[usize],
803        pushed: &[usize],
804    ) {
805        if !self.nested {
806            return;
807        }
808        for at in pushed {
809            self.anchored.all[*at].container = None;
810        }
811        let index = self.pages.len();
812        let geometry = self.paginator.master(index, &self.slot).geometry;
813        for at in landed {
814            let ancestor = self.anchored.all[*at].ancestor;
815            if let Some(node) = ancestor
816                && self.anchored.all[*at].container.is_none()
817                && let Some(area) = within.get(&node)
818            {
819                let border = self.paginator.styles.style(node).border.widths();
820                self.anchored.all[*at].container = Some(Rect {
821                    x: area.x + border.left,
822                    y: area.y + border.top,
823                    w: (area.width - border.inline()).max(0.0),
824                    h: (area.height - border.top - border.bottom).max(0.0),
825                });
826            }
827            for (node, area) in self.anchored.all[*at].boxes(geometry) {
828                within.entry(node).or_insert(area);
829            }
830        }
831    }

833    /// Places one paragraph, and breaks it again where an image
834    /// narrows the bands it is set in.
835    ///
836    /// This is the one thing the flow measures. Everywhere else a
837    /// fragment arrives with its box decided. Here the box depends on
838    /// where the paragraph lands. The flow breaks the paragraph again
839    /// through `LineLayout`, and what comes back is what it places.
840    ///
841    /// A page boundary inside the paragraph starts it over. The flow
842    /// takes the lines that crossed off the fresh column and sets the
843    /// rest of the paragraph again from where they now sit. The page
844    /// index only rises, so the flow breaks the paragraph at most
845    /// twice on any page it tries.
846    pub(super) fn paragraph(&mut self, original: &[Fragment], reflow: &Reflow) {
847        let mut set: Cow<'_, [Fragment]> = Cow::Borrowed(original);
848        let mut ends: Cow<'_, [usize]> = Cow::Borrowed(&reflow.ends);
849        let mut at = 0;
850        // Whether what is in hand was broken beside an image. The
851        // lines a section arrives with fit any page. Lines broken
852        // against a notch fit the page they were broken on, so a page
853        // boundary under them is a break to do again.
854        let mut narrowed = false;
855        while at < set.len() {
856            if set[at].spanning != self.tier().spanning {
857                self.open_tier(set[at].spanning, set[at].lead);
858            }
859            // The profile is read against where the line sits, which
860            // is where `place` is about to put it.
861            let lead = self.lead(&set[at]);
862            let from = if at == 0 { 0 } else { ends[at - 1] };
863            // A line set again keeps none of the gap it was broken
864            // with. That gap was the page it left.
865            let fixed = if from == 0 { original[0].fixed } else { 0.0 };
866            let top = self.cursor + lead + fixed;
867            // The markers of the items the paragraph opens hang before
868            // its first line, and keep clear of an image as it does.
869            let hung = original[0]
870                .markers
871                .as_ref()
872                .filter(|_| from == 0)
873                .map(|markers| {
874                    markers
875                        .iter()
876                        .map(|marker| marker.x)
877                        .fold(f32::MAX, f32::min)
878                        - reflow.setting.x
879                });
880            let profile = self.profile(top, reflow, from == 0, hung);
881            if profile.is_some() || narrowed {
882                narrowed = profile.is_some();
883                let profile = profile.unwrap_or_else(|| Profile::plain(reflow, from == 0));
884                let paginator = self.paginator;
885                if self.paints {
886                    paginator.rebreaks.set(paginator.rebreaks.get() + 1);
887                }
888                let broken = paginator
889                    .lines
890                    .rebreak(&reflow.shaped, &profile.measure, from);
891                if broken.lines.is_empty() {
892                    return;
893                }
894                let mut fresh = set_lines(
895                    self.paginator,
896                    broken.lines,
897                    &profile.measure,
898                    &profile.gaps,
899                    &reflow.setting.wrapped(from == 0, profile.letter),
900                );
901                carry_over(&mut fresh, &set[at..], fixed);
902                if from == 0
903                    && let Some(first) = fresh.first_mut()
904                {
905                    let moved = profile.measure.at(0).origin - reflow.base.at(0).origin;
906                    first.markers = original[0].markers.clone().map(|mut markers| {
907                        for marker in markers.iter_mut() {
908                            marker.x += moved;
909                        }
910                        markers
911                    });
912                }
913                set = Cow::Owned(fresh);
914                ends = Cow::Owned(broken.ends);
915                at = 0;
916            }
917            let mut split = None;
918            for index in at..set.len() {
919                let opened = (self.pages.len(), self.column);
920                self.place(&set[index]);
921                if (self.pages.len(), self.column) != opened {
922                    split = Some(index);
923                    break;
924                }
925            }
926            let Some(index) = split else { return };
927            // What crossed the boundary is this paragraph's again:
928            // the fresh column is where the rest of it starts.
929            let crossed = (index + 1 - at).min(self.placed.len() - self.column_start);
930            at = index + 1 - crossed;
931            self.unplace(crossed);
932        }
933    }

935    /// Takes the last `count` fragments off the column being filled,
936    /// so the paragraph they came from can be set again where they
937    /// now sit.
938    fn unplace(&mut self, count: usize) {
939        let keep = self.placed.len() - count;
940        for placed in self.placed.drain(keep..) {
941            self.pending_anchors.extend(placed.anchors);
942        }
943        self.cursor = self.placed[self.column_start..]
944            .last()
945            .map(|placed| placed.top + placed.height)
946            .unwrap_or(self.tier().top);
947    }

949    /// The images on the page being built that reach the column being
950    /// filled, in that column's coordinates.
951    ///
952    /// An image is placed against the page and a column is a region
953    /// of the page, so which images a column sets around is the
954    /// overlap of the two. An image that reaches no part of this
955    /// column is not one of its holes, and an image that reaches two
956    /// columns is a hole in each of them. A paragraph that spans the
957    /// columns reads the whole content box.
958    fn holes(&self) -> Vec<Hole<'_>> {
959        let index = self.pages.len();
960        let Some(anchored) = self.anchored.by_page.get(&index) else {
961            return Vec::new();
962        };
963        let geometry = self.paginator.master(index, &self.slot).geometry;
964        let (origin, width) = if self.tier().spanning {
965            (geometry.content_origin(), geometry.content_size().0)
966        } else {
967            (geometry.column_origin(self.column), geometry.measure())
968        };
969        let column = Rect {
970            x: origin.0,
971            y: origin.1,
972            w: width,
973            h: self.height,
974        };
975        anchored
976            .iter()
977            .map(|at| &self.anchored.all[*at])
978            .filter(|image| image.wrap != WrapFlow::Auto)
979            .map(|image| (image, image.rect(geometry)))
980            .filter(|(_, rect)| rect.meets(column))
981            .map(|(image, rect)| Hole {
982                rect: rect.within(origin),
983                wrap: image.wrap,
984                shape: image.shape.as_ref(),
985            })
986            .collect()
987    }

989    /// The bands a paragraph that starts at `top` in the column being
990    /// filled is set in, and `None` where no image reaches them.
991    ///
992    /// An image covers whole bands. The flow snaps it to the
993    /// paragraph's own leading, so a line is either set beside it or
994    /// clear of it. A band it covers the whole of is a band nothing
995    /// is set in, and the paragraph goes on below it.
996    ///
997    /// `hung` is where the markers of the list items the paragraph
998    /// opens start, from its leading edge.
999    fn profile(
1000        &self,
1001        top: f32,
1002        reflow: &Reflow,
1003        opening: bool,
1004        hung: Option<f32>,
1005    ) -> Option<Profile> {
1006        // The bands are the paragraph's own, from its leading edge.
1007        // The images are the column's. One of them has to move.
1008        let holes: Vec<Hole> = self
1009            .holes()
1010            .into_iter()
1011            .map(|hole| Hole {
1012                rect: hole.rect.within((reflow.setting.x, 0.0)),
1013                ..hole
1014            })
1015            .collect();
1016        if holes.is_empty() {
1017            return None;
1018        }
1019        let leading = reflow.leading.max(1.0);
1020        let narrowest = reflow.shaped.style().size;
1021        // The band every band past the profile is set in, and the one
1022        // the profile has to list a band that differs from.
1023        let plain = reflow.base.rest();
1024        let cap = reflow.setting.cap.as_ref().filter(|_| opening);
1025        // An initial letter is one box over the bands it is sunk
1026        // over, so it goes where they are clear for the whole of its
1027        // height, with room for a line beside it. Its own bands are
1028        // read against the plain band and give up its column. The
1029        // bands the profile was built with hold the column the letter
1030        // takes with nothing in the way, which is not the column to
1031        // read here.
1032        let column = |cap: &Cap, y: f32| {
1033            let bottom = y + cap.lines as f32 * leading;
1034            clear(plain, &holes, y, bottom, cap.reserved + narrowest)
1035                .first()
1036                .map(|(origin, _)| *origin)
1037        };
1038        let mut letter = None;
1039        let mut spans = Vec::new();
1040        let mut gaps = Vec::new();
1041        // Whether an image reached any band at all, and how far down
1042        // the profile has to be listed: the shorter it is, the fewer
1043        // states the break has to keep.
1044        let mut reached = false;
1045        let mut listed = (0, 0);
1046        let (mut y, mut band, mut gap) = (top, 0, 0.0);
1047        while y < self.height {
1048            let sunk = cap.filter(|cap| band < cap.lines);
1049            // A first-line indent and a drop cap belong to the line
1050            // the paragraph opens on. What is left of it opens on a
1051            // band like any other.
1052            let base = match (opening, sunk) {
1053                (_, Some(_)) => plain,
1054                (true, None) => reflow.base.at(band),
1055                (false, None) => plain,
1056            };
1057            let mut free = match hung.filter(|_| opening && band == 0 && sunk.is_none()) {
1058                Some(hung) => clear_hung(base, hung, &holes, y, y + leading, narrowest),
1059                None => clear(base, &holes, y, y + leading, narrowest),
1060            };
1061            if let Some(cap) = sunk {
1062                // Where the letter goes is settled on the first of
1063                // its bands and held for the rest of them.
1064                let at = match letter {
1065                    Some(at) => Some(at),
1066                    None => column(cap, y),
1067                };
1068                free = match at {
1069                    Some(at) => taking(free, at, cap.reserved, narrowest),
1070                    None => Vec::new(),
1071                };
1072                letter = at;
1073            }
1074            let Some((last, rest)) = free.split_last() else {
1075                // Nothing is set in a band an image covers the whole
1076                // of, so the next band is the first one under it. A
1077                // box covers every band down to its foot; a contour
1078                // may narrow at any of them, so it is asked again one
1079                // band down.
1080                let below = holes
1081                    .iter()
1082                    .filter(|hole| hole.covering(y, y + leading).is_some())
1083                    .fold(y, |below: f32, hole| {
1084                        below.max(match hole.shape {
1085                            None => hole.rect.bottom(),
1086                            Some(_) => y + leading,
1087                        })
1088                    });
1089                if below <= y {
1090                    break;
1091                }
1092                gap += below - y;
1093                y = below;
1094                continue;
1095            };
1096            for (origin, width) in rest {
1097                spans.push(Span {
1098                    origin: *origin,
1099                    width: *width,
1100                    ends_band: false,
1101                });
1102            }
1103            spans.push(Span::band(last.0, last.1));
1104            let above = std::mem::take(&mut gap);
1105            gaps.push(above);
1106            band += 1;
1107            y += leading;
1108            let alone = |span: Span| free.len() == 1 && *last == (span.origin, span.width);
1109            let unmoved = match (opening, sunk) {
1110                (true, Some(_)) => alone(reflow.base.at(band - 1)),
1111                _ => alone(base),
1112            };
1113            reached = reached || above != 0.0 || !unmoved;
1114            if above != 0.0 || !alone(plain) {
1115                listed = (spans.len(), band);
1116            }
1117        }
1118        // A box down to the foot of the column leaves no band under
1119        // it, so the line after the last band goes past the foot.
1120        if gap != 0.0 {
1121            gaps.push(gap);
1122            reached = true;
1123            listed = (spans.len(), band + 1);
1124        }
1125        if !reached {
1126            return None;
1127        }
1128        spans.truncate(listed.0);
1129        gaps.truncate(listed.1);
1130        Some(Profile {
1131            measure: Measure::new(spans, plain),
1132            gaps,
1133            letter,
1134        })
1135    }
1136}

1138/// One image as the column being filled sees it: the rectangle it
1139/// covers, the contour inside that rectangle where it has one, and
1140/// which side of it the prose sets on.
1141#[derive(Debug, Clone, Copy)]
1142struct Hole<'a> {
1143    rect: Rect,
1144    wrap: WrapFlow,
1145    shape: Option<&'a Shape>,
1146}
1147
1148impl Hole<'_> {
1149    /// What the hole covers of the band between `top` and `bottom`,
1150    /// and `None` where it reaches none of it.
1151    ///
1152    /// A box covers the same stretch at every height. A contour is
1153    /// read band by band, over the band grown by the shape margin,
1154    /// and what it gives back is grown by it too.
1155    fn covering(&self, top: f32, bottom: f32) -> Option<(f32, f32)> {
1156        let Some(shape) = self.shape else {
1157            return (self.rect.bottom() > top && self.rect.y < bottom)
1158                .then(|| (self.rect.x, self.rect.right()));
1159        };
1160        let (from, to) = (
1161            top - self.rect.y - shape.margin,
1162            bottom - self.rect.y + shape.margin,
1163        );
1164        let (left, right) = scanline(&shape.rings, from, to)?;
1165        Some((
1166            self.rect.x + left - shape.margin,
1167            self.rect.x + right + shape.margin,
1168        ))
1169    }
1170}

1242/// What one band has left of it where the images on its page cover
1243/// it: the stretches the prose can be set in, in reading order.
1244///
1245/// A stretch narrower than `narrowest` holds nothing worth setting
1246/// and is not one.
1247fn clear(base: Span, holes: &[Hole], top: f32, bottom: f32, narrowest: f32) -> Vec<(f32, f32)> {
1248    let mut free = vec![(base.origin, base.origin + base.width)];
1249    for hole in holes {
1250        let Some((left, right)) = hole.covering(top, bottom) else {
1251            continue;
1252        };
1253        free = free
1254            .iter()
1255            .flat_map(|(start, end)| {
1256                if right <= *start || left >= *end {
1257                    return vec![(*start, *end)];
1258                }
1259                match hole.wrap {
1260                    WrapFlow::Auto => vec![(*start, *end)],
1261                    WrapFlow::Start => vec![(*start, end.min(left))],
1262                    WrapFlow::End => vec![(start.max(right), *end)],
1263                    WrapFlow::Both => vec![(*start, end.min(left)), (start.max(right), *end)],
1264                }
1265            })
1266            .filter(|(start, end)| end - start >= narrowest)
1267            .collect();
1268    }
1269    free.into_iter()
1270        .map(|(start, end)| (start, end - start))
1271        .collect()
1272}

1274/// What one band has left of it for the line a list item opens on,
1275/// where the item's markers start at `hung`. The markers keep their
1276/// distance from the text, so an image in the way of either moves both.
1277fn clear_hung(
1278    base: Span,
1279    hung: f32,
1280    holes: &[Hole],
1281    top: f32,
1282    bottom: f32,
1283    narrowest: f32,
1284) -> Vec<(f32, f32)> {
1285    let before = base.origin - hung;
1286    if before <= 0.0 {
1287        return clear(base, holes, top, bottom, narrowest);
1288    }
1289    let mut free = clear(
1290        Span::band(hung, base.width + before),
1291        holes,
1292        top,
1293        bottom,
1294        narrowest,
1295    );
1296    while let Some(&(start, width)) = free.first() {
1297        if start == hung && width - before >= narrowest {
1298            return clear(base, holes, top, bottom, narrowest);
1299        }
1300        if width - before >= narrowest {
1301            free[0] = (start + before, width - before);
1302            break;
1303        }
1304        free.remove(0);
1305    }
1306    free
1307}

1309/// The leftmost and rightmost point a contour reaches between two
1310/// heights, and `None` where it reaches neither.
1311///
1312/// This is what turns a polygon into per-band spans. A ring is closed
1313/// by its own first point. An edge inside the band contributes its
1314/// ends, and one that crosses the band's edge contributes where it
1315/// crosses.
1316fn scanline(rings: &[Vec<[f32; 2]>], top: f32, bottom: f32) -> Option<(f32, f32)> {
1317    let mut reach: Option<(f32, f32)> = None;
1318    let mut widen = |x: f32| {
1319        reach = Some(match reach {
1320            None => (x, x),
1321            Some((left, right)) => (left.min(x), right.max(x)),
1322        });
1323    };
1324    for ring in rings {
1325        for (at, from) in ring.iter().enumerate() {
1326            let to = ring[(at + 1) % ring.len()];
1327            let (above, below) = (from[1].min(to[1]), from[1].max(to[1]));
1328            if below < top || above > bottom {
1329                continue;
1330            }
1331            for point in [*from, to] {
1332                if point[1] >= top && point[1] <= bottom {
1333                    widen(point[0]);
1334                }
1335            }
1336            for cut in [top, bottom] {
1337                if cut > above && cut < below {
1338                    let along = (cut - from[1]) / (to[1] - from[1]);
1339                    widen(from[0] + along * (to[0] - from[0]));
1340                }
1341            }
1342        }
1343    }
1344    reach
1345}

1347/// What one band has left of it once the initial letter beside it
1348/// takes its column, which starts at `at` and runs `reserved` wide.
1349///
1350/// A stretch narrower than `narrowest` holds nothing worth setting
1351/// and is not one.
1352fn taking(free: Vec<(f32, f32)>, at: f32, reserved: f32, narrowest: f32) -> Vec<(f32, f32)> {
1353    free.into_iter()
1354        .map(|(origin, width)| {
1355            let end = origin + width;
1356            if at + reserved <= origin || at >= end {
1357                return (origin, width);
1358            }
1359            let start = origin.max(at + reserved);
1360            (start, end - start)
1361        })
1362        .filter(|(_, width)| *width >= narrowest)
1363        .collect()
1364}

2889            .iter()

2893                } else {
2894                    measure
2895                }
2896            })
2897            .collect();

2899            .iter()
2900            .map(|(_, runs)| runs.iter().map(|run| run.2).collect::<String>())

2909                .map(|line| paginator.line_width(&line))
2910                .unwrap_or_default()
2911        };

2913        let mut greedy: Vec<String> = Vec::new();
2914        let mut band = 0;

2920                    line.push_str(word);
2921                }

2941                    let gap = (room - width(line.trim())) as f64;
2942                    gap * gap
2943                })

2967    /// Every content run of one page as `(column, baseline, start,
2968    /// end)`, in paint order. The folio is furniture and is not one.
2969    fn column_runs(page: &Page, geometry: crate::style::PageGeometry) -> Vec<(u32, f32, f32, f32)> {
2970        page.items
2971            .iter()
2972            .filter_map(|item| {
2973                let DrawItem::Text {
2974                    x,
2975                    y,
2976                    font_id,
2977                    size,
2978                    glyphs,
2979                    ..
2980                } = item
2981                else {
2982                    return None;
2983                };
2984                if *size == folio_size() {
2985                    return None;
2986                }
2987                let last = glyphs.last()?;
2988                let upem = registry().metrics(*font_id)?.units_per_em as f32;
2989                let advance = registry().advance_width(*font_id, last.id)? as f32;
2990                let column = (0..geometry.column_count())
2991                    .rev()
2992                    .find(|column| *x >= geometry.column_origin(*column).0 - 1e-3)
2993                    .unwrap_or(0);
2994                Some((column, *y, *x, last.x + advance / upem * size))
2995            })
2996            .collect()
2997    }

3022    /// Acceptance: one rectangle over the gutter narrows the column on
3023    /// either side of it, each from its own edge.
3024    ///
3025    /// The image is page geometry and a column is a region of the
3026    /// page, so neither column owns it. The first column gives up its
3027    /// trailing edge and the second gives up its leading edge, out of
3028    /// the one rectangle.

3030    fn an_image_over_the_gutter_narrows_the_column_on_either_side_of_it() {
3031        let css = format!(
3032            "{TWO_COLUMNS} img {{ position: absolute; top: 0; left: 120pt; wrap-flow: both }}"
3033        );
3034        let geometry = two_columns();
3035        let (left, top) = geometry.content_origin();
3036        let measure = geometry.measure();
3037        let (near, far) = (left + 120.0, left + 120.0 + IMAGE);
3038        let foot = top + IMAGE;
3039        let page = &wrapped(&css).pages[0];
3040        assert_eq!(painted(page), vec![(near, top, IMAGE, IMAGE)]);
3041        // One rectangle over the gutter: it reaches into both columns
3042        // and covers the whole of neither.
3043        assert!(near > geometry.column_origin(0).0 && near < geometry.column_origin(0).0 + measure);
3044        assert!(far > geometry.column_origin(1).0 && far < geometry.column_origin(1).0 + measure);
3045
3046        let mut beside = [0, 0];
3047        let mut clear = [0, 0];
3048        for (column, baseline, start, end) in column_runs(page, geometry) {
3049            let origin = geometry.column_origin(column).0;
3050            if baseline <= foot {
3051                beside[column as usize] += 1;
3052                match column {
3053                    0 => assert!(
3054                        end <= near + 1e-3,
3055                        "a line at {baseline} reaches {end}, over the image"
3056                    ),
3057                    _ => assert!(
3058                        start >= far - 1e-3,
3059                        "a line at {baseline} starts at {start}, over the image"
3060                    ),
3061                }
3062                continue;
3063            }
3064            clear[column as usize] += 1;
3065            assert!(
3066                start >= origin - 1e-3 && end <= origin + measure + 1e-3,
3067                "a line at {baseline} runs {start}..{end}, outside column {column}",
3068            );
3069        }
3070        assert!(
3071            beside[0] > 0 && beside[1] > 0,
3072            "no column set beside the image: {beside:?}"
3073        );
3074        assert!(
3075            clear[0] > 0 && clear[1] > 0,
3076            "no column reached past the image: {clear:?}"
3077        );
3078
3079        // Under the image each column has its whole measure back.
3080        let reach = |column: u32| {
3081            column_runs(page, geometry)
3082                .into_iter()
3083                .filter(|(at, baseline, ..)| *at == column && *baseline > foot)
3084                .fold((f32::MAX, f32::MIN), |(from, to), (_, _, start, end)| {
3085                    (from.min(start), to.max(end))
3086                })
3087        };
3088        assert!(
3089            reach(0).1 > near,
3090            "the first column never reached past the image"
3091        );
3092        assert!(
3093            reach(1).0 < far - 1e-3,
3094            "the second column never reached back to its own edge"
3095        );
3096    }
