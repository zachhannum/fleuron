1//! Navigation: the links on each page, with the area each covers and
2//! where it goes, and the outline of the book's headings.
3//!
4//! Both read the destination table the flow built. A link's area is
5//! read off the runs its text was set in, one area for each line, so a
6//! link broken across two lines is followed from either line and not
7//! from the space between them.

19/// One link as the book writes it.
20struct Written<'b> {
21    /// The ids of the link and everything inside it.
22    nodes: std::ops::Range<u32>,
23    url: &'b str,
24    source: Option<&'b str>,
25    position: Option<SourcePos>,
26}

35/// Puts the links of one laid-out book on its pages, and returns the
36/// outline of its headings and what the links had to complain about:
37/// a link whose url names nothing in the book is set as text, and says
38/// so.
39pub(crate) fn navigation(
40    book: &Book,
41    pages: &mut [Page],
42    targets: &BTreeMap<NodeId, PageBox>,
43    registry: &FontRegistry,
44) -> (Navigation, Vec<Warning>) {
45    let mut written = Vec::new();
46    let mut headings = Vec::new();
47    for section in &book.sections {
48        for block in &section.blocks {
49            if let Block::Heading {
50                id, level, inlines, ..
51            } = block
52            {
53                headings.push(Heading {
54                    node: *id,
55                    level: u8::from(*level),
56                    title: title(inlines),
57                });
58            }
59        }
60        links_in_blocks(&section.blocks, section.source.as_deref(), &mut written);
61    }
62    let mut warnings = Vec::new();
63    if !written.is_empty() {
64        let anchors = book.anchors();
65        let places: Vec<Option<LinkTo>> = written
66            .iter()
67            .map(|link| {
68                let to = place(&anchors, targets, link);
69                if let Err(unplaced) = &to {
70                    let message = match unplaced {
71                        Some(node) => format!(
72                            "`{}` names an element on no page. The text is not a link.",
73                            anchors.name(*node).unwrap_or(link.url)
74                        ),
75                        None => format!(
76                            "`{}` names nothing in the book. The text is not a link.",
77                            link.url
78                        ),
79                    };
80                    let at = origin(link.source, link.position);
81                    if !warnings
82                        .iter()
83                        .any(|seen: &Warning| seen.message == message)
84                    {
85                        warnings.push(Warning {
86                            message,
87                            origin: (!at.is_empty()).then_some(at),
88                        });
89                    }
90                }
91                to.ok()
92            })
93            .collect();
94        place_links(pages, &written, &places, registry);
95    }
96    let outline = outline(&headings, targets);
97    (Navigation { outline }, warnings)
98}

100/// Where one link goes, or why it goes nowhere: the element it names
101/// is on no page, or it names nothing in the book.
102fn place(
103    anchors: &Anchors,
104    targets: &BTreeMap<NodeId, PageBox>,
105    link: &Written<'_>,
106) -> Result<LinkTo, Option<NodeId>> {
107    match anchors.resolve(link.url, link.source) {
108        LinkTarget::Outside => Ok(LinkTo::Uri(link.url.to_string())),
109        LinkTarget::Node(node) => targets
110            .get(&node)
111            .map(|place| LinkTo::Place {
112                node,
113                place: *place,
114            })
115            .ok_or(Some(node)),
116        LinkTarget::Missing => Err(None),
117    }
118}

120/// The words of a heading as an outline shows them, on one line.
121fn title(inlines: &[Inline]) -> String {
122    text(inlines)
123        .split_whitespace()
124        .collect::<Vec<_>>()
125        .join(" ")
126}

128fn links_in_blocks<'b>(blocks: &'b [Block], source: Option<&'b str>, out: &mut Vec<Written<'b>>) {
129    for block in blocks {
130        match block {
131            Block::Heading { inlines, .. } | Block::Paragraph { inlines, .. } => {
132                links_in_inlines(inlines, source, out)
133            }
134            Block::Blockquote { blocks, .. } => links_in_blocks(blocks, source, out),
135            Block::List { items, .. } => {
136                for blocks in item_blocks(items) {
137                    links_in_blocks(blocks, source, out);
138                }
139            }
140            Block::Table { head, body, .. } => {
141                for blocks in cell_blocks(head, body) {
142                    links_in_blocks(blocks, source, out);
143                }
144            }
145            Block::CodeBlock { .. }
146            | Block::ThematicBreak { .. }
147            | Block::PageBreak { .. }
148            | Block::ColumnBreak { .. }
149            | Block::Image { .. } => {}
150        }
151    }
152}

154fn links_in_inlines<'b>(
155    inlines: &'b [Inline],
156    source: Option<&'b str>,
157    out: &mut Vec<Written<'b>>,
158) {
159    for inline in inlines {
160        match inline {
161            Inline::Link {
162                id, url, position, ..
163            } => {
164                let first = id.get();
165                out.push(Written {
166                    nodes: first..first + inline_nodes(inline),
167                    url,
168                    source,
169                    position: *position,
170                });
171            }
172            Inline::Emphasis { children, .. }
173            | Inline::Strong { children, .. }
174            | Inline::Strikethrough { children, .. }
175            | Inline::Span { children, .. } => links_in_inlines(children, source, out),
176            Inline::Note { blocks, .. } => links_in_blocks(blocks, source, out),
177            Inline::Text { .. } | Inline::Code { .. } | Inline::Break { .. } => {}
178        }
179    }
180}

182/// Puts on each page the links set on it, with one area for each line.
183///
184/// Runs of one link on one baseline that meet, or nearly meet, are one
185/// area. A link that runs to the foot of one column and on at the head
186/// of the next is two areas. A link that runs on to the next page is on
187/// both pages.
188fn place_links(
189    pages: &mut [Page],
190    written: &[Written<'_>],
191    places: &[Option<LinkTo>],
192    registry: &FontRegistry,
193) {
194    for (index, page) in pages.iter_mut().enumerate() {
195        // Each area with the link it belongs to, the baseline it is on,
196        // and the size of its face, which is how near two runs have to
197        // be to meet.
198        let mut open: Vec<(usize, f32, f32, PageBox)> = Vec::new();
199        for item in &page.items {
200            let DrawItem::Text {
201                y,
202                size,
203                origin: Some(origin),
204                ..
205            } = item
206            else {
207                continue;
208            };
209            let node = origin.node.element().get();
210            let at = written.partition_point(|link| link.nodes.start <= node);
211            let Some(which) = at.checked_sub(1) else {
212                continue;
213            };
214            if !written[which].nodes.contains(&node) || places[which].is_none() {
215                continue;
216            }
217            let Some((_, _, area)) = run_area(registry, index, item) else {
218                continue;
219            };
220            let meets = |other: &PageBox, size: f32| {
221                area.x <= other.x + other.width + size && other.x <= area.x + area.width + size
222            };
223            match open.iter_mut().find(|(link, baseline, near, other)| {
224                *link == which && baseline == y && meets(other, near.max(*size))
225            }) {
226                Some((_, _, near, other)) => {
227                    *near = near.max(*size);
228                    *other = union(other, &area);
229                }
230                None => open.push((which, *y, *size, area)),
231            }
232        }
233        let mut links: Vec<(usize, Link)> = Vec::new();
234        for (which, _, _, area) in open {
235            match links.iter_mut().find(|(link, _)| *link == which) {
236                Some((_, link)) => link.areas.push(area),
237                None => links.push((
238                    which,
239                    Link {
240                        areas: vec![area],
241                        to: places[which]
242                            .clone()
243                            .expect("only a link that goes somewhere is open"),
244                    },
245                )),
246            }
247        }
248        page.links = links.into_iter().map(|(_, link)| link).collect();
249    }
250}

252/// The smallest box that holds both.
253fn union(one: &PageBox, other: &PageBox) -> PageBox {
254    let (left, top) = (one.x.min(other.x), one.y.min(other.y));
255    let right = (one.x + one.width).max(other.x + other.width);
256    let bottom = (one.y + one.height).max(other.y + other.height);
257    PageBox {
258        page: one.page,
259        x: left,
260        y: top,
261        width: right - left,
262        height: bottom - top,
263    }
264}

266/// The headings, nested by level: each takes the headings after it
267/// that are deeper, up to the next one at its level or above. A
268/// heading that is on no page is left out.
269fn outline(headings: &[Heading], targets: &BTreeMap<NodeId, PageBox>) -> Vec<OutlineEntry> {
270    let mut roots: Vec<OutlineEntry> = Vec::new();
271    for heading in headings {
272        let Some(place) = targets.get(&heading.node) else {
273            continue;
274        };
275        let entry = OutlineEntry {
276            title: heading.title.clone(),
277            level: heading.level,
278            place: *place,
279            children: Vec::new(),
280        };
281        let mut siblings = &mut roots;
282        while siblings.last().is_some_and(|last| last.level < entry.level) {
283            siblings = &mut siblings.last_mut().expect("checked above").children;
284        }
285        siblings.push(entry);
286    }
287    roots
288}

290/// The node a run of text was written in, the pseudo-element it was
291/// cut from, and the area the run covers on the page at `index`: its
292/// glyphs across, and its face's ascent and descent down. Nothing for
293/// text the engine wrote itself.
294pub(crate) fn run_area(
295    registry: &FontRegistry,
296    index: usize,
297    item: &DrawItem,
298) -> Option<(NodeId, Option<NodeId>, PageBox)> {
299    let DrawItem::Text {
300        y,
301        font_id,
302        size,
303        glyphs,
304        origin: Some(origin),
305        pseudo_element,
306        ..
307    } = item
308    else {
309        return None;
310    };
311    let metrics = registry.metrics(*font_id)?;
312    let scale = size / metrics.units_per_em as f32;
313    let (first, last) = (glyphs.first()?, glyphs.last()?);
314    let advance = registry.advance_width(*font_id, last.id).unwrap_or(0) as f32 * scale;
315    let (left, right) = (first.x.min(last.x), first.x.max(last.x + advance));
316    let top = y - metrics.ascender as f32 * scale;
317    let bottom = y - metrics.descender as f32 * scale;
318    Some((
319        origin.node,
320        *pseudo_element,
321        PageBox {
322            page: index as u32,
323            x: left,
324            y: top,
325            width: right - left,
326            height: bottom - top,
327        },
328    ))
