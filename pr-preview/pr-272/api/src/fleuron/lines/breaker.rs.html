1//! Knuth-Plass total fit: the set of breaks with the fewest
2//! demerits over the whole paragraph.
3
4use super::justify::{LETTER_SHRINK, LETTER_STRETCH, SPACE_SHRINK, SPACE_STRETCH};
5use super::line::Widths;
6use super::measure::{Measure, Span};
7use super::opportunity::Break;
8use super::paragraph::{HangEnd, LineBreakOptions};

10/// Knuth's demerit weights. A line costs `(line + badness)^2`, so a
11/// paragraph of evenly loose lines beats one tight line and one
12/// gaping one; the rest are the surcharges for breaking a word, for
13/// doing it twice running, and for setting a tight line under a
14/// loose one.
15const LINE_PENALTY: f64 = 10.0;
16
17const HYPHEN_PENALTY: f64 = 50.0;
18
19const DOUBLE_HYPHEN_DEMERITS: f64 = 10_000.0;
20
21const ADJACENT_DEMERITS: f64 = 10_000.0;
22
23/// What breaking before a dash costs. UAX #14 allows a line to end
24/// on either side of one; a book only ever ends on the far side, so
25/// the near side has to be worth something to be taken.
26const DASH_PENALTY: f64 = 200.0;
27
28/// The worst a line may be counted as. Without a ceiling a single
29/// unbreakable line swamps every other term in the paragraph.
30const MAX_BADNESS: f64 = 10_000.0;
31
32/// What a line that overflows the measure costs, per em it overflows
33/// by and once besides. Larger than any feasible paragraph, so text
34/// runs into the margin only where nothing else will set, and the
35/// least of it wins.
36const OVERFULL_DEMERITS: f64 = 1e12;
37
38/// Hyphenated line ends allowed in a row.
39const MAX_CONSECUTIVE_HYPHENS: u8 = 2;
40
41/// What a ragged line may leave at the right, as a fraction of the
42/// measure, before it counts as loose. Ragged setting has no glue to
43/// stretch, so badness has nothing else to read the gap against.
44const RAGGED_STRETCH: f32 = 0.1;
45
46/// A chosen span end, with the adjustment its glue takes.
47#[derive(Debug, Clone, Copy)]
48pub(super) struct Fitted {
49    /// Index into the break list.
50    pub(super) at: usize,
51    /// The slot of the profile the text up to here fills.
52    pub(super) slot: usize,
53    /// The line's adjustment ratio: what fraction of its stretch or
54    /// shrink the glue gives up to reach the measure.
55    pub(super) ratio: f32,
56    /// Font units hanging past the measure at the line's end.
57    pub(super) overhang: f32,
58    /// Font units hanging before the line's origin.
59    pub(super) protrusion: f32,
60}
61
62/// One paragraph's total-fit pass: break list in, chosen line ends
63/// out.
64pub(super) struct Breaker<'a> {
65    pub(super) breaks: &'a [Break],
66    pub(super) widths: &'a Widths,
67    pub(super) measure: &'a Measure,
68    /// The span every slot past the profile's listed ones is set in,
69    /// which is every slot of most paragraphs.
70    pub(super) rest: Span,
71    /// How many spans the profile lists.
72    pub(super) settled: usize,
73    /// The slot the paragraph's first band ends in.
74    pub(super) first_band: usize,
75    pub(super) upem: f32,
76    pub(super) size: f32,
77    /// Font units a hyphenated break is charged for.
78    pub(super) hyphen: f32,
79    pub(super) options: LineBreakOptions,
80    /// The break the first line starts from. Everything before it is
81    /// set already.
82    pub(super) from: usize,
83    /// Where the opening line has to end, when an opening style
84    /// covers exactly that much of the text.
85    pub(super) opening: Option<usize>,
86}
87
88/// How one candidate line comes out: how far its glue is from its
89/// natural width, and what that costs.
90struct Fit {
91    ratio: f32,
92    badness: f64,
93    /// Font units the line runs past the measure by, if it does.
94    overflow: f32,
95    overhang: f32,
96    protrusion: f32,
97    /// Whether the span this fills ends the band it sits in.
98    ends_band: bool,
99}
100
101/// A breakpoint reached by a path, and the best path to it.
102struct Node {
103    /// Index into the break list.
104    at: usize,
105    /// Slots the paragraph has filled to get here.
106    slot: usize,
107    /// Which of the four fitness classes the line ending here fell
108    /// in.
109    fitness: u8,
110    /// Hyphenated line ends in a row up to and including this one.
111    hyphens: u8,
112    demerits: f64,
113    ratio: f32,
114    overhang: f32,
115    protrusion: f32,
116    /// The node this one was reached from; the root has none.
117    previous: Option<usize>,
118}
119
120/// A span end worth keeping, before it becomes a node.
121struct Candidate {
122    at: usize,
123    slot: usize,
124    fitness: u8,
125    hyphens: u8,
126    demerits: f64,
127    ratio: f32,
128    overhang: f32,
129    protrusion: f32,
130    previous: usize,
131}

133impl Breaker<'_> {
134    /// The span slot `index` is set in.
135    fn span(&self, index: usize) -> Span {
136        if index < self.settled {
137            self.measure.at(index)
138        } else {
139            self.rest
140        }
141    }
142
143    /// Points → the paragraph's font units.
144    fn units(&self, points: f32) -> f32 {
145        if self.size > 0.0 {
146            points / self.size * self.upem
147        } else {
148            0.0
149        }
150    }
151
152    /// The last breakpoint, which every path has to reach.
153    pub(super) fn end(&self) -> usize {
154        self.breaks.len() - 1
155    }
156
157    /// Measures the text that runs from break `a` to break `b`, set
158    /// in slot `slot`.
159    fn fit(&self, a: usize, b: usize, slot: usize) -> Fit {
160        let span = self.span(slot - 1);
161        let start = self.breaks[a].next;
162        let end = self.breaks[b].content_end.max(start);
163        let measure = self.units(span.width);
164        let text = self.widths.line_advance(start, end);
165        let spaces = self.widths.spaces[end] - self.widths.spaces[start];
166        let hyphen = if self.breaks[b].hyphen {
167            self.hyphen
168        } else {
169            0.0
170        };
171        // A mark hangs into the margin a band starts at and past the
172        // one it ends at. The gap between two spans of a band is
173        // neither.
174        let protrusion = if self.options.hanging.first && self.opens_band(slot - 1) {
175            self.breaks[a].hang_start
176        } else {
177            0.0
178        };
179        let natural = text + hyphen - protrusion;
180        let overhang = if span.ends_band {
181            self.overhang(b, natural, measure)
182        } else {
183            0.0
184        };
185        let width = natural - overhang;
186
187        // The last line of a paragraph fills whatever it fills, and so
188        // does a line a hard break ends: the glue that finishes it
189        // stretches without limit.
190        let (stretch, shrink) = if self.breaks[b].forced {
191            let shrink = if self.options.justify {
192                spaces * SPACE_SHRINK
193            } else {
194                0.0
195            };
196            (f32::INFINITY, shrink)
197        } else if self.options.justify {
198            let letters = if self.options.inter_character {
199                text - spaces
200            } else {
201                0.0
202            };
203            (
204                spaces * SPACE_STRETCH + letters * LETTER_STRETCH,
205                spaces * SPACE_SHRINK + letters * LETTER_SHRINK,
206            )
207        } else {
208            // Ragged setting has no glue to open, so the gap at the
209            // right is read against a fraction of the measure.
210            (measure * RAGGED_STRETCH, 0.0)
211        };
212
213        let gap = measure - width;
214        let ratio = if gap > 0.0 {
215            if stretch > 0.0 {
216                gap / stretch
217            } else {
218                f32::INFINITY
219            }
220        } else if gap < 0.0 {
221            if shrink > 0.0 {
222                gap / shrink
223            } else {
224                f32::NEG_INFINITY
225            }
226        } else {
227            0.0
228        };
229        Fit {
230            ratio,
231            badness: badness(ratio),
232            overflow: (-gap).max(0.0),
233            overhang,
234            protrusion,
235            ends_band: span.ends_band,
236        }
237    }
238
239    /// The slot the line after `from` fills. A hard break ends the
240    /// band it falls in, so the line after one opens the next band
241    /// rather than the next span of the same band.
242    fn next_slot(&self, from: &Node) -> usize {
243        let mut filled = from.slot;
244        if filled > 0 && self.breaks[from.at].forced {
245            while filled - 1 < self.settled && !self.span(filled - 1).ends_band {
246                filled += 1;
247            }
248        }
249        filled + 1
250    }
251
252    /// Whether the band the span at `index` sits in opens there.
253    fn opens_band(&self, index: usize) -> bool {
254        index == 0 || self.span(index - 1).ends_band
255    }
256
257    /// What hangs past the measure at break `b`, given how wide the
258    /// line would otherwise be.
259    fn overhang(&self, b: usize, natural: f32, measure: f32) -> f32 {
260        let hang = self.breaks[b].hang_end;
261        if hang <= 0.0 {
262            return 0.0;
263        }
264        if b == self.end() && self.options.hanging.last {
265            return hang;
266        }
267        match self.options.hanging.end {
268            HangEnd::Force => hang,
269            HangEnd::Allow if natural > measure && natural - hang <= measure => hang,
270            _ => 0.0,
271        }
272    }

274    /// The chosen line ends, first to last.
275    pub(super) fn run(&self) -> Vec<Fitted> {
276        let mut nodes = vec![Node {
277            at: self.from,
278            slot: 0,
279            fitness: 1,
280            hyphens: 0,
281            demerits: 0.0,
282            ratio: 0.0,
283            overhang: 0.0,
284            protrusion: 0.0,
285            previous: None,
286        }];
287        let mut active = vec![0usize];
288        let mut candidates: Vec<Candidate> = Vec::new();
289
290        for b in self.from + 1..self.breaks.len() {
291            let forced = self.breaks[b].forced;
292            // The cheapest way to break here anyway, for a paragraph
293            // that cannot be set inside the measure at all.
294            let mut overfull: Option<Candidate> = None;
295            // A path a hard break cut before the opening line could
296            // end, for when it is the only path left.
297            let mut stranded: Option<Candidate> = None;
298            let mut index = 0;
299            while index < active.len() {
300                let a = active[index];
301                let slot = self.next_slot(&nodes[a]);
302                // The opening line ends where the style over it does,
303                // so the text set in that style is the text on it.
304                if slot == self.first_band + 1 && self.opening.is_some_and(|end| end != b) {
305                    // Past that end the node reaches nothing, and a
306                    // hard break before it ends the line anyway.
307                    let past = self.opening.is_some_and(|end| b > end);
308                    if forced && !past {
309                        let fit = self.fit(nodes[a].at, b, slot);
310                        stranded.get_or_insert_with(|| self.forced(&nodes[a], a, b, slot, &fit));
311                    }
312                    if forced || past {
313                        active.remove(index);
314                    } else {
315                        index += 1;
316                    }
317                    continue;
318                }
319                let fit = self.fit(nodes[a].at, b, slot);
320                if let Some(candidate) = self.candidate(&nodes[a], a, b, slot, &fit) {
321                    self.keep_best(&mut candidates, candidate);
322                }
323                let long = fit.ratio < -1.0;
324                if long {
325                    let candidate = Candidate {
326                        demerits: nodes[a].demerits
327                            + OVERFULL_DEMERITS * (1.0 + (fit.overflow / self.upem) as f64),
328                        ratio: -1.0,
329                        fitness: 0,
330                        ..self.forced(&nodes[a], a, b, slot, &fit)
331                    };
332                    if overfull
333                        .as_ref()
334                        .is_none_or(|best| candidate.demerits < best.demerits)
335                    {
336                        overfull = Some(candidate);
337                    }
338                }
339                if long || forced {
340                    active.remove(index);
341                } else {
342                    index += 1;
343                }
344            }
345            if candidates.is_empty() {
346                if !active.is_empty() {
347                    continue;
348                }
349                // Nothing fits and nothing is left to try: overflow
350                // the measure rather than drop the text.
351                candidates.push(overfull.or(stranded).expect("a line was too long to set"));
352            }
353            for candidate in candidates.drain(..) {
354                nodes.push(Node {
355                    at: candidate.at,
356                    slot: candidate.slot,
357                    fitness: candidate.fitness,
358                    hyphens: candidate.hyphens,
359                    demerits: candidate.demerits,
360                    ratio: candidate.ratio,
361                    overhang: candidate.overhang,
362                    protrusion: candidate.protrusion,
363                    previous: Some(candidate.previous),
364                });
365                active.push(nodes.len() - 1);
366            }
367        }
368
369        let best = nodes
370            .iter()
371            .enumerate()
372            .filter(|(_, node)| node.at == self.end())
373            .min_by(|(_, one), (_, other)| one.demerits.total_cmp(&other.demerits))
374            .map(|(index, _)| index);
375        let mut chosen = Vec::new();
376        let mut at = best;
377        while let Some(index) = at {
378            let node = &nodes[index];
379            if node.previous.is_none() {
380                break;
381            }
382            chosen.push(Fitted {
383                at: node.at,
384                slot: node.slot - 1,
385                ratio: node.ratio,
386                overhang: node.overhang,
387                protrusion: node.protrusion,
388            });
389            at = node.previous;
390        }
391        chosen.reverse();
392        chosen
393    }
394
395    /// Keeps the cheapest candidate of each kind. Two paths that
396    /// reach the same break and are alike in everything the rest of
397    /// the paragraph can see are interchangeable, so only the cheaper
398    /// survives, which is what keeps the active list from growing
399    /// with the paragraph.
400    fn keep_best(&self, candidates: &mut Vec<Candidate>, candidate: Candidate) {
401        let key = |candidate: &Candidate| {
402            (
403                candidate.fitness,
404                candidate.hyphens,
405                // Only the spans the profile lists make the count
406                // visible from here on; past them every band is the
407                // same one.
408                candidate.slot.min(self.settled + 1),
409            )
410        };
411        match candidates
412            .iter_mut()
413            .find(|kept| key(kept) == key(&candidate))
414        {
415            Some(kept) if kept.demerits <= candidate.demerits => {}
416            Some(kept) => *kept = candidate,
417            None => candidates.push(candidate),
418        }
419    }

421    /// Break `b` reached from node `a` whatever it costs: the line
422    /// between them is wider than the measure, and setting it is
423    /// still better than losing the words.
424    fn forced(&self, from: &Node, a: usize, b: usize, slot: usize, fit: &Fit) -> Candidate {
425        Candidate {
426            at: b,
427            slot,
428            fitness: 0,
429            hyphens: self.hyphens(from, b),
430            demerits: from.demerits,
431            ratio: fit.ratio,
432            overhang: fit.overhang,
433            protrusion: fit.protrusion,
434            previous: a,
435        }
436    }

438    /// Hyphenated line ends in a row, counting the one break `b`
439    /// would add.
440    fn hyphens(&self, from: &Node, b: usize) -> u8 {
441        if self.breaks[b].hyphen {
442            from.hyphens + 1
443        } else {
444            0
445        }
446    }
447
448    /// Break `b` reached from node `a`, if the line between them can
449    /// be set at all.
450    fn candidate(
451        &self,
452        from: &Node,
453        a: usize,
454        b: usize,
455        slot: usize,
456        fit: &Fit,
457    ) -> Option<Candidate> {
458        if fit.ratio < -1.0 {
459            return None;
460        }
461        let hyphens = self.hyphens(from, b);
462        if hyphens > MAX_CONSECUTIVE_HYPHENS {
463            return None;
464        }
465        // The last line, and a line a hard break ends, is not
466        // stretched to the measure, so what is left at its right is
467        // not a fault to be charged for.
468        let ratio = if self.breaks[b].forced && fit.ratio > 0.0 {
469            0.0
470        } else {
471            fit.ratio
472        };
473        let penalty = if self.breaks[b].hyphen {
474            HYPHEN_PENALTY
475        } else if self.breaks[b].dash {
476            DASH_PENALTY
477        } else {
478            0.0
479        };
480        let fitness = fitness(ratio);
481        // Crossing from one span of a band to the next is not a line
482        // break, and the surcharge a line costs is not charged there.
483        let line_penalty = if fit.ends_band { LINE_PENALTY } else { 0.0 };
484        let mut demerits = (line_penalty + fit.badness + penalty).powi(2);
485        if hyphens > 1 {
486            demerits += DOUBLE_HYPHEN_DEMERITS;
487        }
488        if fitness.abs_diff(from.fitness) > 1 {
489            demerits += ADJACENT_DEMERITS;
490        }
491        Some(Candidate {
492            at: b,
493            slot,
494            fitness,
495            hyphens,
496            demerits: from.demerits + demerits,
497            ratio,
498            overhang: fit.overhang,
499            protrusion: fit.protrusion,
500            previous: a,
501        })
502    }
503}

505/// How bad a line of this adjustment ratio is. Cubic, so one gaping
506/// line costs more than several slightly loose ones.
507fn badness(ratio: f32) -> f64 {
508    if !ratio.is_finite() {
509        return MAX_BADNESS;
510    }
511    (100.0 * (ratio.abs() as f64).powi(3)).min(MAX_BADNESS)
512}
513
514/// Knuth's four fitness classes: tight, decent, loose, very loose. A
515/// tight line under a very loose one reads as a mistake even when
516/// each of them on its own does not, which is why the class and not
517/// the ratio is what the demerits compare.
518fn fitness(ratio: f32) -> u8 {
519    if ratio < -0.5 {
520        0
521    } else if ratio <= 0.5 {
522        1
523    } else if ratio < 1.0 {
524        2
525    } else {
526        3
527    }
528}

536    /// A hard break ends the line wherever the measure would have
537    /// put a break. A measure wide enough for the whole stanza still
538    /// sets it as its four lines, and at a narrow one no line runs
539    /// across a break.
540    #[test]
541    fn a_hard_break_ends_the_line_at_any_measure() {
542        let stanza = "Shall I compare thee to a summer's day?\n\
543                      Thou art more lovely and more temperate:\n\
544                      Rough winds do shake the darling buds of May,\n\
545                      And summer's lease hath all too short a date;";
546        let written: Vec<&str> = stanza.lines().collect();
547        let wide: Vec<String> = layout_verse(stanza, 1000.0, Default::default())
548            .iter()
549            .map(line_text)
550            .collect();
551        assert_eq!(wide, written);
552
553        let narrow = layout_verse(stanza, 90.0, Default::default());
554        assert!(narrow.len() > written.len(), "nothing wrapped: {narrow:?}");
555        let mut set = narrow.iter().map(line_text);
556        for verse in written {
557            let mut line = set.next().expect("every verse line is set");
558            while line.len() < verse.len() {
559                let more = set.next().expect("the verse line is set whole");
560                line = format!("{line} {more}");
561            }
562            assert_eq!(line, verse, "a line ran across a hard break");
563        }
564        assert_eq!(set.next(), None);
565    }

627    /// The arithmetic behind the 80pt row above, which is the row
628    /// where the two breakers part.
629    ///
630    /// Ragged badness is `100 * r^3`, where `r` is the gap left at
631    /// the right over a tenth of the measure, and a line costs
632    /// `(10 + badness)^2`, with 10,000 more when its fitness class is
633    /// two off the line before it. The last line fills what it fills
634    /// and costs 100.
635    ///
636    /// Total fit: 59.01 (r 2.62, 3.30M) + 64.70 (r 1.91, 0.50M)
637    /// + 75.57 (100) = 3.82M, two class changes included.
638    ///
639    /// Greedy: 65.60 (r 1.80, 0.35M) + 58.11 (r 2.74, 4.24M)
640    /// + 75.57 (100) = 4.61M, the same two.
641    ///
642    /// Greedy wins line one and loses the paragraph: the word it
643    /// pulls up leaves a hole on line two larger than the one it
644    /// filled.
645    #[test]
646    fn total_fit_beats_greedy_where_they_disagree() {
647        let text = "My father had a small estate in Nottinghamshire:";
648        let got: Vec<String> = layout_body(text, 80.0).iter().map(line_text).collect();
649        assert_eq!(got[0], "My father had", "line 1: {got:?}");
650        assert_ne!(
651            got[0], "My father had a",
652            "line 1 was packed greedily: {got:?}",
653        );
654    }
