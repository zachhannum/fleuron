27/// One node's resolved style: what every downstream pass reads.
28#[derive(Debug, Clone, PartialEq, Serialize)]
29pub struct ComputedStyle {
30    /// The face this style shapes with, resolved against the
31    /// registry. Family, slope and weight chose it; layout only needs
32    /// the answer.
33    pub font_id: u16,
34    /// The families asked for, in order, kept for diagnostics and for
35    /// the snapshot the style tree is reviewed through.
36    pub font_family: Vec<Family>,
37    /// Font size in points.
38    pub font_size: f32,
39    /// Upright or italic.
40    pub font_style: FontStyle,
41    /// Weight on the CSS 1–1000 scale.
42    pub font_weight: u16,
43    /// What text and rules are painted in.
44    pub color: Color,
45    /// Line height as a unitless multiple of the font size.
46    pub line_height: f32,
47    /// Extra advance after every glyph, in points, from
48    /// `letter-spacing`.
49    pub letter_spacing: f32,
50    /// Which capitals the run is drawn with.
51    pub font_variant_caps: FontVariantCaps,
52    /// The features the sheet asked the face for, from
53    /// `font-feature-settings`. They are asked for after the ones the
54    /// `font-variant` longhands name, so a tag in both is set here.
55    #[serde(skip_serializing_if = "Vec::is_empty")]
56    pub font_feature_settings: Vec<FeatureSetting>,
57    /// Which ligatures the run is set with.
58    #[serde(skip_serializing_if = "default_ligatures")]
59    pub font_variant_ligatures: FontVariantLigatures,
60    /// Which figures the run is set with.
61    #[serde(skip_serializing_if = "default_numeric")]
62    pub font_variant_numeric: FontVariantNumeric,
63    /// Which alternate glyphs the run is set with.
64    #[serde(skip_serializing_if = "default_alternates")]
65    pub font_variant_alternates: FontVariantAlternates,
66    /// What the run's letters are transformed to before shaping.
67    pub text_transform: TextTransform,
68    /// Which rules are drawn across the run, from
69    /// `text-decoration-line`.
70    #[serde(skip_serializing_if = "undecorated")]
71    pub text_decoration_line: DecorationLine,
72    /// What those rules are painted in, from
73    /// `text-decoration-color`. `None` is the colour of the text.
74    #[serde(skip_serializing_if = "Option::is_none")]
75    pub text_decoration_color: Option<Color>,
76    /// How each rule is drawn, from `text-decoration-style`.
77    #[serde(skip_serializing_if = "solid")]
78    pub text_decoration_style: DecorationStyle,
79    /// How thick each rule is, in points, from
80    /// `text-decoration-thickness`. `None` is the thickness the face
81    /// declares.
82    #[serde(skip_serializing_if = "Option::is_none")]
83    pub text_decoration_thickness: Option<f32>,
84    /// How lines fill the measure.
85    pub text_align: TextAlign,
86    /// What justification opens up to fill it.
87    pub text_justify: TextJustify,
88    /// Which marks may hang past the measure.
89    pub hanging_punctuation: HangingPunctuation,
90    /// First-line indent in points.
91    pub text_indent: f32,
92    /// Whether words may break at syllable boundaries.
93    pub hyphens: Hyphens,
94    /// Lines that must be left at the bottom of a fragment.
95    pub orphans: u16,
96    /// Lines that must be moved to the top of the next one.
97    pub widows: u16,
98    /// The named page this element's pages take, from `page`.
99    pub page: Option<String>,
100    /// What the element paints in place of children it has none of:
101    /// the ornament a thematic break is set with.
102    pub content: Content,
103    /// The named strings this element sets when the flow reaches it,
104    /// from `string-set`. This is where a running head's text comes
105    /// from.
106    #[serde(skip_serializing_if = "Vec::is_empty")]
107    pub string_set: Vec<StringSet>,
108    /// The folio the page this element opens takes, from
109    /// `counter-reset: page`.
110    #[serde(skip_serializing_if = "Option::is_none")]
111    pub counter_reset: Option<u32>,
112    /// The number the first note under this element takes, from
113    /// `counter-reset: note`. On the footnote area it restarts the
114    /// numbering on every page.
115    #[serde(skip_serializing_if = "Option::is_none")]
116    pub note_reset: Option<u32>,
117    /// Lines an initial letter is sunk over, from `initial-letter`.
118    /// Fewer than two is no drop cap.
119    pub initial_letter: u16,
120    /// Whether the element sits in the flow or against the page.
121    #[serde(skip_serializing_if = "in_flow")]
122    pub position: Position,
123    /// What the insets say, for an element that is against the page.
124    #[serde(skip_serializing_if = "unplaced")]
125    pub inset: Edges<Inset>,
126    /// Which layer the block paints in, from `z-index`. Higher paints
127    /// later, and `auto` is layer 0.
128    #[serde(skip_serializing_if = "ground_layer")]
129    pub z_index: i32,
130    /// How much of what the block paints shows, from `opacity`: 1 is
131    /// all of it and 0 is none. The blocks inside it show that much
132    /// of their own.
133    #[serde(skip_serializing_if = "opaque")]
134    pub opacity: f32,
135    /// How what the element draws is mixed with what is under it,
136    /// from `mix-blend-mode`.
137    #[serde(skip_serializing_if = "BlendMode::is_normal")]
138    pub mix_blend_mode: BlendMode,
139    /// The mode of the nearest element around this one that names
140    /// one. An element is part of what the elements around it draw.
141    #[serde(skip)]
142    pub blend_around: BlendMode,
143    /// Which side of this element the prose sets on, from `wrap-flow`.
144    #[serde(skip_serializing_if = "wraps_nothing")]
145    pub wrap_flow: WrapFlow,
146    /// Which side of its column an image in the flow moves to, from
147    /// `float`.
148    #[serde(skip_serializing_if = "floats_nowhere")]
149    pub float: Float,
150    /// Which floats the block starts below, from `clear`.
151    #[serde(skip_serializing_if = "clears_nothing")]
152    pub clear: Clear,
153    /// The contour the prose sets around, from `shape-outside`.
154    #[serde(skip_serializing_if = "wraps_its_box")]
155    pub shape_outside: ShapeOutside,
156    /// How far the prose keeps off that contour, in points, from
157    /// `shape-margin`.
158    #[serde(skip_serializing_if = "no_shape_margin")]
159    pub shape_margin: f32,
160    /// Margins in points.
161    pub margin: Edges,
162    /// Padding in points, between the border and the content.
163    #[serde(skip_serializing_if = "no_padding")]
164    pub padding: Edges,
165    /// The four border edges.
166    #[serde(skip_serializing_if = "no_border")]
167    pub border: Edges<Border>,
168    /// How far each corner of the border box is rounded, from
169    /// `border-radius`. The background and the border follow the
170    /// corners.
171    #[serde(skip_serializing_if = "square")]
172    pub border_radius: BorderRadius,
173    /// What is painted behind the block: a tint, and an image over
174    /// it.
175    #[serde(skip_serializing_if = "no_background")]
176    pub background: Background,
177    /// What the decoration does where a page break splits the block.
178    #[serde(skip_serializing_if = "sliced")]
179    pub box_decoration_break: BoxDecorationBreak,
180    /// The width the box is asked to take, from `width`. A table reads
181    /// it off the cells of its first row to size its columns.
182    #[serde(skip_serializing_if = "auto_width")]
183    pub width: Width,
184    /// The height the content box is asked to take, from `height`.
185    #[serde(skip_serializing_if = "auto_width")]
186    pub height: Width,
187    /// The least height the content box takes, from `min-height`.
188    #[serde(skip_serializing_if = "auto_width")]
189    pub min_height: Width,
190    /// The most width an image takes, from `max-width`. `Auto` is
191    /// `none`.
192    #[serde(skip_serializing_if = "auto_width")]
193    pub max_width: Width,
194    /// The most height an image takes, from `max-height`. `Auto` is
195    /// `none`.
196    #[serde(skip_serializing_if = "auto_width")]
197    pub max_height: Width,
198    /// Whether a table's cells share their borders, from
199    /// `border-collapse`.
200    #[serde(skip_serializing_if = "separate")]
201    pub border_collapse: BorderCollapse,
202    /// The marker a list item is set with, from `list-style-type`.
203    #[serde(skip_serializing_if = "disc")]
204    pub list_style_type: ListStyleType,
205    /// Where a page break falls before this element.
206    pub break_before: Break,
207    /// Where one falls after it.
208    pub break_after: Break,
209    /// Whether this element may be split across pages.
210    pub break_inside: Break,
211    /// Whether this element is set across every column of a divided
212    /// page, from `column-span`.
213    #[serde(skip_serializing_if = "one_column")]
214    pub column_span: ColumnSpan,
215    /// The custom properties in force, by name, each with every
216    /// `var()` in its value already replaced. A child inherits them.
217    #[serde(skip_serializing_if = "BTreeMap::is_empty")]
218    pub custom: BTreeMap<String, String>,

328    /// Applies one declaration. `parent_size` is what `em` and
329    /// percentages in `font-size` measure against; `root_size` is
330    /// what `rem` does.
331    pub fn apply(&mut self, declaration: &Declaration, parent_size: f32, root_size: f32) {
332        match declaration {
333            Declaration::FontFamily(families) => self.font_family = families.clone(),
334            Declaration::FontSize(length) => {
335                self.font_size = length.to_points(parent_size, root_size).max(0.0)
336            }
337            Declaration::FontStyle(style) => self.font_style = *style,
338            Declaration::FontWeight(weight) => self.font_weight = *weight,
339            Declaration::Color(color) => self.color = *color,
340            Declaration::LineHeight(line_height) => {
341                self.line_height = line_height.to_multiple(self.font_size, root_size)
342            }
343            Declaration::LetterSpacing(length) => {
344                self.letter_spacing = length.to_points(self.font_size, root_size)
345            }
346            Declaration::FontVariantCaps(caps) => self.font_variant_caps = *caps,
347            Declaration::FontFeatureSettings(settings) => {
348                self.font_feature_settings = settings.clone()
349            }
350            Declaration::FontVariantLigatures(ligatures) => {
351                self.font_variant_ligatures = *ligatures
352            }
353            Declaration::FontVariantNumeric(numeric) => self.font_variant_numeric = *numeric,
354            Declaration::FontVariantAlternates(alternates) => {
355                self.font_variant_alternates = *alternates
356            }
357            Declaration::TextTransform(transform) => self.text_transform = *transform,
358            Declaration::TextDecorationLine(line) => self.text_decoration_line = *line,
359            Declaration::TextDecorationColor(color) => self.text_decoration_color = *color,
360            Declaration::TextDecorationStyle(style) => self.text_decoration_style = *style,
361            Declaration::TextDecorationThickness(length) => {
362                self.text_decoration_thickness =
363                    length.map(|length| length.to_points(self.font_size, root_size).max(0.0))
364            }
365            Declaration::TextAlign(align) => self.text_align = *align,
366            Declaration::TextJustify(justify) => self.text_justify = *justify,
367            Declaration::HangingPunctuation(hanging) => self.hanging_punctuation = *hanging,
368            Declaration::TextIndent(length) => {
369                self.text_indent = length.to_points(self.font_size, root_size)
370            }
371            Declaration::Hyphens(hyphens) => self.hyphens = *hyphens,
372            Declaration::Orphans(lines) => self.orphans = *lines,
373            Declaration::Widows(lines) => self.widows = *lines,
374            Declaration::Page(name) => self.page = name.clone(),
375            Declaration::Content(content) => self.content = content.clone(),
376            Declaration::StringSet(sets) => self.string_set = sets.clone(),
377            Declaration::CounterReset(counters) => {
378                self.counter_reset = counters.page;
379                self.note_reset = counters.note;
380            }
381            Declaration::InitialLetter(lines) => self.initial_letter = *lines,
382            Declaration::Position(position) => self.position = *position,
383            Declaration::Inset(edge, length) => {
384                *self.inset.edge(*edge) = match length {
385                    None => Inset::Auto,
386                    Some(Length::Percent(percent)) => Inset::Percent(*percent),
387                    Some(length) => Inset::Points(length.to_points(self.font_size, root_size)),
388                }
389            }
390            Declaration::ZIndex(layer) => self.z_index = *layer,
391            Declaration::Opacity(opacity) => self.opacity = *opacity,
392            Declaration::MixBlendMode(mode) => self.mix_blend_mode = *mode,
393            Declaration::WrapFlow(wrap) => self.wrap_flow = *wrap,
394            Declaration::Float(float) => self.float = *float,
395            Declaration::Clear(clear) => self.clear = *clear,
396            Declaration::ShapeOutside(shape) => {
397                self.shape_outside = match shape {
398                    ShapeSource::None => ShapeOutside::None,
399                    ShapeSource::Auto => ShapeOutside::Auto,
400                    ShapeSource::Polygon(points) => ShapeOutside::Polygon(
401                        points
402                            .iter()
403                            .map(|(x, y)| ShapePoint {
404                                x: Coord::of(*x, self.font_size, root_size),
405                                y: Coord::of(*y, self.font_size, root_size),
406                            })
407                            .collect(),
408                    ),
409                }
410            }
411            Declaration::ShapeMargin(length) => {
412                self.shape_margin = length.to_points(self.font_size, root_size).max(0.0)
413            }
414            Declaration::Margin(edge, length) => {
415                *self.margin.edge(*edge) = length.to_points(self.font_size, root_size)
416            }
417            Declaration::Padding(edge, length) => {
418                *self.padding.edge(*edge) = length.to_points(self.font_size, root_size).max(0.0)
419            }
420            Declaration::BorderStyle(edge, style) => self.border.edge(*edge).style = *style,
421            Declaration::BorderWidth(edge, length) => {
422                self.border.edge(*edge).width = length.to_points(self.font_size, root_size).max(0.0)
423            }
424            Declaration::BorderColor(edge, color) => self.border.edge(*edge).color = *color,
425            Declaration::BorderRadius(corner, x, y) => {
426                *self.border_radius.corner(*corner) = CornerRadius {
427                    x: Coord::of(*x, self.font_size, root_size),
428                    y: Coord::of(*y, self.font_size, root_size),
429                }
430            }
431            Declaration::BackgroundColor(color) => self.background.color = *color,
432            Declaration::BackgroundImage(url) => self.background.image = url.clone(),
433            Declaration::BackgroundRepeat(repeat) => self.background.repeat = *repeat,
434            Declaration::BackgroundSize(size) => {
435                self.background.size = computed_size(*size, self.font_size, root_size)
436            }
437            Declaration::BackgroundPosition(x, y) => {
438                self.background.position = BackgroundPosition {
439                    x: Coord::of(*x, self.font_size, root_size),
440                    y: Coord::of(*y, self.font_size, root_size),
441                }
442            }
443            Declaration::BoxDecorationBreak(value) => self.box_decoration_break = *value,
444            Declaration::Width(width) => self.width = self.size(*width, root_size),
445            Declaration::Height(height) => self.height = self.size(*height, root_size),
446            Declaration::MinHeight(height) => self.min_height = self.size(*height, root_size),
447            Declaration::MaxWidth(width) => self.max_width = self.size(*width, root_size),
448            Declaration::MaxHeight(height) => self.max_height = self.size(*height, root_size),
449            Declaration::BorderCollapse(value) => self.border_collapse = *value,
450            Declaration::ListStyleType(value) => self.list_style_type = *value,
451            Declaration::BreakBefore(value) => self.break_before = *value,
452            Declaration::BreakAfter(value) => self.break_after = *value,
453            Declaration::BreakInside(value) => self.break_inside = *value,
454            Declaration::ColumnSpan(value) => self.column_span = *value,
455            // The cascade resolves both before anything is applied.
456            Declaration::Custom(_) | Declaration::Pending(_) => {}
457        }
458    }

480            Declaration::FontVariantNumeric(_) => {
481                self.font_variant_numeric = base.font_variant_numeric
482            }
483            Declaration::FontVariantAlternates(_) => {
484                self.font_variant_alternates = base.font_variant_alternates
485            }
486            Declaration::TextTransform(_) => self.text_transform = base.text_transform,
487            Declaration::TextDecorationLine(_) => {
488                self.text_decoration_line = base.text_decoration_line

490            Declaration::TextDecorationColor(_) => {

499            Declaration::TextAlign(_) => self.text_align = base.text_align,
500            Declaration::TextJustify(_) => self.text_justify = base.text_justify,
501            Declaration::HangingPunctuation(_) => {
502                self.hanging_punctuation = base.hanging_punctuation
503            }
504            Declaration::TextIndent(_) => self.text_indent = base.text_indent,
505            Declaration::Hyphens(_) => self.hyphens = base.hyphens,
506            Declaration::Orphans(_) => self.orphans = base.orphans,
507            Declaration::Widows(_) => self.widows = base.widows,
508            Declaration::Page(_) => self.page = base.page.clone(),
509            Declaration::Content(_) => self.content = base.content.clone(),

515            Declaration::InitialLetter(_) => self.initial_letter = base.initial_letter,
516            Declaration::Position(_) => self.position = base.position,
517            Declaration::Inset(edge, _) => *self.inset.edge(*edge) = base.inset.get(*edge),
518            Declaration::ZIndex(_) => self.z_index = base.z_index,
519            Declaration::Opacity(_) => self.opacity = base.opacity,
520            Declaration::MixBlendMode(_) => self.mix_blend_mode = base.mix_blend_mode,
521            Declaration::WrapFlow(_) => self.wrap_flow = base.wrap_flow,
522            Declaration::Float(_) => self.float = base.float,
523            Declaration::Clear(_) => self.clear = base.clear,
524            Declaration::ShapeOutside(_) => self.shape_outside = base.shape_outside.clone(),
525            Declaration::ShapeMargin(_) => self.shape_margin = base.shape_margin,

583    /// What this style, computed for `::first-line`, changes about
584    /// the element it was computed beside.
585    pub fn first_line_over(&self, element: &ComputedStyle) -> crate::lines::FirstLine {
586        fn set<T>(differs: bool, value: T) -> Option<T> {
587            differs.then_some(value)
588        }
589        let face = crate::lines::Face {
590            family: set(self.font_family != element.font_family, self.font_id),
591            italic: set(
592                self.font_style != element.font_style,
593                self.font_style == FontStyle::Italic,
594            ),
595            weight: set(self.font_weight != element.font_weight, self.font_weight),
596        };
597        crate::lines::FirstLine {
598            face: (face != crate::lines::Face::default()).then_some(face),
599            size: set(self.font_size != element.font_size, self.font_size),
600            line_height: set(self.line_height != element.line_height, self.line_height),
601            letter_spacing: set(
602                self.letter_spacing != element.letter_spacing,
603                self.letter_spacing,
604            ),
605            caps: set(
606                self.font_variant_caps != element.font_variant_caps,
607                self.font_variant_caps,
608            ),
609            transform: set(
610                self.text_transform != element.text_transform,
611                self.text_transform,
612            ),
613            color: set(self.color != element.color, self.color),
614            decoration: set(self.decoration() != element.decoration(), self.decoration()),
615        }
616    }

618    /// This block's border box inside `measure`, laid out at `x` from
619    /// the enclosing content box's leading edge: `(leading edge,
620    /// width)`, with only the margins taken off.
621    pub fn border_box(&self, x: f32, measure: f32) -> (f32, f32) {
622        (
623            x + self.margin.left,
624            (measure - self.margin.inline()).max(0.0),
625        )
626    }

628    /// Whether prose is set around this element rather than over it:
629    /// the sheet has taken it out of the flow, and it excludes a side.
630    ///
631    /// This is the one place a contour is read, so it is also what
632    /// decides whether one is traced. An element still in the flow
633    /// takes a band of its own and nothing sets beside it, so a
634    /// contour on it would answer a question nobody asks.
635    pub fn excludes(&self) -> bool {
636        self.position == Position::Absolute && self.wrap_flow != WrapFlow::Auto
637    }

639    /// This block's content box inside `measure`: `(leading edge, the
640    /// measure its lines break to)`. Margin, border and padding come
641    /// off both edges, and the leading edge moves in by what they
642    /// take on the left.
643    pub fn content_box(&self, x: f32, measure: f32) -> (f32, f32) {
644        let border = self.border.widths();
645        let leading = self.margin.left + border.left + self.padding.left;
646        let trailing = self.margin.right + border.right + self.padding.right;
647        (x + leading, (measure - leading - trailing).max(0.0))
648    }

650    /// The box this style paints around the runs of an inline
651    /// element, and `None` where it paints none and takes no width.
652    ///
653    /// Padding counts even where nothing is painted over it: it is
654    /// width, and the line breaks against it.
655    pub fn inline_box(&self) -> Option<crate::lines::InlineBox> {
656        let padding = self.padding;
657        let border = self.border.widths();
658        let paints = self.background.paints() || self.border.paints();
659        let spaced = padding != Edges::all(0.0);
660        (paints || spaced).then_some(crate::lines::InlineBox {
661            padding,
662            border,
663            cloned: self.box_decoration_break == BoxDecorationBreak::Clone,
664        })
665    }

717/// What the cascade makes of a written `background-size`: `em` and
718/// `rem` against the font size in force, a percentage kept as one,
719/// because the box it measures against is sized in the layout pass.
720pub(crate) fn computed_size(size: SizeSource, font_size: f32, root_size: f32) -> BackgroundSize {
721    let axis =
722        |length: Option<Length>| length.map(|length| Coord::of(length, font_size, root_size));
723    match size {
724        SizeSource::Auto => BackgroundSize::Auto,
725        SizeSource::Cover => BackgroundSize::Cover,
726        SizeSource::Contain => BackgroundSize::Contain,
727        SizeSource::Fixed(width, height) => BackgroundSize::Fixed {
728            width: axis(width),
729            height: axis(height),
730        },
731    }
732}
