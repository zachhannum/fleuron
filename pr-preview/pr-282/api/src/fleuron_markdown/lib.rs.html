1//! The markdown frontend: source text in, content tree out.
2//!
3//! [`to_sections`] is the primitive, and it is per-source rather than
4//! per-book. One source yields one or more sections, because both
5//! directions are ordinary: a novel may arrive as a single file that
6//! has to become sixty chapters, or as sixty files that each become
7//! one. [`Sections`] is which of those a source is, said out loud.
8//!
9//! Composing sources into a book is [`assemble`], a step of its own.
10//! The caller orders the sources and decides the metadata, so nothing
11//! here has to arbitrate between two files that both claim a title.
12//!
13//! # What the vocabulary cannot express
14//!
15//! The content tree is a book's vocabulary: headings, prose,
16//! blockquotes, scene breaks, page and column breaks, images, lists,
17//! tables. Markdown is wider
18//! than that. Constructs outside it degrade to prose and say so
19//! through the diagnostics channel, with the line and column they
20//! were written at. Text is never dropped, because a manuscript that
21//! quietly loses a paragraph is worse than one that warns about a
22//! code block.
23//!
24//! # Dialects
25//!
26//! [`Dialect`] is a set of switches, so the departures a host's
27//! markdown makes from CommonMark are configuration rather than a
28//! second mapping to keep in step with this one.

42/// How a source is read: where its sections begin, and which
43/// markdown it is written in.
44#[derive(Debug, Clone, Default, PartialEq, Eq, Hash)]
45pub struct Options {
46    /// Where a new section begins.
47    pub sections: Sections,
48    /// The departures from CommonMark the source is allowed.
49    pub dialect: Dialect,
50}
51
52/// Where one source's sections begin.
53///
54/// Sections are what the fragmenter starts a page on, so this is the
55/// decision that sets a book's page count before any styling does.
56#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
57pub enum Sections {
58    /// The whole source is one section: a file per chapter.
59    Whole,
60    /// A heading at this level or shallower opens one: a file per
61    /// book. `AtHeading(H2)` cuts at `#` and `##` alike, so a
62    /// manuscript that opens parts with `#` and chapters with `##`
63    /// starts a page on both.
64    AtHeading(HeadingLevel),
65}
66
67impl Default for Sections {
68    fn default() -> Sections {
69        Sections::AtHeading(HeadingLevel::H1)
70    }
71}
72
73impl Sections {
74    /// Whether a heading at this level opens a section.
75    fn opens(self, level: HeadingLevel) -> bool {
76        match self {
77            Sections::Whole => false,
78            Sections::AtHeading(deepest) => u8::from(level) <= u8::from(deepest),
79        }
80    }
81}

83/// Which markdown a source is written in.
84///
85/// [`Dialect::fleuron`] is the default. The other three are named
86/// after whose markdown they read.
87#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
88pub struct Dialect {
89    /// A leading `---` block is metadata rather than a scene break.
90    pub frontmatter: bool,
91    /// A line of nothing but `{.class #id}` names the block under it,
92    /// and a heading or an image takes the same run written after it.
93    pub attributes: bool,
94    /// Tables, written as GitHub writes them: a header row, a
95    /// delimiter row, and one line per row.
96    pub tables: bool,
97    /// GitHub's other additions: strikethrough and task lists.
98    pub gfm: bool,
99    /// `[[wikilinks]]`, as Obsidian writes them.
100    pub wikilinks: bool,
101    /// `--` and `"` become dashes and curly quotes at parse rather
102    /// than in the manuscript.
103    pub smart_punctuation: bool,
104    /// A line of nothing but `\pagebreak` or `\columnbreak` starts a
105    /// new page or the next column.
106    pub breaks: bool,
107    /// Footnotes, as GitHub writes them: `[^1]` in the prose, and
108    /// `[^1]: the note` on a line of its own.
109    pub footnotes: bool,
110    /// `%%a comment%%`, as Obsidian writes one: in a line of prose
111    /// or over several blocks, and never set.
112    pub comments: bool,
113    /// `==highlighted==`, as Obsidian writes it.
114    pub highlights: bool,
115    /// `^name` at the end of a paragraph or a list item, which is
116    /// what an Obsidian link names the block by, and is never set.
117    pub block_ids: bool,
118    /// A blockquote that opens with `[!type]` is a callout: the type
119    /// is a class on the quotation, and the rest of the line its
120    /// title.
121    pub callouts: bool,
122}
123
124impl Default for Dialect {
125    fn default() -> Dialect {
126        Dialect::fleuron()
127    }
128}
129
130impl Dialect {
131    /// CommonMark, a frontmatter block, attribute lines, tables and
132    /// breaks: the markdown a manuscript for this engine is written
133    /// in, and what a source is read as unless the host says
134    /// otherwise.
135    pub fn fleuron() -> Dialect {
136        Dialect {
137            frontmatter: true,
138            attributes: true,
139            tables: true,
140            gfm: false,
141            wikilinks: false,
142            smart_punctuation: false,
143            breaks: true,
144            footnotes: true,
145            comments: false,
146            highlights: false,
147            block_ids: false,
148            callouts: false,
149        }
150    }
151
152    /// CommonMark, frontmatter included, and nothing besides. A brace
153    /// run, a table and `\pagebreak` are prose here.
154    pub fn common_mark() -> Dialect {
155        Dialect {
156            attributes: false,
157            tables: false,
158            breaks: false,
159            footnotes: false,
160            ..Dialect::fleuron()
161        }
162    }
163
164    /// What an Obsidian vault contains.
165    pub fn obsidian() -> Dialect {
166        Dialect {
167            gfm: true,
168            wikilinks: true,
169            comments: true,
170            highlights: true,
171            block_ids: true,
172            callouts: true,
173            ..Dialect::fleuron()
174        }
175    }
176
177    /// GitHub-flavoured markdown.
178    pub fn gfm() -> Dialect {
179        Dialect {
180            gfm: true,
181            ..Dialect::fleuron()
182        }
183    }
184}

186/// Reads one source into sections, and everything the reading had to
187/// complain about.
188///
189/// `source` names the file for diagnostics and becomes every
190/// section's `source`; it is what [`fleuron::session::Session::replace_source`]
191/// replaces by. Node ids are left unassigned: they are assigned in
192/// document order over a whole book, which is [`assemble`]'s job.
193pub fn to_sections(text: &str, source: &str, options: &Options) -> (Vec<Section>, Vec<Warning>) {
194    convert::run(text, source, options)
195}

197/// Composes ordered sections into a book under the metadata the
198/// caller decided on, and numbers the tree.
199///
200/// The metadata is an argument because a book of many files has no
201/// one file to read it from. Each chapter's frontmatter stays with
202/// the chapter; the work is named here.
203///
204/// ```
205/// use fleuron::content::Metadata;
206/// use fleuron_markdown::{Options, Sections, assemble, to_sections};
207///
208/// // A file per chapter, so nothing is cut at a heading.
209/// let reading = Options {
210///     sections: Sections::Whole,
211///     ..Options::default()
212/// };
213/// let chapters = [
214///     ("ch01.md", "---\ntitle: The Ambassador\n---\n\nHe arrived.\n"),
215///     ("ch02.md", "---\ntitle: A Cold Reception\n---\n\nNobody met him.\n"),
216/// ];
217///
218/// let mut sections = Vec::new();
219/// for (name, text) in chapters {
220///     sections.extend(to_sections(text, name, &reading).0);
221/// }
222///
223/// let book = assemble(
224///     Metadata {
225///         title: Some("The Levant Papers".into()),
226///         ..Metadata::default()
227///     },
228///     sections,
229/// );
230///
231/// assert_eq!(book.metadata.title.as_deref(), Some("The Levant Papers"));
232/// assert_eq!(book.sections[0].title.as_deref(), Some("The Ambassador"));
233/// assert_eq!(book.sections[1].title.as_deref(), Some("A Cold Reception"));
234/// ```
235pub fn assemble(metadata: Metadata, sections: Vec<Section>) -> Book {
236    let mut book = Book { metadata, sections };
237    book.assign_node_ids();
238    book
239}

270    /// Each source read whole as an Obsidian vault holds it, composed
271    /// in the order given, and what the frontend complained about.
272    fn composed(sources: &[(&str, &str)]) -> (Book, Vec<Warning>) {
273        let vault = Options {
274            sections: Sections::Whole,
275            dialect: Dialect::obsidian(),
276        };
277        let mut sections = Vec::new();
278        let mut warnings = Vec::new();
279        for (name, text) in sources {
280            let (read, complaints) = to_sections(text, name, &vault);
281            sections.extend(read);
282            warnings.extend(complaints);
283        }
284        (assemble(Metadata::default(), sections), warnings)
285    }

304    /// The `nth` heading read from `source`, counted from 0.
305    fn heading_in(book: &Book, source: &str, nth: usize) -> LinkTarget {
306        let node = book
307            .sections
308            .iter()
309            .filter(|section| section.source.as_deref() == Some(source))
310            .flat_map(|section| &section.blocks)
311            .filter_map(|block| match block {
312                Block::Heading { id, .. } => Some(*id),
313                _ => None,
314            })
315            .nth(nth)
316            .expect("the source has the heading");
317        LinkTarget::Node(node)
318    }

320    /// The url of every link in the paragraphs read from `source`.
321    fn urls(book: &Book, source: &str) -> Vec<String> {
322        book.sections
323            .iter()
324            .filter(|section| section.source.as_deref() == Some(source))
325            .flat_map(|section| &section.blocks)
326            .filter_map(|block| match block {
327                Block::Paragraph { inlines, .. } => Some(inlines),
328                _ => None,
329            })
330            .flatten()
331            .filter_map(|inline| match inline {
332                Inline::Link { url, .. } => Some(url.clone()),
333                _ => None,
334            })
335            .collect()
336    }

499    /// A chapter file's frontmatter is the chapter's. Book metadata
500    /// is handed to assembly, so nothing here has to guess which of
501    /// sixty files was describing the work.
502    #[test]
503    fn assembly_takes_the_metadata_it_is_given() {
504        let per_file = Options {
505            sections: Sections::Whole,
506            ..Options::default()
507        };
508        let mut sections = Vec::new();
509        for (name, markdown) in [
510            (
511                "ch01.md",
512                "---\ntitle: The Ambassador\n---\n\nHe arrived.\n",
513            ),
514            (
515                "ch02.md",
516                "---\ntitle: A Cold Reception\n---\n\nNobody met him.\n",
517            ),
518        ] {
519            sections.extend(to_sections(markdown, name, &per_file).0);
520        }
521        let book = assemble(
522            Metadata {
523                title: Some("The Levant Papers".into()),
524                author: Some("E. Marsh".into()),
525                ..Metadata::default()
526            },
527            sections,
528        );
529
530        assert_eq!(book.metadata.title.as_deref(), Some("The Levant Papers"));
531        assert_eq!(book.metadata.author.as_deref(), Some("E. Marsh"));
532        let chapters: Vec<Option<&str>> = book
533            .sections
534            .iter()
535            .map(|section| section.title.as_deref())
536            .collect();
537        assert_eq!(chapters, [Some("The Ambassador"), Some("A Cold Reception")]);
