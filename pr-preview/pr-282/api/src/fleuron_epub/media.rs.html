1//! The files a book refers to by url, copied into the container: the
2//! images of the book and its sheets, and the fonts of its sheets.
3//! Nothing here decodes a file. The first bytes name its type.
4
5use std::collections::BTreeMap;
6
7/// What kind of file a url names, which is what decides the type the
8/// bytes have to be.
9#[derive(Debug, Clone, Copy, PartialEq, Eq)]
10pub enum Kind {
11    Image,
12    Font,
13}
14
15/// One file in the container.
16#[derive(Debug, Clone)]
17pub struct Resource {
18    /// Its path in the container, from the package directory.
19    pub href: String,
20    /// Its media type, for the manifest.
21    pub media_type: &'static str,
22    pub bytes: Vec<u8>,
23}
24
25/// Why a url did not become a file in the container.
26#[derive(Debug, Clone, Copy, PartialEq, Eq)]
27pub enum Refused {
28    /// The host had no bytes for it.
29    Missing,
30    /// The bytes are not a type an EPUB can hold.
31    Unknown,
32}
33
34/// Every file the book refers to, in the order it was first asked
35/// for.
36#[derive(Debug, Default)]
37pub struct Resources {
38    files: Vec<Resource>,
39    by_url: BTreeMap<(String, bool), Result<usize, Refused>>,
40}
41
42impl Resources {
43    /// The path of the file behind `url`, loading it the first time
44    /// it is asked for.
45    pub fn resolve(
46        &mut self,
47        url: &str,
48        kind: Kind,
49        load: impl FnOnce(&str) -> Option<Vec<u8>>,
50    ) -> Result<&str, Refused> {
51        let key = (url.to_string(), kind == Kind::Font);
52        if !self.by_url.contains_key(&key) {
53            let found = match load(url) {
54                None => Err(Refused::Missing),
55                Some(bytes) => match media_type(&bytes, kind) {
56                    None => Err(Refused::Unknown),
57                    Some((media_type, extension)) => {
58                        let stem = match kind {
59                            Kind::Image => "image",
60                            Kind::Font => "font",
61                        };
62                        let number = self.files.len() + 1;
63                        self.files.push(Resource {
64                            href: format!("media/{stem}-{number}.{extension}"),
65                            media_type,
66                            bytes,
67                        });
68                        Ok(self.files.len() - 1)
69                    }
70                },
71            };
72            self.by_url.insert(key.clone(), found);
73        }
74        match self.by_url[&key] {
75            Ok(index) => Ok(&self.files[index].href),
76            Err(refused) => Err(refused),
77        }
78    }
79
80    /// Every file, in the order it was first asked for.
81    pub fn files(&self) -> &[Resource] {
82        &self.files
83    }
84
85    /// The same, taken out of the table.
86    pub fn into_files(self) -> Vec<Resource> {
87        self.files
88    }
89}

91/// The media type and a file extension for `bytes`, read from their
92/// first bytes. Only the types an EPUB holds without a fallback.
93fn media_type(bytes: &[u8], kind: Kind) -> Option<(&'static str, &'static str)> {
94    match kind {
95        Kind::Image => {
96            if bytes.starts_with(b"\x89PNG\r\n\x1a\n") {
97                Some(("image/png", "png"))
98            } else if bytes.starts_with(&[0xFF, 0xD8, 0xFF]) {
99                Some(("image/jpeg", "jpg"))
100            } else if bytes.starts_with(b"GIF87a") || bytes.starts_with(b"GIF89a") {
101                Some(("image/gif", "gif"))
102            } else if bytes.len() >= 12 && &bytes[0..4] == b"RIFF" && &bytes[8..12] == b"WEBP" {
103                Some(("image/webp", "webp"))
104            } else if is_svg(bytes) {
105                Some(("image/svg+xml", "svg"))
106            } else {
107                None
108            }
109        }
110        Kind::Font => match bytes.get(0..4)? {
111            [0, 1, 0, 0] | b"true" => Some(("font/ttf", "ttf")),
112            b"OTTO" => Some(("font/otf", "otf")),
113            b"wOFF" => Some(("font/woff", "woff")),
114            b"wOF2" => Some(("font/woff2", "woff2")),
115            _ => None,
116        },
117    }
118}

120/// An SVG file opens with its root element, perhaps after an XML
121/// declaration, a doctype or a comment.
122fn is_svg(bytes: &[u8]) -> bool {
123    let head = &bytes[..bytes.len().min(1024)];
124    let Ok(text) = std::str::from_utf8(head) else {
125        return false;
126    };
127    let text = text.trim_start_matches('\u{feff}').trim_start();
128    (text.starts_with('<')) && text.contains("<svg")
129}

131#[cfg(test)]
132mod tests {
133    use super::*;
134
135    const PNG: &[u8] = b"\x89PNG\r\n\x1a\n\0\0\0\rIHDR";
136
137    /// A url asked for twice is one file, loaded once.
138    #[test]
139    fn a_url_is_loaded_once_and_named_in_order() {
140        let mut resources = Resources::default();
141        let mut loads = 0;
142        let mut load = |_: &str| {
143            loads += 1;
144            Some(PNG.to_vec())
145        };
146        assert_eq!(
147            resources.resolve("a.png", Kind::Image, &mut load),
148            Ok("media/image-1.png")
149        );
150        assert_eq!(
151            resources.resolve("b.png", Kind::Image, &mut load),
152            Ok("media/image-2.png")
153        );
154        assert_eq!(
155            resources.resolve("a.png", Kind::Image, &mut load),
156            Ok("media/image-1.png")
157        );
158        assert_eq!(loads, 2);
159        assert_eq!(resources.files()[0].media_type, "image/png");
160    }
161
162    #[test]
163    fn a_file_the_host_lacks_or_the_format_cannot_hold_is_refused() {
164        let mut resources = Resources::default();
165        assert_eq!(
166            resources.resolve("gone.png", Kind::Image, |_| None),
167            Err(Refused::Missing)
168        );
169        assert_eq!(
170            resources.resolve("a.bmp", Kind::Image, |_| Some(b"BM....".to_vec())),
171            Err(Refused::Unknown)
172        );
173        // A font is not an image, even at the same url.
174        assert_eq!(
175            resources.resolve("a.png", Kind::Font, |_| Some(PNG.to_vec())),
176            Err(Refused::Unknown)
177        );
178        assert!(resources.files().is_empty());
179    }
180
181    #[test]
182    fn the_first_bytes_name_the_type() {
183        let cases: [(&[u8], Kind, &str); 8] = [
184            (PNG, Kind::Image, "image/png"),
185            (&[0xFF, 0xD8, 0xFF, 0xE0], Kind::Image, "image/jpeg"),
186            (b"GIF89a..", Kind::Image, "image/gif"),
187            (b"RIFF\0\0\0\0WEBPVP8 ", Kind::Image, "image/webp"),
188            (
189                b"<?xml version=\"1.0\"?>\n<svg xmlns=\"\">",
190                Kind::Image,
191                "image/svg+xml",
192            ),
193            (&[0, 1, 0, 0, 0, 9], Kind::Font, "font/ttf"),
194            (b"OTTO\0\0", Kind::Font, "font/otf"),
195            (b"wOF2\0\0", Kind::Font, "font/woff2"),
196        ];
197        for (bytes, kind, expected) in cases {
198            assert_eq!(media_type(bytes, kind).map(|(t, _)| t), Some(expected));
199        }
200    }
