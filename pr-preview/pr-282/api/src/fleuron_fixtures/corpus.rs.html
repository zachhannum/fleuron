1//! The corpus: public-domain books, checked in as markdown, handed to
2//! the harness as content trees. Both texts are Project Gutenberg
3//! editions, vendored from the classic-books-markdown collection.
4//!
5//! Two books, sized for the two questions the harness asks. Pride and
6//! Prejudice is the book-scale gate: a complete novel that sets almost
7//! exactly the 300 pages the budgets are written against. The Count of
8//! Monte Cristo is four times that, and answers whether the pipeline
9//! stays linear when a manuscript stops being polite.
10//!
11//! Real prose, not generated: dialogue by the page, em-dashes, long
12//! Latinate words that hyphenation has opinions about, and chapter
13//! structure that makes the fragmenter insert real blanks. Generated
14//! text is uniform, and uniform text hides the tail cases that make a
15//! layout engine slow.
16//!
17//! The markdown is embedded in the binary rather than read at runtime:
18//! the harness runs under wasi with no preopened directories, and a
19//! measurement that depends on a working directory is a measurement
20//! that reports differently depending on where it was started.
21
22use fleuron::content::{Book, HeadingLevel};
23use fleuron_markdown::{Options, Sections};
24
25/// One book in the fixture corpus.
26#[derive(Debug, Clone, Copy, PartialEq, Eq)]
27pub enum Corpus {
28    /// ~122k words. The book-scale gate: a complete novel at the 300
29    /// pages the budgets are written against.
30    PrideAndPrejudice,
31    /// ~459k words. Four times the gate, to catch the superlinearity
32    /// a single book-sized run cannot see.
33    MonteCristo,
34}
35
36impl Corpus {
37    /// Every book in the corpus, smallest first.
38    pub const ALL: [Corpus; 2] = [Corpus::PrideAndPrejudice, Corpus::MonteCristo];
39
40    /// The book the budgets are written against.
41    pub const GATE: Corpus = Corpus::PrideAndPrejudice;
42
43    /// Short identifier, for bench names and gate output.
44    pub fn slug(self) -> &'static str {
45        match self {
46            Corpus::PrideAndPrejudice => "pride-and-prejudice",
47            Corpus::MonteCristo => "monte-cristo",
48        }
49    }
50
51    /// The file the markdown was vendored as, and the `source` every
52    /// section of the parsed book is named by.
53    pub fn source(self) -> &'static str {
54        match self {
55            Corpus::PrideAndPrejudice => "pride-and-prejudice.md",
56            Corpus::MonteCristo => "the-count-of-monte-cristo.md",
57        }
58    }
59
60    /// The vendored markdown.
61    pub fn markdown(self) -> &'static str {
62        match self {
63            Corpus::PrideAndPrejudice => {
64                include_str!("../../../fixtures/corpus/pride-and-prejudice.md")
65            }
66            Corpus::MonteCristo => {
67                include_str!("../../../fixtures/corpus/the-count-of-monte-cristo.md")
68            }
69        }
70    }
71
72    /// How the frontend reads this corpus: both books are one file
73    /// with a heading per chapter, and Monte Cristo opens its volumes
74    /// a level shallower than that.
75    pub fn options() -> Options {
76        Options {
77            sections: Sections::AtHeading(HeadingLevel::H2),
78            ..Options::default()
79        }
80    }
81
82    /// The book as a content tree, node ids assigned. It goes through
83    /// the frontend that ships, so the measured path is the shipped
84    /// path.
85    pub fn book(self) -> Book {
86        self.parse(self.markdown())
87    }
88
89    /// The same, over markdown the caller supplies. The bench and the
90    /// gate both time this, so what they time is one call rather than
91    /// a stage reached through a private method.
92    pub fn parse(self, markdown: &str) -> Book {
93        let (sections, _) =
94            fleuron_markdown::to_sections(markdown, self.source(), &Self::options());
95        fleuron_markdown::assemble(fleuron_markdown::frontmatter(markdown), sections)
96    }
97}

99#[cfg(test)]
100mod tests {
101    use super::*;
102    use fleuron::content::Block;
103
104    /// Both books parse into content trees with the structure the
105    /// harness measures against: metadata off the frontmatter, a
106    /// section per chapter, and prose in every one of them.
107    #[test]
108    fn corpus_books_parse_into_content_trees() {
109        for corpus in Corpus::ALL {
110            let book = corpus.book();
111            assert!(
112                book.metadata.title.is_some(),
113                "{}: no title parsed",
114                corpus.slug()
115            );
116            assert!(
117                book.metadata.author.is_some(),
118                "{}: no author parsed",
119                corpus.slug()
120            );
121            assert!(
122                book.sections.len() > 40,
123                "{}: {} sections, expected a chapter apiece",
124                corpus.slug(),
125                book.sections.len()
126            );
127            for section in &book.sections {
128                assert_eq!(section.source.as_deref(), Some(corpus.source()));
129                assert!(
130                    section
131                        .blocks
132                        .iter()
133                        .any(|b| matches!(b, Block::Paragraph { .. } | Block::Heading { .. })),
134                    "{}: empty section",
135                    corpus.slug()
136                );
137            }
138        }
139    }
140
141    /// The gate book is the smaller one: budgets are written against
142    /// book scale, and the big book exists to catch superlinearity.
143    #[test]
144    fn the_gate_book_is_the_smaller_one() {
145        assert_eq!(Corpus::GATE, Corpus::ALL[0]);
146        assert!(Corpus::PrideAndPrejudice.markdown().len() < Corpus::MonteCristo.markdown().len());
147    }
148}
