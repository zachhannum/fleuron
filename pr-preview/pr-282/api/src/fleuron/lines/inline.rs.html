1//! The boxes the inline elements on a line paint there.
2//!
3//! An inline element has no extent of its own: it has one per line
4//! its runs reach, and the geometry is known only once the line is
5//! broken and its runs are cut. This works that out from the runs of
6//! one span of one line, and charges each run with the edges that
7//! open and close on it, so a painter placing glyphs walks past the
8//! padding rather than under it.
9
10use std::ops::Range;
11
12use crate::linebox::Strut;
13
14use super::LineLayout;
15use super::flatten::{BoxSpan, FlatParagraph};
16use super::line::{InlineFragment, ShapedRun};

18/// One box of `flat` over the runs of one span: the box, and the
19/// first and last of those runs it covers.
20struct Covering<'a> {
21    span: &'a BoxSpan,
22    first: usize,
23    last: usize,
24}
25
26/// A box whose leading edge has been placed and whose trailing edge
27/// has not: the fragment it is filling, and the run it closes on.
28struct Open<'a> {
29    at: usize,
30    span: &'a BoxSpan,
31    last: usize,
32}
33
34impl LineLayout<'_> {
35    /// The boxes the inline elements covering `runs` paint there,
36    /// outermost first.
37    ///
38    /// `runs` are the runs of the span `span` of one line, and `text`
39    /// is the bytes of the paragraph that span holds. Each box is
40    /// placed from the span's own leading edge, so the offset
41    /// alignment gives the span later moves the box with the text.
42    /// Each run comes out charged with the edges that fall on it.
43    pub(super) fn inline_fragments(
44        &self,
45        flat: &FlatParagraph,
46        runs: &mut [ShapedRun],
47        span: usize,
48        text: Range<usize>,
49    ) -> Vec<InlineFragment> {
50        let covering: Vec<Covering<'_>> = flat
51            .boxes
52            .iter()
53            .filter(|span| !span.range.is_empty())
54            .filter_map(|span| {
55                let (first, last) = covered(runs, &span.range)?;
56                Some(Covering { span, first, last })
57            })
58            .collect();
59        if covering.is_empty() {
60            return Vec::new();
61        }
62        let mut fragments: Vec<InlineFragment> = Vec::new();
63        let mut open: Vec<Open<'_>> = Vec::new();
64        let mut x = 0.0;
65        for index in 0..runs.len() {
66            for covering in covering.iter().filter(|covering| covering.first == index) {
67                let box_ = &covering.span.box_;
68                let opens = box_.cloned || covering.span.range.start >= text.start;
69                let closes = box_.cloned || covering.span.range.end <= text.end;
70                let leading = if opens { box_.leading() } else { 0.0 };
71                runs[index].lead += leading;
72                let (above, below) = self.content_area(&runs[index..=covering.last]);
73                fragments.push(InlineFragment {
74                    node: covering.span.node,
75                    span,
76                    x,
77                    width: 0.0,
78                    above: above + box_.above(),
79                    below: below + box_.below(),
80                    opens,
81                    closes,
82                });
83                x += leading;
84                open.push(Open {
85                    at: fragments.len() - 1,
86                    span: covering.span,
87                    last: covering.last,
88                });
89            }
90            x += self.run_width(&runs[index]);
91            // The innermost box closes first, so its trailing edge
92            // falls inside the one around it.
93            while open.last().is_some_and(|open| open.last == index) {
94                let open = open.pop().expect("a box was open");
95                let trailing = match fragments[open.at].closes {
96                    true => open.span.box_.trailing(),
97                    false => 0.0,
98                };
99                runs[index].trail += trailing;
100                x += trailing;
101                fragments[open.at].width = x - fragments[open.at].x;
102            }
103        }
104        fragments
105    }
106
107    /// The content area the runs of one inline element make: the
108    /// tallest ascent and the deepest descent of the faces they are
109    /// set in, in points from the baseline. A face's leading is no
110    /// part of it, so the box holds the letters and nothing more.
111    fn content_area(&self, runs: &[ShapedRun]) -> (f32, f32) {
112        let (mut above, mut below) = (0.0f32, 0.0f32);
113        for run in runs {
114            let Some(metrics) = self.registry.metrics(run.font_id) else {
115                continue;
116            };
117            let area = Strut::content(metrics, run.size);
118            above = above.max(area.above);
119            below = below.max(area.below);
120        }
121        (above, below)
122    }
123
124    /// One run's glyphs in points.
125    fn run_width(&self, run: &ShapedRun) -> f32 {
126        let upem = self
127            .registry
128            .metrics(run.font_id)
129            .map(|metrics| metrics.units_per_em as f32)
130            .unwrap_or(1000.0);
131        run.advance as f32 / upem * run.size
132    }
133}
134
135/// The first and last of `runs` one box covers, and `None` where it
136/// covers none of them.
137fn covered(runs: &[ShapedRun], range: &Range<usize>) -> Option<(usize, usize)> {
138    let mut first = None;
139    let mut last = 0;
140    for (index, run) in runs.iter().enumerate() {
141        let start = run.text_start as usize;
142        if start < range.end && start + run.text.len() > range.start {
143            first.get_or_insert(index);
144            last = index;
145        }
146    }
147    first.map(|first| (first, last))
148}

168    /// Part: horizontal padding and border widths are charged to the
169    /// breaker, so a paragraph that fits on one line without them
170    /// needs a wider measure with them.
171    #[test]
172    fn the_edges_of_a_box_are_width_the_breaker_measures() {
173        assert_eq!(
174            layout_boxed(&tagged(), 200.0, &Boxed(Vec::new())).len(),
175            1,
176            "the fixture did not fit on one line",
177        );
178        for box_ in [padded(4.0), bordered(4.0)] {
179            let styles = Boxed(vec![(tag(), box_)]);
180            let narrowest = |styles: &Boxed| -> f32 {
181                (400..1000)
182                    .map(|steps| steps as f32 * 0.25)
183                    .find(|measure| layout_boxed(&tagged(), *measure, styles).len() == 1)
184                    .expect("the paragraph sets in one line at some measure")
185            };
186            let grew = narrowest(&styles) - narrowest(&Boxed(Vec::new()));
187            assert!(
188                (grew - 8.0).abs() <= 0.25,
189                "the measure grew by {grew}pt, not by the two edges of the box",
190            );
191        }
192    }

194    /// Part: a line yields one border box per inline element it
195    /// covers, outermost first, and a line with no box on it yields
196    /// none.
197    #[test]
198    fn a_line_yields_one_box_per_inline_element() {
199        let outer = NodeId::new(3);
200        let inlines = vec![emphasis(outer, {
201            let mut children = one_run("roll ");
202            children.push(code(tag(), "2d6"));
203            children
204        })];
205        let styles = Boxed(vec![(outer, padded(2.0)), (tag(), padded(4.0))]);
206        let lines = layout_boxed(&inlines, 200.0, &styles);
207        assert_eq!(lines.len(), 1);
208        let boxes = &lines[0].boxes;
209        assert_eq!(
210            boxes.iter().map(|box_| box_.node).collect::<Vec<_>>(),
211            [outer, tag()],
212            "the boxes are not outermost first",
213        );
214        assert!(
215            boxes[0].x < boxes[1].x && boxes[0].width > boxes[1].width,
216            "the nested box is not inside the one around it: {boxes:?}",
217        );
218        assert!(boxes.iter().all(|box_| box_.opens && box_.closes));
219
220        assert!(
221            layout_boxed(&tagged(), 200.0, &Boxed(Vec::new()))[0]
222                .boxes
223                .is_empty(),
224            "a line with no box on it carries one",
225        );
226    }

228    /// Part: padding above and below a run is no part of the line's
229    /// own height, and a box deep enough reaches past the line.
230    #[test]
231    fn a_box_does_not_grow_the_line_it_is_on() {
232        let plain = layout_boxed(&tagged(), 200.0, &Boxed(Vec::new()));
233        let styles = Boxed(vec![(tag(), padded(8.0))]);
234        let padded = layout_boxed(&tagged(), 200.0, &styles);
235        assert_eq!(padded[0].box_, plain[0].box_, "the box grew the line");
236
237        let box_ = padded[0].boxes[0];
238        assert!(
239            box_.above > padded[0].box_.baseline,
240            "a box 8pt deep does not reach past the line above it: {box_:?}",
241        );
242    }

244    /// A cloned box paints an edge at the break, and that edge is
245    /// width. The line under the break opens with a leading edge the
246    /// sliced box does not have, so the cloned tag asks for that much
247    /// more measure to stay in two lines.
248    #[test]
249    fn the_edges_a_cloned_box_adds_at_a_break_are_width_too() {
250        let inlines = vec![code(tag(), "2d6 plus the modifier you wrote down")];
251        let box_ = padded(4.0);
252        let cloned = crate::lines::InlineBox {
253            cloned: true,
254            ..box_
255        };
256        let lines =
257            |box_, measure| layout_boxed(&inlines, measure, &Boxed(vec![(tag(), box_)])).len();
258        let narrowest = |box_| -> f32 {
259            (200..1000)
260                .map(|steps| steps as f32 * 0.25)
261                .find(|measure| lines(box_, *measure) == 2)
262                .expect("the tag sets in two lines at some measure")
263        };
264        let grew = narrowest(cloned) - narrowest(box_);
265        assert!(
266            (grew - 4.0).abs() <= 0.25,
267            "the cloned tag asked for {grew}pt more, not for the edge at the break",
268        );
269    }

271    /// A box that a line break cuts opens on the first of its lines
272    /// and closes on the last, and `clone` closes both edges of both.
273    #[test]
274    fn a_split_box_opens_and_closes_where_the_break_left_it() {
275        let inlines = vec![code(tag(), "2d6 plus the modifier you wrote down")];

283        assert_eq!(edges.first().copied(), Some((true, false)));

286        let cloned = crate::lines::InlineBox {

291        assert!(
292            cloned
293                .iter()

299    /// A band set in two spans places a box from the leading edge
300    /// of the span it is in, and names that span. What moves the span
301    /// later, such as alignment, moves the box with the text.
302    #[test]
303    fn a_box_in_a_divided_band_is_placed_from_its_own_span() {
304        use crate::lines::testing::{body, divided_band, registry, span_text};
305        use crate::lines::{LineBreakOptions, LineLayout, Opening};
306
307        let mut inlines = one_run("roll for the modifier and then ");
308        inlines.push(code(tag(), "2d6"));
309        inlines.extend(one_run(" more"));
310        let lines = LineLayout::new(registry()).layout_styled(
311            &inlines,
312            &body(),
313            &Boxed(vec![(tag(), padded(4.0))]),
314            &divided_band(100.0, 20.0),
315            LineBreakOptions::default(),
316            Opening::default(),
317        );
318        let line = &lines[0];
319        assert_eq!(line.spans.len(), 2, "the band was not divided");
320        assert!(
321            span_text(line, 1).contains("2d6"),
322            "the tag is not in the second span: {:?}",
323            span_text(line, 1),
324        );
325        let box_ = line.boxes[0];
326        assert_eq!(box_.span, 1, "the box does not name the span it is in");
327        assert!(
328            box_.x < line.spans[1].offset,
329            "the box carries the offset of its own span: {box_:?}",
330        );
