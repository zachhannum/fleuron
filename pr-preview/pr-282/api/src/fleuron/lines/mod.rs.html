1//! Line layout: text in, broken lines out.
2//!
3//! Knuth-Plass total fit. A paragraph is flattened into style runs,
4//! shaped, and modelled as boxes, glue and penalties; the breaker
5//! picks the set of breaks with the fewest demerits over the whole
6//! paragraph rather than the most text on each line. The break
7//! source is UAX #14 with word boundaries from UAX #29, plus
8//! optional hyphenation, which enters as a flagged penalty.
9//!
10//! Justified text has its glue stretched or shrunk to the measure
11//! here. The adjustment lands on the glyphs' own advances, so a
12//! painter positions what it is given and never re-derives spacing.
13//! A `Line` is shaped runs: measurement happened here.
14//!
15//! Units: advances come out of the shaper in font units; the measure
16//! arrives in points and converts once, via `units_per_em * size`.
17//!
18//! The stages have a file each: `flatten` makes one string of styled
19//! spans, `shape` turns spans into glyphs, `opportunity` says where a
20//! line may end, `breaker` picks the breaks, `justify` moves the glue
21//! on a line that was set to its measure, and `line` is what comes
22//! out.
23

59/// The layout pass: one paragraph → lines that fit the measure.
60pub struct LineLayout<'a> {
61    registry: &'a FontRegistry,
62    segmenter: WordSegmenterBorrowedStatic,
63    /// The characters shaping found no glyph for, since they were
64    /// last taken.
65    uncovered: RefCell<Found>,
66}

68/// A character the face it was set in has no glyph for, where it was
69/// first written.
70#[derive(Debug, Clone, PartialEq)]
71pub(crate) struct Uncovered {
72    pub(crate) character: char,
73    /// The face that has no glyph for it.
74    pub(crate) font_id: u16,
75    /// `None` where the manuscript does not hold the character.
76    pub(crate) position: Option<SourcePos>,
77}

79/// The characters with no glyph, each one once, in the order they
80/// were shaped.
81#[derive(Default)]
82struct Found {
83    seen: HashSet<char>,
84    list: Vec<Uncovered>,
85}

108    /// The paragraph's strut: the minimum box every one of its lines
109    /// occupies, whatever the runs on it.
110    pub fn strut(&self, style: &ParagraphStyle) -> Strut {
111        self.registry
112            .metrics(style.font_id)
113            .map(|m| Strut::from_metrics(m, style.size, style.line_height))
114            .unwrap_or_default()
115    }
116
117    /// The strut of the line a paragraph opens on. A `line-height`
118    /// on `::first-line` is a multiple of that line's own size, so
119    /// the strut is the opening style's where the rule gave one, and
120    /// the paragraph's own where it did not.
121    pub fn opening_strut(&self, style: &ParagraphStyle, first_line: Option<FirstLine>) -> Strut {
122        match self.opening_style(style, first_line) {
123            Some(opening) => self.strut(&opening),
124            None => self.strut(style),
125        }
126    }
127
128    /// The style the opening line's box is measured in, where
129    /// `::first-line` gave the line a height of its own.
130    fn opening_style(
131        &self,
132        style: &ParagraphStyle,
133        first_line: Option<FirstLine>,
134    ) -> Option<ParagraphStyle> {
135        first_line
136            .filter(|first| first.line_height.is_some())
137            .map(|first| first.over(style, self.registry))
138    }
139
140    /// The box one line occupies: the strut, grown by any run taller
141    /// than it around the shared baseline.
142    pub fn line_box(&self, runs: &[ShapedRun], style: &ParagraphStyle) -> LineBox {
143        let strut = self.strut(style);
144        let mut above = strut.above;
145        let mut below = strut.below;
146        for run in runs {
147            let Some(metrics) = self.registry.metrics(run.font_id) else {
148                continue;
149            };
150            let run_strut = Strut::from_metrics(metrics, run.size, style.line_height);
151            above = above.max(run_strut.above);
152            below = below.max(run_strut.below);
153        }
154        LineBox {
155            baseline: above,
156            height: above + below,
157        }
158    }

214    /// Breaks a paragraph that was already shaped to `measure`,
215    /// starting from the end of the line `from`, which is `0` for the

225    /// Breaks one paragraph, and how many times the breaker ran.
226    ///
227    /// A first-line style takes two runs. What it sets changes the
228    /// width of the opening run, the width moves where the paragraph
229    /// breaks, and the break decides how much text the opening line
230    /// holds. The first run sets the paragraph's leading run in the
231    /// opening style to the end and breaks the whole of it. The
232    /// second sets as far as the extent that break gave and breaks
233    /// again with the opening line ending there, so the style covers
234    /// the text the line comes to hold. The count is fixed at two,
235    /// because a line count that depended on how long a paragraph
236    /// took to settle would not be deterministic.
237    fn broken(
238        &self,
239        inlines: &[Inline],
240        style: &ParagraphStyle,
241        styles: &dyn InlineStyles,
242        measure: &Measure,
243        options: LineBreakOptions,
244        opening: Opening,
245    ) -> (Vec<Line>, u8) {
246        let (broken, runs, _) =
247            self.broken_shaped(inlines, style, styles, measure, options, opening);
248        (broken.lines, runs)
249    }

251    /// The same, handing back what the paragraph was shaped from.
252    fn broken_shaped(
253        &self,
254        inlines: &[Inline],
255        style: &ParagraphStyle,
256        styles: &dyn InlineStyles,
257        measure: &Measure,
258        options: LineBreakOptions,
259        opening: Opening,
260    ) -> (Broken, u8, Option<Shaped>) {
261        let lead = Lead {
262            style: opening.first_line,
263            extent: None,
264            taken: opening.taken,
265            pseudo_element: (opening.first_line.is_some()
266                && opening.node != crate::content::NodeId::UNASSIGNED)
267                .then(|| {
268                    opening
269                        .node
270                        .pseudo(crate::content::PseudoElement::FirstLine)
271                }),
272        };
273        let once = |lead| {
274            let shaped = self.shaped(inlines, style, styles, options, lead);
275            let broken = shaped
276                .as_ref()
277                .map(|shaped| self.break_shaped(shaped, measure, 0, shaped.opening))
278                .unwrap_or_default();
279            (broken, shaped)
280        };
281        let (broken, shaped) = once(lead);
282        if lead.style.is_none() || broken.extent == 0 {
283            return (broken, 1, shaped);
284        }
285        let lead = Lead {
286            extent: Some(broken.extent),
287            ..lead
288        };
289        let (broken, shaped) = once(lead);
290        (broken, 2, shaped)
291    }

309        Some(shaped)
310    }

348    /// One flattened paragraph shaped. `opening` is where the line the
349    /// paragraph opens on has to end.
350    fn shape_flat(
351        &self,
352        flat: flatten::FlatParagraph,
353        style: &ParagraphStyle,
354        options: LineBreakOptions,
355        opening: Option<usize>,
356    ) -> Option<Shaped> {
357        if flat.text.is_empty() {
358            return None;
359        }
360        let upem = self.registry.metrics(style.font_id)?.units_per_em as f32;
361        let spans = self.shape_spans(&flat, style, upem);
362        Some(Shaped {
363            flat,
364            spans,
365            style: style.clone(),
366            options,
367            upem,
368            opening,
369            opening_style: None,
370        })
371    }

373    /// Breaks a shaped paragraph to `measure`, starting from the
374    /// break `from`.
375    ///
376    /// `opening` is where the line the paragraph opens on has to end,
377    /// which is the extent of the style over it.
378    fn break_shaped(
379        &self,
380        shaped: &Shaped,
381        measure: &Measure,
382        from: usize,
383        opening: Option<usize>,
384    ) -> Broken {
385        let Shaped {
386            flat,
387            spans,
388            style,
389            options,
390            upem,
391            opening_style,
392            ..
393        } = shaped;
394        let (options, upem) = (*options, *upem);
395        // The line the paragraph opens on is the first one broken
396        // from its start.
397        let boxed = |opens: bool| opening_style.as_ref().filter(|_| opens).unwrap_or(style);
398        // Points → font units: measure / size gives ems, ems *
399        // units_per_em gives font units.
400        let to_points = |units: f32| units / upem * style.size;
401
402        // Points into the paragraph's own font units: what an
403        // inline box's edges are charged in.
404        let units = if style.size > 0.0 {
405            upem / style.size
406        } else {
407            0.0
408        };
409        let widths = Widths::build(flat, spans, units);
410        let hyphen = self.hyphen_advance(style) as f32;
411        let breaks = self.break_points(&flat.text, &widths, hyphen, options);
412        let breaker = Breaker {
413            breaks: &breaks,
414            widths: &widths,
415            measure,
416            rest: measure.at(measure.settled()),
417            settled: measure.settled(),
418            first_band: (0..).find(|slot| measure.at(*slot).ends_band).unwrap_or(0),
419            upem,
420            size: style.size,
421            hyphen,
422            options,
423            from,
424            opening: opening
425                .and_then(|extent| breaks.iter().position(|at| at.content_end == extent)),
426        };
427
428        let mut broken = Broken::default();
429        let mut start = breaks[from.min(breaks.len() - 1)].next;
430        // The band being filled, and where its first span was set:
431        // a span's offset is from there, so a painter handed the
432        // line's leading edge places the rest. One buffer gathers the
433        // spans of every band.
434        let mut band: Option<(Line, f32)> = None;
435        let mut spans = Vec::new();
436        for fit in breaker.run() {
437            let at = &breaks[fit.at];
438            let span = measure.at(fit.slot);
439            if at.content_end > start {
440                let (line, origin) = band.get_or_insert_with(|| {
441                    let mut line = Line::empty();
442                    line.protrusion = to_points(fit.protrusion);
443                    (line, span.origin)
444                });
445                let first = line.runs.len();
446                line.runs
447                    .extend(cut_runs(flat, &shaped.spans, start, at.content_end));
448                adjust(&mut line.runs[first..], &flat.text, fit.ratio, options);
449                if at.hyphen {
450                    self.hyphenate(&mut line.runs, style);
451                }
452                let mut boxes = self.inline_fragments(
453                    flat,
454                    &mut line.runs[first..],
455                    spans.len(),
456                    start..at.content_end,
457                );
458                line.boxes.append(&mut boxes);
459                let offset = span.origin - *origin;
460                let width = line.runs[first..].iter().map(|run| run.advance).sum();
461                spans.push(LineSpan {
462                    runs: first..line.runs.len(),
463                    offset,
464                    width,
465                });
466                line.width += width;
467            }
468            start = at.next;
469            let hard = at.forced && fit.at != breaker.end();
470            if !span.ends_band && !hard {
471                continue;
472            }
473            // A blank line of preformatted text takes a line of its
474            // own, so what the author wrote under it stays where it
475            // was written.
476            if band.is_none() && options.preformatted && hard {
477                spans.push(LineSpan {
478                    runs: 0..0,
479                    offset: 0.0,
480                    width: 0,
481                });
482                band = Some((Line::empty(), span.origin));
483            }
484            let Some((mut line, _)) = band.take() else {
485                continue;
486            };
487            line.overhang = to_points(fit.overhang);
488            let opens = from == 0 && broken.lines.is_empty();
489            line.box_ = self.line_box(&line.runs, boxed(opens));
490            line.spans = gather(&mut spans);
491            if broken.lines.is_empty() {
492                broken.extent = at.content_end;
493            }
494            broken.lines.push(line);
495            broken.ends.push(fit.at);
496        }
497        // A paragraph that ran out inside a band still sets what it
498        // reached.
499        if let Some((mut line, _)) = band.take() {
500            let opens = from == 0 && broken.lines.is_empty();
501            line.box_ = self.line_box(&line.runs, boxed(opens));
502            line.spans = gather(&mut spans);
503            broken.lines.push(line);
504            broken.ends.push(breaker.end());
505        }
506        tile(&mut broken.lines, flat);
507        broken
508    }
509}

511/// A paragraph shaped once, and everything about it the measure does
512/// not decide: the flattened text, the runs it shaped into, and the
513/// style and options it was set with.
514///
515/// Shaping is the expensive half of setting a paragraph, and it does
516/// not depend on where the lines end. A caller that can break the
517/// same paragraph again against a different profile keeps this. It
518/// breaks the paragraph through [`LineLayout::rebreak`].
519pub struct Shaped {
520    flat: FlatParagraph,
521    spans: Vec<ShapedSpan>,
522    style: ParagraphStyle,
523    options: LineBreakOptions,
524    upem: f32,
525    /// Where the line the paragraph opens on has to end, when an
526    /// opening style covers exactly that much of the text.
527    opening: Option<usize>,
528    /// The style the opening line's box is measured in, where
529    /// `::first-line` gave the line a height of its own.
530    opening_style: Option<ParagraphStyle>,
531}
532
533impl std::fmt::Debug for Shaped {
534    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
535        f.debug_struct("Shaped")
536            .field("text", &self.flat.text)
537            .finish_non_exhaustive()
538    }

763    /// Small capitals on the opening line stop at the break the
764    /// second run chose: the last word of the first line is set in
765    /// them throughout and the first word of the second line is none
766    /// of it. What the author wrote comes back off the runs either
767    /// way.
768    #[test]
769    fn a_first_line_is_small_capitals_as_far_as_the_break() {
770        // The face with no substitutions of its own synthesises its
771        // small capitals, which puts the boundary in the drawn text.
772        let bare = crate::fonts::registry_without_substitutions();
773        let layout = LineLayout::new(&bare);
774        let lines = layout_first(
775            &layout,
776            160.0,
777            Some(FirstLine {
778                caps: Some(FontVariantCaps::SmallCaps),
779                ..FirstLine::default()
780            }),
781        );
782        assert!(lines.len() > 2, "the paragraph did not break");
783
784        let opening = line_text(&lines[0]);
785        let next = line_text(&lines[1]);
786        let last_word = opening.split_whitespace().next_back().expect("a word");
787        let first_word = next.split_whitespace().next().expect("a word");
788        assert_eq!(
789            last_word,
790            last_word.to_uppercase(),
791            "the first line's last word is not small capitals throughout: {opening:?}",
792        );
793        assert_eq!(
794            first_word,
795            first_word.to_lowercase(),
796            "the small capitals ran past the first line: {next:?}",
797        );
798        assert_eq!(opening, opening.to_uppercase());
799        assert_eq!(next, next.to_lowercase());
800        assert!(
801            lines[0]
802                .runs
803                .iter()
804                .any(|run| run.size == body().size * SMALL_CAPS_RATIO),
805            "nothing on the opening line was set at the reduced size",
806        );
807        assert!(
808            lines[1].runs.iter().all(|run| run.size == body().size),
809            "the reduced size reached the second line",
810        );
811
812        // What the author wrote is on the runs beside what was drawn.
813        let written: String = lines[0]
814            .runs
815            .iter()
816            .map(|run| match run.source.is_empty() {
817                true => run.text.as_str(),
818                false => run.source.as_str(),
819            })
820            .collect();
821        assert!(
822            OPENING.starts_with(&written),
823            "the opening line lost the manuscript: {written:?}",
824        );
825    }
