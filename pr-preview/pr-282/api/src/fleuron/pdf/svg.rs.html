1//! An SVG as PDF content: the tree usvg parses, drawn as the paths,
2//! paints, clips and masks it holds, so the art stays vector in the
3//! export.
4//!
5//! Two things an SVG can ask for have no vector form here. Text that
6//! was not converted to outlines is not drawn, since the file names a
7//! face and carries none. A filter effect is a raster operation, so a
8//! group that has one is drawn without it.
9

25/// One SVG, parsed once for the whole export.
26pub(super) struct Vector(usvg::Tree);
27
28impl Vector {
29    /// Parses a file. `None` where it is not an SVG usvg can read.
30    pub(super) fn parse(bytes: &[u8]) -> Option<Vector> {
31        let mut options = usvg::Options::default();
32        // The engine opens no file. An image inside the SVG is drawn
33        // where the SVG carries its bytes, and not where it names a
34        // path.
35        options.image_href_resolver.resolve_string = Box::new(|_, _| None);
36        usvg::Tree::from_data(bytes, &options).ok().map(Vector)
37    }
38
39    /// Draws the art over a box of `size` at the surface's origin.
40    pub(super) fn draw(&self, surface: &mut Surface, size: Size) {
41        let (width, height) = (self.0.size().width(), self.0.size().height());
42        let Some(bounds) = Rect::from_xywh(0.0, 0.0, width, height) else {
43            return;
44        };
45        let (fill, stroke) = (surface.get_fill().cloned(), surface.get_stroke().cloned());
46        surface.push_transform(&Transform::from_scale(
47            size.width() / width,
48            size.height() / height,
49        ));
50        surface.push_clip_path(&rectangle(bounds), &FillRule::NonZero);
51        group(self.0.root(), surface);
52        surface.pop();
53        surface.pop();
54        surface.set_fill(fill);
55        surface.set_stroke(stroke);
56    }
57}

59fn node(node: &usvg::Node, surface: &mut Surface) {
60    match node {
61        usvg::Node::Group(inner) => group(inner, surface),
62        usvg::Node::Path(inner) => path(inner, surface),
63        usvg::Node::Image(inner) => image(inner, surface),
64        usvg::Node::Text(_) => {}
65    }
66}
67
68fn group(group: &usvg::Group, surface: &mut Surface) {
69    let mut pushed = 3;
70    if group.isolate() {
71        surface.push_isolated();
72        pushed += 1;
73    }
74    surface.push_transform(&transform(group.transform()));
75    if let Some(clip) = group.clip_path() {
76        pushed += clip_path(clip, surface);
77    }
78    if let Some(mask) = group.mask() {
79        pushed += self::mask(mask, surface);
80    }
81    surface.push_blend_mode(blend_mode(group.blend_mode()));
82    surface.push_opacity(normalized(group.opacity().get()));
83    for child in group.children() {
84        node(child, surface);
85    }
86    for _ in 0..pushed {
87        surface.pop();
88    }
89}

91fn path(path: &usvg::Path, surface: &mut Surface) {
92    if !path.is_visible() {
93        return;
94    }
95    let Some(outline) = outline(path.data().segments(), usvg::Transform::identity()) else {
96        return;
97    };
98    let mut draw = |fill: Option<&usvg::Fill>, stroke: Option<&usvg::Stroke>| {
99        let fill = fill.map(|fill| self::fill(fill, surface));
100        let stroke = stroke.map(|stroke| self::stroke(stroke, surface));
101        // With neither set, krilla fills in black.
102        if fill.is_some() || stroke.is_some() {
103            surface.set_fill(fill);
104            surface.set_stroke(stroke);
105            surface.draw_path(&outline);
106        }
107    };
108    match path.paint_order() {
109        usvg::PaintOrder::FillAndStroke => draw(path.fill(), path.stroke()),
110        usvg::PaintOrder::StrokeAndFill => {
111            draw(None, path.stroke());
112            draw(path.fill(), None);
113        }
114    }
115}

117/// An image the SVG carries inside it: pixels as an image object, and
118/// another SVG as its own paths.
119fn image(image: &usvg::Image, surface: &mut Surface) {
120    if !image.is_visible() {
121        return;
122    }
123    let data = |bytes: &Arc<Vec<u8>>| krilla::Data::from(bytes.as_ref().clone());
124    let raster = match image.kind() {
125        usvg::ImageKind::JPEG(bytes) => Image::from_jpeg(data(bytes), false),
126        usvg::ImageKind::PNG(bytes) => Image::from_png(data(bytes), false),
127        usvg::ImageKind::GIF(bytes) => Image::from_gif(data(bytes), false),
128        usvg::ImageKind::WEBP(bytes) => Image::from_webp(data(bytes), false),
129        usvg::ImageKind::SVG(tree) => {
130            let Some(bounds) = Rect::from_xywh(0.0, 0.0, tree.size().width(), tree.size().height())
131            else {
132                return;
133            };
134            surface.push_clip_path(&rectangle(bounds), &FillRule::NonZero);
135            group(tree.root(), surface);
136            surface.pop();
137            return;
138        }
139    };
140    if let (Some(raster), Some(size)) = (
141        raster,
142        Size::from_wh(image.size().width(), image.size().height()),
143    ) {
144        surface.draw_image(raster, size);
145    }
146}

148/// Pushes one clip path and reports how many pops undo it.
149///
150/// A PDF clip is one path under one fill rule. A clip path whose
151/// shapes fit that is pushed as a clip. One that does not, because a
152/// shape inside it is clipped again or its shapes disagree about the
153/// rule, is drawn as an alpha mask.
154fn clip_path(clip: &usvg::ClipPath, surface: &mut Surface) -> u16 {
155    let mut rules = Vec::new();
156    clip_rules(clip.root(), &mut rules);
157    let one_rule = rules.iter().all(|rule| *rule == usvg::FillRule::NonZero)
158        || (rules.len() == 1 && rules[0] == usvg::FillRule::EvenOdd);
159    if !(plain(clip.root()) && one_rule) {
160        let mut builder = surface.stream_builder();
161        let mut inner = builder.surface();
162        let mut pushed = 1;
163        if let Some(parent) = clip.clip_path() {
164            pushed += clip_path(parent, &mut inner);
165        }
166        inner.push_transform(&transform(clip.transform()));
167        group(clip.root(), &mut inner);
168        for _ in 0..pushed {
169            inner.pop();
170        }
171        inner.finish();
172        let stream = builder.finish();
173        surface.push_mask(Mask::new(stream, MaskType::Alpha));
174        return 1;
175    }
176    let mut pushed = match clip.clip_path() {
177        Some(parent) => clip_path(parent, surface),
178        None => 0,
179    };
180    let mut builder = PathBuilder::new();
181    clip_segments(clip.root(), clip.transform(), &mut builder);
182    // A clip path with nothing in it hides everything it clips.
183    let path = builder.finish().unwrap_or_else(|| {
184        let mut builder = PathBuilder::new();
185        builder.move_to(0.0, 0.0);
186        builder.line_to(0.0, 0.0);
187        builder.finish().expect("a line is a path")
188    });
189    let rule = rules.first().copied().unwrap_or(usvg::FillRule::NonZero);
190    surface.push_clip_path(&path, &fill_rule(rule));
191    pushed += 1;
192    pushed
193}

195/// Whether no shape inside a clip path is clipped again.
196fn plain(group: &usvg::Group) -> bool {
197    group.children().iter().all(|child| match child {
198        usvg::Node::Group(inner) => inner.clip_path().is_none() && plain(inner),
199        _ => true,
200    })
201}
202
203fn clip_rules(group: &usvg::Group, rules: &mut Vec<usvg::FillRule>) {
204    for child in group.children() {
205        match child {
206            usvg::Node::Path(path) => rules.extend(path.fill().map(usvg::Fill::rule)),
207            usvg::Node::Group(inner) => clip_rules(inner, rules),
208            _ => {}
209        }
210    }
211}
212
213fn clip_segments(group: &usvg::Group, transform: usvg::Transform, builder: &mut PathBuilder) {
214    for child in group.children() {
215        match child {
216            usvg::Node::Path(path) if path.is_visible() => {
217                segments(path.data().segments(), transform, builder);
218            }
219            usvg::Node::Group(inner) => {
220                clip_segments(inner, transform.pre_concat(inner.transform()), builder);
221            }
222            _ => {}
223        }
224    }
225}

227/// Pushes one mask and reports how many pops undo it.
228fn mask(mask: &usvg::Mask, surface: &mut Surface) -> u16 {
229    let mut builder = surface.stream_builder();
230    let mut inner = builder.surface();
231    let mut pushed = 0;
232    if let Some(parent) = mask.mask() {
233        pushed += self::mask(parent, &mut inner);
234    }
235    let bounds = mask.rect().to_rect();
236    if let Some(bounds) =
237        Rect::from_ltrb(bounds.left(), bounds.top(), bounds.right(), bounds.bottom())
238    {
239        inner.push_clip_path(&rectangle(bounds), &FillRule::NonZero);
240        pushed += 1;
241    }
242    group(mask.root(), &mut inner);
243    for _ in 0..pushed {
244        inner.pop();
245    }
246    inner.finish();
247    let kind = match mask.kind() {
248        usvg::MaskType::Luminance => MaskType::Luminosity,
249        usvg::MaskType::Alpha => MaskType::Alpha,
250    };
251    let stream = builder.finish();
252    surface.push_mask(Mask::new(stream, kind));
253    1
254}

256fn fill(fill: &usvg::Fill, surface: &mut Surface) -> Fill {
257    Fill {
258        paint: paint(fill.paint(), surface),
259        opacity: normalized(fill.opacity().get()),
260        rule: fill_rule(fill.rule()),
261    }
262}
263
264fn stroke(stroke: &usvg::Stroke, surface: &mut Surface) -> Stroke {
265    Stroke {
266        paint: paint(stroke.paint(), surface),
267        width: stroke.width().get(),
268        miter_limit: stroke.miterlimit().get(),
269        line_cap: match stroke.linecap() {
270            usvg::LineCap::Butt => LineCap::Butt,
271            usvg::LineCap::Round => LineCap::Round,
272            usvg::LineCap::Square => LineCap::Square,
273        },
274        line_join: match stroke.linejoin() {
275            usvg::LineJoin::Miter | usvg::LineJoin::MiterClip => LineJoin::Miter,
276            usvg::LineJoin::Round => LineJoin::Round,
277            usvg::LineJoin::Bevel => LineJoin::Bevel,
278        },
279        opacity: normalized(stroke.opacity().get()),
280        dash: stroke.dasharray().map(|array| StrokeDash {
281            offset: stroke.dashoffset(),
282            array: array.to_vec(),
283        }),
284    }
285}

287fn paint(paint: &usvg::Paint, surface: &mut Surface) -> Paint {
288    let spread = |method: usvg::SpreadMethod| match method {
289        usvg::SpreadMethod::Pad => SpreadMethod::Pad,
290        usvg::SpreadMethod::Reflect => SpreadMethod::Reflect,
291        usvg::SpreadMethod::Repeat => SpreadMethod::Repeat,
292    };
293    let stops = |stops: &[usvg::Stop]| -> Vec<Stop> {
294        stops
295            .iter()
296            .map(|stop| Stop {
297                offset: normalized(stop.offset().get()),
298                color: rgb::Color::new(stop.color().red, stop.color().green, stop.color().blue)
299                    .into(),
300                opacity: normalized(stop.opacity().get()),
301            })
302            .collect()
303    };
304    match paint {
305        usvg::Paint::Color(color) => rgb::Color::new(color.red, color.green, color.blue).into(),
306        usvg::Paint::LinearGradient(gradient) => LinearGradient {
307            x1: gradient.x1(),
308            y1: gradient.y1(),
309            x2: gradient.x2(),
310            y2: gradient.y2(),
311            transform: transform(gradient.transform()),
312            spread_method: spread(gradient.spread_method()),
313            stops: stops(gradient.stops()),
314            anti_alias: false,
315        }
316        .into(),
317        usvg::Paint::RadialGradient(gradient) => RadialGradient {
318            cx: gradient.cx(),
319            cy: gradient.cy(),
320            cr: gradient.r().get(),
321            fx: gradient.fx(),
322            fy: gradient.fy(),
323            fr: 0.0,
324            transform: transform(gradient.transform()),
325            spread_method: spread(gradient.spread_method()),
326            stops: stops(gradient.stops()),
327            anti_alias: false,
328        }
329        .into(),
330        usvg::Paint::Pattern(pattern) => {
331            let mut builder = surface.stream_builder();
332            let mut inner = builder.surface();
333            group(pattern.root(), &mut inner);
334            inner.finish();
335            Pattern {
336                stream: builder.finish(),
337                transform: transform(pattern.transform().pre_concat(
338                    usvg::Transform::from_translate(pattern.rect().x(), pattern.rect().y()),
339                )),
340                width: pattern.rect().width(),
341                height: pattern.rect().height(),
342            }
343            .into()
344        }
345    }
346}

348fn fill_rule(rule: usvg::FillRule) -> FillRule {
349    match rule {
350        usvg::FillRule::NonZero => FillRule::NonZero,
351        usvg::FillRule::EvenOdd => FillRule::EvenOdd,
352    }

355fn blend_mode(mode: usvg::BlendMode) -> BlendMode {
356    match mode {
357        usvg::BlendMode::Normal => BlendMode::Normal,
358        usvg::BlendMode::Multiply => BlendMode::Multiply,
359        usvg::BlendMode::Screen => BlendMode::Screen,
360        usvg::BlendMode::Overlay => BlendMode::Overlay,
361        usvg::BlendMode::Darken => BlendMode::Darken,
362        usvg::BlendMode::Lighten => BlendMode::Lighten,
363        usvg::BlendMode::ColorDodge => BlendMode::ColorDodge,
364        usvg::BlendMode::ColorBurn => BlendMode::ColorBurn,
365        usvg::BlendMode::HardLight => BlendMode::HardLight,
366        usvg::BlendMode::SoftLight => BlendMode::SoftLight,
367        usvg::BlendMode::Difference => BlendMode::Difference,
368        usvg::BlendMode::Exclusion => BlendMode::Exclusion,
369        usvg::BlendMode::Hue => BlendMode::Hue,
370        usvg::BlendMode::Saturation => BlendMode::Saturation,
371        usvg::BlendMode::Color => BlendMode::Color,
372        usvg::BlendMode::Luminosity => BlendMode::Luminosity,
373    }
374}

391fn rectangle(rect: Rect) -> Path {
392    let mut builder = PathBuilder::new();
393    builder.push_rect(rect);
394    builder.finish().expect("a rectangle is a path")
395}

397fn outline(from: impl Iterator<Item = PathSegment>, transform: usvg::Transform) -> Option<Path> {
398    let mut builder = PathBuilder::new();
399    segments(from, transform, &mut builder);
400    builder.finish()
401}
402
403fn segments(
404    from: impl Iterator<Item = PathSegment>,
405    transform: usvg::Transform,
406    builder: &mut PathBuilder,
407) {
408    for segment in from {
409        match segment {
410            PathSegment::MoveTo(to) => {
411                let mut points = [to];
412                transform.map_points(&mut points);
413                builder.move_to(points[0].x, points[0].y);
414            }
415            PathSegment::LineTo(to) => {
416                let mut points = [to];
417                transform.map_points(&mut points);
418                builder.line_to(points[0].x, points[0].y);
419            }
420            PathSegment::QuadTo(control, to) => {
421                let mut points = [control, to];
422                transform.map_points(&mut points);
423                builder.quad_to(points[0].x, points[0].y, points[1].x, points[1].y);
424            }
425            PathSegment::CubicTo(first, second, to) => {
426                let mut points = [first, second, to];
427                transform.map_points(&mut points);
428                builder.cubic_to(
429                    points[0].x,
430                    points[0].y,
431                    points[1].x,
432                    points[1].y,
433                    points[2].x,
434                    points[2].y,
435                );
436            }
437            PathSegment::Close => builder.close(),
438        }
439    }
