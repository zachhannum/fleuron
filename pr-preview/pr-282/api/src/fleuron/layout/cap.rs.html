12impl Paginator<'_> {
13    /// The initial letter one paragraph opens with, and the inlines
14    /// left after it is taken out.
15    ///
16    /// The sink is what `initial-letter` asked for: the cap's own cap
17    /// height spans that many lines, so its baseline lands on the
18    /// last of them and its top on the first line's cap height.
19    pub(super) fn drop_cap(
20        &self,
21        id: NodeId,
22        computed: &ComputedStyle,
23        inlines: &[Inline],
24    ) -> Option<(Cap, usize)> {
25        let cap_style = self.styles.first_letter(id)?;
26        let sink = cap_style.initial_letter as usize;
27        if sink < 2 {
28            return None;
29        }
30        let initial = take_initial(inlines)?;
31        let body = computed.paragraph();
32        let cap_metrics = self.registry.metrics(cap_style.font_id)?;
33        let cap_units = cap_height(cap_metrics);
34        if cap_units <= 0.0 {
35            return None;
36        }
37        let body_metrics = self.registry.metrics(body.font_id)?;
38        let body_cap = cap_height(body_metrics) / body_metrics.units_per_em as f32 * body.size;
39        // The sunk lines are the paragraph's own height apart, and
40        // the opening line stands off the second by what is under
41        // its own baseline.
42        let strut = self.lines.strut(&body);
43        let opening = self
44            .lines
45            .opening_strut(&body, self.styles.opening_line(id));
46        let sunk = (sink - 1) as f32 * strut.height() + opening.below - strut.below + body_cap;
47        let style = ParagraphStyle {
48            size: sunk * cap_metrics.units_per_em as f32 / cap_units,
49            ..cap_style.paragraph()
50        };
51        let mut line = self.line_of(&initial.text, &style)?;
52        // The cap stands for the letter it was taken from, so a
53        // cursor on the manuscript's first word lands on it.
54        if let Some(run) = line.runs.first_mut() {
55            run.origin = Some(initial.origin);
56            run.pseudo_element = Some(id.pseudo(PseudoElement::FirstLetter));
57        }
58        // A word space of the body text separates the cap from the
59        // lines it is sunk into.
60        let gutter = self
61            .registry
62            .char_glyph(body.font_id, ' ')
63            .and_then(|glyph| self.registry.advance_width(body.font_id, glyph))
64            .unwrap_or(0) as f32
65            / body_metrics.units_per_em as f32
66            * body.size;
67        Some((
68            Cap {
69                reserved: self.line_width(&line) + gutter,
70                line,
71                lines: sink,
72            },
73            initial.taken,
74        ))
75    }
76}

78/// The initial letter of one paragraph, sized and shaped, with the
79/// text it was taken out of.
80#[derive(Debug, Clone)]
81pub(super) struct Cap {
82    /// The letter, shaped at the size the sink works out to.
83    pub(super) line: Line,
84    /// Width the lines beside it give up, the gutter included.
85    pub(super) reserved: f32,
86    /// Lines it is sunk over.
87    pub(super) lines: usize,
88}

90/// The text a drop cap is set from, where it was written, and how far
91/// into the paragraph the rest of the prose starts.
92struct Initial {
93    /// The first letter with the punctuation on either side of it.
94    text: String,
95    /// The node the cap starts in, and the bytes of it the cap holds:
96    /// its text and whatever space stood before it.
97    origin: SourceRange,
98    /// Bytes of the paragraph's text the cap holds.
99    taken: usize,
100}
101
102/// The first letter of a run of inlines and the punctuation next to
103/// it, wherever the markup has put them: a paragraph opening in
104/// italic still opens with a letter.
105fn take_initial(inlines: &[Inline]) -> Option<Initial> {
106    fn texts<'a>(inlines: &'a [Inline], out: &mut Vec<(Option<NodeId>, &'a str)>) {
107        for inline in inlines {
108            match inline {
109                Inline::Text { id, value, .. } | Inline::Code { id, value, .. } => {
110                    out.push((Some(*id), value))
111                }
112                Inline::Break { .. } => out.push((None, "\n")),
113                // The note is set at the foot of the page, so an
114                // initial letter never comes out of one.
115                Inline::Note { .. } => {}
116                Inline::Emphasis { children, .. }
117                | Inline::Strong { children, .. }
118                | Inline::Link { children, .. }
119                | Inline::Strikethrough { children, .. }
120                | Inline::Highlight { children, .. }
121                | Inline::Span { children, .. } => texts(children, out),
122            }
123        }
124    }
125    let mut pieces = Vec::new();
126    texts(inlines, &mut pieces);
127
128    let mut text = String::new();
129    let mut origin: Option<SourceRange> = None;
130    let mut lettered = false;
131    let mut taken = 0;
132    let mut before = 0;
133    'pieces: for (node, value) in pieces {
134        for (at, character) in value.char_indices() {
135            if text.is_empty() && character.is_whitespace() {
136                continue;
137            }
138            let joins = match Kind::of(character) {
139                Kind::Punctuation => true,
140                Kind::Letter => !std::mem::replace(&mut lettered, true),
141                Kind::Other => false,
142            };
143            let Some(node) = node.filter(|_| joins) else {
144                break 'pieces;
145            };
146            let end = (at + character.len_utf8()) as u32;
147            let origin = origin.get_or_insert(SourceRange { node, range: 0..0 });
148            if origin.node == node {
149                origin.range.end = end;
150            }
151            text.push(character);
152            taken = before + end as usize;
153        }
154        before += value.len();
155    }
156    Some(Initial {
157        text,
158        origin: origin.filter(|_| lettered)?,
159        taken,
160    })
161}

163/// What a character is to `::first-letter`.
164enum Kind {
165    /// Opening, closing, quotation, and other punctuation. Dashes and
166    /// connectors are not in it.
167    Punctuation,
168    /// A letter, a number, or a symbol.
169    Letter,
170    Other,
171}
172
173impl Kind {
174    fn of(character: char) -> Kind {
175        match get_general_category(character).abbreviation().as_bytes() {
176            [b'P', b's' | b'e' | b'i' | b'f' | b'o'] => Kind::Punctuation,
177            [b'L' | b'N' | b'S', _] => Kind::Letter,
178            _ => Kind::Other,
179        }
180    }
181}

183/// A face's cap height, falling back to its ascender when the file
184/// declares none.
185fn cap_height(metrics: crate::fonts::FontMetricsTable) -> f32 {
186    if metrics.cap_height > 0 {
187        metrics.cap_height as f32
188    } else {
189        metrics.ascender as f32
190    }
191}

354    /// Acceptance: a drop cap beside a first line with a line height
355    /// of its own still sits on the third baseline, and its top stays
356    /// on that line's cap height, whether the line is taller than the
357    /// paragraph's own or shorter.
358    #[test]
359    fn a_drop_cap_reads_the_line_height_of_the_first_line() {
360        for line_height in ["3", "1"] {
361            let pages = paginate_styled(
362                &format!(
363                    "p::first-letter {{ initial-letter: 3 }}
364                     p::first-line {{ line-height: {line_height} }}"
365                ),
366                vec![section(vec![paragraph(
367                    &"my father had a small estate in nottinghamshire ".repeat(12),
368                )])],
369            );
370            let lines = content_lines(&pages[0]);
371            assert!(lines.len() > 4, "not enough lines to sink into");
372
373            let body = ua().root();
374            let (cap_index, (cap_baseline, cap_size)) = lines
375                .iter()
376                .enumerate()
377                .find(|(_, (_, runs))| runs[0].1 > body.font_size)
378                .map(|(index, (baseline, runs))| (index, (*baseline, runs[0].1)))
379                .expect("a drop cap paints");
380            let prose: Vec<f32> = lines
381                .iter()
382                .enumerate()
383                .filter(|(index, _)| *index != cap_index)
384                .map(|(_, (baseline, _))| *baseline)
385                .collect();
386            let leading = prose[3] - prose[2];
387            assert!(
388                (prose[1] - prose[0] - leading).abs() > 1.0,
389                "line-height: {line_height} left the first line where it was",
390            );
391            assert!(
392                (cap_baseline - prose[2]).abs() < 1e-3,
393                "line-height: {line_height}: the cap is not on the third baseline",
394            );
395
396            let metrics = registry().metrics(body.font_id).unwrap();
397            let cap_height =
398                |size: f32| metrics.cap_height as f32 / metrics.units_per_em as f32 * size;
399            let top = cap_baseline - cap_height(cap_size);
400            assert!(
401                (top - (prose[0] - cap_height(body.font_size))).abs() < 1e-2,
402                "line-height: {line_height}: the cap's top is not the first line's cap height",
403            );
404        }
405    }

407    /// Acceptance: a three-line drop cap sits on the third baseline,
408    /// its top on the first line's cap height, and the lines beside
409    /// it are set to the measure it left them.
410    #[test]
411    fn a_drop_cap_aligns_to_the_third_baseline_and_shortens_three_lines() {
412        let pages = paginate_styled(
413            "p::first-letter { initial-letter: 3 }",
414            vec![section(vec![paragraph(
415                &"my father had a small estate in nottinghamshire ".repeat(12),
416            )])],
417        );
418        let lines = content_lines(&pages[0]);
419        assert!(lines.len() > 4, "not enough lines to sink into");
420
421        let body = ua().root();
422        let (left, _) = origin_of(&pages[0]);
423        // The cap is the one run set larger than the body.
424        let (cap_index, cap) = lines
425            .iter()
426            .enumerate()
427            .find(|(_, (_, runs))| runs[0].1 > body.font_size)
428            .map(|(index, (baseline, runs))| (index, (*baseline, runs[0])))
429            .expect("a drop cap paints");
430        let (cap_baseline, (cap_x, cap_size, _)) = cap;
431
432        // Its baseline is the third line's, and it starts at the
433        // content box's own leading edge.
434        let prose: Vec<&ContentLine<'_>> = lines
435            .iter()
436            .enumerate()
437            .filter(|(index, _)| *index != cap_index)
438            .map(|(_, line)| line)
439            .collect();
440        assert!(
441            (cap_baseline - prose[2].0).abs() < 1e-3,
442            "the cap is not on the third baseline"
443        );
444        assert!(
445            (cap_x - left).abs() < 1e-3,
446            "the cap is not at the leading edge"
447        );
448
449        // Its top sits on the first line's cap height.
450        let cap_height = |font: u16, size: f32| {
451            let metrics = registry().metrics(font).unwrap();
452            metrics.cap_height as f32 / metrics.units_per_em as f32 * size
453        };
454        let top = cap_baseline - cap_height(body.font_id, cap_size);
455        assert!(
456            (top - (prose[0].0 - cap_height(body.font_id, body.font_size))).abs() < 1e-2,
457            "the cap's top is not the first line's cap height",
458        );
459
460        // The three lines beside it start past the cap and are set to
461        // the measure it left them; the fourth is back at the full one.
462        let sunk = prose[0].1[0].0;
463        assert!(sunk > left, "the first line was not moved aside");
464        for line in prose.iter().take(3) {
465            assert!(
466                (line.1[0].0 - sunk).abs() < 1e-3,
467                "a sunk line is not set to the shortened measure",
468            );
469        }
470        assert!(
471            (prose[3].1[0].0 - left).abs() < 1e-3,
472            "the fourth line did not go back to the full measure",
473        );
474        for page in &pages {
475            for item in &page.items {
476                let DrawItem::Text { glyphs, .. } = item else {
477                    continue;
478                };
479                for glyph in glyphs {
480                    assert!(
481                        glyph.x <= left + master(Situation::Body(page.side)).geometry.measure(),
482                        "a sunk line ran past the measure",
483                    );
484                }
485            }
486        }
487    }
