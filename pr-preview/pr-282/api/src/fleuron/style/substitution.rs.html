11/// The custom properties of one element: the ones it inherited, with
12/// the ones its own rules declare, in cascade order, over them.
13///
14/// A custom property whose value names one that holds nothing, or
15/// that depends on itself, holds nothing.
16pub(super) fn resolve(
17    inherited: &BTreeMap<String, String>,
18    declared: &[&Custom],
19    warnings: &mut Vec<Warning>,
20) -> BTreeMap<String, String> {
21    let own: BTreeMap<&str, &Custom> = declared
22        .iter()
23        .map(|custom| (custom.name.as_str(), *custom))
24        .collect();
25    let mut resolver = Resolver {
26        own: &own,
27        inherited,
28        done: BTreeMap::new(),
29        stack: Vec::new(),
30        cyclic: BTreeSet::new(),
31        warnings,
32    };
33    let mut resolved = inherited.clone();
34    for name in own.keys() {
35        match resolver.value(name) {
36            Some(value) => resolved.insert(name.to_string(), value),
37            None => resolved.remove(*name),
38        };
39    }
40    resolved
41}
42
43struct Resolver<'a, 'w> {
44    own: &'a BTreeMap<&'a str, &'a Custom>,
45    inherited: &'a BTreeMap<String, String>,
46    done: BTreeMap<&'a str, Option<String>>,
47    /// The names being resolved, outermost first.
48    stack: Vec<&'a str>,
49    /// Every name found on a cycle.
50    cyclic: BTreeSet<&'a str>,
51    warnings: &'w mut Vec<Warning>,
52}
53
54impl<'a> Resolver<'a, '_> {
55    fn value(&mut self, name: &str) -> Option<String> {
56        let own = self.own;
57        let Some(custom) = own.get(name) else {
58            return self.inherited.get(name).cloned();
59        };
60        if let Some(done) = self.done.get(name) {
61            return done.clone();
62        }
63        let name = custom.name.as_str();
64        if let Some(at) = self.stack.iter().position(|open| *open == name) {
65            self.cyclic.extend(&self.stack[at..]);
66            return None;
67        }
68        self.stack.push(name);
69        let substituted = sheet::substitute(&custom.value, &mut |name| self.value(name));
70        self.stack.pop();
71        let value = if self.cyclic.contains(name) {
72            warn_once(
73                self.warnings,
74                format!("Custom property `{name}` depends on itself. `{name}` has no value."),
75                &custom.origin,
76            );
77            None
78        } else {
79            substituted.ok()
80        };
81        self.done.insert(name, value.clone());
82        value
83    }
84}

86/// What `pending` comes to under the custom properties in `custom`,
87/// read by `read`. `None` is a value that became nothing, and the
88/// warning that says so is already made. The caller then resets
89/// every longhand the property sets.
90pub(super) fn substituted<D>(
91    pending: &Pending,
92    custom: &BTreeMap<String, String>,
93    read: fn(&Pending, &str) -> Option<Vec<D>>,
94    warnings: &mut Vec<Warning>,
95) -> Option<Vec<D>> {
96    let why = match sheet::substitute(&pending.value, &mut |name| custom.get(name).cloned()) {
97        Ok(css) => match read(pending, &css) {
98            Some(declarations) => return Some(declarations),
99            None => Unresolved::Unsupported,
100        },
101        Err(why) => why,
102    };
103    let property = &pending.property;
104    let problem = match why {
105        Unresolved::Missing(name) => format!("Custom property `{name}` has no value."),
106        Unresolved::Unsupported => format!("Unsupported value for `{property}`."),
107    };
108    let inherits = Spec::find(PROPERTIES, property).is_some_and(|spec| spec.inherited);
109    let takes = if inherits {
110        "its inherited value"
111    } else {
112        "its initial value"
113    };
114    warn_once(
115        warnings,
116        format!("{problem} `{property}` takes {takes}."),
117        &pending.origin,
118    );
119    None
120}

122/// Applies a pending declaration of a style rule. `base` is the style
123/// before any declaration applied, which is what a value that became
124/// nothing leaves each longhand at.
125pub(super) fn apply(
126    style: &mut ComputedStyle,
127    pending: &Pending,
128    base: &ComputedStyle,
129    parent_size: f32,
130    root_size: f32,
131    warnings: &mut Vec<Warning>,
132) {
133    match substituted(pending, &style.custom, sheet::read_pending, warnings) {
134        Some(declarations) => {
135            for declaration in &declarations {
136                style.apply(declaration, parent_size, root_size);
137            }
138        }
139        None => {
140            for like in sheet::longhands(&pending.property) {
141                style.reset(&like, base);
142            }
143        }
144    }
145}

147/// Records a warning once. Every element a rule matches reads the
148/// same declaration, and the declaration is what the author fixes.
149fn warn_once(warnings: &mut Vec<Warning>, message: String, origin: &str) {
150    let seen = warnings
151        .iter()
152        .any(|seen| seen.message == message && seen.origin.as_deref() == Some(origin));
153    if !seen {
154        warnings.push(Warning {
155            message,
156            origin: Some(origin.to_string()),
157        });
158    }
159}
